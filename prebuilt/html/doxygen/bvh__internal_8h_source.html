23/* ═══════════════════════════════════════════════════════════════════════════
24 * 1. Float bounds that are always a superset of the double ones
25 *
26 * A plain (float) cast rounds to nearest and can shrink a box by one ulp, which
27 * is enough to exclude a vertex lying on its own leaf's face and report a clean
28 * miss. Rounding the minimum DOWN and the maximum UP keeps the float box a
29 * conservative superset: scaling by (1 -/+ 2.4e-7), i.e. two float ulps
30 * relative, leaves a margin of at least one ulp after both roundings, and an
31 * absolute 1e-30 keeps the rounding outward through zero and the denormals.
32 * ═══════════════════════════════════════════════════════════════════════════ */
33
34/** @brief Relative slack, about two float ulps (\f$2^{-22}\f$). */
35#define MFTL_BVH_FLOAT_SLACK 2.4e-7f
36/** @brief Absolute slack, so the rounding is still outward through zero. */
37#define MFTL_BVH_FLOAT_TINY 1.0e-30f
38
39/**
40 * @brief A float guaranteed to be below @p v (outward rounding of a minimum).
41 * @param v The double to round down.
42 * @return A float strictly below @p v.
43 */
44static inline float mftl_bvh_floor_f(double v)
45{
46 const float f = (float) v;
47 return (f >= 0.0f) ? (f * (1.0f - MFTL_BVH_FLOAT_SLACK) - MFTL_BVH_FLOAT_TINY) : (f * (1.0f + MFTL_BVH_FLOAT_SLACK) - MFTL_BVH_FLOAT_TINY);
48}
49
50/**
51 * @brief A float guaranteed to be above @p v (outward rounding of a maximum).
52 * @param v The double to round up.
53 * @return A float strictly above @p v.
54 */
55static inline float mftl_bvh_ceil_f(double v)
56{
57 const float f = (float) v;
58 return (f >= 0.0f) ? (f * (1.0f + MFTL_BVH_FLOAT_SLACK) + MFTL_BVH_FLOAT_TINY) : (f * (1.0f - MFTL_BVH_FLOAT_SLACK) + MFTL_BVH_FLOAT_TINY);
59}

61/* ═══════════════════════════════════════════════════════════════════════════
62 * 2. AABB overlap, with the periodic image offset added to the B side.
63 * ═══════════════════════════════════════════════════════════════════════════ */
64
65/**
66 * @brief Do two node boxes overlap, with @p b displaced by @p shift?
67 *
68 * @param a Node from the A-side tree.
69 * @param b Node from the B-side tree.
70 * @param shift Periodic image offset added to @p b. All zeros for the identity
71 * image.
72 * @return `true` if the two boxes intersect on all three axes.
73 */
74static inline bool mftl_bvh_aabb_overlap(const MFTL_BVHNode *a, const MFTL_BVHNode *b, const float shift[3])
75{
76 for (int k = 0; k < 3; ++k)
77 {
78 if (a->aabb_max[k] < b->aabb_min[k] + shift[k]) return MFTL_FALSE;
79 if (a->aabb_min[k] > b->aabb_max[k] + shift[k]) return MFTL_FALSE;
80 }
81 return MFTL_TRUE;
82}

84/**
85 * @brief Volume of a node's box; the descent heuristic picks the larger one.
86 * @param n The node.
87 * @return \f$\prod_k (\mathrm{max}_k - \mathrm{min}_k)\f$.
88 */
89static inline float mftl_bvh_node_volume(const MFTL_BVHNode *n)
90{
91 return (n->aabb_max[0] - n->aabb_min[0]) * (n->aabb_max[1] - n->aabb_min[1]) * (n->aabb_max[2] - n->aabb_min[2]);
92}

100/**
101 * @brief Refits every node's box and resolves the tolerance, in three linear
102 * passes with no recursion.
103 *
104 * 1. Reverse scan over `[nodes_used-1 .. 0]`: leaves fit their primitives,
105 * internal nodes merge their children (every child index exceeds its
106 * parent's, so children are always visited first).
107 * 2. `bounding_radius` from the root box; `tolerance` from it (or from
108 * `radius_hint`).
109 * 3. Forward scan inflating every node by the tolerance -- valid because a
110 * uniform inflation commutes with the min/max merge.
111 *
112 * Shared by MFTL_GenerateLBVH() and MFTL_RefitLBVH()
113 * (`test_refit_only_matches_full_rebuild`).
114 *
115 * @param bvh The hierarchy; boxes, `bounding_radius` and `tolerance` rewritten.
116 */
117void MFTLBVHRefit(MFTL_BVH *bvh);
