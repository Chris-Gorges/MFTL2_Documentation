1/**
2 * @file mftl_object.h
3 * @author Christian Gorges
4 * @date 2026-08-14
5 * @ingroup mftl_object
6 *
7 * @brief The mesh container: pool lifecycle, derived connectivity, Morton
8 * ordering, and the hot-loop iterators.
9 *
10 * An #MFTL_Object is one mesh (one body). It owns two pools (mftl_types.h) and
11 * borrows the #MFTL_Domain of the ensemble it was added to.
12 *
13 * **Size has one entry point.** Only src/object/object.c writes a pool's
14 * `capacity`, `block` or sub-array pointers. Counts change through
15 * MFTLObjectSetCounts(), which reserves before it assigns, so `n <= capacity`
16 * always holds; capacity grows through MFTLObjectReserve() /
17 * MFTLObjectReserveAdditional() and is given back only explicitly
18 * (MFTLObjectShrinkToFit(), MFTLObjectReleasePools()).
19 *
20 * **Derived connectivity** (`tris.verts`, `tris.neighbors`) is invalidated by
21 * any topology change and rebuilt by MFTLObjectBuildDerived(). Vertex motion
22 * invalidates nothing: geometric loops gather current positions and form
23 * seam-corrected edges in registers, so no position-derived cache exists.
24 *
25 * Theory: src/object/README.md.
26 */

132/**
133 * @brief Gives back pool capacity a mesh has outgrown, if the waste is large.
134 *
135 * The only way a pool gets smaller. Each pool is resized independently when
136 *
137 * \f[ \mathrm{capacity} > \mathrm{slack} \cdot n , \f]
138 *
139 * to `MFTLGrowCapacity(0, n)` -- the smallest \f$128\cdot2^k \ge n\f$, i.e.

141 * contents, the derived bit and the partition stamp are unchanged: this moves

143 * MFTLObjectSetCounts(), which would stamp the body dirty and defeat the

145 *

147 * @param slack Waste factor, at least 1; use #MFTL_SHRINK_SLACK_DEFAULT (4).
148 * Smaller values thrash against the doubling growth ladder.
149 * @return `true` if either pool was reallocated.
150 *
151 * @warning Every pointer into either pool is dangling afterwards.
152 *
153 * @rank_local
154 */
155bool MFTLObjectShrinkToFit(MFTL_Object *obj, double slack);

202/**
203 * @brief Declares that the current partition no longer describes this mesh.
204 *
205 * MFTLRedistribute() leaves a body untouched when no entity changed owner,
206 * because a rank's post-redistribution state is a function of (global
207 * topology, owner map) only. This stamp reports that something other than the
208 * owner map changed, so the body must be rebuilt anyway. Three kinds of change
209 * need it, and the library stamps all three:
210 *
211 * 1. entity counts -- MFTLObjectSetCounts() stamps;
212 * 2. global ids -- the remeshers' renumbering stamps (ghost triangle ids are
213 * repaired only by being dropped and re-received in a rebuild);
214 * 3. entity flags -- MFTLRemeshMarkFrozen() stamps (a kept ghost vertex keeps
215 * its local record through a redistribution).
216 *
217 * Any other in-place edit that bypasses MFTLObjectSetCounts() owes this call.
218 * A spurious stamp costs one rebuild; a missing one silently corrupts ids.
219 *
220 * @param obj The mesh; only `partition_dirty` is written.
221 *
222 * @rank_local Redistribution reduces the per-body votes before acting.
223 */
224static inline void MFTLObjectMarkPartitionDirty(MFTL_Object *obj) { obj->partition_dirty = MFTL_TRUE; }

245/** @brief Bits per axis of a Morton key: \f$3 \times 21 = 63\f$ fit a `uint64_t`. */
246#define MFTL_MORTON_BITS 21
247/** @brief Largest quantised Morton coordinate, \f$2^{21}-1\f$. */
248#define MFTL_MORTON_MAX 2097151.0
249
250/**
251 * @brief Spreads the low 21 bits of @p x so that bit i moves to bit 3i.
252 *
253 * Five shift-and-mask steps (magic-number bit interleave), \f$O(\log b)\f$.
254 *
255 * @param x Value whose low 21 bits are spread; higher bits are masked off.
256 * @return \f$\sum_i x_i 2^{3i}\f$.
257 */
258static inline uint64_t MFTLMortonSpread3(uint64_t x)
259{
260 x &= UINT64_C(0x1fffff);
261 x = (x | (x << 32)) & UINT64_C(0x1f00000000ffff);
262 x = (x | (x << 16)) & UINT64_C(0x1f0000ff0000ff);
263 x = (x | (x << 8)) & UINT64_C(0x100f00f00f00f00f);
264 x = (x | (x << 4)) & UINT64_C(0x10c30c30c30c30c3);
265 x = (x | (x << 2)) & UINT64_C(0x1249249249249249);
266 return x;
267}

269/**
270 * @brief 63-bit Morton (Z-order) key of three 21-bit coordinates.
271 *
272 * \f[
273 * Z(x,y,z) = \sum_{i=0}^{20} \Big( x_i 2^{3i} + y_i 2^{3i+1} + z_i 2^{3i+2} \Big)
274 * \f]
275 *
276 * Ascending \f$Z\f$ traverses the Z-order curve; points in a common octree
277 * cell share a key prefix, which the renumbering and the LBVH rely on.
278 *
279 * @param x Quantised x coordinate, 21 bits.
280 * @param y Quantised y coordinate, 21 bits.
281 * @param z Quantised z coordinate, 21 bits.
282 * @return The interleaved key.
283 *
284 * @warning Inputs above #MFTL_MORTON_MAX are MASKED, not clamped.
285 */
286static inline uint64_t MFTLMorton3D(uint64_t x, uint64_t y, uint64_t z)
287{
288 return MFTLMortonSpread3(x) | (MFTLMortonSpread3(y) << 1) | (MFTLMortonSpread3(z) << 2);
289}

291/**
292 * @brief Renumbers both pools along a Morton curve, so that neighbours in
293 * space are neighbours in memory.
294 *
295 * The permutation is APPLIED to the pools rather than stored, so there remains
296 * one index space and one copy of every coordinate:
297 *
298 * 1. a Morton key per vertex from `verts.pos`, quantised to 21 bits per axis
299 * against the object's own bounding box with one common scale (cubic cells);
300 * 2. a stable sort of `[0, n_local)` and, separately, of `[n_local, n)`, so the
301 * ownership prefix survives; all vertex arrays are permuted and
302 * `he.vertex_idx` remapped;
303 * 3. triangles are ordered by their smallest (already Morton-ordered) vertex
304 * index, again owned and ghost separately; records, half-edge triples and
305 * geometry arrays move together, `twin_idx` and `out_he_idx` are remapped;
306 * 4. the derived arrays are rebuilt.
307 *
308 * Global ids are untouched; pool indices are not stable across the call. A body
309 * straddling a periodic seam gets a domain-wide box and coarser cells -- a
310 * locality cost only. No time benefit has been measured on the development
311 * machine (docs/analysis.md); its value is the single index space and LBVH
312 * locality.
313 *
314 * @param obj The mesh; both pools are permuted in place.
315 *
316 * @rank_local Each rank orders its own partition; no MPI.
317 */
318void MFTLObjectReorderMorton(MFTL_Object *obj);

320/* ═══════════════════════════════════════════════════════════════════════════
321 * 5. HOT-LOOP ITERATORS
322 *
323 * Macros, because C has no closures: the body is pasted into one flat loop over
324 * contiguous arrays that the optimiser sees whole. Every iterator hoists the
325 * arrays into `restrict` locals and the bound into a local, reads connectivity
326 * sequentially and keeps seam-corrected edges in registers. The body is
327 * variadic (it may contain commas); `break` and `continue` work as in a `for`.
328 * Position reads are indexed gathers, so vectorisation is left to the compiler
329 * and never forced.
330 * ═══════════════════════════════════════════════════════════════════════════ */
331
332/**
333 * @brief Iterates triangle geometry from compact connectivity.
334 *
335 * Streams `tris.verts`, gathers the three current positions and forms two
336 * minimum-image edge vectors in registers. Periodicity is decided once, outside
337 * the loop: the non-periodic branch is gathers and subtractions only, the
338 * periodic one applies \f$d - L\,\mathrm{nearbyint}(d/L)\f$ branch-free, with
339 * a zero reciprocal on non-periodic axes.
340 *
341 * In scope for the body:
342 *
343 * int t triangle pool index
344 * int v0, v1, v2 vertex pool indices (corner k = origin of half-edge 3t+k)
345 * double p0x, p0y, p0z stored position of corner 0
346 * double e1x, e1y, e1z p1 - p0, minimum image
347 * double e2x, e2y, e2z p2 - p0, minimum image
348 *
349 * Requires #MFTL_DERIVED_TOPOLOGY (asserted in Debug).
350 *
351 * @param obj The mesh.
352 * @param count Triangles to visit from index 0: `obj->tris.n_local` for owned
353 * only, `obj->tris.n` to include ghosts.
354 */
355#define MFTL_FOREACH_TRIANGLE_GEOMETRY(obj, count, ...) \
356 do \
357 { \
358 MFTL_ASSERT(MFTLObjectHasTopology(obj) && "MFTL_FOREACH_TRIANGLE_GEOMETRY: stale tris.verts -- rebuild derived connectivity first"); \
359 const int *restrict mftl__tv = (obj)->tris.verts; \
360 const double *restrict mftl__pos = (obj)->verts.pos; \
361 const MFTL_Domain *const mftl__dom = (obj)->domain; \
362 const int mftl__n = (count); \
363 MFTL_ASSERT(mftl__n <= (obj)->tris.n && "MFTL_FOREACH_TRIANGLE_GEOMETRY: count exceeds triangle pool"); \
364 const bool mftl__periodic = mftl__dom != MFTL_NULL && \
365 (mftl__dom->is_periodic[0] || mftl__dom->is_periodic[1] || mftl__dom->is_periodic[2]); \
366 if (!mftl__periodic) \
367 { \
368 for (int t = 0; t < mftl__n; ++t) \
369 { \
370 const int v0 = mftl__tv[3 * t + 0], v1 = mftl__tv[3 * t + 1], v2 = mftl__tv[3 * t + 2]; \
371 const double *const mftl__p0 = &mftl__pos[3 * v0]; \
372 const double *const mftl__p1 = &mftl__pos[3 * v1]; \
373 const double *const mftl__p2 = &mftl__pos[3 * v2]; \
374 const double p0x = mftl__p0[0], p0y = mftl__p0[1], p0z = mftl__p0[2]; \
375 const double e1x = mftl__p1[0] - p0x, e1y = mftl__p1[1] - p0y, e1z = mftl__p1[2] - p0z; \
376 const double e2x = mftl__p2[0] - p0x, e2y = mftl__p2[1] - p0y, e2z = mftl__p2[2] - p0z; \
377 __VA_ARGS__; \
378 } \
379 } \
380 else \
381 { \
382 const double mftl__l0 = mftl__dom->domain_size[0]; \
383 const double mftl__l1 = mftl__dom->domain_size[1]; \
384 const double mftl__l2 = mftl__dom->domain_size[2]; \
385 const double mftl__i0 = mftl__dom->is_periodic[0] ? mftl__dom->inv_domain_size[0] : 0.0; \
386 const double mftl__i1 = mftl__dom->is_periodic[1] ? mftl__dom->inv_domain_size[1] : 0.0; \
387 const double mftl__i2 = mftl__dom->is_periodic[2] ? mftl__dom->inv_domain_size[2] : 0.0; \
388 for (int t = 0; t < mftl__n; ++t) \
389 { \
390 const int v0 = mftl__tv[3 * t + 0], v1 = mftl__tv[3 * t + 1], v2 = mftl__tv[3 * t + 2]; \
391 const double *const mftl__p0 = &mftl__pos[3 * v0]; \
392 const double *const mftl__p1 = &mftl__pos[3 * v1]; \
393 const double *const mftl__p2 = &mftl__pos[3 * v2]; \
394 const double p0x = mftl__p0[0], p0y = mftl__p0[1], p0z = mftl__p0[2]; \
395 const double mftl__d10 = mftl__p1[0] - p0x, mftl__d11 = mftl__p1[1] - p0y, mftl__d12 = mftl__p1[2] - p0z; \
396 const double mftl__d20 = mftl__p2[0] - p0x, mftl__d21 = mftl__p2[1] - p0y, mftl__d22 = mftl__p2[2] - p0z; \
397 const double e1x = mftl__d10 - mftl__l0 * nearbyint(mftl__d10 * mftl__i0); \
398 const double e1y = mftl__d11 - mftl__l1 * nearbyint(mftl__d11 * mftl__i1); \
399 const double e1z = mftl__d12 - mftl__l2 * nearbyint(mftl__d12 * mftl__i2); \
400 const double e2x = mftl__d20 - mftl__l0 * nearbyint(mftl__d20 * mftl__i0); \
401 const double e2y = mftl__d21 - mftl__l1 * nearbyint(mftl__d21 * mftl__i1); \
402 const double e2z = mftl__d22 - mftl__l2 * nearbyint(mftl__d22 * mftl__i2); \
403 __VA_ARGS__; \
404 } \
405 } \
406 } while (0)
