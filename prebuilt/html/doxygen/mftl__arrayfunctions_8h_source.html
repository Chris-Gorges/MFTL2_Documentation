1/**
2 * @file mftl_arrayfunctions.h
3 * @author Christian Gorges
4 * @date 2025-08-12
5 * @ingroup mftl_arrayfunctions
6 *
7 * @brief Flat-array utilities: LSD radix sort and duplicate removal.
8 *
9 * Layer 0, no MFTL types. The sorts are least-significant-digit radix sorts
10 * over 8-bit digits, \f$O(n)\f$ for a fixed key width (a stable insertion sort
11 * below 33 elements); the unique helpers are single scans over sorted input.
12 * Theory and measurements: src/arrayfunctions/README.md.
13 *
14 * **Ordering contract.** The key type is the contract: `int` keys sort in
15 * SIGNED order, `uint64_t` keys in UNSIGNED order. Do not sort unsigned 32-bit
16 * values through an `int *` cast (wrong for values \f$\ge 2^{31}\f$).
17 *
18 * **Stability.** Every sort is stable, so "sort pairs, then MFTLUniquePairs()"
19 * keeps the value that came first -- a deterministic, rank-independent choice.
20 */

101/**
102 * @brief Key-only radix sort, dispatched on the key type at compile time.
103 *
104 * `int *` selects MFTLRadixSort32(), `uint64_t *` selects MFTLRadixSort64().
105 * The selection is a C11 `_Generic` expression, so it costs nothing at run time
106 * and a key type that is neither is a compile error.
107 *
108 * @param keys_in Array to sort, in place.
109 * @param keys_temp Scratch array of the same length and type. Caller-owned;
110 * after the sort its contents are unspecified.
111 * @param size Number of elements.
112 *
113 * @note Under C99 (no `_Generic`) this expands to the 32-bit variant
114 * unconditionally, so passing 64-bit keys is a compile error rather than a
115 * silent mis-sort. Call MFTLRadixSort64() directly there.
116 */
117#define MFTLRadixSort(keys_in, keys_temp, size) _Generic((keys_in), int *: MFTLRadixSort32, uint64_t *: MFTLRadixSort64)((keys_in), (keys_temp), (size))

144/**
145 * @brief Collects the distinct values of an array, preserving first-seen order.
146 *
147 * \f$O(n \cdot u)\f$ for \f$u\f$ distinct values: for tiny sets (a handful of
148 * ranks). For larger inputs sort, then MFTLUniqueIntArray().
149 *
150 * @note Values beyond @p output_capacity are dropped silently; size the output
151 * for the worst case.
152 *
153 * @param input_array Values to scan. Not modified, need not be sorted.
154 * @param input_count Number of values in @p input_array.
155 * @param[out] output_array Receives the distinct values, in first-seen order.
156 * @param output_capacity Elements @p output_array can hold.
157 * @return The number of distinct values written to @p output_array.
158 */
159static inline int MFTLGetUniqueIntegers(const int *input_array, int input_count, int *output_array, int output_capacity)
160{
161 int num_unique = 0;
162 for (int j = 0; j < input_count; ++j)
163 {
164 int d = input_array[j];
165 int found = 0;
166 for (int k = 0; k < num_unique; ++k)
167 {
168 if (output_array[k] == d)
169 {
170 found = 1;
171 break;
172 }
173 }
174 if (!found && num_unique < output_capacity) output_array[num_unique++] = d;
175 }
176 return num_unique;
177}

179/**
180 * @brief Binary search for @p key in an ascending int array.
181 *
182 * Sparse id-keyed directories in the library are sorted arrays, so an
183 * `id -> slot` lookup is this binary search rather than a hash map.
184 *
185 * @param sorted Ascending array.
186 * @param n Number of elements.
187 * @param key The value.
188 * @return An index `i` with `sorted[i] == key`, or -1 (#MFTL_ERROR) if absent.
189 * With duplicate keys it is one of their indices.
190 */
191static inline int MFTLSortedIndexOf(const int *sorted, int n, int key)
192{
193 int lo = 0, hi = n - 1;
194 while (lo <= hi)
195 {
196 const int mid = lo + ((hi - lo) >> 1);
197 if (sorted[mid] == key) return mid;
198 if (sorted[mid] < key)
199 lo = mid + 1;
200 else
201 hi = mid - 1;
202 }
203 return -1;
204}

206/**
207 * @brief Removes adjacent duplicates from a SORTED int array, in place.
208 *
209 * Compares neighbours only, so on unsorted input it merely collapses runs.
210 *
211 * @param arr Sorted array, compacted in place.
212 * @param n Number of elements on entry.
213 * @return The new element count; elements beyond it are left as garbage.
214 */
215static inline int MFTLUniqueIntArray(int *arr, int n)
216{
217 if (n <= 0) return 0;
218 int dest = 0;
219 for (int src = 1; src < n; ++src)
220 {
221 if (arr[src] != arr[dest]) arr[++dest] = arr[src];
222 }
223 return dest + 1;
224}
225
226/**
227 * @brief Removes adjacent duplicate KEYS from a SORTED key/value pair, in place.
228 *
229 * Keeps the first value of each run of equal keys; after a stable sort that is
230 * the entry that came first in the caller's original array.
231 *
232 * @param keys Sorted key array, compacted in place.
233 * @param values Value array, compacted alongside @p keys.
234 * @param n Number of pairs on entry.
235 * @return The new pair count.
236 */
237static inline int MFTLUniquePairs(int *keys, int *values, int n)
238{
239 if (n <= 0) return 0;
240 int dest = 0;
241 for (int src = 1; src < n; ++src)
242 {
243 if (keys[src] != keys[dest])
244 {
245 ++dest;
246 keys[dest] = keys[src];
247 values[dest] = values[src];
248 }
249 }
250 return dest + 1;
251}
