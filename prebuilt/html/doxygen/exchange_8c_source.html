24/* ═══════════════════════════════════════════════════════════════════════════
25 * 1. MPI datatypes
26 *
27 * Every wire struct groups its members into runs of ONE base type
28 * (mftl_communication.h), so each datatype below is one or two CONTIGUOUS blocks
29 * rather than one block per field. The static assertions pin exactly that: they
30 * fail the BUILD if a future field breaks a run, instead of letting MPI silently
31 * transmit the wrong bytes.
32 * ═══════════════════════════════════════════════════════════════════════════ */

44/**
45 * @brief Builds a struct datatype whose extent is pinned to the C `sizeof`
46 * (`MPI_Type_create_resized`), so an array of it has the C stride
47 * regardless of how trailing padding falls.
48 *
49 * @param n_blocks Number of member blocks.
50 * @param blocklengths Element count of each block.
51 * @param offsets Byte offset of each block from the struct base.
52 * @param types MPI datatype of each block's elements.
53 * @param c_size The C `sizeof` of the struct, pinned as the type's extent.
54 * @return The committed datatype.
55 */
56static MPI_Datatype commit_struct(int n_blocks, const int *blocklengths, const MPI_Aint *offsets, const MPI_Datatype *types, size_t c_size)
57{
58 MPI_Datatype raw, out;
59 MPI_Type_create_struct(n_blocks, (int *) blocklengths, (MPI_Aint *) offsets, (MPI_Datatype *) types, &raw);
60 MPI_Type_create_resized(raw, 0, (MPI_Aint) c_size, &out);
61 MPI_Type_free(&raw);
62 MPI_Type_commit(&out);
63 return out;
64}

108void MFTLExchangeStart(const MFTL_Domain *domain, MFTL_Exchange *x, int n_payloads, bool use_global_path, MFTL_ExchangeCtx *ctx)
109{
110 MFTL_ASSERT(n_payloads >= 1 && n_payloads <= MFTL_MAX_EXCHANGE_PAYLOADS && "MFTLExchangeStart: payload count out of range");
111 MFTL_ASSERT((use_global_path || domain->comm_graph != MPI_COMM_NULL) && "MFTLExchangeStart: neighbour path without a graph communicator");
112
113 const int np = n_payloads;
114 const int nt = use_global_path ? domain->nproc : domain->num_neighbors;
115 const MPI_Comm comm = use_global_path ? domain->comm_cart : domain->comm_graph;
116
117 ctx->n_payloads = np;
118 ctx->n_targets = nt;
119 for (int p = 0; p < np; ++p)
120 {
121 ctx->requests[p] = MPI_REQUEST_NULL;
122 ctx->flat_send[p] = MFTL_NULL;
123 }
124
125 /* One allocation holding [send counts | recv counts | send displs | recv displs],
126 * each np*nt ints laid out payload-major. */
127 const size_t stride = (size_t) (nt > 0 ? nt : 1);
128 const size_t block = (size_t) np * stride;
129 ctx->counts = (int *) MFTLCallocOrDie(4 * block, sizeof(int), "MFTLExchangeStart counts");
130 int *const s_cnt = ctx->counts;
131 int *const r_cnt = ctx->counts + block;
132 int *const s_dsp = ctx->counts + 2 * block;
133 int *const r_dsp = ctx->counts + 3 * block;
134
135 /* ── 1. ONE count exchange for all payloads (latency, not volume, is its
136 * cost), staged [target][payload] and de-interleaved. */
137 CS_BEGIN(ts_counts);
138 int *stage = (int *) MFTLMallocOrDie(2 * block * sizeof(int), "MFTLExchangeStart staging");
139 int *stage_recv = stage + block;
140
141 for (int i = 0; i < nt; ++i)
142 {
143 const int rank = use_global_path ? i : domain->neighbors[i];
144 for (int p = 0; p < np; ++p) stage[(size_t) i * np + p] = x[p].send_bufs[rank].count;
145 }
146
147 if (use_global_path)
148 MPI_Alltoall(stage, np, MPI_INT, stage_recv, np, MPI_INT, comm);
149 else
150 MPI_Neighbor_alltoall(stage, np, MPI_INT, stage_recv, np, MPI_INT, comm);
151
152 for (int p = 0; p < np; ++p)
153 {
154 for (int i = 0; i < nt; ++i)
155 {
156 s_cnt[(size_t) p * stride + i] = stage[(size_t) i * np + p];
157 r_cnt[(size_t) p * stride + i] = stage_recv[(size_t) i * np + p];
158 }
159 }
160 free(stage);
161 cs_alloc(4 * block * sizeof(int) + 2 * block * sizeof(int));
162 cs_free();
163 CS_END(ts_counts, t_exchange_counts);
164
165 /* ── 2. Displacements, buffers, and the rank-ordered send image ─────────── */
166 CS_BEGIN(ts_data);
167 for (int p = 0; p < np; ++p)
168 {
169 int *const sc = s_cnt + (size_t) p * stride;
170 int *const rc = r_cnt + (size_t) p * stride;
171 int *const sd = s_dsp + (size_t) p * stride;
172 int *const rd = r_dsp + (size_t) p * stride;
173
174 int s_tot = 0, r_tot = 0;
175 for (int i = 0; i < nt; ++i)
176 {
177 sd[i] = s_tot;
178 s_tot += sc[i];
179 rd[i] = r_tot;
180 r_tot += rc[i];
181 }
182
183 x[p].recv_count = r_tot;
184
185 /* Never hand MPI a null buffer: the standard permits it for zero counts, but
186 * implementations differ on whether they validate the pointer first. */
187 x[p].recv_buf = MFTLMallocOrDie((size_t) (r_tot > 0 ? r_tot : 1) * x[p].type_size, "MFTLExchangeStart recv");
188 ctx->flat_send[p] = (char *) MFTLMallocOrDie((size_t) (s_tot > 0 ? s_tot : 1) * x[p].type_size, "MFTLExchangeStart send");
189 cs_alloc((size_t) ((r_tot > 0 ? r_tot : 1) + (s_tot > 0 ? s_tot : 1)) * x[p].type_size);
190
191 for (int i = 0; i < nt; ++i)
192 {
193 if (sc[i] <= 0) continue;
194 const int rank = use_global_path ? i : domain->neighbors[i];
195 memcpy(ctx->flat_send[p] + (size_t) sd[i] * x[p].type_size, x[p].send_bufs[rank].buffer, (size_t) sc[i] * x[p].type_size);
196 }
197 }
198
199 /* ── 3. Post every payload back to back; the caller may consume one while the
200 * others are in flight. The posting order is fixed by the code. */
201 for (int p = 0; p < np; ++p)
202 {
203 int *const sc = s_cnt + (size_t) p * stride;
204 int *const rc = r_cnt + (size_t) p * stride;
205 int *const sd = s_dsp + (size_t) p * stride;
206 int *const rd = r_dsp + (size_t) p * stride;
207
208 if (use_global_path)
209 MPI_Ialltoallv(ctx->flat_send[p], sc, sd, x[p].type, x[p].recv_buf, rc, rd, x[p].type, comm, &ctx->requests[p]);
210 else
211 MPI_Ineighbor_alltoallv(ctx->flat_send[p], sc, sd, x[p].type, x[p].recv_buf, rc, rd, x[p].type, comm, &ctx->requests[p]);
212 }
213
214 CS_END(ts_data, t_exchange_data);
215}

217void MFTLExchangeWait(MFTL_ExchangeCtx *ctx, int payload)
218{
219 if (payload < 0 || payload >= ctx->n_payloads) return;
220
221 if (ctx->requests[payload] != MPI_REQUEST_NULL) MPI_Wait(&ctx->requests[payload], MPI_STATUS_IGNORE);
222
223 /* The send image is dead the instant the transfer completed. */
224 free(ctx->flat_send[payload]);
225 ctx->flat_send[payload] = MFTL_NULL;
226}
227
228void MFTLExchangeFree(MFTL_ExchangeCtx *ctx)
229{
230 for (int p = 0; p < ctx->n_payloads; ++p) MFTLExchangeWait(ctx, p);
231 free(ctx->counts);
232 ctx->counts = MFTL_NULL;
233 ctx->n_payloads = 0;
234 ctx->n_targets = 0;
235}

237/* ═══════════════════════════════════════════════════════════════════════════
238 * 3. Neighbourhood allgather
239 * ═══════════════════════════════════════════════════════════════════════════ */
240
241double *MFTLNeighborAllgatherDoubles(const MFTL_Domain *domain, const double *send, int n_send, int *rcount, int *rdispl, int *n_recv)
242{
243 const int d = domain->num_neighbors;
244
245 *n_recv = 0;
246 if (domain->nproc < 2 || d < 1) return (double *) MFTLMallocOrDie(sizeof(double), "MFTLNeighborAllgatherDoubles (empty)");
247
248 MPI_Neighbor_allgather(&n_send, 1, MPI_INT, rcount, 1, MPI_INT, domain->comm_graph);
249
250 size_t total = 0;
251 for (int i = 0; i < d; ++i)
252 {
253 rdispl[i] = (int) total;
254 total += (size_t) rcount[i];
255 }
256
257 /* MPI counts are ints (reached only with millions of bodies on one rank). */
258 MFTL_ASSERT(total <= (size_t) 0x7fffffff && "MFTLNeighborAllgatherDoubles: neighbour payload exceeds an MPI int count");
259
260 double *const recv = (double *) MFTLMallocOrDie(MFTL_MAX(total, (size_t) 1) * sizeof(double), "MFTLNeighborAllgatherDoubles");
261 MPI_Neighbor_allgatherv(send, n_send, MPI_DOUBLE, recv, rcount, rdispl, MPI_DOUBLE, domain->comm_graph);
262
263 *n_recv = (int) total;
264 return recv;
265}
