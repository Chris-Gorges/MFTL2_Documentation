1/**
2 * @file remesh_smooth.c
3 * @author Christian Gorges
4 * @date 2026-08-15
5 * @ingroup mftl_remeshing
6 *
7 * @brief Feature classification and tangential smoothing.
8 *
9 * ## The smoother is a projection; the feature class picks the subspace
10 *
11 * Plain Laplacian smoothing shrinks geometry: the pull towards the
12 * neighbourhood centre has a component along the normal ("cut the corner"). So
13 * the displacement is projected onto the subspace the vertex may move in:
14 *
15 * FREE 2D, the tangent plane (remove the normal component);
16 * RIDGE 1D, the feature curve (keep the component along the ridge tangent);
17 * CORNER 0D, locked.
18 *
19 * Moving a ridge vertex TO the midpoint of its feature neighbours is the same
20 * shrinkage on a curve: a regular 24-gon ridge collapses to 0.84 of its radius
21 * in 20 sweeps (`test_a_regular_ridge_is_a_fixed_point`).
22 *
23 * ## The target point
24 *
25 * MFTL_SMOOTH_TRIANGLE_BASED is the AREA-WEIGHTED centroid of the incident
26 * triangles, q = SUM(A_i c_i) / SUM(A_i). The weight equalises triangle areas;
27 * the unweighted mean is nearly the umbrella operator (docs/analysis.md §4:
28 * volume loss 6.0 % unweighted against 3.5 % weighted).
29 *
30 * ## The normal
31 *
32 * MFTLComputeVertexNormals() gives the exact area-weighted normal of every
33 * vertex in one flat, seam-safe triangle pass; a per-vertex fan accumulation
34 * here would be slower (3.4x, src/geometry/vertex_geometry.c) and must handle
35 * open fans itself.
36 */

55/**
56 * @brief Outgoing half-edges of @p v in rotational order, plus fan openness.
57 *
58 * MFTLGetVertexOneRingHalfedges does the walk but does not report whether the
59 * fan closed, and every caller here needs to know: on an OPEN fan there is one
60 * more incident EDGE than there are outgoing half-edges, and it is reachable only
61 * as the incoming half-edge prev(first). A consumer that iterates edges and does
62 * not account for it silently drops one, and then misclassifies every boundary
63 * vertex.
64 *
65 * @param obj The mesh.
66 * @param v Vertex index.
67 * @param[out] out `[MFTL_REMESH_RING_MAX]` outgoing half-edges, rotational order.
68 * @param[out] open Set when the fan is open, i.e. @p v is on a boundary.
69 * @return Number of half-edges written, or 0 if the walk was truncated.
70 */
71static int fan_halfedges(const MFTL_Object *obj, int v, int *out, bool *open)
72{
73 const int first = obj->verts.rec[v].out_he_idx;
74 *open = MFTL_FALSE;
75 if (first == MFTL_ERROR) return 0;
76
77 const int start = MFTLVertexFanStart(obj, first, open);
78
79 int count = 0;
80 int curr = start;
81 do
82 {
83 if (count >= MFTL_REMESH_RING_MAX) return MFTL_ERROR;
84 out[count++] = curr;
85
86 const int twin = obj->tris.he[curr].twin_idx;
87 if (twin == MFTL_ERROR) break;
88 curr = MFTLHeNextIdx(twin);
89 } while (curr != start);
90
91 return count;
92}

94/**
95 * @brief The neighbour an outgoing-half-edge walk cannot reach: the closing edge
96 * of an open fan.
97 *
98 * @p first is the fan's first outgoing half-edge v->w. Its triangle is
99 * (v, w, x), so next(first) targets x, and the edge v-x exists only as the
100 * incoming half-edge prev(first). Without it a boundary vertex is short one
101 * neighbour and lands in the wrong feature class.
102 *
103 * @param obj The mesh.
104 * @param first First outgoing half-edge of the fan.
105 * @return The missing neighbour's vertex index.
106 */
107static inline int open_fan_last_neighbour(const MFTL_Object *obj, int first) { return MFTLHeTarget(obj, MFTLHeNextIdx(first)); }

109/* ═══════════════════════════════════════════════════════════════════════════

113int MFTLDetectFeatures(MFTL_Object *obj, double angle_degrees, double corner_angle_degrees, MFTL_RemeshStats *stats)
114{
115 MFTL_ASSERT(obj != MFTL_NULL);
116
117 const int n_he = 3 * obj->tris.n;
118 const int n_v = obj->verts.n;
119
120 MFTL_Vertex *const vr = obj->verts.rec;
121 MFTL_HalfEdge *const he = obj->tris.he;
122
123 /* Every vertex is reclassified from scratch, so the counts start clean even
124 * when there is no topology left to classify. */
125 for (int v = 0; v < n_v; ++v) vr[v].flags &= ~(MFTL_FLAG_FEATURE | MFTL_FLAG_CORNER);
126
127 if (n_he == 0) return 0;
128
129 /* The normals are recomputed rather than trusted. A detector reading whatever
130 * happens to be in `tris.normal` classifies the mesh as it was one topological
131 * edit ago, and after a split the new triangles' normals have never been
132 * written at all -- so the very edges most likely to be misjudged are the ones
133 * the remesher just created. */
134 MFTLComputeTriangleGeometry(obj, obj->tris.n);
135
136 /* A threshold of 180 degrees or more means "no dihedral angle is sharp". It is
137 * a supported production setting, not a debug switch: on a closed smooth
138 * interface every edge the detector flags is a false positive, and a false
139 * positive here LOCKS a vertex the smoother needed to move. */
140 /* An edge is sharp when the dot product of its two face normals falls BELOW the
141 * threshold, so "never sharp" is a threshold below every possible dot product,
142 * i.e. -2. Writing +2 here disables nothing -- it flags every edge in the mesh,
143 * which locks every vertex and silently turns the smoother off instead. */
144 const bool angle_enabled = (angle_degrees < 180.0);
145 const double threshold_dot = angle_enabled ? MFTL_COS(angle_degrees * MFTL_PI / 180.0) : -2.0;
146
147 const double *const tn = obj->tris.normal;
148
149 int n_feature_edges = 0;
150
151 for (int h = 0; h < n_he; ++h)
152 {
153 const int tw = he[h].twin_idx;
154
155 /* A mesh boundary is a feature of the SURFACE rather than of its curvature,
156 * so it is one at any threshold. It also has no dihedral angle to measure. */
157 if (tw == MFTL_ERROR)
158 {
159 /* "No twin" and "mesh boundary" are different statements, and only one is
160 * partition-independent (mftl_topology.h). On an OWNED triangle they
161 * coincide: the redistribution halo gives every owned triangle all three of
162 * its edge neighbours, so a missing twin really is the edge of the surface.
163 * On a GHOST triangle a missing twin is the outer rim of the halo, and
164 * flagging it would invent a feature curve that follows the partition --
165 * every rank would preserve a different one, and the "curve" would move
166 * with the next migration. MFTL_FLAG_BOUNDARY is the authoritative answer
167 * there, because it was written while the mesh was still complete and
168 * travels with the entity.
169 *
170 * MFTLHeIsGhost is identically false on a serial object, so this costs one
171 * predictable compare and changes nothing about the serial classification. */
172 if (MFTLHeIsGhost(obj, h) && !(he[h].flags & MFTL_FLAG_BOUNDARY))
173 {
174 he[h].flags &= ~MFTL_FLAG_FEATURE;
175 continue;
176 }
177
178 he[h].flags |= MFTL_FLAG_FEATURE;
179 n_feature_edges++;
180 continue;
181 }
182
183 if (h > tw) continue; // one visit per undirected edge
184
185 const int t1 = MFTLHeTriIdx(h);
186 const int t2 = MFTLHeTriIdx(tw);
187
188 const double dot = tn[3 * t1 + 0] * tn[3 * t2 + 0] + tn[3 * t1 + 1] * tn[3 * t2 + 1] + tn[3 * t1 + 2] * tn[3 * t2 + 2];
189
190 if (dot < threshold_dot)
191 {
192 he[h].flags |= MFTL_FLAG_FEATURE;
193 he[tw].flags |= MFTL_FLAG_FEATURE;
194 n_feature_edges++;
195 }
196 else
197 {
198 /* Cleared, not merely left alone: an edge that WAS sharp and has since been
199 * smoothed out must stop being a feature, or the flag set only ever grows
200 * and the mesh gradually freezes solid. */
201 he[h].flags &= ~MFTL_FLAG_FEATURE;
202 he[tw].flags &= ~MFTL_FLAG_FEATURE;
203 }
204 }
205
206 /* ── Vertex classification ────────────────────────────────────────────────
207 *
208 * Counted over UNDIRECTED edges in one linear pass, deliberately not by walking
209 * each vertex's fan. The fan walk visits outgoing half-edges, and the closing
210 * edge of an OPEN fan has no outgoing direction -- so every boundary vertex
211 * comes out one short and lands in the wrong class. mftl_topology.h documents
212 * that trap on MFTLGetVertexOneRingHalfedges. This pass cannot walk into it: it
213 * never asks a vertex anything, it asks the edges and tells the vertices. */
214 const size_t nv_alloc = (size_t) (n_v > 0 ? n_v : 1);
215 int *const feat_count = (int *) MFTLCallocOrDie(nv_alloc, sizeof(int), "remesh feature counts");
216
217 /* The first two feature neighbours of each vertex, recorded during the same
218 * pass. A vertex with exactly two of them needs their DIRECTIONS to decide
219 * whether it is a ridge point or a corner, and picking them up here costs two
220 * stores against a fan walk per candidate vertex afterwards. */
221 int *const feat_nbr = (int *) MFTLMallocOrDie(nv_alloc * 2 * sizeof(int), "remesh feature neighbours");
222 for (size_t i = 0; i < nv_alloc * 2; ++i) feat_nbr[i] = MFTL_ERROR;
223
224 for (int h = 0; h < n_he; ++h)
225 {
226 if (!(he[h].flags & MFTL_FLAG_FEATURE)) continue;
227
228 const int tw = he[h].twin_idx;
229 if (tw != MFTL_ERROR && h > tw) continue; // count the edge, not both of its halves
230
231 const int o = MFTLHeOrigin(obj, h);
232 const int t = he[h].vertex_idx;
233
234 if (feat_count[o] < 2) feat_nbr[2 * o + feat_count[o]] = t;
235 if (feat_count[t] < 2) feat_nbr[2 * t + feat_count[t]] = o;
236
237 feat_count[o]++;
238 feat_count[t]++;
239 }
240
241 /* A feature curve turning by more than corner_angle_degrees is not a curve the
242 * vertex may slide along -- it is a corner OF that curve. Straight-through means
243 * the two edge directions are opposite, i.e. their normalised dot product is
244 * -1, so the test is "further from -1 than the threshold allows". */
245 const double straight_dot = -MFTL_COS(corner_angle_degrees * MFTL_PI / 180.0);
246
247 int n_corner = 0;
248 int n_ridge = 0;
249
250 for (int v = 0; v < n_v; ++v)
251 {
252 const int c = feat_count[v];
253 if (c == 0) continue;
254
255 bool is_ridge = MFTL_FALSE;
256
257 if (c == 2 && feat_nbr[2 * v] != MFTL_ERROR && feat_nbr[2 * v + 1] != MFTL_ERROR)
258 {
259 double d1[3];
260 double d2[3];
261 const double *const p = MFTLVertexPos(obj, v);
262 MFTLGetMinimumImageVector(obj->domain, d1, p, MFTLVertexPos(obj, feat_nbr[2 * v]));
263 MFTLGetMinimumImageVector(obj->domain, d2, p, MFTLVertexPos(obj, feat_nbr[2 * v + 1]));
264
265 const double l1 = d1[0] * d1[0] + d1[1] * d1[1] + d1[2] * d1[2];
266 const double l2 = d2[0] * d2[0] + d2[1] * d2[1] + d2[2] * d2[2];
267
268 if (l1 > 0.0 && l2 > 0.0)
269 {
270 const double dot = (d1[0] * d2[0] + d1[1] * d2[1] + d1[2] * d2[2]) / MFTL_SQRT(l1 * l2);
271 is_ridge = (dot <= straight_dot);
272 }
273 }
274
275 if (is_ridge)
276 {
277 /* Two feature edges continuing roughly straight make a curve passing
278 * THROUGH the vertex: it may slide along that curve. */
279 vr[v].flags |= MFTL_FLAG_FEATURE;
280 n_ridge++;
281 }
282 else
283 {
284 /* One feature edge is the free END of a curve, three or more is a junction,
285 * and two that turn sharply is a corner of the curve itself. All three are
286 * points the curve is pinned to; sliding any of them deforms the feature
287 * rather than redistributing along it. */
288 vr[v].flags |= MFTL_FLAG_CORNER;
289 n_corner++;
290 }
291 }
292
293 free(feat_nbr);
294 free(feat_count);
295
296 if (stats)
297 {
298 stats->n_feature_edges = n_feature_edges;
299 stats->n_corner_vertices = n_corner;
300 stats->n_ridge_vertices = n_ridge;
301 stats->n_free_vertices = n_v - n_corner - n_ridge;
302 }
303
304 return n_feature_edges;
305}

311/**
312 * @brief Displacement of a RIDGE vertex, projected onto its feature curve.
313 *
314 * @param obj The mesh.
315 * @param v A ridge vertex: exactly two feature edges that continue
316 * roughly straight through it.
317 * @param fan Outgoing half-edges of @p v, from fan_halfedges().
318 * @param n_fan Their count.
319 * @param open `true` when the fan is open.
320 * @param[out] disp The displacement, projected onto the ridge TANGENT.
321 * @param[out] min_edge_sq Shortest incident squared edge length, for the step clamp.
322 * @return `true` if a displacement was produced.
323 */
324static bool ridge_displacement(const MFTL_Object *obj, int v, const int *fan, int n_fan, bool open, double disp[3], double *min_edge_sq)
325{
326 const double *const p = MFTLVertexPos(obj, v);
327
328 int nbr[2];
329 int n_nbr = 0;
330
331 for (int i = 0; i < n_fan && n_nbr < 2; ++i)
332 {
333 if (obj->tris.he[fan[i]].flags & MFTL_FLAG_FEATURE) nbr[n_nbr++] = MFTLHeTarget(obj, fan[i]);
334 }
335
336 /* The closing edge of an open fan, which has no outgoing half-edge to carry
337 * its flag -- it is prev(first), reachable only as an incoming edge. */
338 if (open && n_nbr < 2 && n_fan > 0)
339 {
340 const int closing = MFTLHePrevIdx(fan[0]);
341 if (obj->tris.he[closing].flags & MFTL_FLAG_FEATURE) nbr[n_nbr++] = open_fan_last_neighbour(obj, fan[0]);
342 }
343
344 if (n_nbr != 2) return MFTL_FALSE;
345
346 double d1[3];
347 double d2[3];
348 MFTLGetMinimumImageVector(obj->domain, d1, p, MFTLVertexPos(obj, nbr[0]));
349 MFTLGetMinimumImageVector(obj->domain, d2, p, MFTLVertexPos(obj, nbr[1]));
350
351 const double l1 = d1[0] * d1[0] + d1[1] * d1[1] + d1[2] * d1[2];
352 const double l2 = d2[0] * d2[0] + d2[1] * d2[1] + d2[2] * d2[2];
353 *min_edge_sq = MFTL_MIN(l1, l2);
354
355 /* tau spans the curve; delta is the Laplacian pull towards the chord midpoint.
356 * Keeping only delta's component along tau redistributes the vertex BETWEEN its
357 * neighbours -- equalising the two edge lengths -- while the component that
358 * would have pulled it off the curve and onto the chord is discarded. That
359 * discarded component is the whole difference between this and a chord cut, and
360 * it is the one that erodes ridges. */
361 const double tau[3] = {d2[0] - d1[0], d2[1] - d1[1], d2[2] - d1[2]};
362 const double tau_len_sq = tau[0] * tau[0] + tau[1] * tau[1] + tau[2] * tau[2];
363 /* Relative to the two edges, not to 1: the test has to mean "the neighbours
364 * coincide" at a millimetre as much as at a metre. */
365 if (!(tau_len_sq > MFTL_VERYVERYSMALL * (l1 + l2))) return MFTL_FALSE;
366
367 const double delta[3] = {0.5 * (d1[0] + d2[0]), 0.5 * (d1[1] + d2[1]), 0.5 * (d1[2] + d2[2])};
368 const double s = (delta[0] * tau[0] + delta[1] * tau[1] + delta[2] * tau[2]) / tau_len_sq;
369
370 disp[0] = s * tau[0];
371 disp[1] = s * tau[1];
372 disp[2] = s * tau[2];
373 return MFTL_TRUE;
374}

376/**
377 * @brief Displacement of a FREE vertex, projected onto its tangent plane.
378 *
379 * The neighbourhood sums are accumulated as minimum-image OFFSETS from the
380 * vertex, so a fan straddling a periodic seam is averaged correctly.
381 *
382 * @param obj The mesh.
383 * @param v A vertex with no feature edges.
384 * @param fan Outgoing half-edges of @p v, from fan_halfedges().
385 * @param n_fan Their count.
386 * @param open `true` when the fan is open.
387 * @param mode Which neighbourhood the target point is built from.
388 * @param[out] disp The displacement, projected onto the tangent plane.
389 * @param[out] min_edge_sq Shortest incident squared edge length, for the step clamp.
390 * @return `true` if a displacement was produced.
391 */
392static bool free_displacement(const MFTL_Object *obj, int v, const int *fan, int n_fan, bool open, MFTL_SmoothMode mode, double disp[3], double *min_edge_sq)
393{
394 const double *const p = MFTLVertexPos(obj, v);
395
396 double target[3] = {0.0, 0.0, 0.0}; // offset of the target point from p
397 double weight_sum = 0.0;
398 double min_sq = MFTL_DBL_MAX;
399
400 if (mode == MFTL_SMOOTH_VERTEX_BASED)
401 {
402 for (int i = 0; i < n_fan; ++i)
403 {
404 double d[3];
405 MFTLGetMinimumImageVector(obj->domain, d, p, MFTLVertexPos(obj, MFTLHeTarget(obj, fan[i])));
406
407 target[0] += d[0];
408 target[1] += d[1];
409 target[2] += d[2];
410 weight_sum += 1.0;
411 min_sq = MFTL_MIN(min_sq, d[0] * d[0] + d[1] * d[1] + d[2] * d[2]);
412 }
413
414 if (open && n_fan > 0)
415 {
416 double d[3];
417 MFTLGetMinimumImageVector(obj->domain, d, p, MFTLVertexPos(obj, open_fan_last_neighbour(obj, fan[0])));
418
419 target[0] += d[0];
420 target[1] += d[1];
421 target[2] += d[2];
422 weight_sum += 1.0;
423 min_sq = MFTL_MIN(min_sq, d[0] * d[0] + d[1] * d[1] + d[2] * d[2]);
424 }
425 }
426 else
427 {
428 /* One outgoing half-edge per incident triangle, and the triangle's other two
429 * corners are target(o) and target(next(o)). Its centroid offset from p is
430 * (d1 + d2) / 3 and twice its area is |d1 x d2| -- so the area weight costs
431 * one cross product that is not otherwise needed, and no access to
432 * `tris.area`, which the operators have left stale anyway. */
433 for (int i = 0; i < n_fan; ++i)
434 {
435 const int o = fan[i];
436
437 double d1[3];
438 double d2[3];
439 MFTLGetMinimumImageVector(obj->domain, d1, p, MFTLVertexPos(obj, MFTLHeTarget(obj, o)));
440 MFTLGetMinimumImageVector(obj->domain, d2, p, MFTLVertexPos(obj, MFTLHeTarget(obj, MFTLHeNextIdx(o))));
441
442 const double cr[3] = {d1[1] * d2[2] - d1[2] * d2[1], d1[2] * d2[0] - d1[0] * d2[2], d1[0] * d2[1] - d1[1] * d2[0]};
443 const double area2 = MFTL_SQRT(cr[0] * cr[0] + cr[1] * cr[1] + cr[2] * cr[2]);
444
445 target[0] += area2 * (d1[0] + d2[0]) * MFTL_ONE_THIRD;
446 target[1] += area2 * (d1[1] + d2[1]) * MFTL_ONE_THIRD;
447 target[2] += area2 * (d1[2] + d2[2]) * MFTL_ONE_THIRD;
448 weight_sum += area2;
449
450 min_sq = MFTL_MIN(min_sq, d1[0] * d1[0] + d1[1] * d1[1] + d1[2] * d1[2]);
451 }
452
453 if (open && n_fan > 0)
454 {
455 double d[3];
456 MFTLGetMinimumImageVector(obj->domain, d, p, MFTLVertexPos(obj, open_fan_last_neighbour(obj, fan[0])));
457 min_sq = MFTL_MIN(min_sq, d[0] * d[0] + d[1] * d[1] + d[2] * d[2]);
458 }
459 }
460
461 /* Zero only when every fan triangle is exactly degenerate. A small but
462 * non-zero sum is still a well-defined weighted mean -- numerator and
463 * denominator carry the same areas -- so no length scale belongs here. */
464 if (!(weight_sum > 0.0)) return MFTL_FALSE;
465
466 const double inv = 1.0 / weight_sum;
467 double delta[3] = {target[0] * inv, target[1] * inv, target[2] * inv};
468
469 /* MFTLComputeVertexNormals writes exactly zero when a vertex's accumulated
470 * normal had no length -- a deliberate sentinel meaning "the one-ring is
471 * degenerate". Projecting against it would leave the full, un-projected
472 * Laplacian displacement, which is the shrinking operator this function exists
473 * to avoid. Leaving the vertex alone is the right answer. */
474 const double *const n = &obj->verts.normal[3 * v];
475 const double n_len_sq = n[0] * n[0] + n[1] * n[1] + n[2] * n[2];
476 if (n_len_sq < 0.5) return MFTL_FALSE;
477
478 const double dn = delta[0] * n[0] + delta[1] * n[1] + delta[2] * n[2];
479 disp[0] = delta[0] - dn * n[0];
480 disp[1] = delta[1] - dn * n[1];
481 disp[2] = delta[2] - dn * n[2];
482
483 *min_edge_sq = min_sq;
484 return MFTL_TRUE;
485}

487void MFTLSmoothTangential(MFTL_Object *obj, const MFTL_RemeshParameters *params, int iterations)
488{
489 MFTL_ASSERT(obj != MFTL_NULL && params != MFTL_NULL);
490 MFTL_ASSERT(MFTLObjectHasTopology(obj) && "MFTLSmoothTangential: derived topology is stale -- MFTLObjectBuildDerived first");
491
492 const int n_v = obj->verts.n;
493 if (n_v == 0 || obj->tris.n == 0 || iterations <= 0) return;
494 obj->derived &= ~MFTL_DERIVED_GEOMETRY;
495
496 /* Jacobi, not Gauss-Seidel: every new position is computed from the OLD ones
497 * and applied afterwards. Sweeping in place would make the result depend on the
498 * pool ordering, which MFTLObjectReorderMorton is free to change -- so the same
499 * mesh would smooth differently after a reordering that is documented as
500 * changing nothing but locality. */
501 double *const new_pos = (double *) MFTLMallocOrDie((size_t) n_v * 3 * sizeof(double), "remesh smoothing positions");
502
503 const double relax = params->smooth_relaxation;
504 const double step_frac = params->smooth_max_step_fraction;
505
506 for (int sweep = 0; sweep < iterations; ++sweep)
507 {
508 /* Normals are refreshed from current positions every sweep. */
509 MFTLComputeVertexNormals(obj, MFTL_NORMAL_WEIGHT_AREA);
510
511 memcpy(new_pos, obj->verts.pos, (size_t) n_v * 3 * sizeof(double));
512
513 for (int v = 0; v < n_v; ++v) // why also over ghosts? Dont I want to smooth only locals?
514 {
515 const MFTL_Vertex *const rec = &obj->verts.rec[v];
516
517 /* MFTL_FLAG_FROZEN joins the skip mask rather than getting a branch of its
518 * own: a frozen vertex is one another rank holds a copy of, so moving it
519 * would silently give two ranks different positions for one vertex. Ghosts
520 * carry the flag too, which is why the loop bound stays `verts.n` -- READING
521 * a frozen or ghost neighbour's position is not merely allowed, it is
522 * required. It is what makes this pass exact across a partition boundary
523 * with no communication: the one-ring of every unfrozen vertex is complete
524 * and current, because its owner was equally forbidden to move it. */
525 if (rec->flags & (MFTL_FLAG_DEAD | MFTL_FLAG_CORNER | MFTL_FLAG_FROZEN)) continue;
526 if (rec->out_he_idx == MFTL_ERROR) continue;
527
528 int fan[MFTL_REMESH_RING_MAX];
529 bool open = MFTL_FALSE;
530 const int n_fan = fan_halfedges(obj, v, fan, &open);
531 if (n_fan <= 0) continue; // isolated, or a fan too large to traverse
532
533 double disp[3];
534 double min_edge_sq = MFTL_DBL_MAX;
535
536 const bool moved = (rec->flags & MFTL_FLAG_FEATURE) ? ridge_displacement(obj, v, fan, n_fan, open, disp, &min_edge_sq)
537 : free_displacement(obj, v, fan, n_fan, open, params->smooth_mode, disp, &min_edge_sq);
538 if (!moved) continue;
539
540 double scale = relax;
541
542 /* One step may not exceed a fraction of the shortest incident edge. The
543 * tangential projection makes a fold-over unlikely, not impossible: it is
544 * taken against a vertex normal that is itself meaningless where the
545 * one-ring has already degenerated, which is exactly where a large step
546 * would do the most damage. The clamp costs one comparison against a length
547 * the displacement functions have already computed. */
548 const double disp_sq = disp[0] * disp[0] + disp[1] * disp[1] + disp[2] * disp[2];
549 const double limit_sq = step_frac * step_frac * min_edge_sq;
550
551 if (disp_sq * relax * relax > limit_sq && disp_sq > 0.0) scale = MFTL_SQRT(limit_sq / disp_sq);
552
553 double p[3] = {obj->verts.pos[3 * v + 0] + scale * disp[0], obj->verts.pos[3 * v + 1] + scale * disp[1], obj->verts.pos[3 * v + 2] + scale * disp[2]};
554 MFTLApplyPeriodicity(obj->domain, p);
555
556 new_pos[3 * v + 0] = p[0];
557 new_pos[3 * v + 1] = p[1];
558 new_pos[3 * v + 2] = p[2];
559 }
560
561 memcpy(obj->verts.pos, new_pos, (size_t) n_v * 3 * sizeof(double));
562 }
563
564 free(new_pos);
565
566}
