33/**
34 * @brief Emits `tris.verts` and `tris.neighbors` in one streaming pass.
35 *
36 * Both are functions of the half-edge records at the same index:
37 *
38 * verts[3t+k] == he[3t+(k+2)%3].vertex_idx origin of half-edge 3t+k
39 * neighbors[3t+k] == he[3t+k].twin_idx / 3 triangle across it
40 *
41 * They are stored for TRAFFIC: a connectivity-only loop streams 12 B per
42 * triangle from one of them instead of the 48 B of three half-edge records.
43 *
44 * `rebuild_object()` in src/communication/redistribute.c emits the same two
45 * arrays inside its relink loop; `inv_derived_matches_topology()`
46 * (tests/test_mesh_invariants.h) pins both spellings to this convention.
47 *
48 * @param obj The mesh. `tris.verts` and `tris.neighbors` are rewritten.
49 */
50static void build_topology_arrays(MFTL_Object *obj)
51{
52 const int n_tris = obj->tris.n;
53 const MFTL_HalfEdge *restrict he = obj->tris.he;
54 int *restrict tv = obj->tris.verts;
55 int *restrict tn = obj->tris.neighbors;
56
57 for (int t = 0; t < n_tris; ++t)
58 {
59 const int b = 3 * t;
60
61 /* Slot k is the ORIGIN of half-edge b+k: origin(h) == target(prev(h)), the
62 * convention of MFTLGetTriangleVertices(). */
63 tv[b + 0] = he[b + 2].vertex_idx;
64 tv[b + 1] = he[b + 0].vertex_idx;
65 tv[b + 2] = he[b + 1].vertex_idx;
66
67 const int w0 = he[b + 0].twin_idx;
68 const int w1 = he[b + 1].twin_idx;
69 const int w2 = he[b + 2].twin_idx;
70 tn[b + 0] = (w0 == MFTL_ERROR) ? MFTL_ERROR : MFTLHeTriIdx(w0);
71 tn[b + 1] = (w1 == MFTL_ERROR) ? MFTL_ERROR : MFTLHeTriIdx(w1);
72 tn[b + 2] = (w2 == MFTL_ERROR) ? MFTL_ERROR : MFTLHeTriIdx(w2);
73 }
74}

76void MFTLObjectBuildDerived(MFTL_Object *obj)
77{
78 MFTL_ASSERT(obj != MFTL_NULL);
79
80 if (obj->tris.n > 0) build_topology_arrays(obj);
81
82 /* An empty mesh is trivially described, so the bit is set. The geometry bit is
83 * left alone: rebuilding connectivity moves nothing, and every edit that does
84 * change a triangle's corners has already cleared it. */
85 obj->derived = MFTL_DERIVED_TOPOLOGY | (obj->derived & MFTL_DERIVED_GEOMETRY);
86}

92/* Permutation helpers: gather into scratch in the new order, copy back. O(n),
93 * sequential writes and one random read stream. */

95/**
96 * @brief Permutes an interleaved double array by gather-then-copy-back.
97 *
98 * @param field The array, permuted in place.
99 * @param scratch Buffer of at least `n * comp` doubles.
100 * @param old_of_new `old_of_new[i]` is the OLD index of the entity that belongs
101 * at new position `i`.
102 * @param n Number of entities.
103 * @param comp Components per entity: 1 for a scalar, 3 for a vector.
104 */
105static void permute_doubles(double *field, double *scratch, const int *old_of_new, int n, int comp)
106{
107 for (int i = 0; i < n; ++i)
108 {
109 const double *restrict s = &field[(size_t) comp * old_of_new[i]];
110 double *restrict d = &scratch[(size_t) comp * i];
111 for (int c = 0; c < comp; ++c) d[c] = s[c];
112 }
113 memcpy(field, scratch, (size_t) n * (size_t) comp * sizeof(double));
114}

142/**
143 * @brief Moves each triangle's three half-edges as one unit, preserving 3t+k.
144 *
145 * @param field The half-edge array, permuted in place.
146 * @param scratch Buffer of at least `3 * n_tris` half-edges.
147 * @param old_of_new Old TRIANGLE index belonging at each new triangle position.
148 * @param n_tris Number of triangles.
149 *
150 * @note Moving the triples as units is what preserves BOTH the implicit
151 * indexing invariant and the implicit half-edge global ids.
152 */
153static void permute_he_triples(MFTL_HalfEdge *field, MFTL_HalfEdge *scratch, const int *old_of_new, int n_tris)
154{
155 for (int i = 0; i < n_tris; ++i)
156 {
157 const MFTL_HalfEdge *restrict s = &field[3 * old_of_new[i]];
158 MFTL_HalfEdge *restrict d = &scratch[3 * i];
159 d[0] = s[0];
160 d[1] = s[1];
161 d[2] = s[2];
162 }
163 memcpy(field, scratch, 3 * (size_t) n_tris * sizeof(MFTL_HalfEdge));
164}

166/**
167 * @brief Orders the vertices along a Morton curve and remaps every reference.
168 *
169 * @param obj The mesh.
170 * @param[out] old_of_new `[verts.n]` old index per new position.
171 * @param[out] new_of_old `[verts.n]` new index per old position.
172 * @param scratch Buffer large enough for the widest per-vertex array.
173 * @return `true` if a permutation was applied (`false` for fewer than 2 vertices).
174 */
175static bool reorder_vertices(MFTL_Object *obj, int *old_of_new, int *new_of_old, void *scratch)
176{
177 const int nv = obj->verts.n;
178 if (nv < 2) return MFTL_FALSE;
179
180 const double *restrict pos = obj->verts.pos;
181
182 /* Bounding box of THIS object, so the 21 bits per axis are spent where the
183 * mesh is. A seam-straddling object gets a domain-wide box and coarser cells:
184 * a locality cost only. */
185 double lo[3] = {MFTL_DBL_MAX, MFTL_DBL_MAX, MFTL_DBL_MAX};
186 double hi[3] = {-MFTL_DBL_MAX, -MFTL_DBL_MAX, -MFTL_DBL_MAX};
187 for (int v = 0; v < nv; ++v)
188 {
189 for (int k = 0; k < 3; ++k)
190 {
191 const double c = pos[3 * v + k];
192 if (c < lo[k]) lo[k] = c;
193 if (c > hi[k]) hi[k] = c;
194 }
195 }
196
197 /* One scale for all axes keeps the Morton cells cubic. */
198 double range = hi[0] - lo[0];
199 if (hi[1] - lo[1] > range) range = hi[1] - lo[1];
200 if (hi[2] - lo[2] > range) range = hi[2] - lo[2];
201 /* The only condition is a FINITE scale; an absolute floor would collapse every
202 * key of a small object onto one value. */
203 const double inv_range = (range > MFTL_MORTON_MAX / MFTL_DBL_MAX) ? MFTL_MORTON_MAX / range : 0.0;
204
205 uint64_t *keys = (uint64_t *) MFTLMallocOrDie((size_t) nv * sizeof(uint64_t), "MFTLObjectReorderMorton vertex keys");
206
207 for (int v = 0; v < nv; ++v)
208 {
209 const uint64_t ix = (uint64_t) ((pos[3 * v + 0] - lo[0]) * inv_range);
210 const uint64_t iy = (uint64_t) ((pos[3 * v + 1] - lo[1]) * inv_range);
211 const uint64_t iz = (uint64_t) ((pos[3 * v + 2] - lo[2]) * inv_range);
212 keys[v] = MFTLMorton3D(ix, iy, iz);
213 old_of_new[v] = v;
214 }
215
216 /* Owned and ghost vertices are sorted SEPARATELY to keep the ownership prefix. */
217 const int n_local = obj->verts.n_local;
218 MFTLSortPairs64(keys, old_of_new, n_local);
219 MFTLSortPairs64(keys + n_local, old_of_new + n_local, nv - n_local);
220
221 free(keys);
222
223 for (int i = 0; i < nv; ++i) new_of_old[old_of_new[i]] = i;
224
225 /* `normal` is computed geometry but is permuted too, so a reorder never
226 * discards values the caller just computed. */
227 permute_vertex_recs(obj->verts.rec, (MFTL_Vertex *) scratch, old_of_new, nv);
228 permute_doubles(obj->verts.pos, (double *) scratch, old_of_new, nv, 3);
229 permute_doubles(obj->verts.normal, (double *) scratch, old_of_new, nv, 3);
230
231 /* Every reference INTO the vertex pool. (`out_he_idx` is a half-edge index;
232 * it is remapped with the triangles.) */
233 MFTL_HalfEdge *restrict he = obj->tris.he;
234 const int n_he = 3 * obj->tris.n;
235 for (int h = 0; h < n_he; ++h) he[h].vertex_idx = new_of_old[he[h].vertex_idx];
236
237 return MFTL_TRUE;
238}

240/**
241 * @brief Orders the triangles spatially and remaps every reference.
242 *
243 * The key is the SMALLEST of the triangle's three (already Morton-ordered)
244 * vertex indices: spatial by construction, three loads and two compares, no
245 * centroid (and no question of what a centroid means across a periodic seam),
246 * and independent of which corner is called corner 0.
247 *
248 * @param obj The mesh.
249 * @param[out] old_of_new `[tris.n]` old index per new position.
250 * @param[out] new_of_old `[tris.n]` new index per old position.
251 * @param scratch Buffer large enough for the widest per-triangle array.
252 * @return `true` if a permutation was applied.
253 */
254static bool reorder_triangles(MFTL_Object *obj, int *old_of_new, int *new_of_old, void *scratch)
255{
256 const int nt = obj->tris.n;
257 if (nt < 2) return MFTL_FALSE;
258
259 MFTL_HalfEdge *restrict he = obj->tris.he;
260
261 int *keys = (int *) MFTLMallocOrDie((size_t) nt * sizeof(int), "MFTLObjectReorderMorton triangle keys");
262
263 for (int t = 0; t < nt; ++t)
264 {
265 const int a = he[3 * t + 0].vertex_idx;
266 const int b = he[3 * t + 1].vertex_idx;
267 const int c = he[3 * t + 2].vertex_idx;
268 int m = a;
269 if (b < m) m = b;
270 if (c < m) m = c;
271 keys[t] = m;
272 old_of_new[t] = t;
273 }
274
275 const int n_local = obj->tris.n_local;
276 MFTLSortPairs32(keys, old_of_new, n_local);
277 MFTLSortPairs32(keys + n_local, old_of_new + n_local, nt - n_local);
278
279 free(keys);
280
281 for (int i = 0; i < nt; ++i) new_of_old[old_of_new[i]] = i;
282
283 /* `verts` and `neighbors` are not permuted; the caller rebuilds them. */
284 permute_triangle_recs(obj->tris.rec, (MFTL_Triangle *) scratch, old_of_new, nt);
285 permute_he_triples(obj->tris.he, (MFTL_HalfEdge *) scratch, old_of_new, nt);
286 permute_doubles(obj->tris.area, (double *) scratch, old_of_new, nt, 1);
287 permute_doubles(obj->tris.normal, (double *) scratch, old_of_new, nt, 3);
288 permute_doubles(obj->tris.center, (double *) scratch, old_of_new, nt, 3);
289
290 /* Every reference INTO the half-edge array: keep the corner h % 3, remap the
291 * triangle h / 3 (implicit indexing survives the permutation). */
292 he = obj->tris.he;
293 const int n_he = 3 * nt;
294 for (int h = 0; h < n_he; ++h)
295 {
296 const int tw = he[h].twin_idx;
297 if (tw != MFTL_ERROR) he[h].twin_idx = 3 * new_of_old[tw / 3] + tw % 3;
298 }
299
300 MFTL_Vertex *restrict vr = obj->verts.rec;
301 const int nv = obj->verts.n;
302 for (int v = 0; v < nv; ++v)
303 {
304 const int oh = vr[v].out_he_idx;
305 if (oh != MFTL_ERROR) vr[v].out_he_idx = 3 * new_of_old[oh / 3] + oh % 3;
306 }
307
308 return MFTL_TRUE;
309}

311void MFTLObjectReorderMorton(MFTL_Object *obj)
312{
313 MFTL_ASSERT(obj != MFTL_NULL);
314
315 const int nv = obj->verts.n;
316 const int nt = obj->tris.n;
317 if (nv < 2 && nt < 2) return;
318
319 const int n_max = (nv > nt) ? nv : nt;
320
321 /* One scratch block for every permutation, sized for the widest element moved:
322 * a triangle's three half-edge records, 48 bytes. */
323 const size_t widest = 48;
324 void *scratch = MFTLMallocOrDie((size_t) n_max * widest, "MFTLObjectReorderMorton scratch");
325
326 int *old_of_new = (int *) MFTLMallocOrDie(2 * (size_t) n_max * sizeof(int), "MFTLObjectReorderMorton maps");
327 int *new_of_old = old_of_new + n_max;
328
329 reorder_vertices(obj, old_of_new, new_of_old, scratch);
330 reorder_triangles(obj, old_of_new, new_of_old, scratch);
331
332 free(old_of_new);
333 free(scratch);
334
335 /* The derived arrays were deliberately not permuted; rebuild them instead. */
336 MFTLObjectBuildDerived(obj);
337}
