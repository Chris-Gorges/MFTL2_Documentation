1/**
2 * @file mftl_types.h
3 * @author Christian Gorges
4 * @date 2026-08-14
5 * @ingroup mftl_types
6 *
7 * @brief Core data structures of MFTL -- the single place every struct is defined.
8 *
9 * Layer 1: types only, no behaviour, so module headers never include each other.
10 *
11 * ## Layout: topology in AoS, geometry in SoA, one allocation per pool
12 *
13 * - **Topology and identity -> Array of Structures.** #MFTL_Vertex (16 B),
14 * #MFTL_Triangle (12 B), #MFTL_HalfEdge (16 B). A remesher edits one entity at
15 * a time and touches most of its record when it does.
16 * - **Geometry -> Structure of Arrays.** Positions, normals, areas, centroids:
17 * kernels touch one component of every entity, every timestep.
18 *
19 * Both live in ONE allocation per entity class, cut into cache-line-aligned
20 * sub-arrays (mftl_memory.h, src/object/object.c):
21 *
22 * MFTL_VertexPool = one block -> rec[] pos[] normal[]
23 * MFTL_TrianglePool = one block -> rec[] he[] area[] normal[] center[] verts[] neighbors[]
24 *
25 * so a pool has exactly one `capacity`, and "the AoS grew but an SoA array did
26 * not" cannot be expressed.
27 *
28 * ## Three invariants every module may assume
29 *
30 * 1. **Half-edges are triangle-indexed.** Half-edge \f$3t+k\f$ belongs to
31 * triangle \f$t\f$ and \f$\mathrm{next}(3t+k) = 3t+(k+1) \bmod 3\f$, so
32 * `next`, `prev` and `face` are arithmetic (mftl_topology.h), and the
33 * half-edge array lives inside the triangle pool at 3x its capacity. Anything
34 * that creates a triangle creates its half-edges at 3t, 3t+1, 3t+2.
35 * 2. **Ownership is a prefix.** Entities `[0, n_local)` are owned by this rank,
36 * `[n_local, n)` are ghosts. There is no `is_ghost` flag; `owner_proc` only
37 * answers "whose is it" for a ghost.
38 * 3. **Positions live in `verts.pos` and only there.** Spatial (Morton) ordering
39 * is applied to the pool itself (MFTLObjectReorderMorton()), never stored as
40 * a permutation, so there is one index space and one copy of each coordinate.
41 */

68/**
69 * @struct MFTL_Vertex
70 * @brief Identity and topology of one vertex. Its POSITION is not here.
71 *
72 * Its position and normal live in the SoA arrays of #MFTL_VertexPool at the same
73 * index. 16 bytes: four records per cache line.
74 */
75typedef struct MFTL_Vertex
76{
77 int out_he_idx; ///< Index of ONE outgoing half-edge, MFTL_ERROR if isolated. The 1-ring walk starts here.
78 int global_id; ///< Unique across ranks; stable under redistribution and reordering, renumbered by remeshing.
79 int owner_proc; ///< Owning rank; equals this rank exactly for indices < n_local.
80 int flags; ///< Bit flags, see MFTL_FLAG_* in mftl_topology.h.
81} MFTL_Vertex;
82
83/**
84 * @struct MFTL_Triangle
85 * @brief Identity of one triangle. Its topology AND its geometry are implicit.
86 *
87 * Its half-edges are 3t, 3t+1, 3t+2 (invariant 1) and its area, normal and
88 * centroid are SoA arrays of #MFTL_TrianglePool, so what remains is 12 bytes of
89 * identity: a triangle is a name for a cycle of three half-edges.
90 */
91typedef struct MFTL_Triangle
92{
93 int global_id; ///< Unique across all ranks.
94 int owner_proc; ///< The rank that owns this triangle.
95 int flags; ///< Bit flags, see MFTL_FLAG_* in mftl_topology.h.
96} MFTL_Triangle;
97
98/**
99 * @struct MFTL_HalfEdge
100 * @brief One directed edge of one triangle -- the entire mesh topology.
101 *
102 * Each undirected edge of a closed mesh is two opposite half-edges; on an open
103 * surface the boundary edges have no partner and carry `twin_idx == MFTL_ERROR`.
104 *
105 * Only two topological fields are stored: `vertex_idx` (the vertex the
106 * half-edge ENDS at) and `twin_idx`. `next`, `prev` and the triangle are
107 * arithmetic on the index, and \f$\mathrm{origin}(h) =
108 * \mathrm{target}(\mathrm{prev}(h))\f$ (mftl_topology.h). 16 bytes, so a
109 * triangle's three half-edges span one or two cache lines.
110 */
111typedef struct MFTL_HalfEdge
112{
113 int vertex_idx; ///< Vertex at the END (target) of this directed edge.
114 int twin_idx; ///< Oppositely oriented half-edge, MFTL_ERROR if none in THIS partition.
115 int global_id; ///< Unique across all ranks.
116 int flags; ///< Bit flags, see MFTL_FLAG_* in mftl_topology.h.
117} MFTL_HalfEdge;

123/**
124 * @struct MFTL_VertexPool
125 * @brief Every vertex of one object: AoS records plus the per-vertex SoA arrays.
126 *
127 * All pointers address sub-arrays of the one allocation `block`, all sized for
128 * `capacity` vertices. Only src/object/object.c may assign `capacity`, `block`
129 * or any pointer. `rec` and `pos` are authoritative; `normal` is computed
130 * (MFTLComputeVertexNormals()) but is preserved by resizes and reorders.
131 */
132typedef struct MFTL_VertexPool
133{
134 MFTL_Vertex *rec; ///< [capacity] identity + topology, AoS.
135
136 double *pos; ///< [3*capacity] AUTHORITATIVE position, xyz interleaved.
137 double *normal; ///< [3*capacity] unit vertex normal, xyz interleaved; valid only after MFTLComputeVertexNormals().
138
139 int n; ///< Active vertices (owned + ghost).
140 int n_local; ///< Owned vertices; they occupy [0, n_local).
141 int capacity; ///< Entities every array above is sized for. One number, one writer.
142
143 void *block; ///< The single allocation. Freed with free().
144} MFTL_VertexPool;
145
146/**
147 * @struct MFTL_TrianglePool
148 * @brief Every triangle AND every half-edge of one object.
149 *
150 * A triangle has exactly three half-edges at 3t..3t+2, so they share the
151 * triangle block (at 3x its capacity) instead of having a capacity of their own.
152 *
153 * `verts` and `neighbors` are derived, compact connectivity for hot loops, valid
154 * only while #MFTL_DERIVED_TOPOLOGY is set (MFTLObjectBuildDerived()). They are
155 * redundant with the half-edge records -- `verts[3t+k] = he[3t+(k+2)%3].vertex_idx`
156 * and `neighbors[3t+k] = he[3t+k].twin_idx / 3` -- and are stored because a
157 * connectivity-only loop then streams 12 B per triangle instead of 48 B.
158 *
159 * `area`, `normal` and `center` are written by MFTLComputeTriangleGeometry() /
160 * MFTLEnsembleUpdate(); a redistribution leaves them unspecified
161 * (#MFTL_DERIVED_GEOMETRY).
162 */
163typedef struct MFTL_TrianglePool
164{
165 MFTL_Triangle *rec; ///< [capacity] identity, AoS.
166 MFTL_HalfEdge *he; ///< [3*capacity] topology, AoS. Half-edge 3t+k belongs to triangle t.
167
168 /* --- Geometry: written by the per-timestep geometry pass ------------------ */
169 double *area; ///< [capacity] triangle area.
170 double *normal; ///< [3*capacity] unit normal, xyz interleaved.
171 double *center; ///< [3*capacity] centroid, xyz interleaved, WRAPPED into the domain.
172
173 /* --- Derived connectivity: rebuilt, never authoritative ------------------- */
174 int *verts; ///< [3*capacity] vertex pool index; slot k is the ORIGIN of half-edge 3t+k.
175 int *neighbors; ///< [3*capacity] triangle across half-edge 3t+k, MFTL_ERROR at a boundary.
176
177 int n; ///< Active triangles (owned + ghost). Half-edges number 3*n.
178 int n_local; ///< Owned triangles; they occupy [0, n_local).
179 int capacity; ///< Triangles every array above is sized for. `he` holds 3*capacity.
180
181 void *block; ///< The single allocation. Freed with free().
182} MFTL_TrianglePool;

188/**
189 * @struct MFTL_ObjectQualityStatistics
190 * @brief Global mesh-quality summary of one object, filled by MFTLEnsembleUpdate.
191 *
192 * Every field is a value for the WHOLE body: min/max are reduced over the ranks
193 * holding it and averages are divided by global counts over owned entities
194 * (never an average of per-rank averages). The reduction folds contributions in
195 * a fixed rank order, so every holder gets bit-identical values.
196 */
197typedef struct MFTL_ObjectQualityStatistics
198{
199 double min_triangle_area; ///< Smallest triangle area anywhere in the mesh.
200 double max_triangle_area; ///< Largest triangle area anywhere in the mesh.
201 double avg_triangle_area; ///< Total area divided by the GLOBAL triangle count.
202
203 /**
204 * Worst mean-ratio quality in the mesh, \f$q = 4\sqrt{3}A/(a^2+b^2+c^2)\f$
205 * (#MFTL_FOUR_SQRT3): 1 for an equilateral triangle, \f$\to 0\f$ as one
206 * degenerates, never above 1. Despite the name it is a quality, not an aspect
207 * ratio: larger is better.
208 */
209 double min_aspect_ratio;
210 double max_aspect_ratio; ///< Best mean-ratio quality in the mesh. Never exceeds 1.
211 double avg_aspect_ratio; ///< Summed quality divided by the GLOBAL triangle count.
212
213 double min_edge_length_sq; ///< Shortest edge, SQUARED. Squares throughout, so no sqrt in the reduction.
214 double max_edge_length_sq; ///< Longest edge, squared.
215 double avg_edge_length_sq; ///< Summed squared length divided by the global half-edge count.
216} MFTL_ObjectQualityStatistics;

219// The Object (one mesh)
220//------------------------------------------------------------------------------
221
222/** @brief `derived` bit: tris.verts and tris.neighbors are current. */
223#define MFTL_DERIVED_TOPOLOGY 0x1u
224
225/**
226 * @brief `derived` bit: `tris.area/normal/center` were computed over EVERY
227 * triangle, ghosts included, after the last change the library made to
228 * their inputs.
229 *
230 * Set by MFTLComputeTriangleGeometry() over `tris.n` triangles. Cleared by any
231 * change of the entity set (MFTLObjectSetCounts() clears both bits), by the
232 * remeshing operators, by a redistribution's rebuild and by
233 * MFTLUpdateGhostPositions(); kept by MFTLObjectBuildDerived() and
234 * MFTLObjectReorderMorton(), which move no vertex. Moving owned vertices through `verts.pos` does NOT clear it: the
235 * library cannot see that. So a clear bit means "not current", and a set bit
236 * means "current unless the caller moved vertices since". MFTLRedistribute()
237 * recomputes the geometry of a body whose bit is clear before routing it.
238 */
239#define MFTL_DERIVED_GEOMETRY 0x2u

241/**
242 * @struct MFTL_Object
243 * @brief One mesh -- typically one droplet or one body.
244 *
245 * Owns its two pools and borrows the domain of the ensemble it was added to. It
246 * holds no MPI state itself.
247 */
248typedef struct MFTL_Object
249{
250 MFTL_VertexPool verts; ///< Every vertex: AoS records plus the per-vertex SoA arrays.
251 MFTL_TrianglePool tris; ///< Every triangle AND every half-edge.
252
253 /** Which derived arrays are current; see MFTL_DERIVED_* above. */
254 unsigned derived;
255
256 /* Global id counters. Ids are handed out by post-incrementing them, and are
257 * kept DENSE: after generation and after every remeshing pass (which
258 * renumbers) each equals the global entity count of its class. Density
259 * matters because src/communication/ sizes its arrival lookup tables from
260 * these counters. */
261 int next_vertex_gid; ///< Next unused vertex global id. Equals the global vertex count when dense.
262 int next_triangle_gid; ///< Next unused triangle global id.
263 int next_halfedge_gid; ///< Next unused half-edge global id; always 3 x next_triangle_gid, since the id of half-edge k of triangle g is 3g+k.
264
265 int global_n_vertices; ///< Owned vertices summed over all ranks.
266 int global_n_triangles; ///< Owned triangles summed over all ranks.
267
268 int global_id; ///< Body id, dense in [0, n_bodies_global), assigned by MFTLEnsembleAddObject(). The one handle that survives a redistribution.
269
270 /**
271 * Something other than the owner map changed this mesh (entity counts, global
272 * ids or flags), so the current partition no longer describes it and the next
273 * redistribution must rebuild the body even if no entity changed owner.
274 *
275 * Set by MFTLObjectMarkPartitionDirty() -- and by MFTLObjectSetCounts(), the
276 * choke point for every creation and destruction -- and cleared by
277 * redistribution after a rebuild. Conservative by design: a spurious `true`
278 * costs one rebuild, a missing one silently corrupts ghost ids.
279 */
280 bool partition_dirty;
281
282 /**
283 * The body has been rebuilt at least once since it last changed, so its pool
284 * layout is at a fixed point and the next redistribution may skip it.
285 *
286 * Separate from `partition_dirty` because a rebuild is not idempotent: an
287 * owned vertex whose incident triangles are all remote is dropped and
288 * re-received, landing at the end of the pool. A body must therefore be clean
289 * TWICE before a skipped pool equals a rebuilt one slot for slot
290 * (`test_skipping_reproduces_the_full_rebuild`). Kept per body on the object,
291 * so adding or removing a body never un-settles the others.
292 */
293 bool partition_settled;
294
295 struct MFTL_Domain *domain; ///< Borrowed from the ensemble; MFTL_NULL until added.
296
297 MFTL_ObjectQualityStatistics quality; ///< By value: nine doubles do not need their own allocation.
298} MFTL_Object;

304/**
305 * @struct MFTL_Domain
306 * @brief The computational environment shared by every object of an ensemble.
307 *
308 * The MPI communicators, the Cartesian process grid, the physical bounds and the
309 * periodicity, stored once; every object holds a borrowed pointer. Read-only
310 * after MFTLEnsembleCreate().
311 */
312typedef struct MFTL_Domain
313{
314 // --- MPI Topology ---
315 MPI_Comm comm_world; ///< The library's private duplicate of the user's communicator.
316 MPI_Comm comm_cart; ///< Cartesian topology for geometric domain decomposition.
317 MPI_Comm comm_graph; ///< Distributed graph over the (up to) 26 surrounding ranks.
318
319 int rank; ///< My rank in comm_world.
320 int nproc; ///< Size of comm_world.
321
322 int num_neighbors; ///< Degree of comm_graph.
323 int *neighbors; ///< [num_neighbors] neighbour ranks, in the order MPI reports them.
324
325 /**
326 * `[26][3]`, parallel to `neighbors`: the cell offset of each neighbour from
327 * this rank, components in {-1, 0, +1}, so "are two cells 26-neighbours" is a
328 * componentwise test. On thin periodic grids several stencil offsets map to
329 * the same rank (e.g. cuts = {2,1,1}); the FIRST offset found is recorded.
330 */
331 signed char neighbor_offset[26][3];
332
333 // --- Domain Decomposition ---
334 int cuts[3]; ///< Process grid {nx, ny, nz}; the product is nproc.
335
336 // --- Physical Bounds ---
337 double global_min[3]; ///< Lower corner of the domain box, per axis.
338 double global_max[3]; ///< Upper corner of the domain box, per axis.
339 double domain_size[3]; ///< global_max - global_min.
340 double inv_domain_size[3]; ///< 1 / domain_size (0 on a zero-length axis), so the minimum image is a multiply-and-round.
341
342 bool is_periodic[3]; ///< Per-axis periodicity.
343
344 int *rank_map; ///< [nproc] rank of Cartesian cell (i,j,k) at i + j*nx + k*nx*ny; a table lookup instead of MPI_Cart_rank.
345 double grid_factors[3]; ///< cuts[i] / domain_size[i], so owner lookup is a multiply.
346} MFTL_Domain;

348//------------------------------------------------------------------------------
349// The Ensemble (the library handle)
350//------------------------------------------------------------------------------
351
352/**
353 * @struct MFTL_Ensemble
354 * @brief Top-level handle: one domain plus the objects living in it.
355 */
356typedef struct MFTL_Ensemble
357{
358 MFTL_Domain domain; ///< By value: there is exactly one, and every object borrows a pointer to it.
359
360 /**
361 * The bodies THIS RANK holds entities of, sorted by ascending
362 * #MFTL_Object::global_id; owned (MFTLEnsembleDestroy() destroys them). A
363 * redistribution destroys bodies that left the rank and creates bodies that
364 * arrived, so object pointers, list slots and `n_local_objects` are all
365 * invalid afterwards; `global_id` + MFTLEnsembleFindLocalObject() is the
366 * handle that survives.
367 *
368 * @warning Every rank must add the same bodies in the same order, so the same
369 * id names the same body everywhere (checked by a Debug checksum).
370 */
371 MFTL_Object **local_objects;
372 int n_local_objects; ///< Bodies THIS RANK holds. Never a global quantity.
373 int capacity_objects; ///< Slots allocated in `local_objects`.
374
375 /**
376 * Size of the body-id space: every `global_id` is in [0, n_bodies_global).
377 * Identical on every rank and never decreasing -- a topology change raises it
378 * for each new body and retires, never reuses, the ids of vanished ones, so it
379 * may exceed the number of live bodies.
380 *
381 * This, not `n_local_objects`, is the count a collective may be guarded by:
382 * an early return on a local count is a deadlock.
383 */
384 int n_bodies_global;
385
386 int total_global_triangles; ///< Owned triangles summed over all bodies and ranks (MFTLEnsembleUpdateGlobalCounts()).
387
388 /**
389 * Opaque per-call scratch of the communication module (rank-sized arrays,
390 * payload buffers, rebuild scratch), created on the first MFTLRedistribute()
391 * and freed by MFTLEnsembleDestroy(). Never mesh state; kept off
392 * #MFTL_Domain so the domain stays read-only during a redistribution.
393 */
394 void *comm_workspace;
395} MFTL_Ensemble;
