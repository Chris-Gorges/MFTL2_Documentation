1/**
2 * @file ghost_sync.c
3 * @author Christian Gorges
4 * @date 2026-08-14
5 * @ingroup mftl_communication
6 *
7 * @brief Ghost synchronisation of one per-vertex 3-vector (position or normal)
8 * with the topology unchanged.
9 *
10 * A request/reply exchange, because only the HOLDER of a ghost knows it holds
11 * it; no owner-to-holder state is kept between calls. Per ghost vertex 8 B of
12 * request and 32 B of reply; three neighbourhood collectives (counts, requests,
13 * replies). One field per exchange: bundling recomputable fields with the
14 * position would ship bytes the receiver computes more cheaply.
15 *
16 * Per call it builds one `[nproc]` rank-to-neighbour lookup (O(nproc) memory
17 * and time), the only term here that grows with the process count.
18 */
19

62/**
63 * @brief Answers the requests addressed to one object.
64 *
65 * The owner turns a vertex global id into a pool slot through the epoch-stamped
66 * id table (comm_internal.h), filled with this object's OWNED vertices only for
67 * objects that actually received requests. An unanswerable request (the
68 * requester believes this rank owns a vertex it does not) is answered with
69 * `global_id == MFTL_ERROR`, asserted in Debug and skipped by the requester.
70 *
71 * @param obj The mesh whose OWNED vertices answer the requests.
72 * @param field Which per-vertex 3-vector is being synchronised.
73 * @param flat_recv The received requests (local slot, vertex id), all senders.
74 * @param order Request indices belonging to this object.
75 * @param first First slot of @p order to serve.
76 * @param last One past the last slot to serve.
77 * @param[out] replies One payload per request.
78 * @param tbl Epoch-stamped id-to-slot table, reused across objects.
79 * @param[out] n_unserved Incremented once per request this rank could not answer.
80 */
81static void answer_object_requests(const MFTL_Object *obj, ghost_field field, const int *restrict flat_recv, const int *restrict order, int first, int last,
82 MFTL_GhostUpdatePayload *restrict replies, mftl_gid_table *tbl, int *n_unserved)
83{
84 const MFTL_Vertex *const restrict vr = obj->verts.rec;
85 const double *const restrict src = ghost_field_read(obj, field);
86 const int n_local = obj->verts.n_local;
87
88 CS_BEGIN(ts_tbl);
89 /* Owned vertices go in; the mode follows the share of the body held here. */
90 gt_begin(tbl, n_local, MFTL_MAX(obj->next_vertex_gid, 1));
91 for (int v = 0; v < n_local; ++v) gt_put(tbl, vr[v].global_id, v);
92 CS_END(ts_tbl, g_t_table);
93
94 for (int i = first; i < last; ++i)
95 {
96 const int k = order[i];
97 const int gid = flat_recv[2 * k + 1];
98 MFTL_GhostUpdatePayload *const p = &replies[k];
99
100 if (gid < 0 || !gt_has(tbl, gid))
101 {
102 MFTL_ASSERT(0 && "ghost sync: asked for a vertex this rank does not own");
103 memset(p, 0, sizeof *p);
104 p->global_id = MFTL_ERROR;
105 (*n_unserved)++;
106 continue;
107 }
108
109 const int v = gt_get(tbl, gid);
110
111 p->value[0] = src[3 * v + 0];
112 p->value[1] = src[3 * v + 1];
113 p->value[2] = src[3 * v + 2];
114
115 p->global_id = gid; // echoes the request, which is how the requester matches it
116 }
117}

119/**
120 * @brief The shared request/reply exchange behind both public ghost syncs.
121 * @param ens The ensemble.
122 * @param field Which per-vertex 3-vector to move.
123 * @return #MFTL_SUCCESS, or #MFTL_ERROR if the exchange failed.
124 */
125static int ghost_sync_vec3(MFTL_Ensemble *ens, ghost_field field)
126{
127 if (ens == MFTL_NULL || ens->local_objects == MFTL_NULL) return MFTL_ERROR;
128 MFTL_ASSERT(MFTL_MPI_GhostUpdatePayload != MPI_DATATYPE_NULL && "ghost sync: call MFTLRegisterMpiTypes() first");
129
130 MFTL_Domain *const domain = &ens->domain;
131 MFTL_Object *const *const objects = ens->local_objects;
132 const int num_objects = ens->n_local_objects;
133 const int nproc = domain->nproc;
134 const int nn = domain->num_neighbors;
135
136 /* Guards on GLOBAL state (or on a rank with no graph neighbours, which nobody
137 * expects a message from): a rank holding nothing must still enter the
138 * neighbourhood collectives. */
139 if (ens->n_bodies_global == 0 || nn == 0 || nproc == 1) return MFTL_SUCCESS;
140
141 CS_BEGIN(ts_total);
142 MFTLCommStats.g_n_calls++;
143 CS_BEGIN(ts_collect);
144
145 /* ── 1. Collect every ghost (the pool suffix), grouped by owner: a REQUEST
146 * (body id, vertex id) for the wire and a TARGET (local slot, vertex slot)
147 * kept here, so replies are applied by direct index. */
148 mftl_int_buffer *const requests = (mftl_int_buffer *) MFTLMallocOrDie((size_t) nn * sizeof(mftl_int_buffer), "ghost request lists");
149 mftl_int_buffer *const targets = (mftl_int_buffer *) MFTLMallocOrDie((size_t) nn * sizeof(mftl_int_buffer), "ghost target lists");
150 for (int i = 0; i < nn; ++i)
151 {
152 ib_init(&requests[i], 1024);
153 ib_init(&targets[i], 1024);
154 }
155
156 int *const rank_to_neigh = (int *) MFTLMallocOrDie((size_t) nproc * sizeof(int), "ghost neighbour index");
157 for (int r = 0; r < nproc; ++r) rank_to_neigh[r] = MFTL_ERROR;
158 for (int i = 0; i < nn; ++i) rank_to_neigh[domain->neighbors[i]] = i;
159
160 int orphaned = 0;
161 for (int o = 0; o < num_objects; ++o)
162 {
163 const MFTL_Object *const obj = objects[o];
164 const MFTL_Vertex *const vr = obj->verts.rec;
165
166 for (int v = obj->verts.n_local; v < obj->verts.n; ++v)
167 {
168 const int owner = vr[v].owner_proc;
169 const int n_idx = (owner >= 0 && owner < nproc) ? rank_to_neigh[owner] : MFTL_ERROR;
170 if (n_idx == MFTL_ERROR)
171 {
172 /* Owner outside the 26-neighbourhood: unreachable. Skipped (aborting
173 * would deadlock the others) but asserted, because the stale position
174 * later makes two ranks route a triangle differently. There is no global
175 * fallback (unlike MFTLRemeshSyncGhostIds()); a partition whose halo
176 * leaves the neighbourhood must not outlive the call that created it. */
177 MFTL_ASSERT(0 && "ghost sync: ghost vertex whose owner is not a graph neighbour -- its position will not be updated");
178 orphaned++;
179 continue;
180 }
181
182 /* Body by global id on the wire; local slot in the target. */
183 ib_add2(&requests[n_idx], obj->global_id, vr[v].global_id);
184 ib_add2(&targets[n_idx], o, v);
185 MFTLCommStats.g_n_ghosts_requested++;
186 }
187 }
188 free(rank_to_neigh);
189 MFTLCommStats.g_n_orphaned += orphaned;
190
191 if (orphaned > 0)
192 {
193 fprintf(stderr, "[MFTL rank %d] ghost sync: %d ghost vertices whose owner is not a graph neighbour were not updated\n", domain->rank, orphaned);
194 }
195
196 CS_END(ts_collect, g_t_collect);
197
198 /* ── 2. Request counts, non-blocking and overlapped with flattening the
199 * requests; everything after is a strict dependency chain. */
200 int *const n_send = (int *) MFTLMallocOrDie((size_t) nn * sizeof(int), "ghost send counts");
201 int *const n_recv = (int *) MFTLMallocOrDie((size_t) nn * sizeof(int), "ghost recv counts");
202 for (int i = 0; i < nn; ++i) n_send[i] = requests[i].size;
203
204 MPI_Request count_req;
205 CS_BEGIN(ts_counts);
206 MPI_Ineighbor_alltoall(n_send, 1, MPI_INT, n_recv, 1, MPI_INT, domain->comm_graph, &count_req);
207
208 int *const s_displs = (int *) MFTLMallocOrDie((size_t) nn * sizeof(int), "ghost send displs");
209 int total_send = 0;
210 for (int i = 0; i < nn; ++i)
211 {
212 s_displs[i] = total_send;
213 total_send += n_send[i];
214 }
215
216 int *const flat_send = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(total_send, 1) * sizeof(int), "ghost request image");
217 for (int i = 0; i < nn; ++i)
218 {
219 if (n_send[i] > 0) memcpy(flat_send + s_displs[i], requests[i].data, (size_t) n_send[i] * sizeof(int));
220 }
221
222 MPI_Wait(&count_req, MPI_STATUS_IGNORE);
223 CS_END(ts_counts, g_t_counts);
224
225 int *const r_displs = (int *) MFTLMallocOrDie((size_t) nn * sizeof(int), "ghost recv displs");
226 int total_recv = 0;
227 for (int i = 0; i < nn; ++i)
228 {
229 r_displs[i] = total_recv;
230 total_recv += n_recv[i];
231 }
232
233 /* ── 3. Exchange the requests ────────────────────────────────────────────── */
234 CS_BEGIN(ts_req);
235 int *const flat_recv = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(total_recv, 1) * sizeof(int), "ghost incoming requests");
236 MPI_Neighbor_alltoallv(flat_send, n_send, s_displs, MPI_INT, flat_recv, n_recv, r_displs, MPI_INT, domain->comm_graph);
237 free(flat_send);
238 CS_END(ts_req, g_t_requests);
239 MFTLCommStats.g_n_bytes_sent += (long) total_send * (long) sizeof(int);
240 MFTLCommStats.g_n_bytes_recv += (long) total_recv * (long) sizeof(int);
241
242 CS_BEGIN(ts_answer);
243
244 /* ── 4. Answer: group the requests by object (counting sort) so each object's
245 * id table is built once; replies go back at the request's ORIGINAL position,
246 * since the requester matches them positionally. */
247 const int n_replies = total_recv / 2;
248 MFTL_GhostUpdatePayload *const replies =
249 (MFTL_GhostUpdatePayload *) MFTLMallocOrDie((size_t) MFTL_MAX(n_replies, 1) * sizeof(MFTL_GhostUpdatePayload), "ghost replies");
250
251 int *const obj_off = (int *) MFTLCallocOrDie((size_t) num_objects + 1, sizeof(int), "ghost request grouping");
252 int *const order = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(n_replies, 1) * sizeof(int), "ghost request order");
253 int unserved = 0;
254
255 {
256 /* Resolve each body id to this rank's slot once, in place. A body not held
257 * here is a routing error: the requester named this rank as the owner. */
258 int dropped = 0;
259 for (int k = 0; k < n_replies; ++k)
260 {
261 const int slot = MFTLEnsembleFindLocalSlot(ens, flat_recv[2 * k]);
262 flat_recv[2 * k] = slot;
263 if (slot != MFTL_ERROR)
264 obj_off[slot + 1]++;
265 else
266 dropped++;
267 }
268 for (int o = 0; o < num_objects; ++o) obj_off[o + 1] += obj_off[o];
269
270 int *const cursor = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(num_objects, 1) * sizeof(int), "ghost grouping cursors");
271 for (int o = 0; o < num_objects; ++o) cursor[o] = obj_off[o];
272
273 for (int k = 0; k < n_replies; ++k)
274 {
275 const int o = flat_recv[2 * k];
276 if (o == MFTL_ERROR)
277 {
278 MFTL_ASSERT(0 && "ghost sync: request names a body this rank does not hold, but claims this rank owns the vertex");
279 memset(&replies[k], 0, sizeof replies[k]);
280 replies[k].global_id = MFTL_ERROR;
281 continue;
282 }
283 order[cursor[o]++] = k;
284 }
285 free(cursor);
286 unserved += dropped;
287 }
288
289 {
290 /* The ensemble's persistent table, retired per object by an epoch bump. */
291 mftl_gid_table *const tbl = MFTLCommWorkspaceGhostTable(ens);
292 for (int o = 0; o < num_objects; ++o)
293 {
294 if (obj_off[o + 1] == obj_off[o]) continue; // nothing asked of this body
295 answer_object_requests(objects[o], field, flat_recv, order, obj_off[o], obj_off[o + 1], replies, tbl, &unserved);
296 }
297 }
298
299 free(order);
300 free(obj_off);
301 free(flat_recv);
302 CS_END(ts_answer, g_t_answer);
303
304 CS_BEGIN(ts_rep);
305
306 /* ── 5. Replies: counts are the request counts halved, known to both sides. */
307 int *const n_send_rep = n_recv; // what we were asked for, we now answer
308 int *const n_recv_rep = n_send; // what we asked for, we now receive
309 for (int i = 0; i < nn; ++i)
310 {
311 n_send_rep[i] /= 2;
312 n_recv_rep[i] /= 2;
313 s_displs[i] /= 2;
314 r_displs[i] /= 2;
315 }
316
317 const int n_incoming = total_send / 2;
318 MFTL_GhostUpdatePayload *const incoming =
319 (MFTL_GhostUpdatePayload *) MFTLMallocOrDie((size_t) MFTL_MAX(n_incoming, 1) * sizeof(MFTL_GhostUpdatePayload), "ghost updates");
320
321 MPI_Neighbor_alltoallv(replies, n_send_rep, r_displs, MFTL_MPI_GhostUpdatePayload, incoming, n_recv_rep, s_displs, MFTL_MPI_GhostUpdatePayload,
322 domain->comm_graph);
323 free(replies);
324 CS_END(ts_rep, g_t_replies);
325 MFTLCommStats.g_n_bytes_sent += (long) (total_recv / 2) * (long) sizeof(MFTL_GhostUpdatePayload);
326 MFTLCommStats.g_n_bytes_recv += (long) (total_send / 2) * (long) sizeof(MFTL_GhostUpdatePayload);
327 CS_BEGIN(ts_apply);
328
329 /* ── 6. Apply in request order through the target list. */
330 int p = 0;
331 for (int i = 0; i < nn; ++i)
332 {
333 const int *const t = targets[i].data;
334 const int count = targets[i].size;
335
336 for (int k = 0; k < count; k += 2, ++p)
337 {
338 const MFTL_GhostUpdatePayload *const src = &incoming[p];
339 if (src->global_id == MFTL_ERROR) continue; // owner could not serve it
340
341 MFTL_Object *const obj = objects[t[k]];
342 const int idx = t[k + 1];
343 MFTL_ASSERT(obj->verts.rec[idx].global_id == src->global_id && "ghost reply does not match the request order");
344
345 double *const dst = ghost_field_write(obj, field);
346 dst[3 * idx + 0] = src->value[0];
347 dst[3 * idx + 1] = src->value[1];
348 dst[3 * idx + 2] = src->value[2];
349 }
350 }
351
352 free(incoming);
353 CS_END(ts_apply, g_t_apply);
354 CS_BEGIN(ts_free);
355 for (int i = 0; i < nn; ++i)
356 {
357 ib_free(&requests[i]);
358 ib_free(&targets[i]);
359 }
360 free(requests);
361 free(targets);
362 free(n_send);
363 free(n_recv);
364 free(s_displs);
365 free(r_displs);
366 CS_END(ts_free, g_t_teardown);
367
368 CS_END(ts_total, g_t_total);
369
370 /* An orphaned or unserved ghost kept a stale value: report it. */
371 return (unserved == 0 && orphaned == 0) ? MFTL_SUCCESS : MFTL_ERROR;
372}
