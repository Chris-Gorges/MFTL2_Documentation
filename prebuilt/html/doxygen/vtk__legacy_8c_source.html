20/**
21 * Legacy POLYDATA layout written here:
22 *
23 * POINTS <nv> double
24 * <x> <y> <z> x nv
25 * POLYGONS <nt> <4*nt>
26 * 3 <v0> <v1> <v2> x nt
27 *
28 * The second POLYGONS field is the total integer count of the block (four per
29 * record, the leading 3 included). Pool indices are file indices.
30 */
31int MFTLWriteVtk(const MFTL_Object *object, const char *base_filename)
32{
33 if (!object || !base_filename)
34 {
35 fprintf(stderr, "MFTLWriteVtk: NULL object or filename.\n");
36 return MFTL_ERROR;
37 }
38
39 char filename[1024];
40 const int n = snprintf(filename, sizeof filename, "%s.vtk", base_filename);
41 if (n < 0 || (size_t) n >= sizeof filename)
42 {
43 fprintf(stderr, "MFTLWriteVtk: filename too long.\n");
44 return MFTL_ERROR;
45 }
46
47 FILE *fp = fopen(filename, "w");
48 if (!fp)
49 {
50 fprintf(stderr, "MFTLWriteVtk: could not open %s for writing.\n", filename);
51 return MFTL_ERROR;
52 }
53
54 /* A 1 MiB stdio buffer: the default (often 4 KB) makes syscalls dominate. */
55 char *const iobuf = (char *) malloc(1u << 20);
56 if (iobuf) setvbuf(fp, iobuf, _IOFBF, 1u << 20);
57
58 const int nv = object->verts.n;
59 const int nt = object->tris.n;
60
61 fprintf(fp, "# vtk DataFile Version 2.0\n");
62 fprintf(fp, "MFTL object %d\n", object->global_id);
63 fprintf(fp, "ASCII\n");
64 fprintf(fp, "DATASET POLYDATA\n");
65
66 fprintf(fp, "POINTS %d double\n", nv);
67 const double *const pos = object->verts.pos;
68 for (int v = 0; v < nv; ++v) fprintf(fp, "%.10e %.10e %.10e\n", pos[3 * v + 0], pos[3 * v + 1], pos[3 * v + 2]);
69
70 fprintf(fp, "\nPOLYGONS %d %d\n", nt, 4 * nt);
71 for (int t = 0; t < nt; ++t)
72 {
73 int v[3];
74 MFTLGetTriangleVertices(object, t, v);
75 fprintf(fp, "3 %d %d %d\n", v[0], v[1], v[2]);
76 }
77
78 /* Ownership, and the geometry arrays UNVALIDATED (this dump shows the raw
79 * state, including arrays a redistribution left undefined). */
80 fprintf(fp, "\nPOINT_DATA %d\n", nv);
81 fprintf(fp, "SCALARS VertexOwner int 1\nLOOKUP_TABLE default\n");
82 for (int v = 0; v < nv; ++v) fprintf(fp, "%d\n", object->verts.rec[v].owner_proc);
83
84
85 const double *const vn = object->verts.normal;
86 fprintf(fp, "NORMALS vertex_normals double\n");
87 for (int v = 0; v < nv; ++v) fprintf(fp, "%.10e %.10e %.10e\n", vn[3 * v + 0], vn[3 * v + 1], vn[3 * v + 2]);
88
89 fprintf(fp, "\nCELL_DATA %d\n", nt);
90 fprintf(fp, "SCALARS TriangleOwner int 1\nLOOKUP_TABLE default\n");
91 for (int t = 0; t < nt; ++t) fprintf(fp, "%d\n", object->tris.rec[t].owner_proc);
92
93 fprintf(fp, "SCALARS area double 1\nLOOKUP_TABLE default\n");
94 for (int t = 0; t < nt; ++t) fprintf(fp, "%.10e\n", object->tris.area[t]);
95
96 const int err = ferror(fp);
97 const int closed = fclose(fp);
98 free(iobuf); /* only after fclose: stdio uses it until then */
99 if (closed != 0 || err)
100 {
101 fprintf(stderr, "MFTLWriteVtk: write error on %s.\n", filename);
102 return MFTL_ERROR;
103 }
104 return MFTL_SUCCESS;
105}
