1/**
2 * @file topo_cluster.c
3 * @author Christian Gorges
4 * @date 2026-09-25
5 * @ingroup mftl_topology_change
6 *
7 * @brief The reconstruction lattice, the clusters, and each rank's window.
8 *
9 * ## ONE LATTICE, ALIGNED TO EVERYTHING
10 *
11 * The lattice divides every axis into `cuts[k] * m` cells. Two alignments
12 * follow, and every other file relies on them:
13 *
14 * A NODE PLANE ON EVERY WALL. A surface meeting a wall -- the rim of an open
15 * interface -- is cut by marching cubes exactly in the wall plane, so the
16 * reconstructed rim lies ON the wall rather than a fraction of a cell away.
17 *
18 * A NODE PLANE ON EVERY CUT PLANE. Every lattice cell lies inside exactly one
19 * rank box, so "who owns this cell" is integer arithmetic and every triangle
20 * marching cubes produces has exactly one owner, whatever rank computed it.
21 *
22 * ## A WINDOW, NOT A GRID
23 *
24 * No rank stores the lattice. It stores a WINDOW per cluster: its own box, plus
25 * a pad wide enough that every owned triangle's rasterisation footprint fits,
26 * intersected with the cluster's extent. That is the original design's "component grid"
27 * per interacting cluster, cut into per-rank pieces -- memory follows the part
28 * of the cluster near this rank, never the domain.
29 */

48void MFTLTopologyLattice(const MFTL_Domain *domain, double dx, double spacing[3], int cells[3])
49{
50 MFTL_ASSERT(domain != MFTL_NULL && dx > 0.0);
51 for (int k = 0; k < 3; ++k)
52 {
53 const double L = domain->domain_size[k];
54 const int cuts = domain->cuts[k];
55 int per = (int) floor(L / ((double) cuts * dx) + 0.5);
56 if (per < 1) per = 1;
57 cells[k] = per * cuts;
58 spacing[k] = L / (double) cells[k];
59 }
60}

62void MFTLTopoLatticeInit(MFTL_TopoLattice *lat, const MFTL_Domain *dom, double dx)
63{
64 memset(lat, 0, sizeof(*lat));
65 MFTLTopologyLattice(dom, dx, lat->spacing, lat->cells);
66 for (int k = 0; k < 3; ++k)
67 {
68 lat->origin[k] = dom->global_min[k];
69 lat->periodic[k] = dom->is_periodic[k];
70 lat->n_global[k] = lat->periodic[k] ? lat->cells[k] : lat->cells[k] + 1;
71 lat->per_rank[k] = lat->cells[k] / dom->cuts[k];
72 }
73}

79static int uf_find(int *parent, int i)
80{
81 while (parent[i] != i)
82 {
83 parent[i] = parent[parent[i]];
84 i = parent[i];
85 }
86 return i;

89int MFTLTopoBuildClusters(const int *bodies_in, int n_in, const MFTL_TopologyEvent *events, int n_events, MFTL_TopoCluster **out)
90{
91 /* Sorted, unique body list: the cluster order must not depend on the order
92 * the caller listed them in. */
93 int *const b = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(n_in, 1) * sizeof(int), "topology bodies");
94 memcpy(b, bodies_in, (size_t) n_in * sizeof(int));
95 MFTLSortIds(b, n_in);
96 const int nb = MFTLUniqueIntArray(b, n_in);
97
98 int *const parent = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(nb, 1) * sizeof(int), "topology union-find");
99 for (int i = 0; i < nb; ++i) parent[i] = i;
100
101 /* ANY event between two bodies joins them, whatever its kind. Coalescence is
102 * the obvious case; the other is a liquid film between two surfaces that bound
103 * the same liquid from opposite sides -- a drop and a bubble inside it. Each
104 * body's field alone would erode the film from its own side only, and the
105 * film opens only in the union. Union by the smaller index keeps the root
106 * deterministic. */
107 for (int e = 0; e < n_events; ++e)
108 {
109 if (events[e].body_a == events[e].body_b) continue;
110 const int ia = MFTLSortedIndexOf(b, nb, events[e].body_a), ib = MFTLSortedIndexOf(b, nb, events[e].body_b);
111 if (ia < 0 || ib < 0) continue;
112 const int ra = uf_find(parent, ia), rb = uf_find(parent, ib);
113 if (ra < rb)
114 parent[rb] = ra;
115 else if (rb < ra)
116 parent[ra] = rb;
117 }
118
119 /* Clusters in order of their smallest body, which is their root. */
120 int *const cl_of = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(nb, 1) * sizeof(int), "topology cluster index");
121 int n_cl = 0;
122 for (int i = 0; i < nb; ++i) cl_of[i] = (uf_find(parent, i) == i) ? n_cl++ : -1;
123 for (int i = 0; i < nb; ++i) cl_of[i] = cl_of[uf_find(parent, i)];
124
125 MFTL_TopoCluster *const cl = (MFTL_TopoCluster *) MFTLCallocOrDie((size_t) MFTL_MAX(n_cl, 1), sizeof(MFTL_TopoCluster), "topology clusters");
126 for (int i = 0; i < nb; ++i) cl[cl_of[i]].n_bodies++;
127 for (int c = 0; c < n_cl; ++c)
128 {
129 cl[c].bodies = (int *) MFTLMallocOrDie((size_t) cl[c].n_bodies * sizeof(int), "topology cluster bodies");
130 cl[c].n_bodies = 0;
131 }
132 for (int i = 0; i < nb; ++i) cl[cl_of[i]].bodies[cl[cl_of[i]].n_bodies++] = b[i];
133
134 /* Events to the cluster of their first body. */
135 int *const ev_cl = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(n_events, 1) * sizeof(int), "topology event clusters");
136 for (int e = 0; e < n_events; ++e)
137 {
138 const int ia = MFTLSortedIndexOf(b, nb, events[e].body_a);
139 ev_cl[e] = (ia >= 0) ? cl_of[ia] : -1;
140 if (ev_cl[e] >= 0) cl[ev_cl[e]].n_events++;
141 }
142 for (int c = 0; c < n_cl; ++c)
143 {
144 cl[c].events = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(cl[c].n_events, 1) * sizeof(int), "topology cluster events");
145 cl[c].n_events = 0;
146 }
147 for (int e = 0; e < n_events; ++e)
148 if (ev_cl[e] >= 0) cl[ev_cl[e]].events[cl[ev_cl[e]].n_events++] = e;
149
150 free(ev_cl);
151 free(cl_of);
152 free(parent);
153 free(b);
154 *out = cl;
155 return n_cl;
156}

178/* ═══════════════════════════════════════════════════════════════════════════
179 * 3. Extents: centre and half-width on the circle
180 *
181 * The same two-frame trick the spatial halo uses (communication/spatial_halo.c):
182 * a body straddling a periodic seam has a raw min/max box spanning the whole
183 * domain, and in a frame shifted by half a period it does not. Both frames are
184 * reduced -- as a MIN of the lower bound and a MIN of the NEGATED upper bound,
185 * so one reduction does both -- and the tighter one wins per axis.
186 * ═══════════════════════════════════════════════════════════════════════════ */
187
188void MFTLTopoLocalExtents(const MFTL_Ensemble *ens, const MFTL_TopoCluster *cl, int n_cl, double *out)
189{
190 const MFTL_Domain *const dom = &ens->domain;
191 for (int i = 0; i < 12 * n_cl; ++i) out[i] = MFTL_DBL_MAX;
192
193 for (int o = 0; o < ens->n_local_objects; ++o)
194 {
195 const MFTL_Object *const obj = ens->local_objects[o];
196 const int c = MFTLTopoClusterOf(cl, n_cl, obj->global_id);
197 if (c < 0) continue;
198 double *const e = &out[12 * c];
199
200 for (int t = 0; t < obj->tris.n_local; ++t)
201 {
202 double T[3][3];
203 MFTLGetTriangleCorners(obj, t, T);
204 for (int k = 0; k < 3; ++k)
205 {
206 const double lo = fmin(T[0][k], fmin(T[1][k], T[2][k]));
207 const double hi = fmax(T[0][k], fmax(T[1][k], T[2][k]));
208 e[k] = fmin(e[k], lo);
209 e[3 + k] = fmin(e[3 + k], -hi);
210
211 if (dom->is_periodic[k])
212 {
213 /* Shift the triangle so its first corner moves by half a period,
214 * then wrap that corner back; the others follow rigidly. */
215 const double L = dom->domain_size[k];
216 double c0 = T[0][k] + 0.5 * L;
217 c0 -= L * floor((c0 - dom->global_min[k]) / L);
218 const double d = c0 - T[0][k];
219 e[6 + k] = fmin(e[6 + k], lo + d);
220 e[9 + k] = fmin(e[9 + k], -(hi + d));
221 }
222 }
223 }
224 }

227void MFTLTopoResolveExtents(const MFTL_Domain *dom, MFTL_TopoCluster *cl, int n_cl, const double *reduced)
228{
229 for (int c = 0; c < n_cl; ++c)
230 {
231 const double *const e = &reduced[12 * c];
232 cl[c].empty = !(e[0] < MFTL_DBL_MAX);
233 for (int k = 0; k < 3; ++k)
234 {
235 cl[c].full[k] = MFTL_FALSE;
236 if (cl[c].empty)
237 {
238 cl[c].centre[k] = 0.0;
239 cl[c].half[k] = 0.0;
240 continue;
241 }
242 const double lo0 = e[k], hi0 = -e[3 + k];
243 double lo = lo0, hi = hi0;
244 if (dom->is_periodic[k])
245 {
246 const double L = dom->domain_size[k];
247 const double lo1 = e[6 + k], hi1 = -e[9 + k];
248 if (hi1 - lo1 < hi0 - lo0)
249 {
250 lo = lo1 - 0.5 * L;
251 hi = hi1 - 0.5 * L;
252 }
253 if (hi - lo >= L) cl[c].full[k] = MFTL_TRUE;
254 }
255 cl[c].centre[k] = 0.5 * (lo + hi);
256 cl[c].half[k] = 0.5 * (hi - lo);
257 }
258 }
259}

261double MFTLTopoLocalOverhang(const MFTL_Ensemble *ens, const MFTL_TopoCluster *cl, int n_cl, const MFTL_TopoLattice *lat)
262{
263 const MFTL_Domain *const dom = &ens->domain;
264 double worst = 0.0;
265
266 double box_lo[3], box_hi[3];
267 for (int k = 0; k < 3; ++k)
268 {
269 box_lo[k] = lat->origin[k] + (double) (lat->coords[k] * lat->per_rank[k]) * lat->spacing[k];
270 box_hi[k] = box_lo[k] + (double) lat->per_rank[k] * lat->spacing[k];
271 }
272
273 for (int o = 0; o < ens->n_local_objects; ++o)
274 {
275 const MFTL_Object *const obj = ens->local_objects[o];
276 if (MFTLTopoClusterOf(cl, n_cl, obj->global_id) < 0) continue;
277
278 for (int t = 0; t < obj->tris.n_local; ++t)
279 {
280 double T[3][3];
281 MFTLGetTriangleCorners(obj, t, T);
282 for (int j = 0; j < 3; ++j)
283 {
284 for (int k = 0; k < 3; ++k)
285 {
286 /* The corner in the image nearest the box centre. */
287 double x = T[j][k];
288 if (dom->is_periodic[k])
289 {
290 const double L = dom->domain_size[k], mid = 0.5 * (box_lo[k] + box_hi[k]);
291 x -= L * nearbyint((x - mid) / L);
292 }
293 const double out = fmax(box_lo[k] - x, x - box_hi[k]);
294 if (out > worst) worst = out;
295 }
296 }
297 }
298 }
299 return worst;
300}

306void MFTLTopoMakeWindows(const MFTL_Domain *dom, const MFTL_TopoLattice *lat, MFTL_TopoCluster *cl, int n_cl)
307{
308 (void) dom;
309 for (int c = 0; c < n_cl; ++c)
310 {
311 cl[c].has_window = MFTL_FALSE;
312 if (cl[c].refused || cl[c].empty) continue;
313
314 int lo[3], dims[3];
315 bool empty = MFTL_FALSE;
316
317 for (int k = 0; k < 3 && !empty; ++k)
318 {
319 const int n = lat->per_rank[k], N = lat->cells[k];
320 int wlo = lat->coords[k] * n - lat->pad;
321 int whi = (lat->coords[k] + 1) * n + lat->pad; // inclusive node index
322
323 if (lat->periodic[k])
324 {
325 if (whi - wlo + 1 >= N)
326 {
327 wlo = lat->coords[k] * n;
328 whi = wlo + N - 1;
329 }
330 else if (!cl[c].full[k])
331 {
332 /* The cluster's node range, moved to the image nearest the window. */
333 const double margin = lat->r_cut + lat->spacing[k];
334 int clo = (int) floor((cl[c].centre[k] - cl[c].half[k] - margin - lat->origin[k]) / lat->spacing[k]);
335 int chi = (int) ceil((cl[c].centre[k] + cl[c].half[k] + margin - lat->origin[k]) / lat->spacing[k]);
336 if ((chi - clo + 1) + (whi - wlo + 1) <= N)
337 {
338 const int shift = (int) floor(((0.5 * (wlo + whi)) - 0.5 * (clo + chi)) / N + 0.5);
339 clo += shift * N;
340 chi += shift * N;
341 wlo = MFTL_MAX(wlo, clo);
342 whi = MFTL_MIN(whi, chi);
343 if (whi < wlo) empty = MFTL_TRUE;
344 }
345 }
346 }
347 else
348 {
349 wlo = MFTL_MAX(wlo, 0);
350 whi = MFTL_MIN(whi, N);
351 const double margin = lat->r_cut + lat->spacing[k];
352 const int clo = (int) floor((cl[c].centre[k] - cl[c].half[k] - margin - lat->origin[k]) / lat->spacing[k]);
353 const int chi = (int) ceil((cl[c].centre[k] + cl[c].half[k] + margin - lat->origin[k]) / lat->spacing[k]);
354 wlo = MFTL_MAX(wlo, clo);
355 whi = MFTL_MIN(whi, chi);
356 if (whi < wlo) empty = MFTL_TRUE;
357 }
358 lo[k] = wlo;
359 dims[k] = whi - wlo + 1;
360 }
361 if (empty) continue;
362
363 MFTLSdfGridInitWindow(&cl[c].grid, lat->origin, lat->spacing, lat->n_global, lat->periodic, lo, dims);
364 cl[c].has_window = MFTL_TRUE;
365 }
366}

368/* ═══════════════════════════════════════════════════════════════════════════
369 * 5. The one shape a level set cannot reconstruct
370 * ═══════════════════════════════════════════════════════════════════════════ */
371
372bool MFTLTopoRimsOnWalls(const MFTL_Object *obj)
373{
374 const MFTL_Domain *const dom = obj->domain;
375 const int nh = 3 * obj->tris.n_local;
376 for (int h = 0; h < nh; ++h)
377 {
378 if (!(obj->tris.he[h].flags & MFTL_FLAG_BOUNDARY)) continue;
379 const double *const a = MFTLVertexPos(obj, obj->tris.he[h].vertex_idx);
380 const double *const b = MFTLVertexPos(obj, obj->tris.he[MFTLHePrevIdx(h)].vertex_idx);
381
382 /* On a wall: both ends on the same non-periodic boundary plane, to the
383 * tolerance the examples snap rims to. */
384 bool on_wall = MFTL_FALSE;
385 for (int k = 0; k < 3 && !on_wall; ++k)
386 {
387 if (dom == MFTL_NULL || dom->is_periodic[k]) continue;
388 const double tol = 1e-8 * dom->domain_size[k];
389 if (fabs(a[k] - dom->global_min[k]) <= tol && fabs(b[k] - dom->global_min[k]) <= tol) on_wall = MFTL_TRUE;
390 if (fabs(a[k] - dom->global_max[k]) <= tol && fabs(b[k] - dom->global_max[k]) <= tol) on_wall = MFTL_TRUE;
391 }
392 if (!on_wall) return MFTL_FALSE;
393 }
394 return MFTL_TRUE;
395}

397/* ═══════════════════════════════════════════════════════════════════════════
398 * 6. Volume, for the correction that keeps a reconstruction from eating it
399 *
400 * A reconstructed surface loses volume: marching cubes cuts chords across a
401 * convex surface, O(dx^2 * curvature) per unit area, and on a whole jet that is
402 * about one percent PER reconstruction -- measured at -60 of a final 81 over a
403 * run of 45 of them before this correction existed. The liquid a cluster
404 * encloses is therefore measured before and after and restored.
405 *
406 * The measure is the flux of F = (x_k - ref) e_k, which has divergence one, so
407 * for a closed surface it IS the enclosed volume. For an open surface whose rim
408 * lies on walls it is the volume enclosed together with those walls, provided
409 * no wall contributes: a wall perpendicular to another axis has F . n = 0, and
410 * the wall x_k = ref has F = 0. So k is a non-periodic axis whose far wall
411 * carries no rim -- the jet's nozzle wall is x_min and k = x; a sheet with its
412 * rim on the four side walls takes k = z. A closed cluster takes any axis it
413 * does not wrap around, with x_k unwrapped about its centre.
414 * ═══════════════════════════════════════════════════════════════════════════ */
415
416int MFTLTopoRimWalls(const MFTL_Object *obj)
417{
418 const MFTL_Domain *const dom = obj->domain;
419 int mask = 0;
420 const int nh = 3 * obj->tris.n_local;
421 for (int h = 0; h < nh; ++h)
422 {
423 if (!(obj->tris.he[h].flags & MFTL_FLAG_BOUNDARY)) continue;
424 const double *const a = MFTLVertexPos(obj, obj->tris.he[h].vertex_idx);
425 for (int k = 0; k < 3; ++k)
426 {
427 if (dom == MFTL_NULL || dom->is_periodic[k]) continue;
428 const double tol = 1e-8 * dom->domain_size[k];
429 if (fabs(a[k] - dom->global_min[k]) <= tol) mask |= 1 << (2 * k);
430 if (fabs(a[k] - dom->global_max[k]) <= tol) mask |= 1 << (2 * k + 1);
431 }
432 }
433 return mask;
434}

436void MFTLTopoChooseFlux(const MFTL_Domain *dom, MFTL_TopoCluster *cl, int n_cl)
437{
438 for (int c = 0; c < n_cl; ++c)
439 {
440 cl[c].flux_axis = -1;
441 cl[c].flux_ref = 0.0;
442
443 if (cl[c].rim_walls == 0)
444 {
445 /* Closed: any axis the cluster does not wrap, measured about its centre. */
446 for (int k = 0; k < 3 && cl[c].flux_axis < 0; ++k)
447 if (!cl[c].full[k])
448 {
449 cl[c].flux_axis = k;
450 cl[c].flux_ref = cl[c].centre[k];
451 }
452 continue;
453 }
454
455 /* Open: a non-periodic axis with no rim on its FAR wall. The near wall, if
456 * a rim is on it, is where F vanishes. */
457 for (int k = 0; k < 3 && cl[c].flux_axis < 0; ++k)
458 {
459 if (dom->is_periodic[k]) continue;
460 if (cl[c].rim_walls & (1 << (2 * k + 1))) continue;
461 cl[c].flux_axis = k;
462 cl[c].flux_ref = dom->global_min[k];
463 }
464 }

467void MFTLTopoObjectVolume(const MFTL_Object *obj, const MFTL_TopoCluster *c, double *volume, double *area)
468{
469 const MFTL_Domain *const dom = obj->domain;
470 const int k = c->flux_axis;
471 double v = 0.0, a = 0.0;
472
473 for (int t = 0; t < obj->tris.n_local; ++t)
474 {
475 double T[3][3];
476 MFTLGetTriangleCorners(obj, t, T);
477 const double e1[3] = {T[1][0] - T[0][0], T[1][1] - T[0][1], T[1][2] - T[0][2]};
478 const double e2[3] = {T[2][0] - T[0][0], T[2][1] - T[0][1], T[2][2] - T[0][2]};
479 const double n[3] = {0.5 * (e1[1] * e2[2] - e1[2] * e2[1]), 0.5 * (e1[2] * e2[0] - e1[0] * e2[2]), 0.5 * (e1[0] * e2[1] - e1[1] * e2[0])};
480 a += sqrt(n[0] * n[0] + n[1] * n[1] + n[2] * n[2]);
481 if (k < 0) continue;
482
483 double x = (T[0][k] + T[1][k] + T[2][k]) / 3.0;
484 if (dom != MFTL_NULL && dom->is_periodic[k])
485 {
486 /* The image nearest the cluster centre: a closed cluster that crosses the
487 * seam on this axis is measured in one piece. */
488 const double L = dom->domain_size[k];
489 x -= L * nearbyint((x - c->centre[k]) / L);
490 }
491 v += (x - c->flux_ref) * n[k];
492 }
493 *volume = v;
494 *area = a;
495}
