22/**
23 * @brief The fallback path: one MFTLGetTriangleProperties() per triangle.
24 *
25 * That inline function is the definition of a triangle's geometry, so it is
26 * reused rather than re-spelled; outputs are stored straight into the arrays.
27 *
28 * @param obj The mesh. `tris.area`, `tris.normal` and `tris.center` are written.
29 * @param count Triangles to process, from index 0.
30 */
31static void geometry_by_walk(MFTL_Object *obj, int count)
32{
33 for (int t = 0; t < count; ++t)
34 {
35 MFTLGetTriangleProperties(obj, t, &obj->tris.area[t], &obj->tris.normal[3 * t], &obj->tris.center[3 * t]);
36 }
37}

39/**
40 * @brief The steady-state path: compact indices, gathered positions,
41 * periodicity unswitched (#MFTL_FOREACH_TRIANGLE_GEOMETRY).
42 *
43 * The centroid \f$\vect{p}_0 + (\vect{e}_1+\vect{e}_2)/3\f$ is a true centroid
44 * but not necessarily inside the box, so it is wrapped (redistribution feeds it
45 * to MFTLGetOwnerProc()).
46 *
47 * @param obj The mesh. Requires #MFTL_DERIVED_TOPOLOGY.
48 * @param count Triangles to process, from index 0.
49 */
50static void geometry_by_vertices(MFTL_Object *obj, int count)
51{
52 double *const restrict out_area = obj->tris.area;
53 double *const restrict out_normal = obj->tris.normal;
54 double *const restrict out_center = obj->tris.center;
55
56 const MFTL_Domain *const dom = obj->domain;
57
58 MFTL_FOREACH_TRIANGLE_GEOMETRY(
59 obj, count,
60 /* Unnormalised normal; |n| = 2A, so one sqrt serves area and normal. */
61 const double nx = e1y * e2z - e1z * e2y;
62 const double ny = e1z * e2x - e1x * e2z;
63 const double nz = e1x * e2y - e1y * e2x;
64
65 /* Unclamped magnitude, reciprocal guarded against zero -- the same
66 * expression as MFTLGetTriangleProperties(), which
67 * test_triangle_geometry_paths_are_bit_identical compares with memcmp. */
68 const double mag = MFTL_SQRT(nx * nx + ny * ny + nz * nz);
69 const double inv = (mag > 0.0) ? 1.0 / mag : 0.0;
70
71 out_area[t] = 0.5 * mag;
72 out_normal[3 * t + 0] = nx * inv;
73 out_normal[3 * t + 1] = ny * inv;
74 out_normal[3 * t + 2] = nz * inv;
75
76 double c[3] = {p0x + (e1x + e2x) * MFTL_ONE_THIRD, p0y + (e1y + e2y) * MFTL_ONE_THIRD, p0z + (e1z + e2z) * MFTL_ONE_THIRD};
77 MFTLApplyPeriodicity(dom, c);
78
79 out_center[3 * t + 0] = c[0];
80 out_center[3 * t + 1] = c[1];
81 out_center[3 * t + 2] = c[2];);
82}

84void MFTLComputeTriangleGeometry(MFTL_Object *obj, int count)
85{
86 MFTL_ASSERT(obj != MFTL_NULL);
87 MFTL_ASSERT(count <= obj->tris.n && "MFTLComputeTriangleGeometry: count exceeds the triangle pool");
88
89 if (count <= 0) return;
90
91 if ((obj->derived & MFTL_DERIVED_TOPOLOGY) != 0u)
92 geometry_by_vertices(obj, count);
93 else
94 geometry_by_walk(obj, count);
95
96 /* Only a pass over every triangle makes the ghosts' geometry current too. */
97 if (count == obj->tris.n) obj->derived |= MFTL_DERIVED_GEOMETRY;
98}

104/**
105 * @brief Closest point on the SEGMENT [a, b] to @p p.
106 *
107 * @param p Query point.
108 * @param a Segment start.
109 * @param b Segment end.
110 * @param[out] out Closest point.
111 * @param[out] where 0 at @p a, 1 at @p b, 2 strictly inside.
112 * @return Squared distance.
113 */
114static double closest_on_segment(const double p[3], const double a[3], const double b[3], double out[3], int *where)
115{
116 const double ab[3] = {b[0] - a[0], b[1] - a[1], b[2] - a[2]};
117 const double ap[3] = {p[0] - a[0], p[1] - a[1], p[2] - a[2]};
118 const double len_sq = ab[0] * ab[0] + ab[1] * ab[1] + ab[2] * ab[2];
119 const double proj = ab[0] * ap[0] + ab[1] * ap[1] + ab[2] * ap[2];
120
121 double s;
122 if (proj <= 0.0 || len_sq <= 0.0)
123 {
124 s = 0.0;
125 *where = 0;
126 }
127 else if (proj >= len_sq)
128 {
129 s = 1.0;
130 *where = 1;
131 }
132 else
133 {
134 s = proj / len_sq;
135 *where = 2;
136 }
137
138 double d_sq = 0.0;
139 for (int k = 0; k < 3; ++k)
140 {
141 out[k] = (*where == 1) ? b[k] : a[k] + s * ab[k];
142 const double d = p[k] - out[k];
143 d_sq += d * d;
144 }
145 return d_sq;
146}

148/**
149 * @brief The degenerate-triangle answer: the nearest of the three edges (ties to
150 * the lower edge index).
151 *
152 * @param p Query point.
153 * @param tri The three corners.
154 * @param[out] closest Closest point.
155 * @param[out] feature Feature of the closest point.
156 * @return Squared distance.
157 */
158static double closest_on_degenerate(const double p[3], const double tri[3][3], double closest[3], MFTL_TriangleFeature *feature)
159{
160 double best = MFTL_DBL_MAX;
161 for (int e = 0; e < 3; ++e)
162 {
163 double q[3];
164 int where = 0;
165 const double d_sq = closest_on_segment(p, tri[e], tri[(e + 1) % 3], q, &where);
166 if (d_sq < best)
167 {
168 best = d_sq;
169 closest[0] = q[0];
170 closest[1] = q[1];
171 closest[2] = q[2];
172 if (where == 0)
173 *feature = (MFTL_TriangleFeature) (MFTL_TRI_FEATURE_VERTEX0 + e);
174 else if (where == 1)
175 *feature = (MFTL_TriangleFeature) (MFTL_TRI_FEATURE_VERTEX0 + (e + 1) % 3);
176 else
177 *feature = (MFTL_TriangleFeature) (MFTL_TRI_FEATURE_EDGE0 + e);
178 }
179 }
180 return best;
181}

183double MFTLClosestPointOnTriangle(const double p[3], const double tri[3][3], double closest[3], MFTL_TriangleFeature *feature)
184{
185 double q_sink[3];
186 MFTL_TriangleFeature f_sink;
187 double *const q = (closest != MFTL_NULL) ? closest : q_sink;
188 MFTL_TriangleFeature *const f = (feature != MFTL_NULL) ? feature : &f_sink;
189
190 const double *const a = tri[0];
191 const double *const b = tri[1];
192 const double *const c = tri[2];
193
194 const double ab[3] = {b[0] - a[0], b[1] - a[1], b[2] - a[2]};
195 const double ac[3] = {c[0] - a[0], c[1] - a[1], c[2] - a[2]};
196
197 /* Degeneracy, scale-free: |ab x ac|^2 = |ab|^2 |ac|^2 sin^2(angle at a), so
198 * the test is on sin^2 alone. Below it the face formula's denominator is
199 * noise and the triangle is treated as its three edges. */
200 const double cx = ab[1] * ac[2] - ab[2] * ac[1];
201 const double cy = ab[2] * ac[0] - ab[0] * ac[2];
202 const double cz = ab[0] * ac[1] - ab[1] * ac[0];
203 const double cross_sq = cx * cx + cy * cy + cz * cz;
204 const double ab_sq = ab[0] * ab[0] + ab[1] * ab[1] + ab[2] * ab[2];
205 const double ac_sq = ac[0] * ac[0] + ac[1] * ac[1] + ac[2] * ac[2];
206 if (!(cross_sq > MFTL_VERYVERYSMALL * ab_sq * ac_sq)) return closest_on_degenerate(p, tri, q, f);
207
208 /* Vertex region A. */
209 const double ap[3] = {p[0] - a[0], p[1] - a[1], p[2] - a[2]};
210 const double d1 = ab[0] * ap[0] + ab[1] * ap[1] + ab[2] * ap[2];
211 const double d2 = ac[0] * ap[0] + ac[1] * ap[1] + ac[2] * ap[2];
212
213 double w_ab = 0.0, w_ac = 0.0; // closest = a + w_ab*ab + w_ac*ac
214
215 if (d1 <= 0.0 && d2 <= 0.0)
216 {
217 *f = MFTL_TRI_FEATURE_VERTEX0;
218 }
219 else
220 {
221 /* Vertex region B. */
222 const double bp[3] = {p[0] - b[0], p[1] - b[1], p[2] - b[2]};
223 const double d3 = ab[0] * bp[0] + ab[1] * bp[1] + ab[2] * bp[2];
224 const double d4 = ac[0] * bp[0] + ac[1] * bp[1] + ac[2] * bp[2];
225
226 /* Vertex region C. */
227 const double cp[3] = {p[0] - c[0], p[1] - c[1], p[2] - c[2]};
228 const double d5 = ab[0] * cp[0] + ab[1] * cp[1] + ab[2] * cp[2];
229 const double d6 = ac[0] * cp[0] + ac[1] * cp[1] + ac[2] * cp[2];
230
231 const double vc = d1 * d4 - d3 * d2;
232 const double vb = d5 * d2 - d1 * d6;
233 const double va = d3 * d6 - d5 * d4;
234
235 if (d3 >= 0.0 && d4 <= d3)
236 {
237 *f = MFTL_TRI_FEATURE_VERTEX1;
238 w_ab = 1.0;
239 }
240 else if (vc <= 0.0 && d1 >= 0.0 && d3 <= 0.0)
241 {
242 *f = MFTL_TRI_FEATURE_EDGE0;
243 w_ab = d1 / (d1 - d3);
244 }
245 else if (d6 >= 0.0 && d5 <= d6)
246 {
247 *f = MFTL_TRI_FEATURE_VERTEX2;
248 w_ac = 1.0;
249 }
250 else if (vb <= 0.0 && d2 >= 0.0 && d6 <= 0.0)
251 {
252 *f = MFTL_TRI_FEATURE_EDGE2;
253 w_ac = d2 / (d2 - d6);
254 }
255 else if (va <= 0.0 && (d4 - d3) >= 0.0 && (d5 - d6) >= 0.0)
256 {
257 /* On BC: b + s*(c - b) = a + (1-s)*ab + s*ac. */
258 const double s = (d4 - d3) / ((d4 - d3) + (d5 - d6));
259 *f = MFTL_TRI_FEATURE_EDGE1;
260 w_ab = 1.0 - s;
261 w_ac = s;
262 }
263 else
264 {
265 const double denom = va + vb + vc;
266 if (!(denom > 0.0)) return closest_on_degenerate(p, tri, q, f);
267 const double inv = 1.0 / denom;
268 *f = MFTL_TRI_FEATURE_FACE;
269 w_ab = vb * inv;
270 w_ac = vc * inv;
271 }
272 }
273
274 /* Corners are returned bit-exactly, not as a + 1.0*ab (which would round). */
275 switch (*f)
276 {
277 case MFTL_TRI_FEATURE_VERTEX0: q[0] = a[0], q[1] = a[1], q[2] = a[2]; break;
278 case MFTL_TRI_FEATURE_VERTEX1: q[0] = b[0], q[1] = b[1], q[2] = b[2]; break;
279 case MFTL_TRI_FEATURE_VERTEX2: q[0] = c[0], q[1] = c[1], q[2] = c[2]; break;
280 default:
281 for (int k = 0; k < 3; ++k) q[k] = a[k] + w_ab * ab[k] + w_ac * ac[k];
282 break;
283 }
284
285 double d_sq = 0.0;
286 for (int k = 0; k < 3; ++k)
287 {
288 const double d = p[k] - q[k];
289 d_sq += d * d;
290 }
291 return d_sq;
292}
