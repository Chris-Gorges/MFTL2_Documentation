1/**
2 * @file mftl_meshgen.h
3 * @author Christian Gorges
4 * @date 2026-08-14
5 * @ingroup mftl_meshgen
6 *
7 * @brief Mesh construction: indexed triangle lists, icospheres, and marching
8 * cubes on a (distributed, periodic) lattice.
9 *
10 * MFTLBuildIndexedMesh() / MFTLBuildTopologyFromIndexList() are the single place
11 * raw connectivity becomes an MFTL mesh. Every builder leaves the object with
12 * twins linked, #MFTL_FLAG_BOUNDARY recorded (while the mesh is complete) and
13 * derived connectivity built. MFTLBuildIndexedMesh(), and every generator built
14 * on it, also computes the triangle geometry (area, normal, centre), so a new
15 * mesh can go straight into MFTLRedistribute(); MFTLBuildTopologyFromIndexList()
16 * does not, because its caller still owns the positions.
17 *
18 * Sphere generation builds the whole body on rank 0 (other ranks get an empty
19 * object); MFTLRedistribute() spreads it. The collective form
20 * MFTLGenerateSphere() costs two collectives per body; the split form
21 * MFTLGenerateSphereLocal() + MFTLPublishGeneratedBodies() costs two for the
22 * whole ensemble. Theory: src/mesh_creation/README.md.
23 */

31/**
32 * @brief Builds a half-edge mesh from positions and an indexed triangle list.
33 *
34 * Establishes the conventions every module relies on: half-edge 3t+k starts at
35 * `tri_verts[3t+k]` and ends at `tri_verts[3t+(k+1)%3]` (so slot k of
36 * `tris.verts` is the origin of half-edge 3t+k); global ids are
37 * `[0, n)` per class with half-edge id 3t+k, and the `next_*_gid` counters end
38 * one past the last id; every entity is owned by the calling rank. The caller
39 * owns the winding: the normal \f$(\vect{p}_1-\vect{p}_0)\times(\vect{p}_2-\vect{p}_0)\f$
40 * points to the side the triangles are counter-clockwise from.
41 *
42 * Triangle geometry is current on return (when the object has a domain).
43 *
44 * @param object Target object; must be EMPTY and linked to a domain.
45 * @param positions [3 * n_vertices] interleaved xyz.
46 * @param n_vertices Number of vertices, >= 3.
47 * @param tri_verts [3 * n_triangles] vertex indices.
48 * @param n_triangles Number of triangles, >= 1.
49 * @return #MFTL_SUCCESS, or #MFTL_ERROR on invalid arguments (including an index
50 * out of range or a triangle with a repeated corner; checked in Release
51 * before anything is modified).
52 *
53 * @rank_local The collective bookkeeping belongs to the caller.
54 */
55int MFTLBuildIndexedMesh(MFTL_Object *object, const double *positions, int n_vertices, const int *tri_verts, int n_triangles);

57/**
58 * @brief The topology half of MFTLBuildIndexedMesh(), for a vertex pool that is
59 * already filled (positions, global ids, owners).
60 *
61 * Writes triangles and half-edges, re-derives every `out_he_idx`, links twins,
62 * marks the boundary and builds the derived arrays. Triangle and half-edge ids
63 * restart at zero (the previous generation is retired); vertex ids are
64 * untouched. Used by subdivision (which appends midpoints to the pool) and by the
65 * spatial halo's patch builder.
66 *
67 * @param object Object with a filled vertex pool (>= 3 vertices) and a domain.
68 * @param tri_verts [3 * n_triangles] vertex indices, counter-clockwise.
69 * @param n_triangles Number of triangles, >= 1.
70 * @return #MFTL_SUCCESS, or #MFTL_ERROR on invalid arguments (checked in Release
71 * before anything is modified).
72 *
73 * @rank_local
74 */
75int MFTLBuildTopologyFromIndexList(MFTL_Object *object, const int *tri_verts, int n_triangles);

77/**
78 * @brief Generates an icosphere: a regular icosahedron, @p refinements rounds of
79 * 1-to-4 midpoint subdivision, then one central projection onto the
80 * sphere of radius @p radius about @p center.
81 *
82 * On rank 0: \f$V = 10\cdot4^r+2\f$ vertices, \f$F = 20\cdot4^r\f$ triangles;
83 * other ranks get an empty object. Face normals point OUTWARD at every level (the
84 * face table is counter-clockwise seen from outside and subdivision preserves the
85 * parent's rotation), which fixes the sign of the enclosed volume, of surface
86 * tension and of the indicator function. Mean-ratio quality is at least 0.974
87 * (mean 0.997) at every level; triangle areas vary by up to about a factor 2.
88 * Positions are wrapped into a periodic domain, and triangle geometry is current
89 * on return. The result is in subdivision order; MFTLObjectReorderMorton() gives
90 * spatial locality.
91 *
92 * @param object The object to generate into; must already be in an ensemble.
93 * @param center Sphere centre.
94 * @param radius Sphere radius, >= 0.
95 * @param refinements Subdivision levels, >= 0 (0: the bare icosahedron).
96 * @return #MFTL_SUCCESS, or #MFTL_ERROR on invalid arguments.
97 *
98 * @collective One broadcast (id counters) and one reduction (global counts).
99 */
100int MFTLGenerateSphere(MFTL_Object *object, double center[3], double radius, int refinements);

106 *
107 * @code
108 * for (int b = 0; b < n_bodies; ++b) MFTLGenerateSphereLocal(objs[b], c[b], r, refine);
109 * MFTLPublishGeneratedBodies(ens);
110 * @endcode
111 *
112 * @param object The object to generate into; must already be in an ensemble.
113 * @param center Sphere centre.
114 * @param radius Sphere radius, >= 0.
115 * @param refinements Subdivision levels, >= 0.
116 * @return #MFTL_SUCCESS, or #MFTL_ERROR on failure.
117 *
118 * @rank_local Every rank must still call it (non-root ranks mark their empty
119 * object's derived arrays valid).
120 */
121int MFTLGenerateSphereLocal(MFTL_Object *object, double center[3], double radius, int refinements);

138/* ═══════════════════════════════════════════════════════════════════════════
139 * Marching cubes (src/mesh_creation/marching_cubes.c)
140 *
141 * Cube conventions: corner c of a cell is at offset
142 * (c & 1, (c >> 1) & 1, (c >> 2) & 1) from the cell's lower node. Edge e runs
143 * along axis a = e / 4 from its lower corner, whose offsets on the other axes
144 * b = (a+1)%3 and c = (a+2)%3 are (e & 1) and ((e >> 1) & 1). A case MASK has
145 * bit c set when corner c is INSIDE (value < 0).
146 * ═══════════════════════════════════════════════════════════════════════════ */
147

155/**
156 * @struct MFTL_MarchingCubesCase
157 * @brief The triangulation of one of the 256 sign patterns of a cell.
158 *
159 * Generated from one face rule rather than tabulated, which makes the surface
160 * crack-free and consistently oriented across neighbouring cells by
161 * construction (see src/mesh_creation/README.md).
162 */
163typedef struct MFTL_MarchingCubesCase
164{
165 int n_tris; ///< Triangles in this case.
166 /** Corner vertices: a cube edge in [0, 12), or #MFTL_MC_CENTROID + p. Wound so
167 * the normal points from inside (negative) to outside (positive). */
168 signed char tri[MFTL_MC_MAX_CASE_TRIANGLES][3];
169 int n_polys; ///< Separate surface pieces (closed edge loops) in the cell.
170 int poly_len[MFTL_MC_MAX_CASE_POLYGONS]; ///< Vertices of each loop.
171 signed char poly[MFTL_MC_MAX_CASE_POLYGONS][12]; ///< Cube edges of each loop, in winding order.
172 int n_centroids; ///< Loops that needed a centroid vertex. Zero for every case in practice; see the tests.
173} MFTL_MarchingCubesCase;

182/**
183 * @struct MFTL_LevelSetSurface
184 * @brief An extracted iso-surface as a triangle soup with LATTICE identities.
185 *
186 * Every vertex carries the key of the lattice edge it lies on, every triangle
187 * the key of the cell that produced it, both from WRAPPED global lattice
188 * indices. Two extractions of overlapping cell ranges (on two ranks) therefore
189 * name a shared vertex identically and place it at bit-identical coordinates
190 * without comparing positions.
191 *
192 * Keys: node (i, j, k) has index `(k * N_y + j) * N_x + i` over `n_global`; an
193 * edge vertex is `3 * node + axis`; a cell is its lower node's index; a centroid
194 * vertex is `-(16 * cell + polygon) - 2` (negative, disjoint from edge keys).
195 */
196typedef struct MFTL_LevelSetSurface
197{
198 double *pos; ///< [3*n_verts] positions from WRAPPED lattice nodes: bit-identical on every rank; a seam-crossing triangle has corners a period apart.
199 long long *vkey; ///< [n_verts] lattice identity of each vertex.
200 int n_verts; ///< Vertices.
201 int cap_verts; ///< Allocated vertex slots.
202 int *tri; ///< [3*n_tris] vertex indices, normals pointing to phi > 0.
203 long long *tcell; ///< [n_tris] key of the cell each triangle came from.
204 signed char *tslot; ///< [n_tris] triangle's index within its cell's case, < 16.
205 int n_tris; ///< Triangles.
206 int cap_tris; ///< Allocated triangle slots.
207} MFTL_LevelSetSurface;

209/**
210 * @brief Marching cubes over the cells `[cell_lo, cell_hi]` of a grid window.
211 *
212 * A cell is extracted iff its eight corners are in the window and none is
213 * #MFTL_SDF_UNKNOWN. A vertex lies on each crossed lattice edge at the linear
214 * interpolant \f$t = \phi_0/(\phi_0-\phi_1)\f$. Values closer to zero than
215 * `snap * min(spacing)` are pushed out to that distance (sign preserved, zero
216 * counted positive), so no vertex lands on a lattice node. Vertices on one
217 * lattice edge are shared between the cells of one call. On a periodic axis the
218 * range must not exceed one period.
219 *
220 * @param grid The window.
221 * @param cell_lo First cell per axis, as a GLOBAL (unwrapped) lower-node index.
222 * @param cell_hi Last cell per axis, inclusive.
223 * @param snap Relative snap distance (e.g. 1e-2); 0 disables snapping (a
224 * node exactly at zero then still counts as positive).
225 * @param[out] out Appended to; zero-initialise before the first call.
226 * @return The number of triangles appended.
227 * @rank_local
228 */
229int MFTLExtractLevelSet(const MFTL_SdfGrid *grid, const int cell_lo[3], const int cell_hi[3], double snap, MFTL_LevelSetSurface *out);

237/**
238 * @brief Triangulates the zero level set of a whole grid into an EMPTY object.
239 *
240 * MFTLExtractLevelSet() over every cell of the window, positions wrapped into the
241 * object's domain, then MFTLBuildIndexedMesh(). Where the level set leaves the
242 * window the surface is open and its rim carries #MFTL_FLAG_BOUNDARY; on a
243 * periodic axis a window of exactly one period closes it across the seam.
244 * Normals point to \f$\phi > 0\f$. A level set that does not cross the window
245 * leaves the object empty (derived arrays marked current).
246 *
247 * @param obj An EMPTY object linked to a domain.
248 * @param grid The values.
249 * @param snap Relative snap distance; see MFTLExtractLevelSet().
250 * @return #MFTL_SUCCESS, or #MFTL_ERROR on invalid input.
251 * @rank_local
252 */
