43/* The 32-bit paths reinterpret int as uint32_t for digit extraction, and the
44 * pair sorts move int values through a uint32_t shuffle. Both assumptions are
45 * checked here rather than discovered at run time on an exotic target.
46 * (Accessing an int object through an unsigned int lvalue is explicitly allowed
47 * by C11 6.5p7 -- this is not a strict-aliasing violation.) */
48_Static_assert(sizeof(int) == sizeof(uint32_t), "MFTL radix sort assumes a 32-bit int");
49_Static_assert((-1 & 3) == 3, "MFTL radix sort assumes two's complement int");
50
51/* One byte per pass: 4 passes for 32-bit keys, 8 for 64-bit. All histograms
52 * together are 4 KB / 8 KB of stack, small enough for L1, which is what makes
53 * counting every digit in ONE pass over the keys worthwhile. */
54/** @brief Bits consumed per pass. One byte, so a histogram is 256 counters. */
55#define RADIX_BITS 8
56/** @brief Buckets per pass, \f$2^{\texttt{RADIX\_BITS}} = 256\f$. */
57#define HIST_SIZE (1u << RADIX_BITS)
58/** @brief Mask that extracts one digit. */
59#define HIST_MASK (HIST_SIZE - 1u)
60/** @brief Passes for a 32-bit key: \f$32/8\f$. */
61#define PASSES_32 4u
62/** @brief Passes for a 64-bit key: \f$64/8\f$. */
63#define PASSES_64 8u
64
65/**
66 * @brief Order-preserving map from `int32` to `uint32`.
67 *
68 * \f$x \mapsto x \oplus \texttt{0x80000000}\f$ turns signed comparison into
69 * unsigned comparison. Used ONLY for the already-sorted probe; the sort itself
70 * gets signed ordering for free -- see prefix_sum_histograms().
71 */
72#define SIGN_FLIP_32 0x80000000u
73
74/* Below this length the n-independent costs of a radix sort (histogram clear,
75 * 256-entry prefix sums per pass, the wrappers' malloc/free) exceed the
76 * O(n^2/4) moves of an insertion sort that stays in L1. 32 is the largest size
77 * at which insertion sort wins on both random AND descending (its worst case)
78 * input; the measurement is tabulated in src/arrayfunctions/README.md. The
79 * redistribution sorts one small buffer per neighbour rank, so this is the
80 * common case there. */
81/** @brief At or below this element count a stable insertion sort replaces the radix passes. */
82#define INSERTION_CUTOFF 32

91/**
92 * @brief Stable insertion sort of 32-bit signed keys.
93 * @param k Keys, sorted ascending in place.
94 * @param n Number of keys.
95 */
96static void insertion_sort_keys32(int *restrict k, size_t n)
97{
98 for (size_t i = 1; i < n; ++i)
99 {
100 const int key = k[i];
101 size_t j = i;
102 while (j > 0 && k[j - 1] > key)
103 {
104 k[j] = k[j - 1];
105 --j;
106 }
107 k[j] = key;
108 }
109}

179/* ═══════════════════════════════════════════════════════════════════════════
180 * Histogram counting: every digit's histogram in ONE pass over the keys, plus a
181 * branchless "already sorted?" test. Id lists collected in pool order are often
182 * already ascending, and then no scatter pass runs at all.
183 * ═══════════════════════════════════════════════════════════════════════════ */
184
185/**
186 * @return true if the keys are already in ascending SIGNED order.
187 *
188 * @param[out] hist `PASSES_32 * HIST_SIZE` counters, zeroed and filled here.
189 * @param keys The keys to profile.
190 * @param n Number of keys.
191 */
192static bool init_histograms32(uint32_t *restrict hist, const uint32_t *restrict keys, size_t n)
193{
194 memset(hist, 0, sizeof(uint32_t) * PASSES_32 * HIST_SIZE);
195
196 uint32_t sorted = 1u;
197 uint32_t prev = keys[0] ^ SIGN_FLIP_32; /* n >= 1 is a precondition */
198
199 for (size_t i = 0; i < n; ++i)
200 {
201 const uint32_t k = keys[i];
202 hist[0 * HIST_SIZE + (k & HIST_MASK)]++;
203 hist[1 * HIST_SIZE + ((k >> 8) & HIST_MASK)]++;
204 hist[2 * HIST_SIZE + ((k >> 16) & HIST_MASK)]++;
205 hist[3 * HIST_SIZE + ((k >> 24) & HIST_MASK)]++;
206
207 const uint32_t ordered = k ^ SIGN_FLIP_32;
208 sorted &= (uint32_t) (prev <= ordered);
209 prev = ordered;
210 }
211 return sorted != 0u;
212}

214/**
215 * @return true if the keys are already in ascending UNSIGNED order.
216 *
217 * @param[out] hist `PASSES_64 * HIST_SIZE` counters, zeroed and filled here.
218 * @param keys The keys to profile.
219 * @param n Number of keys.
220 */
221static bool init_histograms64(uint32_t *restrict hist, const uint64_t *restrict keys, size_t n)
222{
223 memset(hist, 0, sizeof(uint32_t) * PASSES_64 * HIST_SIZE);
224
225 uint32_t sorted = 1u;
226 uint64_t prev = keys[0]; /* n >= 1 is a precondition */
227
228 for (size_t i = 0; i < n; ++i)
229 {
230 const uint64_t k = keys[i];
231 hist[0 * HIST_SIZE + (uint32_t) (k & HIST_MASK)]++;
232 hist[1 * HIST_SIZE + (uint32_t) ((k >> 8) & HIST_MASK)]++;
233 hist[2 * HIST_SIZE + (uint32_t) ((k >> 16) & HIST_MASK)]++;
234 hist[3 * HIST_SIZE + (uint32_t) ((k >> 24) & HIST_MASK)]++;
235 hist[4 * HIST_SIZE + (uint32_t) ((k >> 32) & HIST_MASK)]++;
236 hist[5 * HIST_SIZE + (uint32_t) ((k >> 40) & HIST_MASK)]++;
237 hist[6 * HIST_SIZE + (uint32_t) ((k >> 48) & HIST_MASK)]++;
238 hist[7 * HIST_SIZE + (uint32_t) ((k >> 56) & HIST_MASK)]++;
239
240 sorted &= (uint32_t) (prev <= k);
241 prev = k;
242 }
243 return sorted != 0u;
244}

246/* ═══════════════════════════════════════════════════════════════════════════
247 * Prefix sums and the pass plan
248 * ═══════════════════════════════════════════════════════════════════════════ */
249/**
250 * @brief Turns each pass's counts into starting OFFSETS, in place.
251 *
252 * Two refinements, both touching only the \f$256 \times\f$ passes counters:
253 *
254 * - **Constant digits are skipped.** If the bucket of @p first_key's digit holds
255 * all @p n keys, the digit is constant and its pass is the identity. Dense
256 * global ids leave the top bytes constant, so typically half the passes of a
257 * 32-bit id sort vanish.
258 * - **Signed order.** For the most significant pass of a signed key the offsets
259 * are accumulated in the rotated bucket order 0x80..0xFF, 0x00..0x7F, which
260 * places negative keys first without modifying any key.
261 *
262 * @param hist `passes * HIST_SIZE` counters, converted from counts to
263 * exclusive prefix sums.
264 * @param passes Number of digit passes.
265 * @param n Number of keys, i.e. the total each histogram sums to.
266 * @param first_key Any one key; its digits identify the constant buckets.
267 * @param signed_msb `true` for signed 32-bit keys: the top pass's prefix sum starts
268 * at bucket 128 and wraps to 0, which places negative keys first
269 * without ever flipping a bit.
270 * @return Bit mask of the passes that must actually run.
271 */
272static uint32_t prefix_sum_histograms(uint32_t *restrict hist, uint32_t passes, size_t n, uint64_t first_key, bool signed_msb)
273{
274 uint32_t active = 0u;
275
276 for (uint32_t b = 0; b < passes; ++b)
277 {
278 uint32_t *restrict off = hist + b * HIST_SIZE;
279
280 const uint32_t bucket0 = (uint32_t) ((first_key >> (b * RADIX_BITS)) & HIST_MASK);
281 if (off[bucket0] == (uint32_t) n) continue; /* digit is constant -> no-op pass */
282
283 /* bias == 128 walks the buckets as 0x80..0xFF, 0x00..0x7F (signed order). */
284 const uint32_t bias = (signed_msb && b == passes - 1u) ? (HIST_SIZE / 2u) : 0u;
285
286 uint32_t running = 0u;
287 for (uint32_t j = 0; j < HIST_SIZE; ++j)
288 {
289 const uint32_t i = (j + bias) & HIST_MASK;
290 const uint32_t count = off[i];
291 off[i] = running;
292 running += count;
293 }
294 active |= 1u << b;
295 }
296 return active;
297}

299/* ═══════════════════════════════════════════════════════════════════════════
300 * Scatter passes: sequential reads, writes to up to 256 streams. Not
301 * vectorisable in portable C (each store depends on a counter the previous
302 * iteration may have incremented), so they stay in their plain form.
303 * ═══════════════════════════════════════════════════════════════════════════ */
304
305/**
306 * @brief One stable counting-sort pass over 32-bit keys.
307 * @param hist Offsets for this pass, advanced as elements are placed.
308 * @param shift Bit position of the digit.
309 * @param in Source keys.
310 * @param out Destination keys.
311 * @param n Number of keys.
312 */
313static inline void radixpass32(uint32_t *restrict hist, uint32_t shift, const uint32_t *restrict in, uint32_t *restrict out, size_t n)
314{
315 for (size_t i = 0; i < n; ++i)
316 {
317 const uint32_t k = in[i];
318 const uint32_t pos = (k >> shift) & HIST_MASK;
319 out[hist[pos]++] = k;
320 }
321}

341/* Key + value, SoA. Values are always uint32_t in the internal shuffle
342 * regardless of key width -- they are int indices, checked by the static
343 * assertion at the top of this file. */
344/**
345 * @brief One stable pass over 32-bit key/value pairs held as two arrays.
346 * @param hist Offsets for this pass, advanced as elements are placed.
347 * @param shift Bit position of the digit.
348 * @param ki Source keys.
349 * @param ko Destination keys.
350 * @param vi Source values.
351 * @param vo Destination values.
352 * @param n Number of pairs.
353 */
354static inline void radixpass_soa32(uint32_t *restrict hist, uint32_t shift, const uint32_t *restrict ki, uint32_t *restrict ko, const uint32_t *restrict vi,
355 uint32_t *restrict vo, size_t n)
356{
357 for (size_t i = 0; i < n; ++i)
358 {
359 const uint32_t k = ki[i];
360 const uint32_t pos = (k >> shift) & HIST_MASK;
361 const uint32_t idx = hist[pos]++;
362 ko[idx] = k;
363 vo[idx] = vi[i];
364 }
365}

369 * @param hist Offsets for this pass, advanced as elements are placed.
370 * @param shift Bit position of the digit.
371 * @param ki Source keys.

392 * ═══════════════════════════════════════════════════════════════════════════ */
393
394void MFTLRadixSort32(int *keys_in, int *keys_temp, int size)
395{
396 if (size < 2) return;
397 if (size <= INSERTION_CUTOFF)
398 {
399 insertion_sort_keys32(keys_in, (size_t) size);
400 return;
401 }
402
403 MFTL_ASSERT(keys_in != MFTL_NULL);
404 MFTL_ASSERT(keys_temp != MFTL_NULL);
405
406 const size_t n = (size_t) size;
407 uint32_t *ui = (uint32_t *) keys_in;
408 uint32_t *ut = (uint32_t *) keys_temp;
409
410 _Alignas(64) uint32_t hist[PASSES_32 * HIST_SIZE]; /* 4 KB stack */
411
412 if (init_histograms32(hist, ui, n)) return; /* already sorted */
413 const uint32_t active = prefix_sum_histograms(hist, PASSES_32, n, (uint64_t) ui[0], true);
414
415 /* Ping-pong between input and scratch; with skipped passes the count can be
416 * odd, in which case the result is copied back from the scratch buffer. */
417 uint32_t *buf[2] = {ui, ut};
418 uint32_t cur = 0u;
419 for (uint32_t b = 0; b < PASSES_32; ++b)
420 {
421 if (!(active & (1u << b))) continue;
422 radixpass32(hist + b * HIST_SIZE, b * RADIX_BITS, buf[cur], buf[cur ^ 1u], n);
423 cur ^= 1u;
424 }
425 if (cur != 0u) memcpy(ui, ut, n * sizeof *ui);
426}

472/* ═══════════════════════════════════════════════════════════════════════════
473 * Public: key/value pair sorts (SoA). The histogram is built before any scratch
474 * is allocated, so sorted or constant input never touches the allocator.
475 * ═══════════════════════════════════════════════════════════════════════════ */
476
477void MFTLSortPairs32(int *keys, int *values, int count)
478{
479 if (count < 2) return;
480 if (count <= INSERTION_CUTOFF)
481 {
482 insertion_sort_pairs32(keys, values, (size_t) count);
483 return;
484 }
485
486 MFTL_ASSERT(keys != MFTL_NULL);
487 MFTL_ASSERT(values != MFTL_NULL);
488
489 const size_t n = (size_t) count;
490 uint32_t *ki = (uint32_t *) keys;
491 uint32_t *vi = (uint32_t *) values;
492
493 _Alignas(64) uint32_t hist[PASSES_32 * HIST_SIZE]; /* 4 KB stack */
494
495 if (init_histograms32(hist, ki, n)) return; /* already sorted (stable: values stay put) */
496 const uint32_t active = prefix_sum_histograms(hist, PASSES_32, n, (uint64_t) ki[0], true);
497 if (active == 0u) return; /* no pass to run; guards against allocating for a no-op */
498
499 uint32_t *kt = (uint32_t *) MFTLMallocOrDie(sizeof(int) * n, "MFTLSortPairs32 keys");
500 uint32_t *vt = (uint32_t *) MFTLMallocOrDie(sizeof(int) * n, "MFTLSortPairs32 values");
501
502 uint32_t *kb[2] = {ki, kt};
503 uint32_t *vb[2] = {vi, vt};
504 uint32_t cur = 0u;
505 for (uint32_t b = 0; b < PASSES_32; ++b)
506 {
507 if (!(active & (1u << b))) continue;
508 radixpass_soa32(hist + b * HIST_SIZE, b * RADIX_BITS, kb[cur], kb[cur ^ 1u], vb[cur], vb[cur ^ 1u], n);
509 cur ^= 1u;
510 }
511 if (cur != 0u)
512 {
513 memcpy(ki, kt, n * sizeof *ki);
514 memcpy(vi, vt, n * sizeof *vi);
515 }
516
517 free(kt);
518 free(vt);
519}
