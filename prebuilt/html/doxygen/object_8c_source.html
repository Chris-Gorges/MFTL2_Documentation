5 * @ingroup mftl_object

7 * @brief Object lifecycle and the single allocation authority for both pools.
8 *
9 * `*_layout_compute` decides where each sub-array starts inside a pool's one
10 * block, `*_pool_bind` publishes those addresses, and `*_pool_resize` is the
11 * only function that allocates, copies or frees a block (`*_pool_reserve` is
12 * its grow-only wrapper, MFTLObjectReleasePools() is a resize to zero, and
13 * MFTLObjectShrinkToFit() is the only way down). Hence the AoS records and every
14 * SoA array of a pool always share one capacity: they are the same `malloc`.

31/** @brief Byte offsets of every vertex sub-array inside one block. */
32typedef struct
33{
34 size_t rec; ///< [capacity] MFTL_Vertex records.
35 size_t pos; ///< [3*capacity] AUTHORITATIVE positions, xyz interleaved.
36 size_t normal; ///< [3*capacity] vertex normals, xyz interleaved.
37} vertex_layout;
38
39/**
40 * @brief Lays out a vertex block for @p cap vertices: `rec`, `pos`, `normal`.
41 *
42 * `rec` and `pos`, which every pass touches, come first.
43 *
44 * @param[out] l Receives one byte offset per field.
45 * @param cap Vertices the block must hold.
46 * @return Total block size in bytes.
47 */
48static size_t vertex_layout_compute(vertex_layout *l, int cap)
49{
50 const size_t n = (size_t) cap;
51 size_t cursor = 0;
52 l->rec = MFTLLayoutAdd(&cursor, n, sizeof(MFTL_Vertex));
53 l->pos = MFTLLayoutAdd(&cursor, 3 * n, sizeof(double));
54 l->normal = MFTLLayoutAdd(&cursor, 3 * n, sizeof(double));
55 return cursor;
56}

58/**
59 * @brief Points every field of @p p into @p blk according to @p l.
60 *
61 * @param[out] p The pool whose pointers are set.
62 * @param blk Base of the allocation.
63 * @param l Offsets from vertex_layout_compute().
64 */
65static void vertex_pool_bind(MFTL_VertexPool *p, unsigned char *blk, const vertex_layout *l)
66{
67 p->rec = (MFTL_Vertex *) (void *) (blk + l->rec);
68 p->pos = (double *) (void *) (blk + l->pos);
69 p->normal = (double *) (void *) (blk + l->normal);
70}

72/**
73 * @brief Resizes the vertex pool to exactly @p capacity vertices, up or down.
74 *
75 * Lays out a new block, zeroes it (so flags and never-written slots start
76 * defined), copies the live prefix of every array and frees the old block --
77 * one code path for growth and shrink. Capacity 0 releases the block and zeroes
78 * the struct (NULL pointers, not a dangling allocation of nothing).
79 *
80 * @param p The pool.
81 * @param capacity Vertices it must hold afterwards; at least `p->n` (asserted:
82 * this moves storage and never discards entities).
83 */
84static void vertex_pool_resize(MFTL_VertexPool *p, int capacity)
85{
86 MFTL_ASSERT(capacity >= p->n && "vertex_pool_resize: would discard live entities");
87
88 if (capacity == p->capacity) return;
89
90 /* Zero capacity is a release: free, and zero the struct so every pointer is
91 * NULL (the state MFTLObjectPoolsAreConsistent() expects of an empty pool). */
92 if (capacity == 0)
93 {
94 free(p->block);
95 memset(p, 0, sizeof *p);
96 return;
97 }
98
99 vertex_layout l;
100 const size_t bytes = vertex_layout_compute(&l, capacity);
101 unsigned char *blk = (unsigned char *) MFTLAlignedAllocOrDie(bytes, "MFTL vertex pool");
102
103 /* Zeroed so never-written fields (flags in particular) start defined. */
104 memset(blk, 0, bytes);
105
106 const size_t n = (size_t) p->n;
107 if (n > 0)
108 {
109 memcpy(blk + l.rec, p->rec, n * sizeof(MFTL_Vertex));
110 memcpy(blk + l.pos, p->pos, 3 * n * sizeof(double));
111 memcpy(blk + l.normal, p->normal, 3 * n * sizeof(double));
112 }
113
114 free(p->block);
115 p->block = blk;
116 vertex_pool_bind(p, blk, &l);
117 p->capacity = capacity;
118}

120/**
121 * @brief Grows the vertex pool to @p capacity vertices; a no-op if it already
122 * holds that many (the ratchet MFTLObjectReserve() publishes).
123 *
124 * @param p The pool.
125 * @param capacity Vertices the pool must hold afterwards.
126 */
127static void vertex_pool_reserve(MFTL_VertexPool *p, int capacity)
128{
129 if (capacity <= p->capacity) return;
130 vertex_pool_resize(p, capacity);
131}

137/** @brief Byte offsets of every triangle/half-edge sub-array inside one block. */
138typedef struct
139{
140 size_t rec; ///< [capacity] MFTL_Triangle records.
141 size_t he; ///< [3*capacity] MFTL_HalfEdge records. Three per triangle, always.
142 size_t area; ///< [capacity] triangle areas.
143 size_t normal; ///< [3*capacity] unit normals, xyz interleaved.
144 size_t center; ///< [3*capacity] centroids, wrapped into the domain.
145 size_t verts; ///< [3*capacity] derived: vertex index per corner.
146 size_t neighbors; ///< [3*capacity] derived: triangle across each half-edge.
147} triangle_layout;
148
149/**
150 * @brief Lays out a triangle block for @p cap triangles and 3 @p cap half-edges:
151 * `rec`, `he`, `area`, `normal`, `center`, `verts`, `neighbors`.
152 *
153 * The half-edges live in this block because a triangle has exactly three, at
154 * 3t..3t+2; a separate half-edge capacity would describe an impossible mesh.
155 *
156 * @param[out] l Receives one byte offset per field.
157 * @param cap Triangles the block must hold.
158 * @return Total block size in bytes.
159 */
160static size_t triangle_layout_compute(triangle_layout *l, int cap)
161{
162 const size_t n = (size_t) cap;
163 size_t cursor = 0;
164 l->rec = MFTLLayoutAdd(&cursor, n, sizeof(MFTL_Triangle));
165 l->he = MFTLLayoutAdd(&cursor, 3 * n, sizeof(MFTL_HalfEdge));
166 l->area = MFTLLayoutAdd(&cursor, n, sizeof(double));
167 l->normal = MFTLLayoutAdd(&cursor, 3 * n, sizeof(double));
168 l->center = MFTLLayoutAdd(&cursor, 3 * n, sizeof(double));
169 l->verts = MFTLLayoutAdd(&cursor, 3 * n, sizeof(int));
170 l->neighbors = MFTLLayoutAdd(&cursor, 3 * n, sizeof(int));
171 return cursor;
172}

192/**
193 * @brief Resizes the triangle pool (and its half-edges) to exactly @p capacity
194 * triangles, up or down. Same contract as vertex_pool_resize().
195 *
196 * @param p The pool.
197 * @param capacity Triangles it must hold afterwards; at least `p->n`.
198 */
199static void triangle_pool_resize(MFTL_TrianglePool *p, int capacity)
200{
201 MFTL_ASSERT(capacity >= p->n && "triangle_pool_resize: would discard live entities");
202
203 if (capacity == p->capacity) return;
204
205 if (capacity == 0)
206 {
207 free(p->block);
208 memset(p, 0, sizeof *p);
209 return;
210 }
211
212 triangle_layout l;
213 const size_t bytes = triangle_layout_compute(&l, capacity);
214 unsigned char *blk = (unsigned char *) MFTLAlignedAllocOrDie(bytes, "MFTL triangle pool");
215 memset(blk, 0, bytes);
216
217 const size_t n = (size_t) p->n;
218 if (n > 0)
219 {
220 memcpy(blk + l.rec, p->rec, n * sizeof(MFTL_Triangle));
221 memcpy(blk + l.he, p->he, 3 * n * sizeof(MFTL_HalfEdge));
222 memcpy(blk + l.area, p->area, n * sizeof(double));
223 memcpy(blk + l.normal, p->normal, 3 * n * sizeof(double));
224 memcpy(blk + l.center, p->center, 3 * n * sizeof(double));
225 memcpy(blk + l.verts, p->verts, 3 * n * sizeof(int));
226 memcpy(blk + l.neighbors, p->neighbors, 3 * n * sizeof(int));
227 }
228
229 free(p->block);
230 p->block = blk;
231 triangle_pool_bind(p, blk, &l);
232 p->capacity = capacity;
233}

251MFTL_Object *MFTLObjectCreate(int global_id)
252{
253 /* calloc: every count, capacity and pointer starts at 0/NULL, which is exactly
254 * the empty-pool state the reserve functions expect. */
255 MFTL_Object *object = (MFTL_Object *) calloc(1, sizeof(MFTL_Object));
256 if (!object)
257 {
258 fprintf(stderr, "MFTL Error: Failed to allocate memory for MFTL_Object struct.\n");
259 return MFTL_NULL;
260 }
261
262 object->global_id = global_id;
263 object->domain = MFTL_NULL; // set by MFTLEnsembleAddObject
264 object->derived = 0u;
265 /* Never partitioned, so no current partition: the first redistribution must
266 * rebuild it. */
267 object->partition_dirty = MFTL_TRUE;
268
269 return object;
270}

306void MFTLObjectSetCounts(MFTL_Object *obj, int n_vertices, int n_local_vertices, int n_triangles, int n_local_triangles)
307{
308 MFTL_ASSERT(obj != MFTL_NULL);
309 MFTL_ASSERT(n_vertices >= 0 && n_triangles >= 0);
310 MFTL_ASSERT(n_local_vertices >= 0 && n_local_vertices <= n_vertices && "MFTLObjectSetCounts: owned vertices must be a prefix of the pool");
311 MFTL_ASSERT(n_local_triangles >= 0 && n_local_triangles <= n_triangles && "MFTLObjectSetCounts: owned triangles must be a prefix of the pool");
312
313 /* Reserve BEFORE assigning: this ordering is what makes "a count larger than
314 * its capacity" unreachable, no matter what the caller passes. */
315 MFTLObjectReserve(obj, n_vertices, n_triangles);
316
317 obj->verts.n = n_vertices;
318 obj->verts.n_local = n_local_vertices;
319 obj->tris.n = n_triangles;
320 obj->tris.n_local = n_local_triangles;
321
322 /* The derived arrays describe a different entity count than they did a moment
323 * ago, so whatever they hold is no longer a description of this mesh. */
324 obj->derived = 0u;
325
326 /* Nor does the partition. Every entity creation or destruction passes through
327 * here, so the redistribution's skip of unchanged bodies cannot miss one. */
328 obj->partition_dirty = MFTL_TRUE;
329}

351size_t MFTLObjectPoolBytes(const MFTL_Object *obj)
352{
353 MFTL_ASSERT(obj != MFTL_NULL);
354
355 /* Re-derived from the layout functions, so it cannot drift from them. */
356 size_t bytes = 0;
357
358 if (obj->verts.capacity > 0)
359 {
360 vertex_layout vl;
361 bytes += vertex_layout_compute(&vl, obj->verts.capacity);
362 }
363 if (obj->tris.capacity > 0)
364 {
365 triangle_layout tl;
366 bytes += triangle_layout_compute(&tl, obj->tris.capacity);
367 }
368
369 return bytes;
370}

372bool MFTLObjectShrinkToFit(MFTL_Object *obj, double slack)
373{
374 MFTL_ASSERT(obj != MFTL_NULL);
375 MFTL_ASSERT(slack >= 1.0 && "MFTLObjectShrinkToFit: a slack below 1 would ask for a capacity under the count");
376
377 bool moved = MFTL_FALSE;
378
379 /* Target: back on the growth ladder. At n == 0 MFTLGrowCapacity() returns 0,
380 * so a full release is the same arithmetic reaching *_pool_resize(p, 0). */
381 const int v_target = MFTLGrowCapacity(0, obj->verts.n);
382 const int t_target = MFTLGrowCapacity(0, obj->tris.n);
383
384 /* GATE (policy): wasteful enough to reallocate? GUARD (idempotence): would
385 * the target actually be smaller? For n <= 128 the target is a flat 128, so
386 * without the guard a capacity-128 pool with n = 4 would "shrink" to itself
387 * on every call. */
388 if ((double) obj->verts.capacity > slack * (double) obj->verts.n && v_target < obj->verts.capacity)
389 {
390 vertex_pool_resize(&obj->verts, v_target);
391 moved = MFTL_TRUE;
392 }
393
394 if ((double) obj->tris.capacity > slack * (double) obj->tris.n && t_target < obj->tris.capacity)
395 {
396 triangle_pool_resize(&obj->tris, t_target);
397 moved = MFTL_TRUE;
398 }
399
400 /* Deliberately not stamped partition-dirty: only storage moved. */
401 return moved;
402}

404void MFTLObjectReleasePools(MFTL_Object *obj)
405{
406 MFTL_ASSERT(obj != MFTL_NULL);
407 MFTL_ASSERT(obj->verts.n == 0 && obj->tris.n == 0 && "MFTLObjectReleasePools: the object still holds entities");
408
409 /* A release is a resize to capacity 0, which also re-NULLs every sub-array
410 * pointer, so the object is indistinguishable from a fresh one. */
411 vertex_pool_resize(&obj->verts, 0);
412 triangle_pool_resize(&obj->tris, 0);
413}

415bool MFTLObjectPoolsAreConsistent(const MFTL_Object *obj)
416{
417 if (obj == MFTL_NULL) return MFTL_FALSE;
418
419 const MFTL_VertexPool *vp = &obj->verts;
420 const MFTL_TrianglePool *tp = &obj->tris;
421
422 /* Counts inside capacity, owned inside active. */
423 if (vp->n < 0 || vp->n > vp->capacity) return MFTL_FALSE;
424 if (tp->n < 0 || tp->n > tp->capacity) return MFTL_FALSE;
425 if (vp->n_local < 0 || vp->n_local > vp->n) return MFTL_FALSE;
426 if (tp->n_local < 0 || tp->n_local > tp->n) return MFTL_FALSE;
427
428 /* An empty pool owns nothing and points nowhere. */
429 if (vp->capacity == 0)
430 {
431 if (vp->block != MFTL_NULL || vp->rec != MFTL_NULL || vp->pos != MFTL_NULL) return MFTL_FALSE;
432 }
433 else
434 {
435 /* Every pointer must sit exactly where the layout for this capacity puts
436 * it -- the strongest form of "all arrays were sized by the same call". */
437 vertex_layout l;
438 (void) vertex_layout_compute(&l, vp->capacity);
439 const unsigned char *blk = (const unsigned char *) vp->block;
440 if (blk == MFTL_NULL) return MFTL_FALSE;
441 if ((const unsigned char *) vp->rec != blk + l.rec) return MFTL_FALSE;
442 if ((const unsigned char *) vp->pos != blk + l.pos) return MFTL_FALSE;
443 if ((const unsigned char *) vp->normal != blk + l.normal) return MFTL_FALSE;
444 }
445
446 if (tp->capacity == 0)
447 {
448 if (tp->block != MFTL_NULL || tp->rec != MFTL_NULL || tp->he != MFTL_NULL) return MFTL_FALSE;
449 }
450 else
451 {
452 triangle_layout l;
453 (void) triangle_layout_compute(&l, tp->capacity);
454 const unsigned char *blk = (const unsigned char *) tp->block;
455 if (blk == MFTL_NULL) return MFTL_FALSE;
456 if ((const unsigned char *) tp->rec != blk + l.rec) return MFTL_FALSE;
457 if ((const unsigned char *) tp->he != blk + l.he) return MFTL_FALSE;
458 if ((const unsigned char *) tp->area != blk + l.area) return MFTL_FALSE;
459 if ((const unsigned char *) tp->normal != blk + l.normal) return MFTL_FALSE;
460 if ((const unsigned char *) tp->center != blk + l.center) return MFTL_FALSE;
461 if ((const unsigned char *) tp->verts != blk + l.verts) return MFTL_FALSE;
462 if ((const unsigned char *) tp->neighbors != blk + l.neighbors) return MFTL_FALSE;
463 }
464
465 return MFTL_TRUE;
466}
