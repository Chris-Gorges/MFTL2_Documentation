1/**
2 * @file remesh_dispersed.c
3 * @author Christian Gorges
4 * @date 2026-08-17
5 * @ingroup mftl_remeshing
6 *
7 * @brief Distributed remeshing for bodies SMALLER than a rank box: move the
8 * bodies, not the cut planes.
9 *
10 * ## The same problem from the opposite end
11 *
12 * An edge whose endpoints belong to two ranks can be collapsed by neither.
13 * remesh_parallel.c moves the cut planes until every entity has been interior
14 * somewhere; for a body that fits in a rank box it is simpler to migrate the
15 * whole body onto one rank, where no edge is shared. Both are exact; R/h
16 * decides which is cheaper:
17 *
18 * R >> h R << h
19 * move the planes the bodies
20 * phases about 4 at budget 4 (np = 2..8) 1
21 * migrations one per placement, plus one exactly 2
22 * frozen O(delta/h) of the mesh nothing
23 * id cost per-phase holder prefix sums one per-body MAX, total
24 * weakness phase count, band precondition load balance at N_bodies < P
25 *
26 * ## Why the ids need almost nothing
27 *
28 * After centering exactly ONE rank holds any entity of a body, and ids live in a
29 * per-body space (`next_*_gid` is per object), so that rank's counter is already
30 * unique for the body. The operators run untouched, MFTLRemeshRenumberGlobalIds()
31 * makes the ids dense on the owner, and the other holders learn the counters by
32 * one per-body MAX over the holders (a rank holding nothing contributes zero).
33 *
34 * ## Ordering
35 *
36 * Admission is decided BEFORE anything is modified: if a body is too large, the
37 * whole ensemble goes to the spanning driver, whose per-phase id scan assumes
38 * all holders enter with the same id counters -- which a half-remeshed
39 * ensemble would violate. The test uses radii already reduced for the strategy
40 * decision, so it costs one small reduction.
41 */

63/**
64 * @brief The box width a body's radius is judged against: the rank box, or the
65 * COARSENED box when the fallback is armed and the fine one is too small.
66 *
67 * ═══════════════════════════════════════════════════════════════════════════
68 * WHY THE YARDSTICK MOVES AND THE ROUTING DOES NOT
69 * ═══════════════════════════════════════════════════════════════════════════
70 *
71 * `r_max > frac * h_min` sends the whole ensemble to the spanning driver. Raise
72 * the rank count on a fixed domain and `h_min` shrinks, so a spray of droplets
73 * that are ABSOLUTELY tiny becomes RELATIVELY large and is classified SPANNING
74 * -- for a reason that has nothing to do with the bodies and everything to do
75 * with how many ranks were asked for. The spanning driver is the wrong answer
76 * there by roughly two migrations against thirty-four.
77 *
78 * What the test is actually asking is "can one rank reasonably hold this body",
79 * and that question is answered by whatever box the pass is willing to work in.
80 * So when `parallel_coarsen` is armed, the yardstick becomes the coarsened box
81 * `group[k] * h_k` -- the same relaxation the spanning driver would apply to
82 * itself, capped by the same knob.
83 *
84 * ── The yardstick moves; the DESTINATIONS do not ──────────────────────────
85 *
86 * Nothing here changes where a body is sent. Routing through coarse group
87 * representatives would restrict destinations to `P/G` ranks and undo the load
88 * balancer, which needs every rank to be reachable. The coarse box appears
89 * ONLY as the size a rank is deemed able to hold, which is a memory statement
90 * and not a routing one.
91 *
92 * ══ THE RELAXATION IS BOUNDED, AND THE BOUND IS NOT GEOMETRIC ══════════════
93 *
94 * An unbounded version of this is worse than useless, and it is worth saying
95 * exactly how because it is not obvious and it is what the tests caught. The
96 * factors clamp at `cuts[k]`, so on a 2x1x1 grid the coarse box is the WHOLE
97 * DOMAIN -- every body fits, SPANNING becomes unreachable, and a jet spanning
98 * the domain gets serialised onto one rank while reporting itself perfectly
99 * centred.
100 *
101 * The real precondition for relaxing is not about the body at all. Dispersed
102 * puts one body on one rank, so it uses `min(N_bodies, P)` ranks and no more --
103 * and the relaxation is worth having exactly when that is still all of them.
104 * With fewer bodies than ranks the strategy is already leaving ranks idle, the
105 * relaxation buys no parallelism, and all it does is take a large body that the
106 * spanning schedule would have shared and give it to one rank. So:
107 *
108 * N_bodies >= nproc there are enough bodies to keep the machine busy, and
109 * the fine box has stopped being a statement about the
110 * bodies and started being one about the rank count
111 * G < nproc never merge the entire grid into one box. That is the
112 * gather, it is a spanning-path decision, and it has its
113 * own parameter with its own memory bound
114 *
115 * ── It must give the same answer to the chooser and to the driver ─────────
116 *
117 * Both call this. If they disagreed, the chooser would elect DISPERSED on a body
118 * the driver then rejects, and the pass would pay a wasted measurement and fall
119 * back to spanning -- correct, slower, and reported only as
120 * `n_bodies_not_centred`.
121 *
122 * @param dom The MPI/physical world.
123 * @param params May be #MFTL_NULL, in which case no relaxation is applied.
124 * @param r_max The largest body radius anywhere.
125 * @param n_bodies Bodies in the ensemble. Below `nproc` no relaxation happens.
126 * @return The box width to compare `r_max / frac` against; 0.0 when no axis is
127 * cut, which the callers already treat as "the question is meaningless".
128 * @rank_local Pure arithmetic on replicated data, which is what lets the chooser
129 * use it without a second collective.
130 */
131static double dispersed_box_min(const MFTL_Domain *dom, const MFTL_RemeshParameters *params, double r_max, int n_bodies)
132{
133 const double h_min = MFTLDomainGetHMin(dom);
134 if (h_min <= 0.0) return h_min;
135
136 if (params == MFTL_NULL || params->parallel_coarsen == 0) return h_min;
137 if (n_bodies < dom->nproc) return h_min; // not enough bodies for the relaxation to buy anything
138
139 const double frac = (params->dispersed_radius_fraction > 0.0) ? params->dispersed_radius_fraction : 0.4;
140 if (!(r_max > frac * h_min)) return h_min; // the fine box is already enough
141
142 /* The smallest per-axis merge that would make the body fit, by the same rule
143 * and with the same clamp the spanning fallback uses: never more groups than
144 * there are cells, because merging every cell on an axis is the most a
145 * coarsening can do and is still a legal decomposition. */
146 int group[3] = {1, 1, 1};
147 for (int k = 0; k < 3; ++k)
148 {
149 const int cuts = dom->cuts[k];
150 if (cuts < 2 || dom->domain_size[k] <= 0.0) continue;
151
152 const double h = dom->domain_size[k] / (double) cuts;
153 const double ratio = r_max / (frac * h);
154 if (ratio <= 1.0) continue;
155
156 int g = (int) ceil(ratio);
157 if (g > cuts) g = cuts;
158 group[k] = g;
159 }
160
161 MFTLRemeshCapGroups(group, params->parallel_coarsen);
162
163 /* Merging the whole grid into one box is the GATHER, and the gather is not
164 * this function's decision to take: it belongs to the spanning driver, which
165 * bounds it by a triangle count because one rank holding everything is a
166 * memory question. Refusing here leaves such an ensemble on the spanning
167 * path, where `serial_gather_max_triangles` can answer it properly. */
168 if (group[0] * group[1] * group[2] >= dom->nproc) return h_min;
169
170 double box = MFTL_DBL_MAX;
171 for (int k = 0; k < 3; ++k)
172 {
173 const int cuts = dom->cuts[k];
174 if (cuts < 2 || dom->domain_size[k] <= 0.0) continue;
175 box = MFTL_MIN(box, (double) group[k] * dom->domain_size[k] / (double) cuts);
176 }
177
178 return (box == MFTL_DBL_MAX) ? 0.0 : box;
179}

204MFTL_RemeshParallelStrategy MFTLRemeshChooseStrategyMeasured(MFTL_Ensemble *ens, const MFTL_RemeshParameters *params, MFTL_RemeshMeasurement *m)
205{
206 MFTL_ASSERT(ens != MFTL_NULL && m != MFTL_NULL);
207
208 /* At one rank nothing is split and both strategies degenerate to the serial
209 * kernel. Answering SPANNING keeps MFTLRemeshEnsemble's nproc == 1 path the
210 * only special case there is. */
211 /* GLOBAL body count: MFTLRemeshMeasureBodies() below is collective, and both
212 * this function's callers dispatch on its answer -- two ranks returning
213 * different strategies is an immediate deadlock. */
214 if (ens->domain.nproc < 2 || ens->n_bodies_global == 0) return MFTL_REMESH_PARALLEL_SPANNING;
215
216 const double frac = (params && params->dispersed_radius_fraction > 0.0) ? params->dispersed_radius_fraction : 0.4;
217
218 /* The measurement is taken even when h_min says the answer is already SPANNING,
219 * because the caller may be MFTLRemeshEnsemble, which owns the buffers and
220 * would otherwise have to know whether they were filled. Filling them
221 * unconditionally is one collective on a path that is about to run a dozen.
222 * The caller has computed the geometry it reads. */
223 MFTLRemeshMeasureBodies(ens, m->centroid, m->radius, m->weight, &m->r_max, m->lowest_holder);
224
225 if (MFTLDomainGetHMin(&ens->domain) <= 0.0) return MFTL_REMESH_PARALLEL_SPANNING;
226
227 /* ONE SCALAR, identical to the bit on every rank (a scalar MPI_MAX inside the
228 * measurement). A maximum over this rank's own radii would differ between
229 * ranks and split them between two drivers -- a deadlock. */
230 const double box = dispersed_box_min(&ens->domain, params, m->r_max, ens->n_bodies_global);
231 return (m->r_max > frac * box) ? MFTL_REMESH_PARALLEL_SPANNING : MFTL_REMESH_PARALLEL_DISPERSED;
232}

247/* ═══════════════════════════════════════════════════════════════════════════
248 * 1b. Load balancing
249 * ═══════════════════════════════════════════════════════════════════════════
250 *
251 * ## One rank decides each body and tells the holders
252 *
253 * MFTL_OwnerMap is undefined if two ranks name different destinations for one
254 * entity, so a body's destination must be AGREED. Replicating every body's
255 * centroid and weight on every rank and running one deterministic greedy would
256 * cost O(bodies in the ensemble) payload and memory per rank. Instead one rank
257 * decides and sends the verdict: O(local bodies x degree), degree <= 26.
258 *
259 * ## The precondition is a local test
260 *
261 * Let home(o) = MFTLGetOwnerProc(centroid(o)), identical on every holder because
262 * the centroid is reduced. Required:
263 *
264 * for every body o and every holder H of o: home(o) is H or a neighbour of H
265 *
266 * Each holder tests its own half without communication; one reduction agrees
267 * it. It gives exactly the two messages needed: home can be told about o
268 * (lowest_holder(o), a genuine holder, is its neighbour; the graph is
269 * symmetric), and home can tell every holder. Not required: a clique of holders,
270 * a holder owning an entity, or home holding the body
271 * (MFTLEnsembleBodyHoldersFitTheGraph() is strictly stronger and deliberately
272 * not used). On failure the pass reports n_bodies_unreachable and uses plain
273 * centroid ownership -- never an O(B) replicated fallback.
274 *
275 * ## A negotiation, because a local rule cannot see the global load
276 *
277 * Every overloaded rank sees the same idle neighbour. So each rank advertises
278 * its room (cap - load); an overloaded rank asks a FEW neighbours, emptiest
279 * first, for all of their room until its excess (plus one body) is covered; each
280 * neighbour grants its real room to its askers neediest first, so
281 * SUM(grants) <= room and nobody is pushed past the cap however many ship at
282 * once. Two failure modes shaped this, both silent: a proportional ask over 26
283 * neighbours yields grants smaller than one body, so nothing moves (108 bodies
284 * over 27 ranks stayed at imbalance 2.00); and a tie-break identical on every
285 * rank starves one target, so ties break on the neighbour's cell offset
286 * relative to the asker. Rounds are bounded (MFTL_DISPERSED_BALANCE_ROUNDS):
287 * running out leaves an imbalance that is reported, never a wrong mesh.
288 *
289 * ## What one hop costs
290 *
291 * Weight moves at most one cell per pass: a pile-up in one rank box is fixed
292 * when its neighbours have room; one spanning many cells only partly.
293 *
294 * Every body is decided by exactly one rank, so a flawed tie-break yields a
295 * different assignment, never a corrupt mesh; the tie-breaks are total anyway,
296 * for run-to-run reproducibility.
297 */

314typedef struct

316 double w; ///< Weight: the body's global owned-triangle count, floored at one.

320/**
321 * @brief Orders bodies heaviest first, ties by ascending global id.
322 *
323 * Heaviest first because a large body placed late finds only fragments of the
324 * grants left; the id tie-break makes the order TOTAL, so the result does not
325 * depend on qsort's (unspecified) stability and one rank's assignment repeats
326 * exactly from run to run.
327 *
328 * @param a First element.
329 * @param b Second element.
330 * @return Negative, zero or positive in the usual qsort sense.
331 */
332static int by_weight_desc(const void *a, const void *b)
333{
334 const homed_body *const x = (const homed_body *) a;
335 const homed_body *const y = (const homed_body *) b;
336
337 if (x->w > y->w) return -1;
338 if (x->w < y->w) return 1;
339 return (x->gid < y->gid) ? -1 : ((x->gid > y->gid) ? 1 : 0);
340}

379 *
380 * Imbalance reached against a slack of 1.25 (`examples/remesh_regime_sweep

417/**
418 * @brief Is neighbour slot @p a preferred to slot @p b at equal usefulness?
419 *
420 * ══ A TIE-BREAK THAT IS THE SAME ON EVERY RANK STARVES THE SAME RANK ═══════
421 *
422 * The obvious tie-break -- lowest rank id -- is deterministic, total, and wrong
423 * here, because it is the SAME ORDERING ON EVERY SENDER. Four overloaded ranks
424 * looking at four equally empty neighbours all pick the same one, then the same
425 * second one, and the fourth is never chosen by anybody: measured as 1 of 8
426 * ranks left idle on a spread fixture that has thirty-two bodies for eight
427 * ranks. Nothing is wrong with the mesh and the imbalance still reports 1.25 --
428 * this is a tie-break, not a correctness rule -- but a rank was left with
429 * nothing to do for no reason.
430 *
431 * The fix has to make different senders prefer different targets, and the
432 * MEANING is already available: the cell OFFSET of a neighbour is relative to
433 * the rank doing the looking, so a nearer destination is both a different choice
434 * on every rank and a genuinely cheaper migration -- redistribution's stencil is
435 * a box, and a body sent one cell stays inside more senders' halos than one sent
436 * across a corner. Manhattan distance first, rank id second so the order stays
437 * total.
438 *
439 * @param dom The MPI/physical world.
440 * @param a A neighbour slot.
441 * @param b A neighbour slot.
442 * @return #MFTL_TRUE when @p a should be tried before @p b.
443 * @rank_local
444 */
445static bool nearer_neighbour(const MFTL_Domain *dom, int a, int b)
446{
447 int da = 0, db = 0;
448 for (int k = 0; k < 3; ++k)
449 {
450 da += (dom->neighbor_offset[a][k] < 0) ? -dom->neighbor_offset[a][k] : dom->neighbor_offset[a][k];
451 db += (dom->neighbor_offset[b][k] < 0) ? -dom->neighbor_offset[b][k] : dom->neighbor_offset[b][k];
452 }
453 if (da != db) return da < db;
454 return dom->neighbors[a] < dom->neighbors[b];
455}

457/**
458 * @brief Orders the neighbour slots by DESCENDING @p key, ties by nearness.
459 *
460 * An insertion sort over at most 26 entries, which is the whole point: the
461 * balancer's per-rank work must not grow with `nproc`, and an ordering of the
462 * neighbourhood is the largest thing it is allowed to sort. Slot order itself is
463 * never used as a tie-break -- it is whatever MPI reported for this rank, and
464 * ordering on it would make the assignment depend on that.
465 *
466 * @param dom The MPI/physical world.
467 * @param key `[d]` the quantity to order by, per neighbour slot.
468 * @param d Neighbours, at most 26.
469 * @param[out] out `[d]` slot indices, largest key first.
470 * @return How many were written, i.e. @p d.
471 * @rank_local
472 */
473static int order_neighbours(const MFTL_Domain *dom, const double *key, int d, int out[26])
474{
475 for (int i = 0; i < d; ++i)
476 {
477 int j = i - 1;
478 const int v = i;
479 while (j >= 0 && (key[out[j]] < key[v] || (key[out[j]] == key[v] && nearer_neighbour(dom, v, out[j]))))
480 {
481 out[j + 1] = out[j];
482 j--;
483 }
484 out[j + 1] = v;
485 }
486 return d;
487}

489/**
490 * @brief Every body this rank is HOME to, gathered from its lowest holders.
491 *
492 * One neighbourhood gather of three doubles per body this rank SPEAKS FOR --
493 * `lowest_holder[o] == rank`, so each body is published exactly once in the job
494 * however many ranks hold it.
495 *
496 * @param ens The ensemble.
497 * @param m The local measurement; row `o` describes local slot `o`.
498 * @param nh The resolved neighbourhood.
499 * @param[out] n_out Bodies homed here.
500 * @return A newly allocated `[*n_out]` array, sorted heaviest-first. The caller
501 * frees it. Never #MFTL_NULL.
502 * @collective One MFTLRemeshNeighborGather().
503 */
504static homed_body *gather_homed_bodies(MFTL_Ensemble *ens, const MFTL_RemeshMeasurement *m, const nbhood *nh, int *n_out)
505{
506 const MFTL_Domain *const dom = &ens->domain;
507
508 /* ── What this rank publishes ────────────────────────────────────────────
509 *
510 * `(gid, home, weight)`. The gid is what a holder matches the verdict against,
511 * `home` is the address, and the weight is what home balances on -- it cannot
512 * be recomputed there, because home need hold no entity of the body. */
513 int n_send = 0;
514 double *const send = (double *) MFTLMallocOrDie(3 * (size_t) MFTL_MAX(m->n_local, 1) * sizeof(double), "dispersed home publish");
515
516 for (int o = 0; o < m->n_local; ++o)
517 {
518 if ((int) m->lowest_holder[o] != dom->rank) continue;
519 send[n_send + 0] = (double) ens->local_objects[o]->global_id;
520 send[n_send + 1] = (double) MFTLGetOwnerProc(dom, &m->centroid[3 * o]);
521 send[n_send + 2] = balance_weight(m->weight[o]);
522 n_send += 3;
523 }
524
525 int *const rcount = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(nh->d, 1) * sizeof(int), "dispersed home counts");
526 int *const rdispl = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(nh->d, 1) * sizeof(int), "dispersed home displacements");
527
528 g_balancer_payload += n_send;
529
530 int n_recv = 0;
531 double *const recv = MFTLRemeshNeighborGather(dom, send, n_send, rcount, rdispl, &n_recv);
532
533 /* Capacity: everything this rank published plus everything it received could
534 * in principle be homed here. Bounded by `local + degree * their locals`, i.e.
535 * by what the neighbourhood holds and never by the ensemble. */
536 const int cap = n_send / 3 + n_recv / 3;
537 homed_body *const mine = (homed_body *) MFTLMallocOrDie((size_t) MFTL_MAX(cap, 1) * sizeof(homed_body), "dispersed homed bodies");
538 int n = 0;
539
540 for (int i = 0; i < n_send; i += 3)
541 {
542 if ((int) send[i + 1] != dom->rank) continue;
543 mine[n].gid = (int) send[i];
544 mine[n].w = send[i + 2];
545 n++;
546 }
547
548 for (int i = 0; i < nh->d; ++i)
549 {
550 /* THE SELF-LOOP. On a grid with one cut on a periodic axis this rank is its
551 * own graph neighbour, so its block comes back to it -- and it was already
552 * folded in above. Counting it twice doubles the load of every body homed
553 * here, which no MIN or MAX would notice and every SUM would carry. */
554 if (i == nh->self) continue;
555
556 for (int j = rdispl[i]; j + 2 < rdispl[i] + rcount[i]; j += 3)
557 {
558 if ((int) recv[j + 1] != dom->rank) continue;
559 mine[n].gid = (int) recv[j];
560 mine[n].w = recv[j + 2];
561 n++;
562 }
563 }
564
565 free(recv);
566 free(rdispl);
567 free(rcount);
568 free(send);
569
570 qsort(mine, (size_t) n, sizeof(homed_body), by_weight_desc);
571
572 *n_out = n;
573 return mine;
574}

576/**
577 * @brief Chooses a destination rank for every body this rank holds.
578 *
579 * Six steps, and every payload is bounded by the neighbourhood: publish who is
580 * home to what, agree two scalars, advertise room, ask for room, grant room,
581 * publish the verdicts. See the section banner for the argument.
582 *
583 * @param ens The ensemble.
584 * @param params Supplies `dispersed_balance_slack`; `<= 0` reports the
585 * balance without changing anything.
586 * @param m The local measurement, describing the CURRENT local list.
587 * @param veto Non-zero anywhere in the job means the reachability
588 * precondition failed; balancing is skipped.
589 * @param[out] dest `[m->n_local]` destination rank per LOCAL SLOT.
590 * @param[out] st Receives `n_bodies_reassigned`, `n_ranks_idle` and
591 * `load_imbalance`.
592 * @collective Two neighbourhood gathers (the home publish and the verdict
593 * publish), one neighbourhood allgather of a single double, three
594 * directed exchanges per balancing round, and four scalar
595 * `comm_world` reductions. Nothing is proportional to
596 * `n_bodies_global` or to `nproc`.
597 */
598static void assign_destinations(MFTL_Ensemble *ens, const MFTL_RemeshParameters *params, const MFTL_RemeshMeasurement *m, int veto, int *dest,
599 MFTL_RemeshStats *st)
600{
601 const MFTL_Domain *const dom = &ens->domain;
602
603 nbhood nh;
604 nh.d = dom->num_neighbors;
605 nh.self = MFTLDomainNeighborIndex(dom, dom->rank);
606
607 /* The unbalanced answer, and the one every body keeps unless its home rank
608 * says otherwise. Every holder computes it from its own reduced centroid, so
609 * they agree without being told. */
610 for (int o = 0; o < m->n_local; ++o) dest[o] = MFTLGetOwnerProc(dom, &m->centroid[3 * o]);
611
612 /* ── 1. Who is home to what ──────────────────────────────────────────── */
613 int n_mine = 0;
614 homed_body *const mine = gather_homed_bodies(ens, m, &nh, &n_mine);
615
616 double load = 0.0;
617 double heaviest = 0.0;
618 for (int i = 0; i < n_mine; ++i)
619 {
620 load += mine[i].w;
621 if (mine[i].w > heaviest) heaviest = mine[i].w;
622 }
623
624 /* ── 2. The two scalars a cap needs: `total` for the mean and `heaviest` for
625 * the floor -- no per-rank load table. */
626 double total = load;
627 MFTLRemeshAgreeDouble(dom, &total, 1);
628 MFTLRemeshAgreeDoubleMax(dom, &heaviest, 1);
629
630 const double slack = params->dispersed_balance_slack;
631 /* Every term is REPLICATED -- the slack is a checked parameter, the rank count
632 * is the world and the veto came out of a reduction -- so every rank enters
633 * and leaves the collectives below together. `nh.d` deliberately does NOT
634 * appear: it is per rank, and gating a neighbourhood collective on it is a
635 * deadlock on any grid where one rank is isolated. (None is, above one rank:
636 * `nproc > 1` puts at least two cells on some axis and the +/-1 offset along
637 * it is in range, so the degree is at least one.) */
638 const bool balancing = (slack > 0.0 && dom->nproc > 1 && veto == 0);
639
640 /* `MAX(mean * slack, heaviest)` -- the second term is not defensive padding. A
641 * body is indivisible here, so a cap below the heaviest body asks for an
642 * assignment that does not exist and every rank holding that body would try,
643 * and fail, to shed it every pass. */
644 const double cap = MFTL_MAX(total / (double) dom->nproc * slack, heaviest);
645
646 int reassigned = 0;
647 double shipped_out = 0.0;
648 double received_in = 0.0;
649
650 /* Every buffer of the negotiation is sized by the graph degree (at most 26),
651 * in one allocation. */
652 double *const nb_buf = (double *) MFTLCallocOrDie(7 * (size_t) MFTL_MAX(nh.d, 1), sizeof(double), "dispersed neighbourhood scalars");
653 double *const spare_nb = nb_buf; // [d] room each neighbour last advertised
654 double *const demand = nb_buf + nh.d; // [d] what this rank asks of each
655 double *const dem_in = nb_buf + 2 * nh.d; // [d] what each neighbour asks of this rank
656 double *const reply = nb_buf + 3 * nh.d; // [2d] (granted, room still unpromised) to each asker
657 double *const got = nb_buf + 5 * nh.d; // [2d] the same, received
658
659 verdict *moved = MFTL_NULL;
660 bool *placed = MFTL_NULL;
661
662 if (balancing)
663 {
664 moved = (verdict *) MFTLMallocOrDie((size_t) MFTL_MAX(n_mine, 1) * sizeof(verdict), "dispersed verdicts");
665 placed = (bool *) MFTLCallocOrDie((size_t) MFTL_MAX(n_mine, 1), sizeof(bool), "dispersed placed flags");
666
667 /* ── 3. Advertise room ────────────────────────────────────────────────
668 *
669 * Room and not load, because what a sender must respect is the TARGET's cap
670 * and the cap is the target's own arithmetic. Once, before the rounds; from
671 * then on each grant carries the granter's remaining room back with it, so
672 * no further advertisement is needed. */
673 double left = MFTL_MAX(cap - load, 0.0);
674 /* The whole fixed cost of the negotiation, counted once so that
675 * MFTLRemeshBalancerPayload() is the honest total and not just its
676 * body-proportional half: one advertisement, then four directed doubles per
677 * neighbour per round (ask, two of grant-and-room, ship). */
678 g_balancer_payload += 1 + 4 * nh.d * MFTL_DISPERSED_BALANCE_ROUNDS;
679 MFTLRemeshNeighborAllgather(dom, &left, 1, spare_nb);
680 if (nh.self != MFTL_ERROR) spare_nb[nh.self] = 0.0; // shipping to oneself moves nothing
681
682 double cur = load;
683
684 for (int round = 0; round < MFTL_DISPERSED_BALANCE_ROUNDS; ++round)
685 {
686 /* ── 4. Ask FEW neighbours for a lot, and ask again next round ─────────
687 *
688 * Bodies are indivisible: splitting the excess over all neighbours in
689 * proportion to their room returns grants smaller than one body, nothing
690 * is placeable, and the pass reports an unchanged imbalance with
691 * `n_bodies_reassigned == 0` -- indistinguishable from an ensemble that
692 * needed nothing (measured: 108 bodies over 27 ranks stuck at 2.00). So
693 * the ask is CONCENTRATED: neighbours emptiest first, each asked for all
694 * it advertises, until the budget is covered.
695 *
696 * Concentrating makes every overloaded rank ask the same emptiest
697 * neighbour, which can serve one of them; hence ROUNDS -- served targets
698 * advertise less next time and the others move on.
699 *
700 * The budget is `cur - cap` plus the heaviest unplaced body, because only
701 * whole bodies leave and asking for exactly the excess is routinely one
702 * body short. A target cannot be pushed over its cap: a grant is bounded
703 * by its own room. */
704 for (int i = 0; i < nh.d; ++i) demand[i] = 0.0;
705
706 if (cur > cap)
707 {
708 double biggest = 0.0;
709 for (int i = 0; i < n_mine; ++i)
710 {
711 if (!placed[i])
712 {
713 biggest = mine[i].w; // sorted heaviest first, so the first unplaced is the largest
714 break;
715 }
716 }
717
718 int ord[26];
719 order_neighbours(dom, spare_nb, nh.d, ord);
720
721 int n_open = 0;
722 for (int i = 0; i < nh.d; ++i)
723 {
724 if (spare_nb[i] > 0.0) n_open++;
725 }
726
727 const double budget = (cur - cap) + biggest;
728
729 /* AS MANY NEIGHBOURS AS THE BUDGET CAN GIVE A WHOLE BODY EACH, and no
730 * more. Fewer and ranks that could have been given work stay idle;
731 * more and every chunk is a fraction of a body and nothing is placeable.
732 * `budget / biggest` is exactly that count, clamped to the neighbours
733 * that have room and to one. */
734 int k = (biggest > 0.0) ? (int) (budget / biggest) : n_open;
735 if (k > n_open) k = n_open;
736 if (k < 1) k = 1;
737
738 const double share = budget / (double) k;
739 for (int j = 0; j < k; ++j)
740 {
741 const int i = ord[j];
742 if (spare_nb[i] <= 0.0) break; // ordered by descending room, so the rest are full too
743 demand[i] = MFTL_MIN(spare_nb[i], share);
744 }
745 }
746
747 MFTLRemeshNeighborAlltoall(dom, demand, 1, dem_in);
748 if (nh.self != MFTL_ERROR) dem_in[nh.self] = 0.0;
749
750 /* ── 5. Grant, neediest first, and never more room than exists ───────
751 *
752 * `SUM_s granted_sr <= room_r` by construction, so NO RANK CAN BE PUSHED
753 * PAST ITS CAP however many neighbours ship at once -- the property a
754 * greedy over stale loads does not have, and the reason this is a
755 * negotiation rather than a race. Serving in priority order rather than
756 * proportionally is the granting half of the indivisibility argument
757 * above: a room divided among all askers is a room divided into pieces
758 * none of them can use.
759 *
760 * The second field is what the next round is ordered by, which is why no
761 * second advertisement is needed. It is the room left after this round's
762 * PROMISES, so it under-states whenever a grant goes unused -- and `left`
763 * itself is corrected below from what was actually shipped, so the
764 * under-statement costs at most one round of a smaller ask. */
765 {
766 int ord[26];
767 order_neighbours(dom, dem_in, nh.d, ord);
768
769 double unpromised = left;
770 for (int j = 0; j < nh.d; ++j)
771 {
772 const int i = ord[j];
773 const double give = MFTL_MIN(dem_in[i], unpromised);
774 reply[2 * i] = (give > 0.0) ? give : 0.0;
775 unpromised -= reply[2 * i];
776 }
777 for (int i = 0; i < nh.d; ++i) reply[2 * i + 1] = unpromised;
778 }
779
780 MFTLRemeshNeighborAlltoall(dom, reply, 2, got);
781 if (nh.self != MFTL_ERROR)
782 {
783 got[2 * nh.self] = 0.0;
784 got[2 * nh.self + 1] = 0.0;
785 }
786
787 /* ── 6. Place the bodies, heaviest first ─────────────────────────────
788 *
789 * Into the neighbour with the most room left, which keeps the largest body
790 * still to come placeable, and among equals into the NEAREST -- see
791 * nearer_neighbour() for why a tie-break that is identical on every sender
792 * leaves a rank idle. */
793 for (int i = 0; i < nh.d; ++i) demand[i] = 0.0; // reused: weight shipped to each neighbour
794
795 for (int i = 0; i < n_mine && cur > cap; ++i)
796 {
797 if (placed[i]) continue;
798 const double w = mine[i].w;
799
800 int best = MFTL_ERROR;
801 for (int j = 0; j < nh.d; ++j)
802 {
803 if (got[2 * j] < w) continue;
804 if (best == MFTL_ERROR || got[2 * j] > got[2 * best] || (got[2 * j] == got[2 * best] && nearer_neighbour(dom, j, best))) best = j;
805 }
806 if (best == MFTL_ERROR) continue; // no neighbour granted room for one this large
807
808 moved[reassigned].gid = mine[i].gid;
809 moved[reassigned].dest = dom->neighbors[best];
810 moved[reassigned].w = w;
811 reassigned++;
812 placed[i] = MFTL_TRUE;
813
814 got[2 * best] -= w;
815 demand[best] += w;
816 cur -= w;
817 shipped_out += w;
818 }
819
820 /* ── 7. Say what was actually shipped ────────────────────────────────
821 *
822 * A promise is not a delivery: a grant of two bodies' worth that fits only
823 * one leaves room the granter must get back, or the rounds would leak
824 * capacity and the imbalance report would be wrong about the inflow. One
825 * more directed double closes both. */
826 MFTLRemeshNeighborAlltoall(dom, demand, 1, dem_in);
827 if (nh.self != MFTL_ERROR) dem_in[nh.self] = 0.0;
828
829 double inflow = 0.0;
830 for (int i = 0; i < nh.d; ++i) inflow += dem_in[i];
831 received_in += inflow;
832 left = MFTL_MAX(left - inflow, 0.0);
833
834 for (int i = 0; i < nh.d; ++i) spare_nb[i] = got[2 * i + 1];
835 if (nh.self != MFTL_ERROR) spare_nb[nh.self] = 0.0;
836 }
837 }
838
839 /* ── 7. Publish the verdicts, and read everyone else's ───────────────────
840 *
841 * `(gid, dest, weight)`, three doubles per body that actually moved -- which
842 * is a small fraction of a balanced ensemble and exactly zero when the pass
843 * changed nothing. Only the HOLDERS need it and they are all one hop from
844 * home, which is the precondition this driver checked before it started. */
845 if (balancing)
846 {
847 double *const send = (double *) MFTLMallocOrDie(3 * (size_t) MFTL_MAX(reassigned, 1) * sizeof(double), "dispersed verdict publish");
848 for (int i = 0; i < reassigned; ++i)
849 {
850 send[3 * i + 0] = (double) moved[i].gid;
851 send[3 * i + 1] = (double) moved[i].dest;
852 send[3 * i + 2] = moved[i].w;
853 }
854
855 int *const rcount = (int *) MFTLMallocOrDie((size_t) nh.d * sizeof(int), "dispersed verdict counts");
856 int *const rdispl = (int *) MFTLMallocOrDie((size_t) nh.d * sizeof(int), "dispersed verdict displacements");
857
858 g_balancer_payload += 3 * reassigned;
859
860 int n_recv = 0;
861 double *const recv = MFTLRemeshNeighborGather(dom, send, 3 * reassigned, rcount, rdispl, &n_recv);
862
863 /* Everything this rank must apply: its own verdicts plus its neighbours'. A
864 * body appears at most once -- only home publishes it -- so the sorted list
865 * is a strictly increasing key and a binary search per local body resolves
866 * it in `O(local log moved)`. */
867 const int cap_all = reassigned + n_recv / 3;
868 verdict *const all = (verdict *) MFTLMallocOrDie((size_t) MFTL_MAX(cap_all, 1) * sizeof(verdict), "dispersed verdict table");
869 int n_all = 0;
870
871 for (int i = 0; i < reassigned; ++i) all[n_all++] = moved[i];
872
873 for (int i = 0; i < nh.d; ++i)
874 {
875 if (i == nh.self) continue; // own verdicts, already in
876 for (int j = rdispl[i]; j + 2 < rdispl[i] + rcount[i]; j += 3)
877 {
878 all[n_all].gid = (int) recv[j];
879 all[n_all].dest = (int) recv[j + 1];
880 all[n_all].w = recv[j + 2];
881 n_all++;
882 }
883 }
884
885 qsort(all, (size_t) n_all, sizeof(verdict), by_gid_asc);
886
887 for (int o = 0; o < m->n_local; ++o)
888 {
889 const int g = ens->local_objects[o]->global_id;
890 int lo = 0, hi = n_all - 1;
891 while (lo <= hi)
892 {
893 const int mid = lo + (hi - lo) / 2;
894 if (all[mid].gid == g)
895 {
896 dest[o] = all[mid].dest;
897 break;
898 }
899 if (all[mid].gid < g)
900 lo = mid + 1;
901 else
902 hi = mid - 1;
903 }
904 }
905
906 free(all);
907 free(recv);
908 free(rdispl);
909 free(rcount);
910 free(send);
911 }
912
913 /* ── 8. The report, in four scalars ──────────────────────────────────────
914 *
915 * The load a rank ends with is what it was home to, less what it shed, plus
916 * what it was given -- all three known locally by now. `total` is conserved by
917 * construction, so only the peak and the idle count have to be reduced. */
918 const double post = load - shipped_out + received_in;
919
920 double peak = post;
921 MFTLRemeshAgreeDoubleMax(dom, &peak, 1);
922
923 int counts[2] = {(post <= 0.0) ? 1 : 0, reassigned};
924 MFTLRemeshSumInts(dom, counts, 2);
925
926 st->n_ranks_idle = counts[0];
927 st->n_bodies_reassigned = counts[1];
928 st->load_imbalance = (total > 0.0) ? (peak * (double) dom->nproc / total) : 1.0;
929
930 free(placed);
931 free(moved);
932 free(nb_buf);
933 free(mine);
934}

948
949 /* This form owes its own measurement. MFTLRemeshEnsemble's AUTO path does not
950 * come through here -- it measured once for the strategy choice and hands that
951 * measurement straight to the Measured form below. */

953 MFTLRemeshMeasurementInit(&m, ens);
954
955 if (ens->n_bodies_global >= 1)
956 {

958

960 * it (see MFTLRemeshSpanningPass()). */

963 MFTLRemeshMeasurementFree(&m);

969 }
970 return MFTL_ERROR;
971 }
972 MFTLRemeshMeasureBodies(ens, m.centroid, m.radius, m.weight, &m.r_max, m.lowest_holder);

976 MFTLRemeshMeasurementFree(&m);
977 return rc;

980int MFTLRemeshEnsembleDispersedMeasured(MFTL_Ensemble *ens, const MFTL_RemeshParameters *params, MFTL_RemeshStats *stats, const MFTL_RemeshMeasurement *m)
981{
982 MFTL_ASSERT(ens != MFTL_NULL && params != MFTL_NULL && m != MFTL_NULL);
983
984 MFTL_Domain *const dom = &ens->domain;
985
986 /* The GLOBAL body count for the guard: everything below is collective, so a
987 * rank holding nothing must still reach it.
988 *
989 * No cached local count: migrations destroy and create local objects, so
990 * `ens->n_local_objects` and `ens->local_objects` change under this function
991 * and every loop re-reads them. */
992 const int n = ens->n_bodies_global;
993
994 MFTL_RemeshStats sink;
995 MFTL_RemeshStats *const st = stats ? stats : &sink;
996 memset(st, 0, sizeof(*st));
997 st->strategy = MFTL_REMESH_PARALLEL_DISPERSED;
998 st->n_coarse_groups = 1; // no bands, so nothing to coarsen; 1 rather than 0 so the field always reads as a group size
999
1000 if (n == 0) return MFTL_SUCCESS;
1001
1002 int status = MFTL_SUCCESS;
1003
1004 /* Free-running per-rank accumulation, reduced once on the way out. Same six
1005 * sections as the spanning driver, so the two are directly comparable -- which
1006 * is the whole point of measuring them at all. */
1007 const double t0_call = MFTLRemeshWallTime();
1008 double t_mark;
1009
1010 MFTL_RemeshMigrator mig;
1011 MFTLRemeshMigratorInit(&mig);
1012
1013 /* ── 1. Where is each body, and how big is it? ───────────────────────────
1014 *
1015 * Handed in by the caller, who already measured to choose the strategy.
1016 * The radius is the ADMISSION TEST (step 2); the rows describe this rank's
1017 * bodies, which is all the driver and the balancer need. */
1018 MFTL_ASSERT(m->n_local == ens->n_local_objects &&
1019 "MFTLRemeshEnsembleDispersedMeasured: the measurement describes a different local list -- something migrated between the two calls");
1020
1021 const int n_local = m->n_local;
1022 int *const dest = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(n_local, 1) * sizeof(int), "dispersed destinations");
1023
1024 /* ── 2. Is this strategy applicable at all? ──────────────────────────────
1025 *
1026 * ── Why the test is on the RADIUS and not on the result of the migration ──
1027 *
1028 * The obvious check -- migrate, then see whether every body came out with no
1029 * ghosts -- is vacuous, and quietly so. An MFTL_OwnerMap names one destination
1030 * for every entity of a body, so redistribution puts the body wholly on that
1031 * rank WHATEVER ITS SIZE: a jet spanning the whole domain would arrive
1032 * complete on one rank, report itself perfectly local, and remesh correctly.
1033 * The check would pass and the run would have serialised the largest object in
1034 * the simulation onto a single rank's memory.
1035 *
1036 * So the thing that has to be tested is the precondition itself, and it is
1037 * geometric: a body wider than a rank box has no rank that can hold it as a
1038 * matter of design, not of outcome. Two consequences, both real:
1039 *
1040 * MEMORY one rank holds the entire body while the others hold none of it.
1041 * MESSAGES the centering stops being a 26-neighbour exchange, because a
1042 * triangle further than one box from the centroid has a
1043 * destination outside the stencil, and redistribution correctly
1044 * falls back to the GLOBAL communicator -- an all-to-all where the
1045 * design promised a neighbourhood.
1046 *
1047 * Taking the decision here also means NOTHING HAS MOVED YET when the fallback
1048 * fires. That matters: the spanning driver's per-phase id scan assumes every
1049 * holder entered with the same id counters, which a half-migrated,
1050 * half-remeshed ensemble would not have.
1051 *
1052 * The threshold is MFTLRemeshChooseStrategy's, so a caller who let AUTO decide
1053 * can never reach this branch -- only one who forced the strategy, or whose
1054 * bodies grew since they measured them. */
1055 const double frac = (params->dispersed_radius_fraction > 0.0) ? params->dispersed_radius_fraction : 0.4;
1056
1057 /* THE SAME YARDSTICK THE CHOOSER USED, and it has to be derived the same way
1058 * rather than passed in: `r_max` is the largest radius, so the box it implies
1059 * is a property of the ensemble and every rank recomputes it identically from
1060 * data it already holds. A chooser and a driver disagreeing here would elect
1061 * DISPERSED and then reject it, costing a measurement and a fall back to
1062 * spanning -- visible only as `n_bodies_not_centred`. */
1063 const double h_min = dispersed_box_min(dom, params, m->r_max, n);
1064
1065 /* Counted over THIS RANK'S bodies, once per body rather than once per holder:
1066 * every holder reduced to the same radius, so an unqualified sum reports a
1067 * body held by four ranks four times. `lowest_holder` designates one. */
1068 /* TWO VERDICTS ON ONE MESSAGE, because both are gates and both are sums of a
1069 * local count. `gate[0]` decides whether this driver runs at all; `gate[1]`
1070 * decides whether it may BALANCE, and it is the reachability precondition of
1071 * the neighbourhood balancer -- see the section banner above. Folding the
1072 * second in here is what makes the balancer's precondition free: it is one
1073 * more int on a reduction the pass was taking anyway. */
1074 int gate[2] = {0, 0};
1075 if (dom->nproc > 1)
1076 {
1077 for (int o = 0; o < n_local; ++o)
1078 {
1079 if (h_min > 0.0 && m->radius[o] > frac * h_min && (int) m->lowest_holder[o] == dom->rank) gate[0]++;
1080
1081 /* Counted once per HOLDER that cannot reach the body's home rank, not once
1082 * per body: that is what makes it a purely local count, and the number is
1083 * a diagnostic rather than a body count. Zero is the only value that
1084 * changes what happens. */
1085 if (neighbor_slot(dom, MFTLGetOwnerProc(dom, &m->centroid[3 * o])) == MFTL_ERROR) gate[1]++;
1086 }
1087 }
1088 MFTLRemeshSumInts(dom, gate, 2);
1089 const int too_large = gate[0];
1090 st->n_bodies_not_centred = gate[0];
1091 st->n_bodies_unreachable = gate[1];
1092
1093 if (too_large > 0)
1094 {
1095 free(dest);
1096 MFTLRemeshMigratorFree(&mig);
1097
1098 if (dom->rank == 0)
1099 {
1100 fprintf(stderr,
1101 "[MFTL] dispersed remeshing: %d of %d bodies are wider than %.2f * h_min (%.4g) and cannot be "
1102 "centred on one rank. Falling back to the spanning driver for the whole ensemble.\n",
1103 too_large, n, frac, frac * h_min);
1104 }
1105
1106 /* The spanning driver zeroes and refills the stats, so the one field that
1107 * describes THIS decision has to be carried across by hand. Nothing else
1108 * should survive: the pass being reported is the one that ran. */
1109 const int rc = MFTLRemeshSpanningPass(ens, params, st, MFTL_NULL);
1110 st->n_bodies_not_centred = too_large;
1111 return rc;
1112 }
1113
1114 /* ── Centroid ownership, then balance ────────────────────────────────────
1115 *
1116 * A body goes to the rank owning its centroid unless that rank is over the
1117 * cap and a neighbour granted room; decided on ONE rank, delivered one hop.
1118 * With the balancer off the same call only reports the imbalance. */
1119 assign_destinations(ens, params, m, gate[1], dest, st);
1120
1121 /* ── 3. Move each body whole ─────────────────────────────────────────────── */
1122 t_mark = MFTLRemeshWallTime();
1123 MFTL_RemeshMigrationReport mig_report;
1124 if (MFTLRemeshMigrateToRanks(&mig, ens, dest, &mig_report) != MFTL_SUCCESS) status = MFTL_ERROR;
1125 st->t_migrate += MFTLRemeshWallTime() - t_mark;
1126 st->n_entities_migrated += mig_report.n_moved;
1127 st->n_migrations++;
1128
1129 free(dest);
1130 MFTLRemeshMigratorFree(&mig);
1131
1132 /* One destination per body means no rank can end up holding part of one, so
1133 * the serial kernel's own precondition is met by construction rather than by
1134 * inspection. Asserted because it is the hinge of the whole strategy and it is
1135 * free in Release. */
1136#ifndef NDEBUG
1137 for (int o = 0; o < ens->n_local_objects; ++o)
1138 {
1139 const MFTL_Object *const obj = ens->local_objects[o];
1140 MFTL_ASSERT(obj->verts.n_local == obj->verts.n && obj->tris.n_local == obj->tris.n &&
1141 "dispersed remeshing: a body kept ghosts after centering -- the owner map did not name one rank");
1142 }
1143#endif
1144
1145 /* ── 4. The untouched serial kernel, on whole bodies ─────────────────────
1146 *
1147 * MFTLRemeshObject, not MFTLRemeshIterate: a body with no ghosts is exactly
1148 * what the serial entry point is specified for, down to its assertion. A rank
1149 * holding nothing of a body enters with tris.n == 0 and returns immediately,
1150 * which is why this loop needs no ownership test of its own. */
1151 t_mark = MFTLRemeshWallTime();
1152 int local_ops = 0;
1153 for (int o = 0; o < ens->n_local_objects; ++o)
1154 {
1155 MFTL_Object *const obj = ens->local_objects[o];
1156 if (obj->tris.n == 0) continue;
1157
1158 MFTL_RemeshStats one;
1159 memset(&one, 0, sizeof(one));
1160
1161 if (MFTLRemeshObject(obj, params, &one) != MFTL_SUCCESS) status = MFTL_ERROR;
1162 remesh_stats_accumulate(st, &one);
1163 local_ops += one.n_split + one.n_collapse + one.n_flip;
1164
1165 /* Unconditionally, whatever `renumber_global_ids` says. In serial that flag
1166 * is a caller preference; here density is a REQUIREMENT of the migration
1167 * that follows, because redistribution sizes its arrival table from
1168 * next_*_gid and a pass that leaves holes grows that table without bound
1169 * over a run. Idempotent when MFTLRemeshObject already did it. */
1170 MFTLRemeshRenumberGlobalIds(obj);
1171 }
1172 st->t_sweep += MFTLRemeshWallTime() - t_mark;
1173
1174 /* ── 5. Make the id counters agree ───────────────────────────────────────
1175 *
1176 * Exactly one rank holds entities of a body, so its counter is the answer
1177 * and every other holder contributes zero: a per-body MAX over the holders
1178 * picks it without anyone knowing who the owner is. Contributing the stale
1179 * pre-pass value would be wrong for a body that shrank. */
1180 /* Re-read, not `n_local`: the migration in step 3 changed which bodies this
1181 * rank holds, which is the whole point of it. */
1182 const int n_held = ens->n_local_objects;
1183 MFTL_BodyReduceOp cops[2] = {MFTL_BODY_REDUCE_MAX, MFTL_BODY_REDUCE_MAX};
1184 double *const counters = (double *) MFTLCallocOrDie((size_t) MFTL_MAX(2 * n_held, 1), sizeof(double), "dispersed id counters");
1185
1186 for (int o = 0; o < n_held; ++o)
1187 {
1188 const MFTL_Object *const obj = ens->local_objects[o];
1189 counters[2 * o + 0] = (obj->verts.n > 0) ? (double) obj->next_vertex_gid : 0.0;
1190 counters[2 * o + 1] = (obj->tris.n > 0) ? (double) obj->next_triangle_gid : 0.0;
1191 }
1192
1193 /* Per body, over its holders, indexed by LOCAL SLOT. Normally one holder per
1194 * body at this point, but a rank can still hold an EMPTY object for a body
1195 * and size its next arrival table from `next_*_gid`; such an empty holder
1196 * vetoes the neighbourhood route, and the reduction takes the global one. */
1197 t_mark = MFTLRemeshWallTime();
1198 MFTLEnsembleReduceBodies(ens, counters, 2, cops, MFTL_BODY_REDUCE_SCOPE_AUTO);
1199
1200 /* One int of pure latency, and it cannot ride the per-body message: it is a
1201 * maximum over RANKS, not over the holders of any one body. */
1202 int busiest = local_ops;
1203 MFTLRemeshAgreeIntMax(dom, &busiest, 1);
1204 st->t_reduce += MFTLRemeshWallTime() - t_mark;
1205
1206 for (int o = 0; o < n_held; ++o)
1207 {
1208 MFTL_Object *const obj = ens->local_objects[o];
1209 obj->next_vertex_gid = (int) counters[2 * o + 0];
1210 obj->next_triangle_gid = (int) counters[2 * o + 1];
1211 obj->next_halfedge_gid = 3 * obj->next_triangle_gid;
1212 }
1213
1214 /* MAX rather than SUM, and it is the right reduction here: exactly one rank
1215 * holds any entity of a body, so no two ranks operate on the same body and the
1216 * maximum is the busiest rank's count. It is reported as such -- see the
1217 * locality note on MFTL_RemeshStats. */
1218 st->n_ops_busiest_rank = busiest;
1219 free(counters);
1220
1221 /* One pass over whole bodies IS the serial answer, so there is nothing left
1222 * for a second placement to reveal. Unlike the shift schedule, this driver has
1223 * no notion of an entity it was not allowed to touch. */
1224 st->n_phases = 1;
1225 st->converged = MFTL_TRUE;
1226 st->n_edges_hidden = 0;
1227
1228 /* ── 6. Back to the decomposition the flow solver expects ────────────────── */
1229 t_mark = MFTLRemeshWallTime();
1230 MFTLEnsembleComputeTriangleGeometry(ens);
1231 st->t_geometry += MFTLRemeshWallTime() - t_mark;
1232
1233 t_mark = MFTLRemeshWallTime();
1234 if (MFTLRedistribute(ens) != MFTL_SUCCESS) status = MFTL_ERROR;
1235 st->t_migrate += MFTLRemeshWallTime() - t_mark;
1236 st->n_migrations++;
1237
1238 MFTLEnsembleUpdateGlobalCounts(ens);
1239
1240 /* Redistribution leaves `tris.area/normal/center` unspecified by design, and
1241 * MFTLRemeshObject promises its caller current triangle geometry. This entry
1242 * point promises the same thing rather than being a different contract. */
1243 t_mark = MFTLRemeshWallTime();
1244 MFTLEnsembleComputeTriangleGeometry(ens);
1245 st->t_geometry += MFTLRemeshWallTime() - t_mark;
1246
1247 /* The maximum over ranks, and unconditional -- see the same block at the end
1248 * of MFTLRemeshEnsembleParallel. Here the imbalance it exposes is the POINT:
1249 * this strategy sends every body to one rank, so at N_bodies < nproc most
1250 * ranks contribute a `t_sweep` of zero and the max is one rank's whole cost. */
1251 st->t_total = MFTLRemeshWallTime() - t0_call;
1252 MFTLRemeshAgreeProfile(dom, st, MFTL_NULL, 0);
1253
1254 return status;
1255}
