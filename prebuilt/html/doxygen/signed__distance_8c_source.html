1/**
2 * @file signed_distance.c
3 * @author Christian Gorges
4 * @date 2026-09-25
5 * @ingroup mftl_geometry
6 *
7 * @brief Bounded rasterisation of a triangle mesh's signed distance onto a
8 * lattice window, signed by angle-weighted pseudo-normals.
9 *
10 * A SCATTER over triangles: each triangle visits only the nodes inside its box
11 * dilated by the cutoff and keeps, per node, the contribution of smallest
12 * magnitude (MFTLSdfMerge()). Cost \f$O(T\,(h+2r)^3/\Delta x^3)\f$ for triangle
13 * size h and cutoff r, independent of the rest of the mesh. The sign comes from
14 * the pseudo-normal of the closest FEATURE (face, edge or vertex) -- see
15 * MFTLRasterizeSignedDistance().
16 */
17

28void MFTLSdfGridInitWindow(MFTL_SdfGrid *grid, const double origin[3], const double spacing[3], const int n_global[3], const bool periodic[3], const int lo[3],
29 const int dims[3])
30{
31 MFTL_ASSERT(grid != MFTL_NULL);
32
33 size_t n = 1;
34 for (int k = 0; k < 3; ++k)
35 {
36 MFTL_ASSERT(spacing[k] > 0.0 && dims[k] >= 1 && "MFTLSdfGridInitWindow: empty axis or non-positive spacing");
37 grid->origin[k] = origin[k];
38 grid->spacing[k] = spacing[k];
39 grid->n_global[k] = n_global[k];
40 grid->periodic[k] = (periodic != MFTL_NULL) ? periodic[k] : MFTL_FALSE;
41 grid->lo[k] = lo[k];
42 grid->dims[k] = dims[k];
43 MFTL_ASSERT((!grid->periodic[k] || dims[k] <= n_global[k]) && "MFTLSdfGridInitWindow: a periodic window may not exceed one period");
44 n *= (size_t) dims[k];
45 }
46
47 grid->phi = (double *) MFTLMallocOrDie(n * sizeof(double), "SDF grid");
48 for (size_t i = 0; i < n; ++i) grid->phi[i] = MFTL_SDF_UNKNOWN;
49}

64/**
65 * @brief Unit normal of a triangle from its corners, or zero if it has no area.
66 *
67 * Recomputed rather than read from `tris.normal`, which a redistribution leaves
68 * unspecified.
69 *
70 * @param c The three corners, as MFTLGetTriangleCorners() returns them.
71 * @param[out] n The unit normal.
72 */
73static void face_normal(const double c[3][3], double n[3])
74{
75 const double e1[3] = {c[1][0] - c[0][0], c[1][1] - c[0][1], c[1][2] - c[0][2]};
76 const double e2[3] = {c[2][0] - c[0][0], c[2][1] - c[0][1], c[2][2] - c[0][2]};
77 n[0] = e1[1] * e2[2] - e1[2] * e2[1];
78 n[1] = e1[2] * e2[0] - e1[0] * e2[2];
79 n[2] = e1[0] * e2[1] - e1[1] * e2[0];
80 const double mag = sqrt(n[0] * n[0] + n[1] * n[1] + n[2] * n[2]);
81 const double inv = (mag > 0.0) ? 1.0 / mag : 0.0;
82 n[0] *= inv;
83 n[1] *= inv;
84 n[2] *= inv;
85}

87long MFTLRasterizeSignedDistance(MFTL_SdfGrid *grid, const MFTL_Object *obj, int n_tris, double cutoff)
88{
89 MFTL_ASSERT(grid != MFTL_NULL && obj != MFTL_NULL);
90 MFTL_ASSERT(cutoff > 0.0);
91 MFTL_ASSERT(n_tris <= obj->tris.n);
92 MFTL_ASSERT((n_tris == 0 || MFTLObjectHasTopology(obj)) && "MFTLRasterizeSignedDistance: stale tris.verts -- MFTLObjectBuildDerived first");
93
94 if (grid->phi == MFTL_NULL || n_tris <= 0) return 0;
95
96 const double cut_sq = cutoff * cutoff;
97 const int nx = grid->dims[0], ny = grid->dims[1];
98 long n_updates = 0;
99
100 for (int t = 0; t < n_tris; ++t)
101 {
102 double T[3][3];
103 MFTLGetTriangleCorners(obj, t, T);
104
105 /* ── The node box this triangle can reach ───────────────────────────── */
106 int g_lo[3], g_hi[3];
107 bool empty = MFTL_FALSE;
108 for (int k = 0; k < 3; ++k)
109 {
110 const double mn = fmin(T[0][k], fmin(T[1][k], T[2][k])) - cutoff;
111 const double mx = fmax(T[0][k], fmax(T[1][k], T[2][k])) + cutoff;
112 g_lo[k] = (int) ceil((mn - grid->origin[k]) / grid->spacing[k]);
113 g_hi[k] = (int) floor((mx - grid->origin[k]) / grid->spacing[k]);
114 if (g_hi[k] < g_lo[k]) empty = MFTL_TRUE;
115 }
116 if (empty) continue;
117
118 /* Pseudo-normals: the face normal always; each edge's lazily. Edge k of
119 * MFTLClosestPointOnTriangle() is half-edge 3t+k, and corner k is
120 * tris.verts[3t+k]. */
121 double n_face[3];
122 face_normal(T, n_face);
123 double n_edge[3][3];
124 bool have_edge[3] = {MFTL_FALSE, MFTL_FALSE, MFTL_FALSE};
125
126 for (int gz = g_lo[2]; gz <= g_hi[2]; ++gz)
127 {
128 const int lz = MFTLSdfGridLocal(grid, 2, gz);
129 if (lz < 0) continue;
130 const double z = MFTLSdfGridCoord(grid, 2, gz);
131
132 for (int gy = g_lo[1]; gy <= g_hi[1]; ++gy)
133 {
134 const int ly = MFTLSdfGridLocal(grid, 1, gy);
135 if (ly < 0) continue;
136 const double y = MFTLSdfGridCoord(grid, 1, gy);
137
138 for (int gx = g_lo[0]; gx <= g_hi[0]; ++gx)
139 {
140 const int lx = MFTLSdfGridLocal(grid, 0, gx);
141 if (lx < 0) continue;
142
143 const double x[3] = {MFTLSdfGridCoord(grid, 0, gx), y, z};
144 double q[3];
145 MFTL_TriangleFeature f;
146 const double d_sq = MFTLClosestPointOnTriangle(x, T, q, &f);
147 if (d_sq > cut_sq) continue;
148
149 double *const node = &grid->phi[((size_t) lz * ny + ly) * nx + lx];
150 const double d = sqrt(d_sq);
151 /* A node already holding something strictly nearer cannot change. */
152 if (d > fabs(*node)) continue;
153
154 const double *n = n_face;
155 if (f >= MFTL_TRI_FEATURE_EDGE0)
156 {
157 const int e = f - MFTL_TRI_FEATURE_EDGE0;
158 if (!have_edge[e])
159 {
160 const int tw = obj->tris.he[3 * t + e].twin_idx;
161 double nt[3] = {0.0, 0.0, 0.0};
162 if (tw != MFTL_ERROR)
163 {
164 double Tw[3][3];
165 MFTLGetTriangleCorners(obj, MFTLHeTriIdx(tw), Tw);
166 face_normal(Tw, nt);
167 }
168 for (int k = 0; k < 3; ++k) n_edge[e][k] = n_face[k] + nt[k];
169 have_edge[e] = MFTL_TRUE;
170 }
171 n = n_edge[e];
172 }
173 else if (f >= MFTL_TRI_FEATURE_VERTEX0)
174 {
175 n = &obj->verts.normal[3 * obj->tris.verts[3 * t + (f - MFTL_TRI_FEATURE_VERTEX0)]];
176 }
177
178 const double dot = (x[0] - q[0]) * n[0] + (x[1] - q[1]) * n[1] + (x[2] - q[2]) * n[2];
179 const double value = (dot < 0.0) ? -d : d;
180 const double merged = MFTLSdfMerge(*node, value);
181 if (merged != *node)
182 {
183 *node = merged;
184 n_updates++;
185 }
186 }
187 }
188 }
189 }
190 return n_updates;
191}
