1/**
2 * @file topo_comm.c
3 * @author Christian Gorges
4 * @date 2026-09-25
5 * @ingroup mftl_topology_change
6 *
7 * @brief EVERY MPI call of the topology-change module.
8 *
9 * Seven shapes, and how each scales:
10 *
11 * topo_comm_allreduce_* a handful of values PER CLUSTER or per listed body.
12 * Clusters exist only where events were detected, so
13 * this is bounded by what happened, not by the
14 * ensemble.
15 * MFTLTopoCommAllgather the event list, and one record per (component,
16 * holder). Both are paid only on a step with events,
17 * and both are proportional to what changed. It is a
18 * comm_world collective ON PURPOSE: every rank must
19 * agree on the event list, including a rank two boxes
20 * away that will reconstruct part of a body it does
21 * not yet hold.
22 * MFTLTopoCommExchange / 26-neighbour personalised exchanges: the lattice
23 * MFTLTopoCommReply overlap, the component labels, the ghost ids. The
24 * global communicator only when some destination is
25 * provably outside the neighbourhood, which the caller
26 * detects and this file agrees on.
27 */

66double *MFTLTopoCommAllgather(const MFTL_Domain *dom, const double *mine, int n_mine, int stride, int *n_out)
67{
68 const int np = dom->nproc;
69 int *const counts = (int *) MFTLMallocOrDie(2 * (size_t) np * sizeof(int), "topology allgather counts");
70 int *const displs = counts + np;
71
72 const int my_count = n_mine * stride;
73 MPI_Allgather(&my_count, 1, MPI_INT, counts, 1, MPI_INT, dom->comm_world);
74
75 int total = 0;
76 for (int r = 0; r < np; ++r)
77 {
78 displs[r] = total;
79 total += counts[r];
80 }
81
82 double *const all = (double *) MFTLMallocOrDie((size_t) MFTL_MAX(total, 1) * sizeof(double), "topology allgather records");
83 MPI_Allgatherv(mine, my_count, MPI_DOUBLE, all, counts, displs, MPI_DOUBLE, dom->comm_world);
84
85 free(counts);
86 *n_out = total / stride;
87 return all;
88}

90/**
91 * @brief The shared body of the exchange and the reply: an alltoallv of
92 * doubles, on the graph or the global communicator.
93 *
94 * On the graph path the per-rank counts are gathered into neighbour order and
95 * the result is scattered back into rank order, so callers see ranks either
96 * way. A rank reached through two stencil offsets (a thin periodic grid) is ONE
97 * entry of `neighbors` per offset; only the first occurrence carries data, the
98 * rest carry zero, so nothing is delivered twice. The graph's self-loop, if it
99 * has one, carries zero as well: callers merge their own contributions locally.
100 */
101static double *alltoallv_doubles(const MFTL_Domain *dom, const double *send, const int *send_counts_by_rank, int stride, bool global_path,
102 int *recv_counts_by_rank, int *n_recv)
103{
104 const int np = dom->nproc;
105
106 /* Agree on the route: one rank's far destination forces everybody global. */
107 int g = global_path ? 1 : 0;
108 MPI_Allreduce(MPI_IN_PLACE, &g, 1, MPI_INT, MPI_LOR, dom->comm_world);
109
110 /* Send displacements in RANK order, which is how callers packed. */
111 int *const sdispl_rank = (int *) MFTLMallocOrDie((size_t) np * sizeof(int), "topology send displacements");
112 int total_send = 0;
113 for (int r = 0; r < np; ++r)
114 {
115 sdispl_rank[r] = total_send;
116 total_send += send_counts_by_rank[r] * stride;
117 }
118
119 double *recv = MFTL_NULL;
120 int total_recv = 0;
121
122 if (g)
123 {
124 int *const sc = (int *) MFTLMallocOrDie(4 * (size_t) np * sizeof(int), "topology global counts");
125 int *const rc = sc + np, *const sd = sc + 2 * np, *const rd = sc + 3 * np;
126 for (int r = 0; r < np; ++r)
127 {
128 sc[r] = (r == dom->rank) ? 0 : send_counts_by_rank[r] * stride;
129 sd[r] = sdispl_rank[r];
130 }
131 MPI_Alltoall(sc, 1, MPI_INT, rc, 1, MPI_INT, dom->comm_world);
132 for (int r = 0; r < np; ++r)
133 {
134 rd[r] = total_recv;
135 total_recv += rc[r];
136 }
137 recv = (double *) MFTLMallocOrDie((size_t) MFTL_MAX(total_recv, 1) * sizeof(double), "topology global receive");
138 MPI_Alltoallv(send, sc, sd, MPI_DOUBLE, recv, rc, rd, MPI_DOUBLE, dom->comm_world);
139 if (recv_counts_by_rank)
140 for (int r = 0; r < np; ++r) recv_counts_by_rank[r] = rc[r] / stride;
141 free(sc);
142 }
143 else
144 {
145 const int nn = dom->num_neighbors;
146 int *const sc = (int *) MFTLCallocOrDie(4 * (size_t) MFTL_MAX(nn, 1), sizeof(int), "topology graph counts");
147 int *const rc = sc + nn, *const sd = sc + 2 * nn, *const rd = sc + 3 * nn;
148
149 /* First occurrence of each rank in the neighbour list carries its data. */
150 for (int i = 0; i < nn; ++i)
151 {
152 const int r = dom->neighbors[i];
153 bool first = MFTL_TRUE;
154 for (int j = 0; j < i; ++j)
155 if (dom->neighbors[j] == r) first = MFTL_FALSE;
156 sc[i] = (first && r != dom->rank) ? send_counts_by_rank[r] * stride : 0;
157 sd[i] = sdispl_rank[r];
158 }
159 MPI_Neighbor_alltoall(sc, 1, MPI_INT, rc, 1, MPI_INT, dom->comm_graph);
160 for (int i = 0; i < nn; ++i)
161 {
162 rd[i] = total_recv;
163 total_recv += rc[i];
164 }
165
166 /* Receive in neighbour order, then regroup into rank order. */
167 double *const tmp = (double *) MFTLMallocOrDie((size_t) MFTL_MAX(total_recv, 1) * sizeof(double), "topology graph receive");
168 MPI_Neighbor_alltoallv(send, sc, sd, MPI_DOUBLE, tmp, rc, rd, MPI_DOUBLE, dom->comm_graph);
169
170 int *const by_rank = (int *) MFTLCallocOrDie((size_t) np, sizeof(int), "topology graph per-rank counts");
171 for (int i = 0; i < nn; ++i) by_rank[dom->neighbors[i]] += rc[i];
172 int *const cursor = (int *) MFTLMallocOrDie((size_t) np * sizeof(int), "topology graph cursors");
173 int acc = 0;
174 for (int r = 0; r < np; ++r)
175 {
176 cursor[r] = acc;
177 acc += by_rank[r];
178 }
179 recv = (double *) MFTLMallocOrDie((size_t) MFTL_MAX(total_recv, 1) * sizeof(double), "topology graph regrouped");
180 for (int i = 0; i < nn; ++i)
181 {
182 if (rc[i] == 0) continue;
183 memcpy(recv + cursor[dom->neighbors[i]], tmp + rd[i], (size_t) rc[i] * sizeof(double));
184 cursor[dom->neighbors[i]] += rc[i];
185 }
186 if (recv_counts_by_rank)
187 for (int r = 0; r < np; ++r) recv_counts_by_rank[r] = by_rank[r] / stride;
188
189 free(cursor);
190 free(by_rank);
191 free(tmp);
192 free(sc);
193 }
194
195 free(sdispl_rank);
196 *n_recv = total_recv / stride;
197 return recv;
198}
