34/**
35 * @brief One-ring scratch buffer size.
36 *
37 * Larger than MFTL_MAX_VALENCE (32) on purpose. The library's walkers stop at
38 * MFTL_MAX_VALENCE and a boundary vertex reports one MORE neighbour than it has
39 * triangles, so a buffer of exactly MFTL_MAX_VALENCE cannot distinguish "valence
40 * 32" from "the walk was truncated". With headroom, `count == MFTL_MAX_VALENCE`
41 * is an honest answer and truncation is detectable -- and every operator below
42 * REFUSES rather than proceeds on a truncated ring.
43 */
44#define MFTL_REMESH_RING_MAX 64

79/**
80 * @brief Writes triangle @p t as the CCW vertex triple (v0, v1, v2).
81 *
82 * Half-edge 3t+k runs from v[k] to v[(k+1)%3] and stores only its TARGET, so the
83 * triple lands as (v1, v2, v0). This is the single point at which that rotation
84 * is expressed; MFTLGetTriangleVertices() and `tris.verts` read it back out and
85 * the three must agree, or triangles come out silently rotated.
86 *
87 * Deliberately does NOT touch twins, flags or global ids: a rewrite has to place
88 * each logical edge's flags in its NEW slot, and only the caller knows which
89 * slot that is.
90 *
91 * @param obj The mesh.
92 * @param t Triangle index.
93 * @param v0 First corner.
94 * @param v1 Second corner.
95 * @param v2 Third corner. The three must be distinct (asserted in Debug).
96 */
97static inline void tri_write(MFTL_Object *obj, int t, int v0, int v1, int v2)
98{
99 MFTL_ASSERT(v0 != v1 && v1 != v2 && v2 != v0 && "tri_write: degenerate triangle");
100 MFTL_HalfEdge *const he = &obj->tris.he[3 * t];
101 he[0].vertex_idx = v1;
102 he[1].vertex_idx = v2;
103 he[2].vertex_idx = v0;
104}

120 * @brief Re-points a rim half-edge at @p outer, which may be MFTL_ERROR.
121 *
122 * The asymmetry is the point: @p inner has just been written into a new slot, so
123 * its twin link must be set unconditionally, while @p outer is an untouched
124 * half-edge of a neighbouring triangle that only needs its back-link updated.
125 *
126 * @param obj The mesh.
127 * @param inner The rewritten half-edge on the patch rim.

130static inline void he_pair_outer(MFTL_Object *obj, int inner, int outer)

145/**
146 * @brief Appends a vertex at @p pos with a fresh global id.
147 *
148 * Ownership is an array prefix and an append lands at the END of the pool. On a
149 * ghost-free object those coincide and `n_local` grows with the pool. On a
150 * partitioned object the new vertex stays past the ghosts, recognisable by
151 * `owner_proc`, until MFTLRemeshCompactObject() restores the prefix; nothing in
152 * between may decide ownership by index, which is why the parallel gate is
153 * #MFTL_FLAG_FROZEN. A new entity is owned, unreplicated and unfrozen.
154 *
155 * @param obj The mesh. **May reallocate both pool blocks.**
156 * @param pos Position of the new vertex.
157 * @return Its pool index. `out_he_idx` is #MFTL_ERROR; the caller links it.
158 */
159static inline int remesh_new_vertex(MFTL_Object *obj, const double pos[3])
160{
161 const int v = obj->verts.n;
162 const bool partitioned = (obj->verts.n_local < obj->verts.n);
163
164 MFTLObjectReserveAdditional(obj, 1, 0);
165 MFTLObjectSetCounts(obj, v + 1, partitioned ? obj->verts.n_local : v + 1, obj->tris.n, obj->tris.n_local);
166
167 MFTL_Vertex *const rec = &obj->verts.rec[v];
168 rec->out_he_idx = MFTL_ERROR; // the caller links it
169 rec->global_id = obj->next_vertex_gid++;
170 rec->owner_proc = (obj->domain != MFTL_NULL) ? obj->domain->rank : 0;
171 rec->flags = 0;
172
173 obj->verts.pos[3 * v + 0] = pos[0];
174 obj->verts.pos[3 * v + 1] = pos[1];
175 obj->verts.pos[3 * v + 2] = pos[2];
176
177 return v;
178}

180/**
181 * @brief Appends a triangle AND its three half-edges.
182 *
183 * There is no separate half-edge allocator, and there cannot be one: the three
184 * half-edges of triangle t are 3t..3t+2 in the same block, so they come into
185 * existence exactly when it does. Their global ids are 3 * (triangle gid) + k,
186 * which is the invariant redistribution relies on to avoid transmitting them.
187 *
188 * @param obj The mesh. **May reallocate both pool blocks.**
189 * @return The new triangle's pool index. Its half-edges are at 3t, 3t+1, 3t+2
190 * with cleared twins and flags.
191 */
192static inline int remesh_new_triangle(MFTL_Object *obj)
193{
194 const int t = obj->tris.n;
195 const bool partitioned = (obj->tris.n_local < obj->tris.n); // see remesh_new_vertex
196
197 MFTLObjectReserveAdditional(obj, 0, 1);
198 MFTLObjectSetCounts(obj, obj->verts.n, obj->verts.n_local, t + 1, partitioned ? obj->tris.n_local : t + 1);
199
200 MFTL_Triangle *const rec = &obj->tris.rec[t];
201 rec->global_id = obj->next_triangle_gid++;
202 rec->owner_proc = (obj->domain != MFTL_NULL) ? obj->domain->rank : 0;
203 rec->flags = 0;
204
205 MFTL_HalfEdge *const he = &obj->tris.he[3 * t];
206 for (int k = 0; k < 3; ++k)
207 {
208 he[k].vertex_idx = MFTL_ERROR;
209 he[k].twin_idx = MFTL_ERROR;
210 he[k].global_id = 3 * rec->global_id + k;
211 he[k].flags = 0;
212 }
213
214 /* Kept in step so that inv_global_ids_are_dense's "next_halfedge_gid == 3 * n"
215 * clause still means something after a pass that only creates triangles. */
216 obj->next_halfedge_gid = 3 * obj->next_triangle_gid;
217
218 return t;
219}

225/**
226 * @brief Squared sine of a corner angle at or below which a fold test treats the
227 * triangle as degenerate.
228 *
229 * DIMENSIONLESS, so the verdict is the same at every length scale. An absolute
230 * threshold on the squared area vector (dimension L^4, compared with 1e-15)
231 * refused every collapse and flip for edges below about 1.8e-4 -- a millimetre
232 * droplet in SI units -- so the mesh could only be refined
233 * (`test_remeshing_is_independent_of_the_length_unit`). 1e-15 as a ratio is the
234 * floor below which the cross product is rounding noise.
235 */
236#define MFTL_REMESH_DEGENERATE_SIN2 MFTL_VERYVERYSMALL
237
238/**
239 * @brief \f$\vect{e}_1 \times \vect{e}_2\f$, or the zero vector when the
240 * triangle the two edges span is degenerate.
241 *
242 * The zero vector is the "not computable" sentinel area_vector_cos() rejects, so
243 * a degenerate triangle fails a fold test exactly as an inverted one does.
244 *
245 * @param e1 First edge vector.
246 * @param e2 Second edge vector, from the same corner.
247 * @param[out] out The cross product, or zero.
248 */
249static inline void fold_area_vector(const double e1[3], const double e2[3], double out[3])
250{
251 out[0] = e1[1] * e2[2] - e1[2] * e2[1];
252 out[1] = e1[2] * e2[0] - e1[0] * e2[2];
253 out[2] = e1[0] * e2[1] - e1[1] * e2[0];
254
255 const double cr_sq = out[0] * out[0] + out[1] * out[1] + out[2] * out[2];
256 const double ref = (e1[0] * e1[0] + e1[1] * e1[1] + e1[2] * e1[2]) * (e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2]);
257 if (!(cr_sq > MFTL_REMESH_DEGENERATE_SIN2 * ref)) out[0] = out[1] = out[2] = 0.0;
258}

260/**
261 * @brief Twice the area vector of triangle @p t, optionally with one or two of
262 * its corners moved to @p sub_pos.
263 *
264 * @param obj The mesh.
265 * @param t Triangle index.
266 * @param sub_a Vertex index to substitute, or #MFTL_ERROR.
267 * @param sub_b Second vertex index to substitute, or #MFTL_ERROR.
268 * @param sub_pos Position both substituted corners take.
269 * @param[out] out \f$\vect{e}_1 \times \vect{e}_2\f$, un-normalised, so its
270 * magnitude is twice the area -- or zero when the
271 * triangle is degenerate (fold_area_vector()).
272 *
273 * The substitution is what makes a fold-over test possible without editing the
274 * mesh first: an operator asks "what would this triangle's normal be afterwards"
275 * and compares. Passing MFTL_ERROR for both @p sub_a and @p sub_b gives the
276 * current normal, from the same code path, so the comparison cannot be skewed by
277 * two slightly different formulas.
278 *
279 * Un-normalised on purpose: only its direction and whether it is zero are read.
280 */
281static inline void tri_area_vector_substituted(const MFTL_Object *obj, int t, int sub_a, int sub_b, const double sub_pos[3], double out[3])
282{
283 int v[3];
284 MFTLGetTriangleVertices(obj, t, v);
285
286 const double *p[3];
287 for (int k = 0; k < 3; ++k)
288 {
289 p[k] = (v[k] == sub_a || v[k] == sub_b) ? sub_pos : MFTLVertexPos(obj, v[k]);
290 }
291
292 double e1[3];
293 double e2[3];
294 MFTLGetMinimumImageVector(obj->domain, e1, p[0], p[1]);
295 MFTLGetMinimumImageVector(obj->domain, e2, p[0], p[2]);
296 fold_area_vector(e1, e2, out);
297}

299/**
300 * @brief cos of the angle between two area vectors, 0 if either is zero.
301 *
302 * Returning 0 rather than 1 for a degenerate input is what makes this usable as
303 * a rejection test: an operation that produces a zero-area triangle should fail
304 * the same threshold an inverting one fails, not pass it by accident. WHAT counts
305 * as degenerate is decided where the vector is built, relative to its own edges
306 * (fold_area_vector()); here the test is against zero, which has no length scale.
307 *
308 * @param a First area vector.
309 * @param b Second area vector.
310 * @return \f$\cos\theta\f$ between them, or 0 if either is zero.
311 */
312static inline double area_vector_cos(const double a[3], const double b[3])
313{
314 const double den = MFTL_SQRT((a[0] * a[0] + a[1] * a[1] + a[2] * a[2]) * (b[0] * b[0] + b[1] * b[1] + b[2] * b[2]));
315 if (!(den > 0.0)) return 0.0;
316 return (a[0] * b[0] + a[1] * b[1] + a[2] * b[2]) / den;
317}

319/**
320 * @brief Midpoint of @p pa and @p pb, minimum-image correct and wrapped.
321 * @param domain Supplies the periodicity.
322 * @param pa First position.
323 * @param pb Second position.
324 * @param[out] out The midpoint, wrapped back into the domain.
325 */
326static inline void midpoint_periodic(const MFTL_Domain *domain, const double pa[3], const double pb[3], double out[3])
327{
328 double d[3];
329 MFTLGetMinimumImageVector(domain, d, pa, pb);
330 out[0] = pa[0] + 0.5 * d[0];
331 out[1] = pa[1] + 0.5 * d[1];
332 out[2] = pa[2] + 0.5 * d[2];
333 MFTLApplyPeriodicity(domain, out);
334}

336/* ═══════════════════════════════════════════════════════════════════════════
337 * The parallel gate
338 *
339 * Every one of these asks the SAME question -- "is every entity this operation
340 * would make another rank's copy of stale inside my exclusive set?" -- of the
341 * three different modified sets. What counts as modified is what TRAVELS on the
342 * wire (mftl_communication.h): a vertex's position and identity, a triangle's
343 * identity and its three vertex global ids. `twin_idx` and `out_he_idx` are
344 * local index space, rebuilt at every receiver, so re-pointing the rim twins of
345 * a frozen neighbour is not a modification and does not need gating.
346 *
347 * On a serial object no flag is ever set and all three collapse to `false`, so
348 * the sweeps that call them are bit-identical to the serial remesher.
349 * ═══════════════════════════════════════════════════════════════════════════ */
350
351/**
352 * @brief Would splitting or flipping @p h modify something this rank does not own?
353 *
354 * Both operators rewrite the two triangles incident to the edge and nothing else
355 * -- a split additionally APPENDS, and an appended entity is by definition not
356 * replicated anywhere. So the gate is the two incident triangles, and their
357 * corners come along for free: a triangle is frozen whenever any of its corners
358 * is a ghost, so an unfrozen triangle's corners are all owned.
359 *
360 * @param obj The mesh.
361 * @param h Half-edge index.
362 * @return `true` if the operation must be refused. Always `false` on a
363 * ghost-free object, which is what makes the gated sweeps bit-identical
364 * to the serial remesher.
365 */
366static inline bool freeze_blocks_pair(const MFTL_Object *obj, int h)
367{
368 if (MFTLTriangleIsFrozen(obj, MFTLHeTriIdx(h))) return MFTL_TRUE;
369
370 const int tw = obj->tris.he[h].twin_idx;
371 return (tw != MFTL_ERROR) && MFTLTriangleIsFrozen(obj, MFTLHeTriIdx(tw));
372}
373
374/**
375 * @brief Would collapsing @p h modify something this rank does not own?
376 *
377 * The modified set is larger than it looks: the origin vertex dies, the target
378 * MOVES, and every triangle in the origin's fan has that corner rewritten. Two
379 * flag tests cover all of it, and the reason is worth stating because it is what
380 * makes a fan walk unnecessary here:
381 *
382 * a vertex is unfrozen => every triangle incident to it is unfrozen
383 *
384 * which holds by construction -- MFTLRemeshMarkFrozen freezes the corners of
385 * every triangle it freezes. So "both endpoints unfrozen" already asserts that
386 * the whole fan is this rank's to rewrite, and it additionally guarantees that
387 * the fan is locally COMPLETE, because an unfrozen vertex is owned and the
388 * redistribution halo is vertex-complete. The link condition, the wing valences
389 * and the fold-over test can therefore all be evaluated exactly.
390 *
391 * @param obj The mesh.
392 * @param h Half-edge index; its ORIGIN is the vertex that would be retired.
393 * @return `true` if the collapse must be refused.
394 */
395static inline bool freeze_blocks_collapse(const MFTL_Object *obj, int h)
396{
397 return MFTLVertexIsFrozen(obj, MFTLHeOrigin(obj, h)) || MFTLVertexIsFrozen(obj, MFTLHeTarget(obj, h));
398}

404/**
405 * @brief Adds one object's per-pass operation counts into a running total.
406 *
407 * The one place the per-object counters of #MFTL_RemeshStats are summed, used by
408 * all three drivers. The counters are sums; `iterations` is the maximum, because
409 * the objects of one pass iterate side by side rather than one after another.
410 * Fields that describe the PASS rather than an object (phases, migrations, the
411 * profile) are the driver's own and are not touched.
412 *
413 * @param[in,out] dst The running total.
414 * @param src One object's statistics.
415 */
416static inline void remesh_stats_accumulate(MFTL_RemeshStats *dst, const MFTL_RemeshStats *src)
417{
418 dst->n_split += src->n_split;
419 dst->n_collapse += src->n_collapse;
420 dst->n_flip += src->n_flip;
421 dst->n_collapse_rejected_link += src->n_collapse_rejected_link;
422 dst->n_collapse_rejected_fold += src->n_collapse_rejected_fold;
423 dst->n_collapse_rejected_feature += src->n_collapse_rejected_feature;
424 dst->n_flip_rejected_fold += src->n_flip_rejected_fold;
425 dst->n_feature_edges += src->n_feature_edges;
426 dst->n_corner_vertices += src->n_corner_vertices;
427 dst->n_ridge_vertices += src->n_ridge_vertices;
428 dst->n_free_vertices += src->n_free_vertices;
429 dst->n_vertices_removed += src->n_vertices_removed;
430 dst->n_orphans_removed += src->n_orphans_removed;
431 dst->iterations = MFTL_MAX(dst->iterations, src->iterations);
432}

461/**
462 * @struct MFTL_RemeshMeasurement
463 * @brief What MFTLRemeshMeasureBodies() produced, BY LOCAL SLOT, plus the one
464 * number that has to be ensemble-wide.
465 *
466 * Row `o` describes `ens->local_objects[o]` at the moment the measurement was
467 * taken, so it is invalidated by anything that changes the local list -- which
468 * in this module means the first migration. Every consumer reads it before one.
469 *
470 * `r_max` is separate because it is the only quantity here that every rank must
471 * agree about: the strategy ratio and the dispersed admission test are both
472 * taken against it, and two ranks answering differently is two ranks entering
473 * different drivers.
474 */
475typedef struct
476{
477 double *centroid; ///< [3*n_local] area-weighted, seam-consistent, per local slot.
478 double *radius; ///< [n_local] max corner distance from it.
479 double *weight; ///< [n_local] global owned-triangle count.
480 /**
481 * @brief [n_local] lowest-numbered rank holding each body.
482 *
483 * `lowest_holder[o] == domain->rank` designates THIS rank as the one that may
484 * count body `o` in a sum over ranks. Every holder reduces to the same radius
485 * and the same weight, so without it a body held by four ranks is counted four
486 * times -- which is what made the dispersed driver report "4 of 1 bodies are
487 * wider than a rank box".
488 */
489 double *lowest_holder;
490 double r_max; ///< Largest radius over the WHOLE ensemble.
491 int n_local; ///< Local bodies when the measurement was taken.
492} MFTL_RemeshMeasurement;

521/**
522 * @brief MFTLRemeshEnsembleParallel(), with the parameter check optional.
523 *
524 * The check (MFTLRemeshAgreeParams()) is taken at the pass's first collective,
525 * after the local work before it, so it adds a latency rather than a
526 * synchronisation point. Callers that already agreed the parameters in this
