1/**
2 * @file mftl_remeshing.h
3 * @author Christian Gorges
4 * @date 2026-08-15
5 * @ingroup mftl_remeshing
6 *
7 * @brief Isotropic surface remeshing -- split, collapse, flip, tangential
8 * smoothing -- serial and distributed.
9 *
10 * Layer 3. The operators, the smoother and the per-object driver are serial and
11 * contain no MPI; the two distributed drivers are schedules around the same
12 * kernel. Every MPI call of the module is in src/remeshing/remesh_comm.c
13 * (enforced by `test_remeshing_mpi_is_confined`).
14 *
15 * **Algorithm** (Botsch & Kobbelt, SGP 2004). One iteration against a target
16 * edge length \f$L\f$:
17 *
18 * 1. detect features (MFTLDetectFeatures());
19 * 2. SPLIT every edge longer than \f$\tfrac43 L\f$ at its midpoint;
20 * 3. COLLAPSE every edge shorter than \f$\tfrac45 L\f$, unless a resulting edge
21 * would be longer than \f$\tfrac43 L\f$;
22 * 4. compact (MFTLRemeshCompactObject());
23 * 5. FLIP every edge whose flip strictly lowers the squared valence deviation;
24 * 6. detect features again and smooth tangentially (MFTLSmoothTangential()).
25 *
26 * A split of an edge longer than \f$\tfrac43 L\f$ gives halves longer than
27 * \f$\tfrac23 L\f$, i.e. no further from \f$L\f$ than the original edge; the
28 * collapse's stretch clause keeps split and collapse from undoing each other.
29 *
30 * **Operators under implicit indexing.** Half-edge 3t+k is corner k of triangle
31 * t, so an operator rewrites whole triangles and re-links the twins on the rim of
32 * its patch: a flip allocates nothing, a split appends one triangle (with its
33 * three half-edges) per side, and a collapse flags two triangles and one vertex
34 * #MFTL_FLAG_DEAD for the compaction to remove. Edge flags (FEATURE, BOUNDARY)
35 * are carried to the logical edge's new slot explicitly. A half-edge's global
36 * id, \f$3\,t_\mathrm{gid}+k\f$, depends on the slot only and stays valid.
37 *
38 * **Global ids** stay unique but become sparse (splits create, collapses retire).
39 * Redistribution and the ghost sync size lookup tables from `next_*_gid`, so
40 * every pass ends dense again: MFTLRemeshRenumberGlobalIds() on a whole object,
41 * per-body prefix sums over each body's holders in the spanning driver.
42 *
43 * **Distributed remeshing.** An edge whose endpoints two ranks own can be
44 * collapsed by neither, and a lock/ask protocol inside the sweeps would make the
45 * result depend on message timing. Two exact strategies need no negotiation;
46 * AUTO chooses collectively from one ratio, the largest body radius R against
47 * the narrowest rank box h:
48 *
49 * - **SPANNING** (MFTLRemeshEnsembleParallel()): the kernel runs only on
50 * entities no other rank holds a copy of (the others are #MFTL_FLAG_FROZEN,
51 * derived locally by MFTLRemeshMarkFrozen()), and the cut planes are shifted
52 * by half a box over the placements \f$\{0,h/2\}^3\f$ so that every entity
53 * is interior in some placement. Requires \f$h > 4\delta\f$, \f$\delta\f$
54 * the half-width of the frozen band.
55 * - **DISPERSED** (MFTLRemeshEnsembleDispersed(), \f$R \le 0.4\,h\f$ by
56 * default): each body is migrated whole onto one rank (its centroid's owner,
57 * load-balanced) and remeshed serially; two migrations per pass, nothing
58 * frozen.
59 *
60 * On a ghost-free object nothing is frozen, so the serial path is unchanged to
61 * the bit. Theory, call graphs and measurements: src/remeshing/README.md and
62 * docs/analysis.md §4-5.
63 */

82 *

85 * when any incident triangle is (one more): three edges. The band actually
86 * measured is about \f$0.84\f$ `max_edge_length`

92/**
93 * @brief The eight placements, in units of \f$h_k/2\f$ per axis.
94 *
95 * Ordered so that every prefix is as useful as possible: rows 0-1 ((0,0,0),
96 * (1,1,1)) make every entity near the planes of ONE axis interior at least once
97 * (all but \f$O((\delta/h)^2)\f$ of the mesh); rows 0 and 2-4 form the
98 * even-parity set {000, 011, 101, 110}, which covers every entity near the
99 * planes of TWO axes; an entity near planes of all three axes needs its unique
100 * complementary placement, so all eight rows are required. Placements differing

102 * de-duplicates them.

666// Each operator is an approval predicate plus an execution function. The

671/**
672 * @brief Is half-edge @p h a legal and wanted split candidate?
673 *
674 * Wanted: longer than `max_edge_length`. Legal: alive, not touched earlier in
675 * this sweep, and the canonical half of its edge (the smaller half-edge index,
676 * or the only one at a boundary), so every undirected edge is offered once.
677 *
678 * @param obj The mesh.
679 * @param h Half-edge index.
680 * @param params Supplies `max_edge_length_sq`.
681 * @return `true` if the split sweep should split this edge.
682 */
683bool MFTLSplitApproved(const MFTL_Object *obj, int h, const MFTL_RemeshParameters *params);

685/**
686 * @brief Splits the edge of half-edge @p h at its midpoint.
687 *
688 * C C
689 * / \ /|\
690 * / \ / | \
691 * A-----B -> A--M--B
692 * \ / \ | /
693 * \ / \|/
694 * D D
695 *
696 * Rewrites the incident triangle(s) and appends one triangle per side. M is the
697 * minimum-image midpoint, wrapped into the domain, and inherits the edge's
698 * FEATURE/BOUNDARY classification, as do both halves. M and the halves are
699 * marked visited, so the same sweep neither re-splits nor collapses them.
700 *
701 * @param obj The mesh. **May reallocate the pools.**
702 * @param h Half-edge index, normally one MFTLSplitApproved() accepted.
703 * @return Pool index of M, or #MFTL_ERROR if @p h is not a live half-edge.
704 */
705int MFTLSplitEdge(MFTL_Object *obj, int h);
706
707/**
708 * @brief Is collapsing half-edge @p h (its origin into its target) legal and
709 * wanted?
710 *
711 * Wanted: shorter than `min_edge_length`. Each clause prevents a result that is
712 * a valid data structure but not a valid surface, and none of them fails loudly:
713 *
714 * - **Features:** a corner is never retired; a feature (boundary) vertex may
715 * only collapse into a feature (boundary) vertex; two feature (boundary)
716 * vertices joined by a non-feature (interior) edge may not collapse -- that
717 * edge is a chord, and collapsing it pinches the curve.
718 * - **Link condition:** the endpoints share exactly two neighbours (one at a
719 * boundary edge); otherwise two sheets are welded into a non-manifold vertex.
720 * - **Valence:** the merged vertex has valence in [3, #MFTL_MAX_VALENCE], and
721 * each wing vertex has at least 4 neighbours (a tetrahedron passes the link
722 * condition and would dissolve).
723 * - **Stretch:** no edge of the merged vertex exceeds `max_edge_length`, or
724 * split and collapse oscillate.
725 * - **Fold-over:** no surviving triangle's normal turns by more than
726 * `max_normal_deviation_cos` or becomes degenerate (the test is relative to
727 * the triangle's own edges, so it is independent of the length unit).
728 *
729 * Truncated one-ring walks (valence beyond the scratch size) are refused.
730 *
731 * @param obj The mesh.
732 * @param h Half-edge index; its ORIGIN would be retired.
733 * @param params Supplies the length band, the fold-over threshold
734 * and the feature settings.
735 * @param[out] reject_reason May be #MFTL_NULL. Receives a
736 * `MFTL_REMESH_REJECT_*` code.
737 * @return `true` if the collapse is legal and wanted.
738 */
739bool MFTLCollapseApproved(const MFTL_Object *obj, int h, const MFTL_RemeshParameters *params, int *reject_reason);
740
741/**
742 * @name Rejection codes
743 *
744 * Reported through `reject_reason` of MFTLCollapseApproved() and
745 * MFTLFlipApproved() and tallied into #MFTL_RemeshStats.
746 * @{
747 */
748/** @brief Not rejected. */
749#define MFTL_REMESH_REJECT_NONE 0
750/** @brief Link condition or valence bound failed. */
751#define MFTL_REMESH_REJECT_LINK 1
752/** @brief A triangle would turn past `max_normal_deviation_cos` or degenerate. */
753#define MFTL_REMESH_REJECT_FOLD 2
754/** @brief Blocked by a feature, boundary or corner rule. */
755#define MFTL_REMESH_REJECT_FEATURE 3
756/** @brief Any other reason: not wanted, dead or visited entity, stretch bound, truncated one-ring. */
757#define MFTL_REMESH_REJECT_OTHER 4
758/** @} */
759
760/**
761 * @brief Collapses the edge of half-edge @p h, retiring its ORIGIN vertex.
762 *
763 * The target survives and moves to the position MFTLCollapseApproved()
764 * evaluated: its own position if it is a corner or if it alone lies on a
765 * feature curve or boundary, otherwise the minimum-image midpoint. The one or
766 * two incident triangles and the origin are flagged #MFTL_FLAG_DEAD; pool
767 * indices do not change until MFTLRemeshCompactObject().
768 *
769 * @param obj The mesh. Allocates nothing.
770 * @param h Half-edge index, normally one MFTLCollapseApproved() accepted.
771 * @return #MFTL_SUCCESS, or #MFTL_ERROR if @p h was not collapsible.
772 */
773int MFTLCollapseEdge(MFTL_Object *obj, int h);
774
775/**
776 * @brief Would flipping half-edge @p h strictly lower the squared valence
777 * deviation of the four quad vertices?
778 *
779 * \f[
780 * E = \sum_{v \in \{a,b,c,d\}} \big(\deg(v) - \deg^{*}(v)\big)^{2},
781 * \qquad \deg^{*} = \begin{cases} 6 & \text{interior} \\ 4 & \text{boundary} \end{cases}
782 * \f]
783 *
784 * (\f$\deg\f$ counts neighbours). Only a STRICT decrease is accepted, or two
785 * flips could alternate forever. Refused for boundary and feature edges, when
786 * an endpoint has three neighbours or fewer, when the opposite vertices are
787 * already adjacent (a duplicate edge), when a valence would exceed
788 * #MFTL_MAX_VALENCE, and when either new triangle would turn past
789 * `max_normal_deviation_cos` against the quad's normal (a non-convex quad).
790 *
791 * @param obj The mesh.
792 * @param h Half-edge index.
793 * @param params Supplies `max_normal_deviation_cos`.
794 * @param[out] reject_reason May be #MFTL_NULL. #MFTL_REMESH_REJECT_FOLD only when
795 * the valence criterion wanted the flip and the
796 * geometry refused it.
797 * @return `true` if the flip strictly lowers \f$E\f$ and is geometrically legal.
798 */
799bool MFTLFlipApproved(const MFTL_Object *obj, int h, const MFTL_RemeshParameters *params, int *reject_reason);

857/**
858 * @brief Flags feature edges and classifies every vertex as free, ridge or
859 * corner.
860 *
861 * Recomputes `tris.normal` first (after a split, new triangles have none). Then,
862 * in linear passes over undirected edges (no fan walks, which would miss the
863 * closing edge of an open fan):
864 *
865 * - **edges:** #MFTL_FLAG_FEATURE on both halves of every edge whose face
866 * normals differ by more than @p angle_degrees, and on every mesh-boundary
867 * edge; cleared elsewhere, so a smoothed-out fold stops being a feature. A
868 * missing twin counts as a boundary only on an owned triangle -- on a ghost it
869 * is the rim of the halo, and #MFTL_FLAG_BOUNDARY decides;
870 * - **vertices:** #MFTL_FLAG_FEATURE (ridge point) for exactly two feature edges
871 * continuing within @p corner_angle_degrees of straight; #MFTL_FLAG_CORNER for
872 * one, three or more, or two that turn more sharply; neither for none.
873 *
874 * @param obj The mesh. Flags rewritten, `tris.normal`
875 * refreshed.
876 * @param angle_degrees Normal-angle threshold in degrees;
877 * \f$\ge 180\f$ disables it, leaving only
878 * boundaries as features.
879 * @param corner_angle_degrees Turn angle above which a two-edge vertex is a
880 * corner.
881 * @param[out] stats May be #MFTL_NULL. Receives the edge and
882 * vertex-class counts.
883 * @return The number of undirected feature edges.
884 * @rank_local
885 */
886int MFTLDetectFeatures(MFTL_Object *obj, double angle_degrees, double corner_angle_degrees, MFTL_RemeshStats *stats);
887
888/**
889 * @brief Tangential smoothing: @p iterations Jacobi sweeps in which each vertex
890 * moves towards its neighbourhood centre, PROJECTED onto the subspace its
891 * feature class allows.
892 *
893 * - **Corner:** locked (0D).
894 * - **Ridge:** the pull towards the midpoint of its two feature neighbours,
895 * projected onto the ridge tangent (1D). Moving to the midpoint itself cuts
896 * the chord and erodes the ridge.
897 * - **Free:** the pull towards the target point of `smooth_mode`, minus its
898 * component along the area-weighted vertex normal (2D). Without the
899 * projection, Laplacian smoothing shrinks the surface.
900 *
901 * Vertex normals come from MFTLComputeVertexNormals(), refreshed every sweep; a
902 * vertex whose normal is zero (degenerate one-ring) stays put. Steps are scaled
903 * by `smooth_relaxation` and capped at `smooth_max_step_fraction` of the
904 * shortest incident edge. All differences are minimum-image vectors and
905 * positions are wrapped, so the smoother is seam-safe. Frozen vertices and
906 * ghosts are not moved but are read, which makes a distributed sweep exact
907 * without communication. Jacobi updates make the result independent of pool
908 * order.
909 *
910 * @param obj The mesh; positions updated in place.
911 * @param params Supplies `smooth_mode`, `smooth_relaxation` and
912 * `smooth_max_step_fraction`.
913 * @param iterations Sweeps to perform.
914 *
915 * @note Requires #MFTL_DERIVED_TOPOLOGY and a current feature classification.
916 * @rank_local
917 */
918void MFTLSmoothTangential(MFTL_Object *obj, const MFTL_RemeshParameters *params, int iterations);

1086 * holder to count a body in sums over ranks.
