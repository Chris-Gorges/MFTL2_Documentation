21/**
22 * @brief Normalises a 3-vector in place: the result has unit length, or is zero.
23 *
24 * \f$\vect{v} \leftarrow \vect{v}/\norm{\vect{v}}\f$, with the magnitude computed
25 * UNCLAMPED and the zero vector mapped to itself:
26 * `inv = (mag > 0) ? 1/mag : 0`. The result is unit to rounding whenever
27 * \f$\norm{\vect{v}}^2\f$ is a normal double, i.e. for
28 * \f$1.5\cdot10^{-154} \lesssim \norm{\vect{v}} \lesssim 1.3\cdot10^{154}\f$;
29 * outside it the squared sum is subnormal, zero or infinite and the result loses
30 * precision or becomes the zero vector. Zero is a detectable sentinel: a unit
31 * vector never has zero length.
32 *
33 * Every normalisation in the library uses this same expression. The two
34 * alternatives are both wrong for small vectors: clamping \f$\norm{\vect{v}}^2\f$
35 * before the square root floors the divisor and returns vectors SHORTER than
36 * unit (a vector of length \f$10^{-9}\f$ came out at \f$10^{-3}\f$), and
37 * `1/max(mag, tiny)` merely moves that floor. A short normal still points the
38 * right way, so every sign test passes while it silently scales whatever it
39 * multiplies. The select costs one instruction (+1.1 % in the per-triangle
40 * kernel). Guarded by `test_normalize_is_unit_across_magnitudes`.
41 *
42 * @param vec The vector, normalised in place.
43 */
44static inline void MFTLNormalizeVector(double vec[3])
45{
46 const double mag = MFTL_SQRT(MFTL_SQ(vec[0]) + MFTL_SQ(vec[1]) + MFTL_SQ(vec[2]));
47 const double inv_mag = (mag > 0.0) ? 1.0 / mag : 0.0;
48 vec[0] *= inv_mag;
49 vec[1] *= inv_mag;
50 vec[2] *= inv_mag;
51}

53/**
54 * @brief Dot product of two 3-vectors.
55 * @param v1 First vector.
56 * @param v2 Second vector.
57 * @return \f$\vect{v}_1 \cdot \vect{v}_2\f$.
58 */
59static inline double MFTLVectorInProd(const double v1[3], const double v2[3]) { return v1[0] * v2[0] + v1[1] * v2[1] + v1[2] * v2[2]; }

61/**
62 * @brief Squared Euclidean length of a 3-vector.
63 *
64 * Prefer it to MFTLVectorNorm() wherever lengths are only compared.
65 *
66 * @param v The vector.
67 * @return \f$\norm{\vect{v}}^2\f$.
68 */
69static inline double MFTLVectorNormSq(const double v[3]) { return v[0] * v[0] + v[1] * v[1] + v[2] * v[2]; }
70
71/**
72 * @brief Euclidean length of a 3-vector.
73 * @param v The vector.
74 * @return \f$\norm{\vect{v}}\f$.
75 */
76static inline double MFTLVectorNorm(const double v[3]) { return MFTL_SQRT(MFTLVectorNormSq(v)); }

78/**
79 * @brief Cross product, \f$\vect{out} = \vect{v}_1 \times \vect{v}_2\f$.
80 *
81 * @param[out] out Result. **Must not alias** @p v1 or @p v2 -- component 0 is
82 * written before components 1 and 2 are read.
83 * @param v1 Left operand.
84 * @param v2 Right operand.
85 */
86static inline void MFTLVectorCrossVector(double out[3], const double v1[3], const double v2[3])
87{
88 out[0] = v1[1] * v2[2] - v1[2] * v2[1];
89 out[1] = v1[2] * v2[0] - v1[0] * v2[2];
90 out[2] = v1[0] * v2[1] - v1[1] * v2[0];
91}
