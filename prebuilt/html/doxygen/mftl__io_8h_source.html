1/**
2 * @file mftl_io.h
3 * @author Christian Gorges
4 * @date 2026-08-14
5 * @ingroup mftl_io
6 *
7 * @brief Mesh output in VTK formats: a serial legacy debug dump and a parallel
8 * VTK XML (`.vtu`) writer.
9 *
10 * Nothing is read back by the library. Four decisions define MFTLWriteVtu():
11 *
12 * 1. **Raw binary by default.** Decimal formatting of a double costs about
13 * 100 ns and ~17 bytes, so an ASCII writer is bound by `printf`, not by the
14 * disk; the appended raw section is a memcpy (23-32x faster, docs/analysis.md
15 * §3). ASCII is kept for inspection.
16 * 2. **Two file layouts.** One piece per rank plus a `.pvtu` index (no
17 * contention, but P+1 files per dump -- what overloads a parallel file
18 * system's metadata server first), or one shared file written collectively
19 * with MPI-IO. On one local SSD the per-rank layout was faster (the shared
20 * file: 1.0x at P = 1, 0.45x at P = 4); re-measure on the target system.
21 * 3. **Only owned entities are written.** A ghost is a copy of a triangle another
22 * rank owns; writing it would count every halo triangle once per holder and
23 * bias every integral a visualisation tool computes (10-30 % in a typical
24 * run). `include_ghosts` exists for inspecting the halo.
25 * 4. **Periodic unwrapping at write time.** A triangle straddling a periodic face
26 * gets three private points at its unwrapped corners; the mesh in memory is
27 * untouched.
28 *
29 * Theory: src/io/README.md.
30 */

41/** @brief How the numbers are encoded in a .vtu file. */
42typedef enum
43{
44 /** Raw binary in an appended section, in the host byte order (declared in
45 * the header). The default: 23-32x faster to write than ASCII and about half
46 * the size; read natively by ParaView and VisIt. */

50 * Not available with #MFTL_IO_SINGLE_FILE. */
51 MFTL_VTK_ASCII = 1

54/** @brief How the output is spread over files. */
55typedef enum
56{
57 /** `<prefix>_%05d.vtu` from every rank plus `<prefix>.pvtu` from rank 0.
58 * No shared file; P+1 files per dump. */
59 MFTL_IO_FILE_PER_RANK = 0,
60
61 /** One `<prefix>.vtu` holding a single Piece, written collectively with
62 * MPI-IO. Implies MFTL_VTK_BINARY. */
63 MFTL_IO_SINGLE_FILE = 1
64} MFTL_IoLayout;

66/**
67 * @struct MFTL_VtkOptions
68 * @brief What to write and how; start from MFTLVtkDefaults().
69 *
70 * The optional arrays are identity and ownership, which are always
71 * well-defined. There is deliberately no option for `tris.area` or
72 * `tris.normal`: nothing records whether they are current and a redistribution
73 * leaves them unspecified, so a file containing them could be stale while
74 * looking right. Compute and write them yourself if needed.
75 */
76typedef struct MFTL_VtkOptions
77{
78 MFTL_IoLayout layout; ///< Default MFTL_IO_FILE_PER_RANK.
79 MFTL_VtkEncoding encoding; ///< Default MFTL_VTK_BINARY.
80
81 /** Write ghost entities too. Default false. The merged dataset then contains
82 * every halo triangle once per rank holding it, so use it to LOOK at the
83 * halo, never to integrate over the surface. */
84 bool include_ghosts;
85
86 /** `VertexOwner` / `TriangleOwner` (Int32): which rank owns each entity.
87 * Default true -- this is how you audit a domain decomposition visually. */
88 bool write_owner;
89
90 /** `BodyID` (Int32): the object's `global_id`, per cell. Default true; it is
91 * what lets a visualisation tool re-separate the bodies after aggregation. */
92 bool write_body_id;
93
94 /** `VertexGID` / `TriangleGID` (Int32): the entity global ids. Default false.
95 * Useful for following one entity across timesteps or across a
96 * redistribution; 8 bytes per triangle of file you usually do not want. */
97 bool write_global_id;
98} MFTL_VtkOptions;

111/**
112 * @brief Writes the calling rank's pool of one object to a legacy ASCII `.vtk`
113 * file: the raw debug dump.
114 *
115 * Every entity the rank holds, ghosts included, with no filtering, no periodic
116 * unwrapping and no communication. It also writes `VertexOwner`,
117 * `TriangleOwner`, `verts.normal` and `tris.area` UNVALIDATED -- its contract is
118 * to show what is in memory, including arrays a redistribution left undefined.
119 *
120 * Strictly single-rank: several ranks writing the same path race. Call it from
121 * one rank, or give every rank its own name.
122 *
123 * @param object The object (valid topology required).
124 * @param base_filename Base path; ".vtk" is appended.
125 * @return #MFTL_SUCCESS, or #MFTL_ERROR on a NULL argument or an I/O failure.
126 */
127int MFTLWriteVtk(const MFTL_Object *object, const char *base_filename);
128
129/**
130 * @brief Writes every local object of an ensemble as one VTK XML unstructured
131 * grid (all bodies in one point/cell list per piece).
132 *
133 * Points are the vertices referenced by the written triangles (owned triangles
134 * may reference ghost vertices), plus three private points per triangle that
135 * straddles a periodic face; a seam vertex is therefore written by every rank
136 * with a triangle on it. Cells carry `TriangleOwner`, `BodyID` and optionally
137 * `TriangleGID`; points `VertexOwner` and optionally `VertexGID`. Uses compact
138 * connectivity when current, the half-edge records otherwise; no geometry
139 * precondition.
140 *
141 * - #MFTL_IO_FILE_PER_RANK writes `<prefix>_%05d.vtu` per rank and
142 * `<prefix>.pvtu` on rank 0.
143 * - #MFTL_IO_SINGLE_FILE writes one `<prefix>.vtu` with one Piece: one
144 * non-blocking `MPI_Exscan` + `MPI_Allreduce` pair places every block, rank 0
145 * writes header, length prefixes and footer, and every rank writes its slice of
146 * each block with `MPI_File_write_at_all` (enabling two-phase collective
147 * buffering). The file is truncated to its exact size. Requires
148 * #MFTL_VTK_BINARY and at most \f$2^{31}-1\f$ points and
149 * \f$(2^{31}-1)/3\f$ cells (Int32 connectivity and offsets).
150 *
151 * @param ensemble The ensemble; zero objects yield a valid empty dataset.
152 * @param filename_prefix Base path/name; suffixes are appended.
153 * @param options #MFTL_NULL for MFTLVtkDefaults().
154 * @return #MFTL_SUCCESS if every rank succeeded, #MFTL_ERROR otherwise -- the same
155 * value on every rank.
156 *
157 * @collective In both layouts (the return code is reduced).
158 */
159int MFTLWriteVtu(MFTL_Ensemble *ensemble, const char *filename_prefix, const MFTL_VtkOptions *options);
