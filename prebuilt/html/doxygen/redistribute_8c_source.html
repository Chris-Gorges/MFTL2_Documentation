1/**
2 * @file redistribute.c
3 * @author Christian Gorges
4 * @date 2026-08-14
5 * @ingroup mftl_communication
6 *
7 * @brief Redistribution: re-partitioning a distributed mesh over the process grid.
8 *
9 * Both public entry points run redistribute_core(); MFTLRedistribute() is
10 * MFTLRedistributeWithOwners() with the geometric owner rule.
11 *
12 * Most of a partition does not move, so kept entities never touch the wire
13 * format:
14 *
15 * kept -> a PERMUTATION of the pool: records and positions are gathered
16 * into scratch in final order and written back.
17 * arrivals -> de-duplicated through a global-id table (gt_* in
18 * comm_internal.h) and appended.
19 * twins -> REPAIRED, not rebuilt: a twin between two kept triangles is an
20 * index remap; only half-edges that lost their partner are
21 * re-paired.
22 *
23 * **Why the incremental twin repair is complete.** Kept-ness is decided per
24 * triangle, so a kept half-edge's twin survives iff the twin's triangle is kept,
25 * and then both are remapped consistently. A half-edge whose twin did not
26 * survive can only pair with another such half-edge, so the repair set is closed
27 * under the twin relation and bucketing it against itself finds every pair a
28 * full MFTLRebuildTwins() would. tests/test_redistribution.c re-runs
29 * MFTLRebuildTwins() after a redistribution and asserts that nothing changes.
30 */

52/**
53 * @brief Per-destination starting capacity of the id lists: a fifth of this
54 * rank's fair share of the mesh, clamped to [1024, 10^6].
55 *
56 * @param n_tris_global Triangles in the whole mesh.
57 * @param n_verts_global Vertices in the whole mesh.
58 * @param nproc Ranks in the communicator.
59 * @param[out] tri_hint Starting triangle capacity per destination.
60 * @param[out] vert_hint Starting vertex capacity per destination.
61 */
62static void calc_buffer_hints(int n_tris_global, int n_verts_global, int nproc, int *tri_hint, int *vert_hint)
63{
64 /* Assume no single destination takes more than a fifth of this rank's fair
65 * share; the buffers grow if that is wrong. */
66 const double factor = 0.2;
67 const int t = (int) (factor * (double) n_tris_global / (double) nproc);
68 const int v = (int) (factor * (double) n_verts_global / (double) nproc);
69
70 *tri_hint = MFTL_MIN(MFTL_MAX(1024, t), 1000000);
71 *vert_hint = MFTL_MIN(MFTL_MAX(1024, v), 1000000);
72}

74/**
75 * @brief Grows a send buffer geometrically, in elements of @p elem bytes.
76 *
77 * @param sb The send buffer.
78 * @param needed Additional elements that must fit.
79 * @param elem Size of one element in bytes.
80 */
81static void sb_reserve(MFTL_SendBuffer *sb, int needed, size_t elem)
82{
83 if (sb->count + needed <= sb->capacity) return;
84
85 int cap = (sb->capacity == 0) ? MFTL_MAX(needed, 256) : sb->capacity;
86 while (cap < sb->count + needed) cap *= 2;
87
88 sb->buffer = MFTLReallocOrDie(sb->buffer, (size_t) cap * elem, "MFTL redistribution send buffer");
89 sb->capacity = cap;
90}

145/**
146 * @brief Recomputes the triangle geometry of a body whose centres the library
147 * knows to be out of date, before routing by them.
148 *
149 * Called when #MFTL_DERIVED_GEOMETRY is clear: no pass over every triangle
150 * since the last ghost sync, topology change or rebuild. That catches the
151 * inconsistent cases -- no pass at all, a pass over the owned triangles only
152 * (ghost centres stale), a pass before the ghost sync, a second redistribution
153 * without one (arrivals unspecified) -- in O(1) per body. It cannot catch owned

167
168 MFTLComputeTriangleGeometry(obj, obj->tris.n);
169 MFTLCommStats.n_objects_centres_recomputed++;

171 static bool warned = MFTL_FALSE;

182/**
183 * @brief Assigns every entity (ghosts included) its NEW owner and writes it into
184 * `owner_proc`.
185 *
186 * Two streams over `verts.pos` and `tris.center` through MFTLGetOwnerProc(), or
187 * the caller's #MFTL_OwnerMap (tested once per object, outside the loops). The
188 * centre-currency assertion belongs to the geometric rule only. Each loop also
189 * counts owner changes from the value it overwrites -- one compare against a
190 * word it stores anyway -- and that verdict is half of the skip precondition.
191 * Idempotent: a second call rewrites the same values and reports no change.
192 *
193 * @param obj The mesh.
194 * @param map Explicit owner map, or NULL for the geometric rule.
195 * @param[out] tri_owner `[tris.n]` destination rank per triangle.
196 * @param[out] vert_owner `[verts.n]` destination rank per vertex.
197 * @param[out] vert_gid `[verts.n]` vertex global ids, gathered contiguously
198 * for the packing loops.
199 * @param record_stats `false` on a re-classification, so counters describe
200 * each entity once.
201 * @return Non-zero if any entity of @p obj changed owner.
202 */
203static int compute_owners(MFTL_Object *obj, const MFTL_OwnerMap *map, int *restrict tri_owner, int *restrict vert_owner, int *restrict vert_gid,
204 bool record_stats)
205{
206 const MFTL_Domain *const dom = obj->domain;
207
208 const double *const restrict pos = obj->verts.pos;
209 MFTL_Vertex *const restrict vr = obj->verts.rec;
210 const int nv = obj->verts.n;
211
212 const double *const restrict ctr = obj->tris.center;
213 MFTL_Triangle *const restrict tr = obj->tris.rec;
214 const int nt = obj->tris.n;
215
216 if (map != MFTL_NULL)
217 {
218 const int *const restrict vo = map->vert_owner;
219 const int *const restrict to = map->tri_owner;
220 MFTL_ASSERT(vo != MFTL_NULL && to != MFTL_NULL && "redistribute: an owner map must cover both entity classes");
221
222 long changed_v = 0, changed_t = 0;
223
224 for (int v = 0; v < nv; ++v)
225 {
226 const int owner = vo[v];
227 MFTL_ASSERT(owner >= 0 && owner < dom->nproc && "redistribute: owner map names a rank outside the communicator");
228 vert_owner[v] = owner;
229 vert_gid[v] = vr[v].global_id;
230 changed_v += (vr[v].owner_proc != owner);
231 vr[v].owner_proc = owner;
232 }
233
234 for (int t = 0; t < nt; ++t)
235 {
236 const int owner = to[t];
237 MFTL_ASSERT(owner >= 0 && owner < dom->nproc && "redistribute: owner map names a rank outside the communicator");
238 tri_owner[t] = owner;
239 changed_t += (tr[t].owner_proc != owner);
240 tr[t].owner_proc = owner;
241 }
242
243 if (record_stats) stats_owner_scan(nv, nt, changed_v, changed_t);
244 return (changed_v | changed_t) != 0;
245 }
246
247 MFTL_ASSERT(MFTLTriangleCentresAreCurrent(obj) &&
248 "redistribute: stale triangle centres -- call MFTLComputeTriangleGeometry(obj, obj->tris.n) (ghosts included) first");
249
250 long changed_v = 0, changed_t = 0;
251
252 for (int v = 0; v < nv; ++v)
253 {
254 const int owner = MFTLGetOwnerProc(dom, &pos[3 * v]);
255 vert_owner[v] = owner;
256 vert_gid[v] = vr[v].global_id;
257 /* Unconditional: the answer decides whether this body is rebuilt at all. */
258 changed_v += (vr[v].owner_proc != owner);
259 vr[v].owner_proc = owner;
260 }
261
262 for (int t = 0; t < nt; ++t)
263 {
264 const int owner = MFTLGetOwnerProc(dom, &ctr[3 * t]);
265 tri_owner[t] = owner;
266 changed_t += (tr[t].owner_proc != owner);
267 tr[t].owner_proc = owner;
268 }
269
270 if (record_stats) stats_owner_scan(nv, nt, changed_v, changed_t);
271 return (changed_v | changed_t) != 0;
272}

274/**
275 * @struct vertex_dedup
276 * @brief De-duplicates each REMOTE destination's vertex list on construction.
277 *
278 * A vertex is a candidate once per incident triangle bound for a destination.
279 * One bit per (vertex, destination slot) makes the test-and-set O(1) instead of
280 * a later sort + unique. Slots are handed out on first contact; 32 suffice on
281 * the neighbour route (at most 26 remote destinations). On the global route a
282 * destination beyond 32 gets #MFTL_DEDUP_NO_BIT and its list is sorted instead;
283 * correctness never depends on the bound.
284 */
285/** @brief De-duplication bit slots available, one per remote destination. */
286#define MFTL_DEDUP_BITS 32
287/** @brief Slot value meaning "no bit available"; that destination falls back to a sort. */
288#define MFTL_DEDUP_NO_BIT MFTL_DEDUP_BITS
289
290typedef struct
291{
292 uint32_t *mask; ///< [verts.n] zeroed per object.
293 int *rank_bit; ///< [nproc] bit slot per destination, MFTL_ERROR = unassigned.
294 int n_bits; ///< Slots handed out for the current object.
295} vertex_dedup;

297/**
298 * @brief Destination analysis: which ranks each OWNED triangle goes to, and what
299 * this rank keeps.
300 *
301 * Destinations of triangle t: its owner, the owners of its three corners and
302 * the owners of its three edge neighbours (the halo rule). This rank is one
303 * destination like any other, recorded as keep flags instead of a copy. Ghost
304 * triangles are never forwarded. Interior triangles, whose seven candidates all
305 * name one rank, take a register-only fast path (an XOR-OR test); otherwise the
306 * candidates are collected in first-seen order, which fixes the wire order and
307 * hence the arrival slot order.
308 *
309 * @param obj The mesh.
310 * @param my_rank This rank.
311 * @param is_neighbor `[nproc]` flags marking the 26-neighbourhood.
312 * @param tri_owner `[tris.n]` destination rank per triangle.
313 * @param vert_owner `[verts.n]` destination rank per vertex.
314 * @param dd Per-destination vertex de-duplication state.
315 * @param[out] keep_t `[tris.n]` flags: triangle stays on this rank.
316 * @param[out] keep_v `[verts.n]` flags: vertex stays on this rank.
317 * @param[out] tri_ids `[nproc]` per-destination triangle index lists.
318 * @param[out] vert_ids `[nproc]` per-destination vertex index lists.
319 * @param[out] touched Destination ranks that received anything.
320 * @param[out] n_touched Number of entries written to @p touched.
321 * @return 1 if any destination lies outside the 26-neighbourhood (forcing the
322 * global route).
323 */
324static int collect_destinations(const MFTL_Object *obj, int my_rank, const char *restrict is_neighbor, const int *restrict tri_owner,
325 const int *restrict vert_owner, vertex_dedup *dd, char *restrict keep_t, char *restrict keep_v, mftl_int_buffer *tri_ids,
326 mftl_int_buffer *vert_ids, int *touched, int *n_touched)
327{
328 MFTL_ASSERT(MFTLObjectHasTopology(obj) && "redistribute: tris.verts/neighbors are stale -- MFTLObjectBuildDerived first");
329
330 const int n_owned = obj->tris.n_local;
331 if (n_owned < 1) return 0;
332
333 /* 12 B per triangle each, streaming (48 B from the half-edge records). */
334 const int *const restrict tv = obj->tris.verts;
335 const int *const restrict tn = obj->tris.neighbors;
336
337 uint32_t *const restrict mask = dd->mask;
338 int *const restrict rank_bit = dd->rank_bit;
339
340 int requires_global = 0;
341
342 for (int t = 0; t < n_owned; ++t)
343 {
344 const int base = 3 * t;
345 const int v0 = tv[base + 0], v1 = tv[base + 1], v2 = tv[base + 2];
346
347 /* Fast path: "is any of the seven candidates different from the owner" is
348 * the OR of six XORs, answered in registers; cand[] (a variable-index
349 * array that cannot live in registers) is built only when the answer is yes.
350 * A missing edge neighbour contributes `r`, so its term vanishes. cand[]
351 * keeps first-seen order, which fixes the wire and arrival order. */
352 const int r = tri_owner[t];
353 const int o0 = vert_owner[v0], o1 = vert_owner[v1], o2 = vert_owner[v2];
354 const int a0 = tn[base + 0], a1 = tn[base + 1], a2 = tn[base + 2];
355 const int r0 = (a0 != MFTL_ERROR) ? tri_owner[a0] : r;
356 const int r1 = (a1 != MFTL_ERROR) ? tri_owner[a1] : r;
357 const int r2 = (a2 != MFTL_ERROR) ? tri_owner[a2] : r;
358
359 int dests[7];
360 int n_dests = 1;
361 dests[0] = r;
362
363 if (((o0 ^ r) | (o1 ^ r) | (o2 ^ r) | (r0 ^ r) | (r1 ^ r) | (r2 ^ r)) != 0)
364 {
365 int cand[7];
366 int n_cand = 0;
367 cand[n_cand++] = r;
368 cand[n_cand++] = o0;
369 cand[n_cand++] = o1;
370 cand[n_cand++] = o2;
371 if (a0 != MFTL_ERROR) cand[n_cand++] = r0;
372 if (a1 != MFTL_ERROR) cand[n_cand++] = r1;
373 if (a2 != MFTL_ERROR) cand[n_cand++] = r2;
374 n_dests = MFTLGetUniqueIntegers(cand, n_cand, dests, 7);
375 }
376
377 for (int k = 0; k < n_dests; ++k)
378 {
379 const int dst = dests[k];
380
381 if (dst == my_rank)
382 {
383 keep_t[t] = 1;
384 keep_v[v0] = 1;
385 keep_v[v1] = 1;
386 keep_v[v2] = 1;
387 continue;
388 }
389
390 if (!is_neighbor[dst]) requires_global = 1;
391
392 int b = rank_bit[dst];
393 if (b == MFTL_ERROR)
394 {
395 touched[(*n_touched)++] = dst; // first contact and first triangle coincide
396 MFTL_ASSERT(tri_ids[dst].size == 0 && "collect_destinations: bit slot and buffer state disagree");
397 b = (dd->n_bits < MFTL_DEDUP_BITS) ? dd->n_bits++ : MFTL_DEDUP_NO_BIT;
398 if (b == MFTL_DEDUP_NO_BIT) MFTLCommStats.n_dedup_overflow++;
399 rank_bit[dst] = b;
400 }
401
402 ib_add(&tri_ids[dst], t);
403
404 if (b == MFTL_DEDUP_NO_BIT)
405 {
406 ib_add3(&vert_ids[dst], v0, v1, v2); // append_destination will sort this one
407 continue;
408 }
409
410 const uint32_t m = (uint32_t) 1 << b;
411 if (!(mask[v0] & m))
412 {
413 mask[v0] |= m;
414 ib_add(&vert_ids[dst], v0);
415 }
416 if (!(mask[v1] & m))
417 {
418 mask[v1] |= m;
419 ib_add(&vert_ids[dst], v1);
420 }
421 if (!(mask[v2] & m))
422 {
423 mask[v2] |= m;
424 ib_add(&vert_ids[dst], v2);
425 }
426 }
427 }
428
429 return requires_global;
430}

436/**
437 * @brief Packs one triangle and its three half-edges into a transfer unit: ten
438 * ints, all of them information the receiver cannot derive.
439 *
440 * @param obj The mesh.
441 * @param t Triangle index.
442 * @param tv `obj->tris.verts`.
443 * @param vgid `[verts.n]` vertex global ids, contiguous, from compute_owners().
444 * @param[out] dst The wire record.
445 * @param body_gid #MFTL_Object::global_id of the owning body.
446 */
447static inline void pack_triangle(const MFTL_Object *obj, int t, const int *restrict tv, const int *restrict vgid, MFTL_PackedTriangleComplex *dst, int body_gid)
448{
449 const MFTL_Triangle *const src = &obj->tris.rec[t];
450 const MFTL_HalfEdge *const he = &obj->tris.he[3 * t];
451 const int base = 3 * t;
452
453 dst->global_id = src->global_id;
454 dst->owner_proc = src->owner_proc;
455 dst->flags = src->flags;
456 dst->body_gid = body_gid;
457
458 /* tris.verts[3t+k] is the ORIGIN of half-edge k; the wire carries its TARGET,
459 * target(k) == origin((k+1)%3). */
460 dst->vertex_gid[0] = vgid[tv[base + 1]];
461 dst->vertex_gid[1] = vgid[tv[base + 2]];
462 dst->vertex_gid[2] = vgid[tv[base + 0]];
463
464 dst->he_flags[0] = he[0].flags;
465 dst->he_flags[1] = he[1].flags;
466 dst->he_flags[2] = he[2].flags;
467
468 MFTL_ASSERT(dst->vertex_gid[0] == obj->verts.rec[he[0].vertex_idx].global_id && "pack_triangle: half-edge target rotation is wrong");
469 MFTL_ASSERT(he[0].global_id == 3 * src->global_id && he[1].global_id == 3 * src->global_id + 1 && he[2].global_id == 3 * src->global_id + 2 &&
470 "pack_triangle: half-edge global ids are not 3*triangle_gid + k -- the receiver reconstructs them from that");
471}

510/**
511 * @brief Packs one object's contribution for one REMOTE destination rank,
512 * appending to that destination's send buffers.
513 *
514 * Triangle lists need no de-duplication (each triangle is appended at most once
515 * per distinct destination, in ascending order); vertex lists do only for a
516 * destination without a de-duplication bit.
517 *
518 * @param obj The mesh.
519 * @param body_gid #MFTL_Object::global_id of the owning body.
520 * @param vgid `[verts.n]` vertex global ids.
521 * @param tri_ids Triangle indices bound for this destination.
522 * @param vert_ids Vertex indices bound for this destination.
523 * @param out_tris Send buffer the packed triangles are appended to.
524 * @param out_verts Send buffer the packed vertices are appended to.
525 * @param needs_dedup `true` when the vertex list must be sorted and uniqued here.
526 */
527static void append_destination(const MFTL_Object *obj, int body_gid, const int *restrict vgid, mftl_int_buffer *tri_ids, mftl_int_buffer *vert_ids,
528 MFTL_SendBuffer *out_tris, MFTL_SendBuffer *out_verts, bool needs_dedup)
529{
530 /* Already strictly increasing: see collect_destinations(). */
531 MFTL_ASSERT(is_strictly_increasing(tri_ids->data, tri_ids->size) && "append_destination: triangle list is not already unique");
532
533 if (tri_ids->size > 0)
534 {
535 sb_reserve(out_tris, tri_ids->size, sizeof(MFTL_PackedTriangleComplex));
536 MFTL_PackedTriangleComplex *const buf = (MFTL_PackedTriangleComplex *) out_tris->buffer;
537 const int *const restrict tv = obj->tris.verts;
538 for (int i = 0; i < tri_ids->size; ++i) pack_triangle(obj, tri_ids->data[i], tv, vgid, &buf[out_tris->count++], body_gid);
539 }
540
541 if (vert_ids->size > 0)
542 {
543 int n = vert_ids->size;
544 if (needs_dedup)
545 {
546 MFTLSortIds(vert_ids->data, n);
547 n = MFTLUniqueIntArray(vert_ids->data, n);
548 }
549
550 sb_reserve(out_verts, n, sizeof(MFTL_PackedVertex));
551 MFTL_PackedVertex *const buf = (MFTL_PackedVertex *) out_verts->buffer;
552 for (int i = 0; i < n; ++i) pack_vertex(obj, vert_ids->data[i], &buf[out_verts->count++], body_gid);
553 }

564 * The kept lists are OLD pool indices in final relative order: the owned run
565 * first, then the ghost run. Both runs are ascending, which keeps the gather in
566 * the rebuild a forward scan over the pool rather than a random walk.
567 */
568typedef struct
569{
570 int *kept_v; ///< [n_kept_v] old vertex indices (arena slice).
571 int n_kept_v_owned; ///< Kept vertices this rank OWNS; they lead the rebuilt prefix.
572 int n_kept_v; ///< Kept vertices in total, owned plus retained ghosts.
573
574 int *kept_t; ///< [n_kept_t] old triangle indices (arena slice).
575 int n_kept_t_owned; ///< Kept triangles this rank OWNS.
576 int n_kept_t; ///< Kept triangles in total.
577
578 /* Arrival slices: indices into the flat receive buffers, grouped by body. */
579 const int *recv_v_idx; ///< [n_recv_v] arriving vertex slots.
580 int n_recv_v; ///< Vertices arriving for this body.
581 const int *recv_t_idx; ///< [n_recv_t] arriving triangle slots.
582 int n_recv_t; ///< Triangles arriving for this body.
583} body_state;

585/**
586 * @brief Splits one entity class's keep flags into the owned run and the ghost
587 * run, both ascending, in ONE pass.
588 *
589 * The owned run is filled forward from 0 and the ghost run backward from the
590 * end of the body's arena slice (sized for the whole class, so the cursors
591 * cannot meet); the ghost run is then reversed and moved down. O(halo) fix-up,
592 * byte-identical to two separate scans. Ownership is read from the dense
593 * `owner` array written by compute_owners() (4 B per entity) rather than from
594 * the AoS records.
595 *
596 * @param keep `[n]` keep flags from collect_destinations().
597 * @param owner `[n]` destination rank per entity.
598 * @param n Entities: `verts.n`, or `tris.n_local` (only owned
599 * triangles can be kept).
600 * @param my_rank This rank.
601 * @param[out] out `[n]` kept indices: owned run then ghost run.
602 * @param[out] n_owned Length of the owned run.
603 * @return Kept entities in total.
604 */
605static int split_kept_run(const char *restrict keep, const int *restrict owner, int n, int my_rank, int *restrict out, int *n_owned)
606{
607 int lo = 0, hi = n;
608
609 for (int i = 0; i < n; ++i)
610 {
611 if (!keep[i]) continue;
612 if (owner[i] == my_rank)
613 out[lo++] = i;
614 else
615 out[--hi] = i;
616 }
617
618 *n_owned = lo;
619 const int n_ghost = n - hi;
620
621 for (int a = 0, b = n_ghost - 1; a < b; ++a, --b)
622 {
623 const int tmp = out[hi + a];
624 out[hi + a] = out[hi + b];
625 out[hi + b] = tmp;
626 }
627 if (n_ghost > 0 && hi != lo) memmove(out + lo, out + hi, (size_t) n_ghost * sizeof(int));
628
629 return lo + n_ghost;
630}

633 * @brief Splits the keep flags into the two ascending runs the rebuild wants.
634 *
635 * @param obj The mesh, for the two class sizes.
636 * @param keep_v `[verts.n]` flags from collect_destinations().
637 * @param keep_t `[tris.n_local]` flags from collect_destinations().
638 * @param vert_owner `[verts.n]` destination rank per vertex.
639 * @param tri_owner `[tris.n]` destination rank per triangle.
640 * @param my_rank This rank.
641 * @param[out] bs Per-body state; its kept lists and counts are filled in.
642 */
643static void build_kept_lists(const MFTL_Object *obj, const char *restrict keep_v, const char *restrict keep_t, const int *restrict vert_owner,
644 const int *restrict tri_owner, int my_rank, body_state *bs)

648}

700 *
701 * Every rank packs body by body, and the (neighbour) alltoallv concatenates the

704 * the headers recovers the runs without sorting the arrivals.

715/**
716 * @brief Resolves a body id off the wire to this call's work-list slot.
717 *
718 * The one place a wire body id becomes a local index. The work list is sorted
719 * for its first @p n_sorted entries (the ensemble's directory) and followed by
720 * the bodies arriving on this rank for the first time; binary search, then a
721 * scan of the short tail.
722 *
723 * @param work The work list: sorted prefix, then arrivals.
724 * @param n_sorted Length of the sorted prefix.
725 * @param n_work Total entries in @p work.
726 * @param gid Body id off the wire.
727 * @return The slot, or #MFTL_ERROR if this call has no such body yet.
728 */
729static inline int slot_of_body(MFTL_Object *const *work, int n_sorted, int n_work, int gid)
730{
731 /* Binary search over the part that came from the sorted directory. */
732 int lo = 0, hi = n_sorted - 1;
733 while (lo <= hi)
734 {
735 const int mid = lo + ((hi - lo) >> 1);
736 const int g = work[mid]->global_id;
737 if (g == gid) return mid;
738 if (g < gid)
739 lo = mid + 1;
740 else
741 hi = mid - 1;
742 }
743
744 /* Then the (short, unsorted) tail of first-time arrivals. */
745 for (int i = n_sorted; i < n_work; ++i)
746 if (work[i]->global_id == gid) return i;
747
748 return MFTL_ERROR;
749}

753 *

862/**
863 * @brief Grows the per-body arrays, PRESERVING `states` (it must survive growth
864 * within a call when arrivals extend the work list) and zeroing its tail.
865 *
866 * @param ws The workspace.
867 * @param n Bodies the arrays must hold.
868 */
869static void ws_bodies_reserve(comm_workspace *ws, int n)
870{
871 if (n <= ws->cap_objects) return;

873 const int c = MFTLGrowCapacity(ws->cap_objects, n);
874 const int had = ws->cap_objects;
875
876 MFTL_Object **const work = (MFTL_Object **) MFTLMallocOrDie((size_t) c * sizeof(MFTL_Object *), "redistribution work list");

893 cs_alloc((size_t) c * (sizeof(MFTL_Object *) + sizeof(body_state) + 3 * sizeof(int)) + 2 * sizeof(int));

896/**
897 * @brief Sizes the workspace for this call; every array only grows. The
898 * `[nproc]` block is built on the first call only.
899 *
900 * @param ws The workspace.
901 * @param nproc Ranks in the communicator.
902 * @param num_objects Bodies this rank holds, before any arrival.
903 * @param sum_verts Total vertices over all bodies, for the kept-list arena.
904 * @param sum_tris Total owned triangles over all bodies.
905 * @param max_verts Largest body's vertex count.
906 * @param max_tris Largest body's triangle count.
907 * @param tri_hint First-touch size for a destination's triangle id list.
908 * @param vert_hint Ditto for vertices.
909 */
910static void ws_reserve(comm_workspace *ws, int nproc, int num_objects, int sum_verts, int sum_tris, int max_verts, int max_tris, int tri_hint, int vert_hint)
911{
912 if (ws->nproc != nproc)
913 {
914 /* Only on the first call for this ensemble, when nothing is allocated yet. */
915 ws_free_rank_arrays(ws);
916
917 ws->is_neighbor = (char *) MFTLCallocOrDie((size_t) nproc, 1, "redistribution neighbour map");
918 ws->tri_ids = (mftl_int_buffer *) MFTLMallocOrDie((size_t) nproc * sizeof(mftl_int_buffer), "redistribution id lists");
919 ws->vert_ids = (mftl_int_buffer *) MFTLMallocOrDie((size_t) nproc * sizeof(mftl_int_buffer), "redistribution id lists");
920 ws->send_tris = (MFTL_SendBuffer *) MFTLCallocOrDie((size_t) nproc, sizeof(MFTL_SendBuffer), "redistribution send buffers");
921 ws->send_verts = (MFTL_SendBuffer *) MFTLCallocOrDie((size_t) nproc, sizeof(MFTL_SendBuffer), "redistribution send buffers");
922 ws->send_counts = (MFTL_SendBuffer *) MFTLCallocOrDie((size_t) nproc, sizeof(MFTL_SendBuffer), "redistribution body counts");
923 ws->touched = (int *) MFTLMallocOrDie((size_t) nproc * sizeof(int), "redistribution touched ranks");
924 ws->rank_bit = (int *) MFTLMallocOrDie((size_t) nproc * sizeof(int), "redistribution dedup bit slots");
925 for (int r = 0; r < nproc; ++r)
926 {
927 ib_init(&ws->tri_ids[r], tri_hint);
928 ib_init(&ws->vert_ids[r], vert_hint);
929 ws->rank_bit[r] = MFTL_ERROR;
930 }
931 ws->nproc = nproc;
932 cs_alloc((size_t) nproc * (1 + 2 * sizeof(mftl_int_buffer) + 3 * sizeof(MFTL_SendBuffer) + 2 * sizeof(int)));
933 }
934 else
935 {
936 /* The hint is derived from the mesh density, which changes as the mesh does.
937 * Only untouched lists can adopt a new one; a grown list keeps its capacity. */
938 for (int r = 0; r < nproc; ++r)
939 {
940 ws->tri_ids[r].hint = tri_hint;
941 ws->vert_ids[r].hint = vert_hint;
942 }
943 }
944
945 ws_bodies_reserve(ws, num_objects);
946
947 ws_bytes((void **) &ws->kept_arena_v, &ws->cap_arena_v, MFTL_MAX(sum_verts, 1), sizeof(int), "redistribution kept vertex lists");
948 ws_bytes((void **) &ws->kept_arena_t, &ws->cap_arena_t, MFTL_MAX(sum_tris, 1), sizeof(int), "redistribution kept triangle lists");
949
950 /* One block, two slices (local votes, reduced votes), sized by bodies held. */
951 ws_bytes((void **) &ws->dirty_votes, &ws->cap_votes, MFTL_MAX(num_objects, 1), 2 * sizeof(double), "redistribution skip votes");
952
953 if (max_verts > ws->cap_verts)
954 {
955 const int c = MFTLGrowCapacity(ws->cap_verts, max_verts);
956 free(ws->vert_owner);
957 free(ws->vert_gid);
958 free(ws->keep_v);
959 free(ws->dedup_mask);
960 ws->vert_owner = (int *) MFTLMallocOrDie((size_t) c * sizeof(int), "redistribution vertex owners");
961 ws->vert_gid = (int *) MFTLMallocOrDie((size_t) c * sizeof(int), "redistribution vertex gids");
962 ws->keep_v = (char *) MFTLMallocOrDie((size_t) c, "redistribution vertex keep flags");
963 ws->dedup_mask = (uint32_t *) MFTLMallocOrDie((size_t) c * sizeof(uint32_t), "redistribution vertex dedup mask");
964 ws->cap_verts = c;
965 cs_alloc((size_t) c * (2 * sizeof(int) + 1 + sizeof(uint32_t)));
966 }
967
968 if (max_tris > ws->cap_tris)
969 {
970 const int c = MFTLGrowCapacity(ws->cap_tris, max_tris);
971 free(ws->tri_owner);
972 free(ws->keep_t);
973 ws->tri_owner = (int *) MFTLMallocOrDie((size_t) c * sizeof(int), "redistribution triangle owners");
974 ws->keep_t = (char *) MFTLMallocOrDie((size_t) c, "redistribution triangle keep flags");
975 ws->cap_tris = c;
976 cs_alloc((size_t) c * (sizeof(int) + 1));
977 }
978
979 /* Footprint by what it scales with, read off the capacities. The payload
980 * buffers are not counted per rank: they are only touched for destinations
981 * that received something. */
982 MFTLCommStats.ws_bytes_per_rank = (long) ((size_t) ws->nproc * (1 + 2 * sizeof(mftl_int_buffer) + 3 * sizeof(MFTL_SendBuffer) + 2 * sizeof(int)));
983
984 /* Every per-body array is sized by bodies HELD, so this should stay flat in a
985 * weak-scaling sweep. */
986 MFTLCommStats.ws_bytes_per_body = (long) ((size_t) ws->cap_objects * (sizeof(MFTL_Object *) + sizeof(body_state) + 3 * sizeof(int)) + 2 * sizeof(int) +
987 (size_t) ws->cap_votes * 2 * sizeof(double));
988
989 MFTLCommStats.ws_bytes_per_entity = (long) ((size_t) ws->cap_arena_v * sizeof(int) + (size_t) ws->cap_arena_t * sizeof(int) +
990 (size_t) ws->cap_verts * (2 * sizeof(int) + 1 + sizeof(uint32_t)) + (size_t) ws->cap_tris * (sizeof(int) + 1));
991}

1051/**
1052 * @brief Re-pairs exactly the half-edges that lost their twin.
1053 *
1054 * The bucketing of MFTLRebuildTwins() restricted to the repair set (closed under
1055 * the twin relation, see the file comment); only unlinked candidates are paired,
1056 * keeping the relation symmetric on non-manifold input. The matching
1057 * `tris.neighbors` entries are written with each linked pair.
1058 *
1059 * @param obj The mesh being rebuilt.
1060 * @param u_list Candidate half-edges whose twin is unresolved.
1061 * @param n_u Number of entries in @p u_list.
1062 * @param v_head `[verts.n]` per-vertex bucket heads, scratch.
1063 * @param u_next `[n_u]` bucket chain links, scratch.
1064 */
1065static void repair_twins(MFTL_Object *obj, const int *restrict u_list, int n_u, int *restrict v_head, int *restrict u_next)
1066{
1067 MFTL_HalfEdge *const he = obj->tris.he;
1068 int *const restrict tn = obj->tris.neighbors;
1069
1070 for (int j = 0; j < n_u; ++j) v_head[he[MFTLHePrevIdx(u_list[j])].vertex_idx] = MFTL_ERROR;
1071
1072 for (int j = 0; j < n_u; ++j)
1073 {
1074 const int origin = he[MFTLHePrevIdx(u_list[j])].vertex_idx;
1075 u_next[j] = v_head[origin];
1076 v_head[origin] = j;
1077 }
1078
1079 for (int j = 0; j < n_u; ++j)
1080 {
1081 const int h = u_list[j];
1082 if (he[h].twin_idx != MFTL_ERROR) continue; // already linked by its partner
1083
1084 const int u = he[MFTLHePrevIdx(h)].vertex_idx;
1085 const int v = he[h].vertex_idx;
1086
1087 for (int c = v_head[v]; c != MFTL_ERROR; c = u_next[c])
1088 {
1089 const int h2 = u_list[c];
1090 if (he[h2].vertex_idx != u || he[h2].twin_idx != MFTL_ERROR) continue;
1091
1092 he[h].twin_idx = h2;
1093 he[h2].twin_idx = h;
1094 tn[h] = MFTLHeTriIdx(h2);
1095 tn[h2] = MFTLHeTriIdx(h);
1096 break;
1097 }
1098 }
1099}

1101/**
1102 * @brief Rebuilds one object from its kept entities and its arrivals (no sort,
1103 * no full twin rebuild).
1104 *
1105 * Steps: (0) a body with nothing kept and nothing arriving is emptied and its
1106 * pools released; (1) gather kept records and positions; (2) de-duplicate the
1107 * arrivals by global id (a vertex arrives once per sender sharing it, and the
1108 * copies are identical; a triangle has one owner and cannot arrive twice);
1109 * (3) map logical to final slots; (4) size the pools (geometric pre-growth, so
1110 * jittering counts do not reallocate every step) and place the kept entities;
1111 * (5-6) remap kept half-edges, retiring twins that did not survive into the
1112 * repair list, and emit `out_he_idx` and the derived arrays inline; (7) write
1113 * the arrivals; (8) repair twins.
1114 *
1115 * @param obj The mesh, rewritten in place. On return: valid topology and
1116 * positions, #MFTL_DERIVED_TOPOLOGY set, cached geometry
1117 * unspecified, `partition_dirty` cleared.
1118 * @param bs Per-body state from build_kept_lists().
1119 * @param recv_v Received vertex records.
1120 * @param recv_t Received triangle records.
1121 * @param my_rank This rank.
1122 * @param s Reusable scratch, grown monotonically across calls.
1123 */
1124static void rebuild_object(MFTL_Object *obj, const body_state *bs, const MFTL_PackedVertex *recv_v, const MFTL_PackedTriangleComplex *recv_t, int my_rank,
1125 rebuild_scratch *s)
1126{
1127 const int n_old_v = obj->verts.n;
1128 const int n_old_t = obj->tris.n;
1129
1130 const int n_kept_v = bs->n_kept_v;
1131 const int n_kept_t = bs->n_kept_t;
1132
1133 /* ── 0. Bodies this rank no longer has any part of: emptied here, destroyed
1134 * at the end of the call. */
1135 if (n_kept_v == 0 && n_kept_t == 0 && bs->n_recv_v == 0 && bs->n_recv_t == 0)
1136 {
1137 MFTLCommStats.n_objects_empty_local++;
1138
1139 if (n_old_v != 0 || n_old_t != 0)
1140 {
1141 /* It has left entirely: release its pools (after generation this recovers
1142 * most of rank 0's footprint -- measured 80 MB -> 33.5 MB at 256 bodies,
1143 * refine 3, np = 4). */
1144 MFTLObjectSetCounts(obj, 0, 0, 0, 0);
1145 MFTLObjectReleasePools(obj);
1146 }
1147
1148 /* An empty mesh is trivially described. */
1149 obj->derived = MFTL_DERIVED_TOPOLOGY;
1150 /* This call is the rebuild the dirty stamp asked for. */
1151 obj->partition_dirty = MFTL_FALSE;
1152 return;
1153 }
1154
1155 CS_TIMELINE(tm);
1156 MFTLCommStats.n_verts_kept += n_kept_v;
1157 MFTLCommStats.n_tris_kept += n_kept_t;
1158 MFTLCommStats.n_verts_received += bs->n_recv_v;
1159 MFTLCommStats.n_tris_received += bs->n_recv_t;
1160
1161 /* ── 1. Gather the kept records and positions (geometry is not copied: it is
1162 * unspecified after a redistribution). Also finds the largest kept id, which
1163 * gt_begin() needs to choose its indexing mode. */
1164 ensure_bytes(&s->v_rec, &s->v_rec_cap, (size_t) MFTL_MAX(n_kept_v, 1) * sizeof(MFTL_Vertex));
1165 ensure_bytes(&s->v_pos, &s->v_pos_cap, (size_t) MFTL_MAX(n_kept_v, 1) * 3 * sizeof(double));
1166 ensure_bytes(&s->t_rec, &s->t_rec_cap, (size_t) MFTL_MAX(n_kept_t, 1) * sizeof(MFTL_Triangle));
1167 ensure_bytes(&s->t_he, &s->t_he_cap, (size_t) MFTL_MAX(n_kept_t, 1) * 3 * sizeof(MFTL_HalfEdge));
1168
1169 MFTL_Vertex *const kv_rec = (MFTL_Vertex *) s->v_rec;
1170 double *const kv_pos = (double *) s->v_pos;
1171 MFTL_Triangle *const kt_rec = (MFTL_Triangle *) s->t_rec;
1172 MFTL_HalfEdge *const kt_he = (MFTL_HalfEdge *) s->t_he;
1173
1174 int v_range = obj->next_vertex_gid;
1175 int t_range = obj->next_triangle_gid;
1176
1177 for (int i = 0; i < n_kept_v; ++i)
1178 {
1179 const int o = bs->kept_v[i];
1180 kv_rec[i] = obj->verts.rec[o];
1181 if (kv_rec[i].global_id >= v_range) v_range = kv_rec[i].global_id + 1;
1182 /* Re-derived below; the old value indexes the old pool. */
1183 kv_rec[i].out_he_idx = MFTL_ERROR;
1184 kv_pos[3 * i + 0] = obj->verts.pos[3 * o + 0];
1185 kv_pos[3 * i + 1] = obj->verts.pos[3 * o + 1];
1186 kv_pos[3 * i + 2] = obj->verts.pos[3 * o + 2];
1187 }
1188
1189 for (int i = 0; i < n_kept_t; ++i)
1190 {
1191 const int o = bs->kept_t[i];
1192 kt_rec[i] = obj->tris.rec[o];
1193 kt_he[3 * i + 0] = obj->tris.he[3 * o + 0];
1194 kt_he[3 * i + 1] = obj->tris.he[3 * o + 1];
1195 kt_he[3 * i + 2] = obj->tris.he[3 * o + 2];
1196 if (kt_rec[i].global_id >= t_range) t_range = kt_rec[i].global_id + 1;
1197 }
1198
1199 CS_LAP(tm, t_kept_gather);
1200
1201 /* ── 2. Classify the arrivals against the kept entities ──────────────────── */
1202
1203 /* Entry count and id range decide the table's indexing mode (comm_internal.h). */
1204 for (int i = 0; i < bs->n_recv_v; ++i)
1205 {
1206 const int g = recv_v[bs->recv_v_idx[i]].global_id;
1207 if (g >= v_range) v_range = g + 1;
1208 }
1209 for (int i = 0; i < bs->n_recv_t; ++i)
1210 {
1211 const int g = recv_t[bs->recv_t_idx[i]].global_id;
1212 if (g >= t_range) t_range = g + 1;
1213 }
1214
1215 gt_begin(&s->v_tbl, n_kept_v + bs->n_recv_v, MFTL_MAX(v_range, 1));
1216 gt_begin(&s->t_tbl, n_kept_t + bs->n_recv_t, MFTL_MAX(t_range, 1));
1217 for (int i = 0; i < n_kept_v; ++i) gt_put(&s->v_tbl, kv_rec[i].global_id, i);
1218 for (int i = 0; i < n_kept_t; ++i) gt_put(&s->t_tbl, kt_rec[i].global_id, i);
1219
1220 ensure_ints(&s->recv_v_logical, &s->recv_v_logical_cap, MFTL_MAX(bs->n_recv_v, 1));
1221 ensure_ints(&s->recv_t_logical, &s->recv_t_logical_cap, MFTL_MAX(bs->n_recv_t, 1));
1222
1223 /* A vertex arrives once per sender sharing it; the copies are bit-identical,
1224 * so the first wins. */
1225 int n_new_v = 0, n_new_v_owned = 0;
1226 for (int i = 0; i < bs->n_recv_v; ++i)
1227 {
1228 const MFTL_PackedVertex *const p = &recv_v[bs->recv_v_idx[i]];
1229 if (gt_has(&s->v_tbl, p->global_id))
1230 {
1231 s->recv_v_logical[i] = MFTL_ERROR;
1232 continue;
1233 }
1234 const int L = n_kept_v + n_new_v++;
1235 gt_put(&s->v_tbl, p->global_id, L);
1236 s->recv_v_logical[i] = L;
1237 if (p->owner_proc == my_rank) n_new_v_owned++;
1238 }
1239
1240 /* Triangles cannot duplicate (only the single owner forwards one); the check
1241 * guards against silent corruption if that is ever violated. */
1242 int n_new_t = 0, n_new_t_owned = 0;
1243 for (int i = 0; i < bs->n_recv_t; ++i)
1244 {
1245 const MFTL_PackedTriangleComplex *const p = &recv_t[bs->recv_t_idx[i]];
1246 if (gt_has(&s->t_tbl, p->global_id))
1247 {
1248 MFTL_ASSERT(0 && "redistribute: the same triangle arrived twice -- two ranks believe they own it");
1249 s->recv_t_logical[i] = MFTL_ERROR;
1250 continue;
1251 }
1252 const int L = n_kept_t + n_new_t++;
1253 gt_put(&s->t_tbl, p->global_id, L);
1254 s->recv_t_logical[i] = L;
1255 if (p->owner_proc == my_rank) n_new_t_owned++;
1256 }
1257
1258 /* ── 3. Logical -> pool slot ─────────────────────────────────────────────── */
1259 ensure_ints(&s->v_final_of_logical, &s->v_final_of_logical_cap, MFTL_MAX(n_kept_v + n_new_v, 1));
1260 ensure_ints(&s->t_final_of_logical, &s->t_final_of_logical_cap, MFTL_MAX(n_kept_t + n_new_t, 1));
1261
1262 /* Ownership of an arrival is re-read from the payload (same visiting order). */
1263 {
1264 const int n_kept_v_ghost = n_kept_v - bs->n_kept_v_owned;
1265 for (int L = 0; L < bs->n_kept_v_owned; ++L) s->v_final_of_logical[L] = L;
1266 for (int L = bs->n_kept_v_owned; L < n_kept_v; ++L) s->v_final_of_logical[L] = L + n_new_v_owned;
1267
1268 int o = bs->n_kept_v_owned;
1269 int g = bs->n_kept_v_owned + n_new_v_owned + n_kept_v_ghost;
1270 for (int i = 0; i < bs->n_recv_v; ++i)
1271 {
1272 const int L = s->recv_v_logical[i];
1273 if (L == MFTL_ERROR) continue;
1274 const MFTL_PackedVertex *const p = &recv_v[bs->recv_v_idx[i]];
1275 s->v_final_of_logical[L] = (p->owner_proc == my_rank) ? o++ : g++;
1276 }
1277 MFTL_ASSERT(o == bs->n_kept_v_owned + n_new_v_owned && "rebuild_object: owned vertex block overrun");
1278 }
1279
1280 {
1281 const int n_kept_t_ghost = n_kept_t - bs->n_kept_t_owned;
1282 for (int L = 0; L < bs->n_kept_t_owned; ++L) s->t_final_of_logical[L] = L;
1283 for (int L = bs->n_kept_t_owned; L < n_kept_t; ++L) s->t_final_of_logical[L] = L + n_new_t_owned;
1284
1285 int o = bs->n_kept_t_owned;
1286 int g = bs->n_kept_t_owned + n_new_t_owned + n_kept_t_ghost;
1287 for (int i = 0; i < bs->n_recv_t; ++i)
1288 {
1289 const int L = s->recv_t_logical[i];
1290 if (L == MFTL_ERROR) continue;
1291 const MFTL_PackedTriangleComplex *const p = &recv_t[bs->recv_t_idx[i]];
1292 s->t_final_of_logical[L] = (p->owner_proc == my_rank) ? o++ : g++;
1293 }
1294 MFTL_ASSERT(o == bs->n_kept_t_owned + n_new_t_owned && "rebuild_object: owned triangle block overrun");
1295 }
1296
1297 const int n_v = n_kept_v + n_new_v;
1298 const int n_v_local = bs->n_kept_v_owned + n_new_v_owned;
1299 const int n_t = n_kept_t + n_new_t;
1300 const int n_t_local = bs->n_kept_t_owned + n_new_t_owned;
1301
1302 /* ── 4. Size the pools, then place the kept entities ─────────────────────── */
1303 CS_LAP(tm, t_arrivals);
1304
1305 /* Grow GEOMETRICALLY before MFTLObjectSetCounts() (whose reserve is exact),
1306 * so counts jittering under advection do not reallocate both pools every
1307 * step. This leaves capacity in [n, 2n), which is why the shrink gate in
1308 * phase 5b is 4 and not 2 (MFTL_SHRINK_SLACK_DEFAULT). */
1309 if (n_v > obj->verts.capacity || n_t > obj->tris.capacity)
1310 {
1311 MFTLObjectReserve(obj, MFTLGrowCapacity(obj->verts.capacity, n_v), MFTLGrowCapacity(obj->tris.capacity, n_t));
1312 }
1313 MFTLObjectSetCounts(obj, n_v, n_v_local, n_t, n_t_local);
1314 CS_LAP(tm, t_reserve);
1315
1316 {
1317 const int a = bs->n_kept_v_owned;
1318 const int b = n_kept_v - a;
1319 if (a > 0)
1320 {
1321 memcpy(&obj->verts.rec[0], &kv_rec[0], (size_t) a * sizeof(MFTL_Vertex));
1322 memcpy(&obj->verts.pos[0], &kv_pos[0], (size_t) a * 3 * sizeof(double));
1323 }
1324 if (b > 0)
1325 {
1326 const int dst = a + n_new_v_owned;
1327 memcpy(&obj->verts.rec[dst], &kv_rec[a], (size_t) b * sizeof(MFTL_Vertex));
1328 memcpy(&obj->verts.pos[3 * dst], &kv_pos[3 * a], (size_t) b * 3 * sizeof(double));
1329 }
1330 }
1331 {
1332 const int a = bs->n_kept_t_owned;
1333 const int b = n_kept_t - a;
1334 if (a > 0)
1335 {
1336 memcpy(&obj->tris.rec[0], &kt_rec[0], (size_t) a * sizeof(MFTL_Triangle));
1337 memcpy(&obj->tris.he[0], &kt_he[0], (size_t) a * 3 * sizeof(MFTL_HalfEdge));
1338 }
1339 if (b > 0)
1340 {
1341 const int dst = a + n_new_t_owned;
1342 memcpy(&obj->tris.rec[dst], &kt_rec[a], (size_t) b * sizeof(MFTL_Triangle));
1343 memcpy(&obj->tris.he[3 * dst], &kt_he[3 * a], (size_t) b * 3 * sizeof(MFTL_HalfEdge));
1344 }
1345 }
1346
1347 CS_LAP(tm, t_kept_place);
1348
1349 /* ── 5. Old pool index -> new pool slot, for the kept entities ───────────── */
1350 ensure_ints(&s->v_final_of_old, &s->v_final_of_old_cap, MFTL_MAX(n_old_v, 1));
1351 ensure_ints(&s->t_final_of_old, &s->t_final_of_old_cap, MFTL_MAX(n_old_t, 1));
1352
1353 /* The triangle sentinel is load-bearing (it retires twins whose triangle was
1354 * not kept); the vertex one is read only by an assertion, because a kept
1355 * triangle's corners are always kept -- so it is cleared in Debug only. */
1356#ifndef NDEBUG
1357 for (int v = 0; v < n_old_v; ++v) s->v_final_of_old[v] = MFTL_ERROR;
1358#endif
1359 for (int t = 0; t < n_old_t; ++t) s->t_final_of_old[t] = MFTL_ERROR;
1360 for (int i = 0; i < n_kept_v; ++i) s->v_final_of_old[bs->kept_v[i]] = s->v_final_of_logical[i];
1361 for (int i = 0; i < n_kept_t; ++i) s->t_final_of_old[bs->kept_t[i]] = s->t_final_of_logical[i];
1362
1363 /* ── 6. Re-link the kept half-edges: remap the target vertex, remap or retire
1364 * the twin (retired twins form the repair list for free), re-derive
1365 * `out_he_idx`, and emit `tris.verts` / `tris.neighbors` -- whose inputs are
1366 * in registers here -- instead of a closing MFTLObjectBuildDerived() pass.
1367 * The rotation is thus spelled in two places; inv_derived_matches_topology()
1368 * (tests/test_mesh_invariants.h) pins them together on every redistributed
1369 * mesh. Every slot is written exactly once, by forward streams. */
1370 ib_reset(&s->u_list);
1371 MFTL_HalfEdge *const he = obj->tris.he;
1372 MFTL_Vertex *const vr = obj->verts.rec;
1373 int *const restrict tv = obj->tris.verts;
1374 int *const restrict tn = obj->tris.neighbors;
1375
1376 for (int i = 0; i < n_kept_t; ++i)
1377 {
1378 const int slot = s->t_final_of_logical[i];
1379 MFTL_HalfEdge *const h3 = &he[3 * slot];
1380
1381 for (int k = 0; k < 3; ++k)
1382 {
1383 const int nv_idx = s->v_final_of_old[h3[k].vertex_idx];
1384 MFTL_ASSERT(nv_idx != MFTL_ERROR && "rebuild_object: a kept triangle references a vertex that was not kept");
1385 h3[k].vertex_idx = nv_idx;
1386
1387 const int tw = h3[k].twin_idx;
1388 const int tw_tri = (tw != MFTL_ERROR) ? s->t_final_of_old[tw / 3] : MFTL_ERROR;
1389 if (tw_tri != MFTL_ERROR)
1390 {
1391 /* Keep the corner h % 3, remap the triangle h / 3. */
1392 h3[k].twin_idx = 3 * tw_tri + tw % 3;
1393 }
1394 else
1395 {
1396 h3[k].twin_idx = MFTL_ERROR;
1397 ib_add(&s->u_list, 3 * slot + k);
1398 }
1399
1400 /* twin_idx / 3, or MFTL_ERROR until repair_twins() finds a partner. */
1401 tn[3 * slot + k] = tw_tri;
1402 }
1403
1404 /* origin(3s+k) == target(3s+(k+2)%3), and all three targets are final now. */
1405 vr[h3[2].vertex_idx].out_he_idx = 3 * slot + 0;
1406 vr[h3[0].vertex_idx].out_he_idx = 3 * slot + 1;
1407 vr[h3[1].vertex_idx].out_he_idx = 3 * slot + 2;
1408
1409 /* Slot k is the ORIGIN of half-edge 3s+k (as in build_topology_arrays()). */
1410 tv[3 * slot + 0] = h3[2].vertex_idx;
1411 tv[3 * slot + 1] = h3[0].vertex_idx;
1412 tv[3 * slot + 2] = h3[1].vertex_idx;
1413 }
1414
1415 CS_LAP(tm, t_relink);
1416
1417 /* ── 7. Write the arrivals ───────────────────────────────────────────────── */
1418 for (int i = 0; i < bs->n_recv_v; ++i)
1419 {
1420 const int L = s->recv_v_logical[i];
1421 if (L == MFTL_ERROR) continue;
1422
1423 const MFTL_PackedVertex *const p = &recv_v[bs->recv_v_idx[i]];
1424 const int slot = s->v_final_of_logical[L];
1425
1426 obj->verts.pos[3 * slot + 0] = p->pos[0];
1427 obj->verts.pos[3 * slot + 1] = p->pos[1];
1428 obj->verts.pos[3 * slot + 2] = p->pos[2];
1429
1430 MFTL_Vertex *const rec = &vr[slot];
1431 rec->global_id = p->global_id;
1432 rec->owner_proc = p->owner_proc;
1433 rec->flags = p->flags;
1434 rec->out_he_idx = MFTL_ERROR; // filled in by the triangle pass below
1435 }
1436
1437 for (int i = 0; i < bs->n_recv_t; ++i)
1438 {
1439 const int L = s->recv_t_logical[i];
1440 if (L == MFTL_ERROR) continue;
1441
1442 const MFTL_PackedTriangleComplex *const p = &recv_t[bs->recv_t_idx[i]];
1443 const int slot = s->t_final_of_logical[L];
1444
1445 MFTL_Triangle *const rec = &obj->tris.rec[slot];
1446 rec->global_id = p->global_id;
1447 rec->owner_proc = p->owner_proc;
1448 rec->flags = p->flags;
1449
1450 MFTL_HalfEdge *const h3 = &he[3 * slot];
1451 for (int k = 0; k < 3; ++k)
1452 {
1453 MFTL_ASSERT(gt_has(&s->v_tbl, p->vertex_gid[k]) &&
1454 "rebuild_object: a received triangle references a vertex that never arrived -- the sender's vertex list is incomplete");
1455 h3[k].vertex_idx = s->v_final_of_logical[gt_get(&s->v_tbl, p->vertex_gid[k])];
1456 /* Reconstructed, not transmitted: half-edge 3t+k carries 3*tri_gid + k. */
1457 h3[k].global_id = 3 * p->global_id + k;
1458 h3[k].flags = p->he_flags[k];
1459 h3[k].twin_idx = MFTL_ERROR;
1460 tn[3 * slot + k] = MFTL_ERROR; // every arriving twin is resolved by repair_twins
1461 ib_add(&s->u_list, 3 * slot + k);
1462 }
1463
1464 vr[h3[2].vertex_idx].out_he_idx = 3 * slot + 0;
1465 vr[h3[0].vertex_idx].out_he_idx = 3 * slot + 1;
1466 vr[h3[1].vertex_idx].out_he_idx = 3 * slot + 2;
1467
1468 tv[3 * slot + 0] = h3[2].vertex_idx;
1469 tv[3 * slot + 1] = h3[0].vertex_idx;
1470 tv[3 * slot + 2] = h3[1].vertex_idx;
1471 }
1472
1473 /* ── 8. Repair, then rebuild what is genuinely derived ───────────────────── */
1474 CS_LAP(tm, t_arrivals);
1475
1476 if (s->u_list.size > 0)
1477 {
1478 ensure_ints(&s->v_head, &s->v_head_cap, MFTL_MAX(n_v, 1));
1479 ensure_ints(&s->u_next, &s->u_next_cap, s->u_list.size);
1480 repair_twins(obj, s->u_list.data, s->u_list.size, s->v_head, s->u_next);
1481 }
1482
1483#ifndef NDEBUG
1484 for (int v = 0; v < n_v; ++v)
1485 {
1486 MFTL_ASSERT(vr[v].out_he_idx != MFTL_ERROR && "rebuild_object: a vertex ended up with no incident triangle");
1487 }
1488#endif
1489
1490 CS_LAP(tm, t_twins);
1491
1492 /* Steps 6-8 emitted both derived arrays; restore the bit SetCounts cleared. */
1493 obj->derived = MFTL_DERIVED_TOPOLOGY;
1494
1495 /* Cleared last: MFTLObjectSetCounts() set it on the way through. */
1496 obj->partition_dirty = MFTL_FALSE;
1497}

1499/* ═══════════════════════════════════════════════════════════════════════════
1500 * 6. Multi-body: separating the interleaved arrival stream
1501 * ═══════════════════════════════════════════════════════════════════════════ */
1502
1503/**
1504 * @brief Ints in the per-(body, destination) HEADER payload.
1505 *
1506 * `{gid, n_tris, n_verts}` name and delimit the run; the four global fields
1507 * (`next_*_gid`, `global_n_*`) let a rank that held nothing of a body create it
1508 * with correct id counters (arrival tables are sized from them). They are the
1509 * same on every holder, so a receiver that already has the body asserts them.
1510 */

1528/**
1529 * @brief Groups the arrival runs by object with a counting sort (no comparisons);
1530 * within an object the arrival order is preserved, which makes "the first
1531 * copy of a vertex wins" well defined.
1532 *
1533 * @param runs Per-sender runs of arriving entities.
1534 * @param n_runs Number of runs.
1535 * @param num_objects Objects in the work list.
1536 * @param use_tris `true` to group the triangle runs, `false` for vertices.
1537 * @param[out] out_off `[num_objects+1]` CSR offsets into @p out_idx.
1538 * @param[out] out_idx Arrival indices, grouped by object.
1539 * @param cursor `[num_objects]` scratch write cursors.
1540 * @return The number of indices written to @p out_idx.
1541 */
1542static int group_runs_by_object(const body_run *runs, int n_runs, int num_objects, bool use_tris, int *out_off, int *out_idx, int *cursor)
1543{
1544 for (int o = 0; o <= num_objects; ++o) out_off[o] = 0;
1545
1546 for (int k = 0; k < n_runs; ++k)
1547 {
1548 const int o = runs[k].obj_idx;
1549 if (o != MFTL_ERROR) out_off[o + 1] += use_tris ? runs[k].tri_len : runs[k].vert_len;
1550 }
1551
1552 for (int o = 0; o < num_objects; ++o) out_off[o + 1] += out_off[o];
1553 for (int o = 0; o < num_objects; ++o) cursor[o] = out_off[o];
1554
1555 /* Arrival order is preserved within each object's slice. */
1556 for (int k = 0; k < n_runs; ++k)
1557 {
1558 const int o = runs[k].obj_idx;
1559 if (o == MFTL_ERROR) continue;
1560
1561 const int start = use_tris ? runs[k].tri_start : runs[k].vert_start;
1562 const int len = use_tris ? runs[k].tri_len : runs[k].vert_len;
1563 int p = cursor[o];
1564 for (int i = 0; i < len; ++i) out_idx[p + i] = start + i;
1565 cursor[o] = p + len;
1566 }
1567
1568 return out_off[num_objects];
1569}

1571/* ════════════════════════════════════════════════════════════════════════════
1572 * 6b. Leaving a body alone
1573 *
1574 * A rank's state after a redistribution is a pure function of the global
1575 * topology and the owner map: read backwards, the halo rule DEFINES the output
1576 * ("rank r holds t iff r owns t, one of its corners or one of its edge
1577 * neighbours", plus the corners of what it holds). Positions only decide the
1578 * owner map. So a body for which neither input changed already holds the
1579 * answer, and its whole rebuild can be skipped (docs/analysis.md §2.4).
1580 *
1581 * The precondition is NOT "did topology change": under the remesher's freeze
1582 * discipline a topology change touches only unreplicated entities and is
1583 * invisible to the halo rule. What a skip breaks is GLOBAL IDS and FLAGS of
1584 * replicated entities -- e.g. renumbering leaves ghost-triangle ids to be
1585 * repaired by the rebuild. Hence: skip iff no entity changed owner AND the body
1586 * is not `partition_dirty`.
1587 *
1588 * The decision needs agreement (a rank may be the receiver of a change it
1589 * cannot see), and one collective per body would cost more than it saves. So
1590 * the object loop is split: pass 1a classifies every body and packs those this
1591 * rank saw change; ONE per-body MAX reduction over each body's holders agrees
1592 * the votes; pass 1c packs the bodies dirty elsewhere, re-running the idempotent
1593 * compute_owners() instead of storing owner arrays between the passes.
1594 *
1595 * rebuild_object() is not idempotent: an owned vertex whose incident triangles
1596 * are all remote is dropped and re-received at the end of the owned run, so a
1597 * second rebuild of an unchanged body can still move entities. A skip is
1598 * therefore taken only when the body was clean in this call AND the previous one
1599 * (D = "some rank saw it change"):
1600 *
1601 * D now D before action layout afterwards
1602 * 1 - rebuild not necessarily a fixed point
1603 * 0 1 rebuild ("settle") a fixed point
1604 * 0 0 SKIP unchanged, still a fixed point
1605 *
1606 * which makes a skipped pool identical, slot for slot, to a rebuilt one, at the
1607 * cost of one rebuild per dirty-to-clean transition. "D before" is
1608 * MFTL_Object::partition_settled, so a body created on arrival starts unsettled
1609 * without affecting any other body.
1610 * ════════════════════════════════════════════════════════════════════════════ */
1611
1612/** @brief Whether unchanged bodies may be left alone. See MFTLRedistributeSetSkipUnchangedBodies(). */
1613static bool g_skip_unchanged = MFTL_TRUE;
1614
1615void MFTLRedistributeSetSkipUnchangedBodies(bool on) { g_skip_unchanged = on; }
1616
1617bool MFTLRedistributeGetSkipUnchangedBodies(void) { return g_skip_unchanged; }

1619/**
1620 * @brief Waste factor at which a rebuilt body's pools are given back; 0 = off.
1621 *
1622 * Not part of the Debug consistency checksum: it changes no message and no mesh,
1623 * so ranks may legitimately differ.
1624 */
1625static double g_shrink_slack = 0.0;
1626
1627void MFTLRedistributeSetShrinkSlack(double slack) { g_shrink_slack = (slack > 0.0) ? slack : 0.0; }
1628
1629double MFTLRedistributeGetShrinkSlack(void) { return g_shrink_slack; }

1631/**
1632 * @brief Destination analysis, kept lists and packing for ONE body -- shared by
1633 * both passes of the object loop. compute_owners() must have run on
1634 * @p obj immediately before.
1635 *
1636 * @param ws The workspace; supplies every scratch array.
1637 * @param obj The mesh (its global id is the body name on the wire).
1638 * @param my_rank This rank.
1639 * @param dd Per-destination vertex de-duplication state.
1640 * @param[out] bs This body's kept lists and counts.
1641 * @return 1 if any destination lies outside the 26-neighbourhood.
1642 */
1643static int analyze_and_pack_body(comm_workspace *ws, MFTL_Object *obj, int my_rank, vertex_dedup *dd, body_state *bs)
1644{
1645 CS_TIMELINE(tm);
1646
1647 memset(ws->keep_v, 0, (size_t) obj->verts.n);
1648 memset(ws->keep_t, 0, (size_t) obj->tris.n_local);
1649 memset(dd->mask, 0, (size_t) obj->verts.n * sizeof(uint32_t));
1650 dd->n_bits = 0;
1651 CS_LAP(tm, t_masks);
1652
1653 int n_touched = 0;
1654 const int requires_global = collect_destinations(obj, my_rank, ws->is_neighbor, ws->tri_owner, ws->vert_owner, dd, ws->keep_t, ws->keep_v, ws->tri_ids,
1655 ws->vert_ids, ws->touched, &n_touched);
1656
1657 build_kept_lists(obj, ws->keep_v, ws->keep_t, ws->vert_owner, ws->tri_owner, my_rank, bs);
1658 CS_LAP(tm, t_destinations);
1659
1660 MFTLCommStats.n_destinations += n_touched;
1661
1662 /* Only the ranks this object actually reached (at most 27 on the neighbour
1663 * route), not all nproc. */
1664 for (int i = 0; i < n_touched; ++i)
1665 {
1666 const int r = ws->touched[i];
1667 const int tri_before = ws->send_tris[r].count;
1668 const int vert_before = ws->send_verts[r].count;
1669
1670 append_destination(obj, obj->global_id, ws->vert_gid, &ws->tri_ids[r], &ws->vert_ids[r], &ws->send_tris[r], &ws->send_verts[r],
1671 dd->rank_bit[r] == MFTL_DEDUP_NO_BIT);
1672
1673 sb_reserve(&ws->send_counts[r], MFTL_BODY_COUNT_INTS, sizeof(int));
1674 int *const c = (int *) ws->send_counts[r].buffer + ws->send_counts[r].count;
1675 c[MFTL_BODY_HDR_GID] = obj->global_id; // the body GID, not this rank's slot for it
1676 c[MFTL_BODY_HDR_N_TRIS] = ws->send_tris[r].count - tri_before;
1677 c[MFTL_BODY_HDR_N_VERTS] = ws->send_verts[r].count - vert_before;
1678 c[MFTL_BODY_HDR_NEXT_VGID] = obj->next_vertex_gid;
1679 c[MFTL_BODY_HDR_NEXT_TGID] = obj->next_triangle_gid;
1680 c[MFTL_BODY_HDR_GLOBAL_V] = obj->global_n_vertices;
1681 c[MFTL_BODY_HDR_GLOBAL_T] = obj->global_n_triangles;
1682 ws->send_counts[r].count += MFTL_BODY_COUNT_INTS;
1683
1684 ib_reset(&ws->tri_ids[r]);
1685 ib_reset(&ws->vert_ids[r]);
1686 dd->rank_bit[r] = MFTL_ERROR;
1687 }
1688
1689 CS_LAP(tm, t_pack);
1690 return requires_global;
1691}

1695 * @brief Debug-only: do all ranks agree on the size of the body-id space and on
1696 * the skip switch?
1697 *
1698 * Equal id spaces are the observable consequence of "every rank added the same
1699 * bodies in the same order"; disagreement would merge entities into the wrong
1700 * body. The skip switch changes the number of collectives, so disagreement

1703 *
1704 * @param dom Supplies the communicator.
1705 * @param n_bodies_global Size of the body-id space.

1709
1710 unsigned sig = (unsigned) n_bodies_global * 2u + (g_skip_unchanged ? 1u : 0u);
1711
1712
1713 unsigned v[2] = {sig, ~sig};
1714 MPI_Allreduce(MPI_IN_PLACE, v, 2, MPI_UNSIGNED, MPI_MIN, dom->comm_world);
1715 MFTL_ASSERT(v[0] == sig && v[1] == ~sig &&
1716 "redistribute: ranks disagree about the body-id space or about the skip switch -- every rank must add the same bodies in the same "
1717 "order, and MFTLRedistributeSetSkipUnchangedBodies is collective");
1718}

1722 * 7. The driver
1723 * ═══════════════════════════════════════════════════════════════════════════ */
1724
1725/**
1726 * @brief The redistribution implementation behind both public entry points:
1727 * setup, pass 1a, skip vote, pass 1c, route decision, exchange, arrival
1728 * grouping, rebuild, optional shrink, directory update.
1729 *
1730 * @param ens The ensemble: its domain, local object list and persistent
1731 * workspace.
1732 * @param maps `[ens->n_local_objects]` explicit owner maps in local-slot order,
1733 * or NULL for the geometric rule.
1734 * @return #MFTL_SUCCESS.
1735 */
1736static int redistribute_core(MFTL_Ensemble *ens, const MFTL_OwnerMap *maps)
1737{
1738 MFTL_ASSERT(MFTL_MPI_PackedVertex != MPI_DATATYPE_NULL && "MFTLRedistribute: call MFTLRegisterMpiTypes() first");
1739
1740 MFTL_Domain *const domain = &ens->domain;
1741 MFTL_Object **const objects = ens->local_objects;
1742 const int num_objects = ens->n_local_objects;
1743
1744 CS_BEGIN(ts_total);
1745 MFTLCommStats.n_calls++;
1746 MFTLCommStats.n_objects_visited += num_objects;
1747
1748 const int nproc = domain->nproc;
1749 const int my_rank = domain->rank;
1750
1751#ifndef NDEBUG
1752 check_object_lists_agree(domain, ens->n_bodies_global);
1753#endif
1754
1755 if (nproc < 2)
1756 {
1757 /* One rank: nothing moves; only establish the derived arrays if missing. */
1758 for (int o = 0; o < num_objects; ++o)
1759 {
1760 MFTL_Object *const obj = objects[o];
1761 MFTL_ASSERT(obj->verts.n_local == obj->verts.n && obj->tris.n_local == obj->tris.n && "redistribute: single-rank object holds ghosts");
1762 if (!MFTLObjectHasTopology(obj)) MFTLObjectBuildDerived(obj);
1763 }
1764
1765 /* Same postcondition as in parallel: an empty body is held by nobody.
1766 * Walked backwards so an erase does not move a slot still to be visited. */
1767 for (int o = num_objects - 1; o >= 0; --o)
1768 {
1769 MFTL_Object *gone = objects[o];
1770 if (gone->tris.n > 0 || gone->verts.n > 0) continue;
1771
1772 MFTLEnsembleEraseLocalSlot(ens, MFTLEnsembleFindLocalSlot(ens, gone->global_id));
1773 MFTLObjectDestroy(&gone);
1774 MFTLCommStats.n_objects_destroyed++;
1775 }
1776
1777 CS_END(ts_total, t_total);
1778 return MFTL_SUCCESS;
1779 }
1780
1781 /* ── Phase 0: setup ────────────────────────────────────────────────────── */
1782 CS_BEGIN(ts_setup);
1783
1784 int global_tris = 0, global_verts = 0;
1785 int sum_verts = 0, sum_tris = 0, max_verts = 1, max_tris = 1;
1786 for (int o = 0; o < num_objects; ++o)
1787 {
1788 MFTL_ASSERT(objects[o]->domain == domain && "redistribute: object belongs to a different domain");
1789 global_tris += objects[o]->global_n_triangles;
1790 global_verts += objects[o]->global_n_vertices;
1791 sum_verts += objects[o]->verts.n;
1792 sum_tris += objects[o]->tris.n_local;
1793 max_verts = MFTL_MAX(max_verts, objects[o]->verts.n);
1794 max_tris = MFTL_MAX(max_tris, objects[o]->tris.n);
1795 }
1796
1797 int tri_hint, vert_hint;
1798 calc_buffer_hints(global_tris, global_verts, nproc, &tri_hint, &vert_hint);
1799
1800 /* The header payload always travels (it is what lets an arrival create its
1801 * body), so the payload count -- collective state -- is constant. */
1802 if (ens->comm_workspace == MFTL_NULL) ens->comm_workspace = MFTLCallocOrDie(1, sizeof(comm_workspace), "MFTL communication workspace");
1803 comm_workspace *const ws = (comm_workspace *) ens->comm_workspace;
1804
1805 /* A LOCAL SLOT indexes this call's work list and every workspace array; a
1806 * BODY ID indexes anything that goes on the wire.
1807 *
1808 * The pre-exchange holder test must be taken HERE: compute_owners() below
1809 * overwrites every `owner_proc` with the DESTINATION rank, after which the
1810 * pools describe the partition about to exist rather than the one the vote
1811 * has to reach. */
1812 const bool holders_fit = MFTLEnsembleBodyHoldersFitTheGraph(ens);
1813
1814 /* At least one slot: the arrival path grows the arrays from here. */
1815 ws_reserve(ws, nproc, MFTL_MAX(num_objects, 1), sum_verts, sum_tris, max_verts, max_tris, tri_hint, vert_hint);
1816
1817 char *const is_neighbor = ws->is_neighbor;
1818 memset(is_neighbor, 0, (size_t) nproc);
1819 for (int i = 0; i < domain->num_neighbors; ++i) is_neighbor[domain->neighbors[i]] = 1;
1820
1821 MFTL_SendBuffer *const send_tris = ws->send_tris;
1822 MFTL_SendBuffer *const send_verts = ws->send_verts;
1823 MFTL_SendBuffer *const send_counts = ws->send_counts;
1824 int *const kept_arena_v = ws->kept_arena_v;
1825 int *const kept_arena_t = ws->kept_arena_t;
1826 int *const tri_owner = ws->tri_owner;
1827 int *const vert_owner = ws->vert_owner;
1828 int *const vert_gid = ws->vert_gid;
1829
1830 /* The work list: bodies held now; phase 4 appends first-time arrivals. */
1831 const int n_pre = num_objects;
1832 int n_work = n_pre;
1833 for (int o = 0; o < n_pre; ++o) ws->work[o] = objects[o];
1834
1835 /* The payload buffers keep their allocations; only the fill level is reset. */
1836 for (int r = 0; r < nproc; ++r)
1837 {
1838 send_tris[r].count = 0;
1839 send_verts[r].count = 0;
1840 send_counts[r].count = 0;
1841 }
1842 memset(ws->states, 0, (size_t) n_pre * sizeof(body_state));
1843
1844 vertex_dedup dd;
1845 dd.mask = ws->dedup_mask;
1846 dd.rank_bit = ws->rank_bit;
1847 dd.n_bits = 0;
1848 CS_END(ts_setup, t_setup);
1849
1850 /* ── Phase 1a: ownership everywhere, analysis where THIS rank sees change
1851 * (section 6b). */
1852 int requires_global = 0;
1853 int arena_v = 0, arena_t = 0;
1854
1855 double *const dirty_local = ws->dirty_votes;
1856 double *const dirty_global = ws->dirty_votes + ws->cap_votes;
1857
1858 memset(dirty_local, 0, (size_t) n_pre * sizeof(double));
1859
1860 for (int o = 0; o < n_pre; ++o)
1861 {
1862 MFTL_Object *const obj = ws->work[o];
1863 body_state *const bs = &ws->states[o];
1864
1865 bs->kept_v = kept_arena_v + arena_v;
1866 bs->kept_t = kept_arena_t + arena_t;
1867 arena_v += obj->verts.n;
1868 arena_t += obj->tris.n_local;
1869
1870 /* Nothing of this body is here: no opinion. */
1871 if (obj->tris.n < 1 || obj->verts.n < 1) continue;
1872
1873 CS_TIMELINE(tm);
1874 if (maps == MFTL_NULL && (obj->derived & MFTL_DERIVED_GEOMETRY) == 0u) refresh_geometry_before_routing(obj);
1875 int dirty = compute_owners(obj, maps ? &maps[o] : MFTL_NULL, tri_owner, vert_owner, vert_gid, MFTL_TRUE);
1876 CS_LAP(tm, t_owners);
1877
1878 /* The dirty stamp, and a body never given derived arrays, force a rebuild. */
1879 dirty |= obj->partition_dirty;
1880 dirty |= !MFTLObjectHasTopology(obj);
1881 if (!g_skip_unchanged) dirty = 1;
1882
1883 if (!dirty) continue;
1884
1885 dirty_local[o] = 1.0;
1886 requires_global |= analyze_and_pack_body(ws, obj, my_rank, &dd, bs);
1887 }
1888
1889 /* ── Phase 1b: one per-body MAX over each body's HOLDERS. Not in place:
1890 * pass 1c needs this rank's own votes to know what it already did. */
1891 CS_BEGIN(ts_vote);
1892 {
1893 static const MFTL_BodyReduceOp vote_op = MFTL_BODY_REDUCE_MAX;
1894 memcpy(dirty_global, dirty_local, (size_t) n_pre * sizeof(double));
1895 MFTLEnsembleReduceBodiesPrescanned(ens, dirty_global, 1, &vote_op, holders_fit);
1896 }
1897 CS_END(ts_vote, t_skip_vote);
1898
1899 /* ── Phase 1c: bodies dirty elsewhere, and clean bodies still settling. */
1900 for (int o = 0; o < n_pre; ++o)
1901 {
1902 MFTL_Object *const obj = ws->work[o];
1903
1904 if (dirty_local[o] != 0.0) continue; // already done in pass 1a
1905 if (dirty_global[o] == 0.0 && obj->partition_settled) continue; // skipped
1906
1907 if (obj->tris.n < 1 || obj->verts.n < 1) continue; // nothing here to send
1908
1909 /* Idempotent re-classification; counters not updated twice. */
1910 CS_TIMELINE(tm);
1911 (void) compute_owners(obj, maps ? &maps[o] : MFTL_NULL, tri_owner, vert_owner, vert_gid, MFTL_FALSE);
1912 CS_LAP(tm, t_owners);
1913
1914 requires_global |= analyze_and_pack_body(ws, obj, my_rank, &dd, &ws->states[o]);
1915 MFTLCommStats.n_objects_reclassified++;
1916 if (dirty_global[o] == 0.0) MFTLCommStats.n_objects_settled++;
1917 }
1918
1919
1920
1921 /* ── Phase 2: agree on the route. Collective by necessity: a non-neighbour
1922 * receiver cannot know it must listen on the global path. */
1923 CS_BEGIN(ts_route);
1924 int global_decision = 0;
1925 MPI_Allreduce(&requires_global, &global_decision, 1, MPI_INT, MPI_LOR, domain->comm_cart);
1926 CS_END(ts_route, t_route);
1927 if (global_decision)
1928 MFTLCommStats.n_route_global++;
1929 else
1930 MFTLCommStats.n_route_neighbor++;
1931
1932 /* ── Phase 3: exchange ─────────────────────────────────────────────────── */
1933 MFTL_Exchange xfer[3];
1934 xfer[0].send_bufs = send_verts;
1935 xfer[0].type = MFTL_MPI_PackedVertex;
1936 xfer[0].type_size = sizeof(MFTL_PackedVertex);
1937 xfer[1].send_bufs = send_tris;
1938 xfer[1].type = MFTL_MPI_PackedTriangleComplex;
1939 xfer[1].type_size = sizeof(MFTL_PackedTriangleComplex);
1940 xfer[2].send_bufs = send_counts;
1941 xfer[2].type = MPI_INT;
1942 xfer[2].type_size = sizeof(int);
1943
1944 for (int r = 0; r < nproc; ++r)
1945 {
1946 const long bytes = (long) send_verts[r].count * (long) sizeof(MFTL_PackedVertex) + (long) send_tris[r].count * (long) sizeof(MFTL_PackedTriangleComplex) +
1947 (long) send_counts[r].count * (long) sizeof(int);
1948 if (bytes > 0)
1949 {
1950 MFTLCommStats.n_bytes_sent += bytes;
1951 MFTLCommStats.n_messages++;
1952 }
1953 }
1954
1955 MFTL_ExchangeCtx ctx;
1956 MFTLExchangeStart(domain, xfer, 3, global_decision != 0, &ctx);
1957
1958 /* The send buffers were copied; they persist for the next call. */
1959
1960 /* ── Phase 4: reconstruct the per-body arrival slices ───────────────────── */
1961 CS_BEGIN(ts_w2);
1962 MFTLExchangeWait(&ctx, 2);
1963 CS_END(ts_w2, t_wait);
1964
1965 CS_BEGIN(ts_g2);
1966 const int *const rc = (const int *) xfer[2].recv_buf;
1967 const int n_runs = xfer[2].recv_count / MFTL_BODY_COUNT_INTS;
1968
1969 ws_bytes((void **) &ws->runs, &ws->cap_runs, MFTL_MAX(n_runs, 1), sizeof(body_run), "redistribution body runs");
1970 body_run *const runs = ws->runs;
1971
1972 int t_off = 0, v_off = 0;
1973 for (int k = 0; k < n_runs; ++k)
1974 {
1975 const int *const tp = &rc[MFTL_BODY_COUNT_INTS * k];
1976 const int gid = tp[MFTL_BODY_HDR_GID];
1977
1978 int slot = slot_of_body(ws->work, n_pre, n_work, gid);
1979 if (slot == MFTL_ERROR)
1980 {
1981 /* A body arriving where this rank held nothing: created and appended to
1982 * the work list (merged into the directory at the end); it starts
1983 * unsettled. */
1984 MFTL_ASSERT(gid >= 0 && gid < ens->n_bodies_global && "redistribute: received a body gid outside the id space");
1985
1986 ws_bodies_reserve(ws, n_work + 1);
1987 MFTL_Object *const arrived = MFTLObjectCreate(gid);
1988 arrived->domain = domain;
1989 arrived->derived = MFTL_DERIVED_TOPOLOGY; // an empty mesh is trivially described
1990
1991 /* The persistent slot may hold the previous call's state: clear it. */
1992 memset(&ws->states[n_work], 0, sizeof(body_state));
1993
1994 ws->work[n_work] = arrived;
1995 slot = n_work++;
1996 MFTLCommStats.n_objects_created++;
1997 }
1998
1999 /* The four global fields: identical on every holder (asserted), applied. */
