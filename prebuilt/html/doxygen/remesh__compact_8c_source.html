1/**
2 * @file remesh_compact.c
3 * @author Christian Gorges
4 * @date 2026-08-15
5 * @ingroup mftl_remeshing
6 *
7 * @brief Removing dead entities, and re-densifying global ids.
8 *
9 * ## WHY A COLLAPSE CANNOT FREE ANYTHING, AND WHAT FOLLOWS
10 *
11 * A triangle's three half-edges ARE slots 3t..3t+2 of the triangle pool. There
12 * is no half-edge free list because there is no half-edge allocation, so an
13 * operator that removes a triangle cannot hand anything back -- it can only mark
14 * it. Compaction is therefore not an optimisation here, it is the deallocator.
15 *
16 * Two properties of this library make the pass simpler than a general compaction:
17 *
18 * HALF-EDGES MOVE AS TRIPLES. `he[3*t_new + k] = he[3*t_old + k]`, so the
19 * implicit indexing invariant survives by construction and so do the implicit
20 * half-edge global ids (3 * triangle gid + k) -- neither needs a fix-up pass.
21 * Compacting half-edges as an independent array with a map of their own is
22 * what forces `next_idx` and `triangle_idx` to be remapped as well.
23 *
24 * ONE ALLOCATION PER POOL. Compacting a pool is a loop over its sub-arrays,
25 * and a sub-array that someone forgot cannot exist -- it would have to have
26 * been allocated separately, and nothing outside src/object/object.c can do
27 * that.
28 *
29 * ## out_he_idx IS RE-DERIVED, NOT REMAPPED
30 *
31 * Remapping it would be one array lookup per vertex and would faithfully
32 * preserve whatever each operator left behind, INCLUDING an index the operator
33 * failed to repair. Re-deriving it from the surviving half-edges is the same
34 * single-pass rule MFTLBuildTopologyFromIndexList uses, costs O(H) rather than
35 * O(V), and cannot be stale. It gives up nothing: the fan walkers rewind to the
36 * start of an open fan themselves (MFTLVertexFanStart), so which outgoing
37 * half-edge a vertex names has never mattered.
38 *
39 * ## COMPACTION IS ALSO WHERE THE OWNERSHIP PREFIX IS RE-ESTABLISHED
40 *
41 * An operator appends, and an append lands at the end of the pool -- past the
42 * ghosts. So during a parallel remeshing sweep the pool is
43 *
44 * [ owned ...... | ghosts ...... | entities created this sweep ...... ]
45 * ^ n_local
46 *
47 * which violates invariant 2 of mftl_types.h (ownership is a prefix) for exactly
48 * as long as the sweep runs. This function is what ends that window, and it is
49 * the natural place: it is already a full permutation of both pools with a full
50 * index remap behind it, so restoring the prefix costs one more block move and
51 * one more term in the map -- not a second pass.
52 *
53 * The permutation is not a forward compaction any more, because a ghost may have
54 * to move RIGHT to make room for the entities created after it. It is a forward
55 * compaction FOLLOWED BY a swap of the two trailing blocks, and that is the whole
56 * of it: the forward compaction preserves relative order, so the survivors come
57 * out as [owned][ghosts][created] and the prefix is restored by exchanging the
58 * last two. A general permutation would need cycle-following or a full copy of
59 * both pools; two adjacent blocks need one buffer the size of the smaller of
60 * them and two memmoves.
61 *
62 * On a serial object there are no ghosts, an append increments `n_local` with
63 * the pool, and both trailing blocks are empty.
64 */

77/**
78 * @brief Exchanges the two adjacent blocks [first, first+n1) and [first+n1, +n2).
79 *
80 * @param base Start of the array.
81 * @param elem Size of one element in bytes.
82 * @param first Index the first run starts at.
83 * @param n1 Elements in the first run.
84 * @param n2 Elements in the second run, which follows it immediately.
85 * @param scratch Buffer of at least `MIN(n1, n2) * elem` bytes.
86 */
87static void swap_adjacent_blocks(void *base, size_t elem, int first, int n1, int n2, void *scratch)
88{
89 if (n1 == 0 || n2 == 0) return;
90
91 char *const p = (char *) base + (size_t) first * elem;
92
93 if (n1 <= n2)
94 {
95 memcpy(scratch, p, (size_t) n1 * elem);
96 memmove(p, p + (size_t) n1 * elem, (size_t) n2 * elem);
97 memcpy(p + (size_t) n2 * elem, scratch, (size_t) n1 * elem);
98 }
99 else
100 {
101 memcpy(scratch, p + (size_t) n1 * elem, (size_t) n2 * elem);
102 memmove(p + (size_t) n2 * elem, p, (size_t) n1 * elem);
103 memcpy(p, scratch, (size_t) n2 * elem);
104 }
105}

107/**
108 * @struct pool_partition
109 * @brief The three runs a compacted pool falls into, and the map term they imply.
110 *
111 * `n_kept_owned` survivors from the ownership prefix, then `n_ghost` surviving
112 * ghosts, then `n_created` entities this sweep appended. Swapping the last two
113 * restores the prefix, and `partition_remap` is that swap expressed on indices.
114 */
115typedef struct
116{
117 int n_kept_owned; ///< Surviving entities that were owned before the pass.
118 int n_ghost; ///< Surviving ghosts.
119 int n_created; ///< Entities this rank created during the pass, past the old prefix.
120} pool_partition;
121
122/**
123 * @brief Where a compacted index lands once the two trailing blocks are
124 * swapped, i.e. the map that re-establishes the ownership prefix.
125 *
126 * @param p The three run lengths of the pool.
127 * @param i Index after the forward compaction.
128 * @return Index after the block swap.
129 */
130static inline int partition_remap(const pool_partition *p, int i)
131{
132 if (i < p->n_kept_owned) return i;
133 if (i < p->n_kept_owned + p->n_ghost) return i + p->n_created;
134 return i - p->n_ghost;
135}

137int MFTLRemeshCompactObject(MFTL_Object *obj, MFTL_RemeshStats *stats)
138{
139 MFTL_ASSERT(obj != MFTL_NULL);
140
141 const int n_v_old = obj->verts.n;
142 const int n_t_old = obj->tris.n;
143
144 /* The prefix as it stands BEFORE this call: everything below it is owned,
145 * everything above it is a ghost or was created by this sweep, and only
146 * `owner_proc` tells those two apart. */
147 const int lv_old = obj->verts.n_local;
148 const int lt_old = obj->tris.n_local;
149 const int my_rank = (obj->domain != MFTL_NULL) ? obj->domain->rank : 0;
150
151 if (n_t_old == 0 && n_v_old == 0) return 0;
152
153 /* One allocation for both maps plus the reference marks: same lifetime, and
154 * three separate mallocs of a few hundred kilobytes each is three first-touch
155 * fault storms per remeshing iteration. */
156 int *const scratch = (int *) MFTLMallocOrDie(sizeof(int) * ((size_t) n_v_old * 2 + (size_t) n_t_old), "remesh compaction maps");
157 int *const v_map = scratch;
158 int *const v_referenced = scratch + n_v_old;
159 int *const t_map = scratch + 2 * (size_t) n_v_old;
160
161 /* ── 1. Which vertices does the surviving topology actually use? ──────────
162 *
163 * Not the same question as "which are not flagged dead". A vertex can be left
164 * with no incident triangle by a sequence of collapses around it without ever
165 * being the one retired. It is still a valid record pointing at a half-edge
166 * that no longer exists, so every later traversal from it walks into freed
167 * territory. Counting these separately is deliberate -- an orphan is a symptom,
168 * and a silent removal would hide it. */
169 memset(v_referenced, 0, (size_t) n_v_old * sizeof(int));
170
171 for (int t = 0; t < n_t_old; ++t)
172 {
173 if (tri_is_dead(obj, t)) continue;
174 for (int k = 0; k < 3; ++k)
175 {
176 const int v = obj->tris.he[3 * t + k].vertex_idx;
177 MFTL_ASSERT(v >= 0 && v < n_v_old && "MFTLRemeshCompactObject: live half-edge targets a vertex outside the pool");
178 v_referenced[v] = 1;
179 }
180 }
181
182 /* ── 2. Build the maps, and classify the survivors into the three runs ───── */
183 pool_partition vpart = {0, 0, 0};
184 pool_partition tpart = {0, 0, 0};
185
186 int n_v_new = 0;
187 int n_dead_v = 0;
188 int n_orphan_v = 0;
189
190 for (int v = 0; v < n_v_old; ++v)
191 {
192 const bool dead = vert_is_dead(obj, v);
193
194 if (dead || !v_referenced[v])
195 {
196 MFTL_ASSERT((v < lv_old || obj->verts.rec[v].owner_proc == my_rank) &&
197 "MFTLRemeshCompactObject: a GHOST vertex was retired. Ghosts are frozen for the whole sweep, so "
198 "this means the freeze classification and the operators disagree about what is replicated.");
199 v_map[v] = MFTL_ERROR;
200 if (dead)
201 n_dead_v++;
202 else
203 n_orphan_v++;
204 continue;
205 }
206
207 if (v < lv_old)
208 vpart.n_kept_owned++;
209 else if (obj->verts.rec[v].owner_proc != my_rank)
210 {
211 vpart.n_ghost++;
212 MFTL_ASSERT(vpart.n_created == 0 && "MFTLRemeshCompactObject: a created vertex precedes a ghost -- the pool is not [owned|ghost|created]");
213 }
214 else
215 vpart.n_created++;
216
217 v_map[v] = n_v_new++;
218 }
219
220 int n_t_new = 0;
221 for (int t = 0; t < n_t_old; ++t)
222 {
223 if (tri_is_dead(obj, t))
224 {
225 MFTL_ASSERT((t < lt_old || obj->tris.rec[t].owner_proc == my_rank) && "MFTLRemeshCompactObject: a GHOST triangle was retired -- see the vertex case");
226 t_map[t] = MFTL_ERROR;
227 continue;
228 }
229
230 if (t < lt_old)
231 tpart.n_kept_owned++;
232 else if (obj->tris.rec[t].owner_proc != my_rank)
233 {
234 tpart.n_ghost++;
235 MFTL_ASSERT(tpart.n_created == 0 && "MFTLRemeshCompactObject: a created triangle precedes a ghost -- the pool is not [owned|ghost|created]");
236 }
237 else
238 tpart.n_created++;
239
240 t_map[t] = n_t_new++;
241 }
242
243 /* Both blocks non-empty is exactly "a partitioned object grew this sweep". */
244 const bool swap_v = (vpart.n_ghost > 0 && vpart.n_created > 0);
245 const bool swap_t = (tpart.n_ghost > 0 && tpart.n_created > 0);
246
247 const int removed = (n_v_old - n_v_new) + (n_t_old - n_t_new);
248 if (removed == 0 && !swap_v && !swap_t)
249 {
250 /* Nothing moved and no block needs exchanging, so the prefix is already
251 * right: either the object has no ghosts (and the appends grew `n_local`
252 * with the pool) or it grew by nothing. */
253 free(scratch);
254 if (!MFTLObjectHasTopology(obj)) MFTLObjectBuildDerived(obj);
255 return 0;
256 }
257
258 /* ── 3. Compact the pools ────────────────────────────────────────────────
259 *
260 * Forward scan, so the destination index never exceeds the source index and
261 * the moves cannot overlap destructively. */
262 MFTL_VertexPool *const vp = &obj->verts;
263 for (int v = 0; v < n_v_old; ++v)
264 {
265 const int nv = v_map[v];
266 if (nv == MFTL_ERROR || nv == v) continue;
267
268 vp->rec[nv] = vp->rec[v];
269 for (int k = 0; k < 3; ++k)
270 {
271 vp->pos[3 * nv + k] = vp->pos[3 * v + k];
272 vp->normal[3 * nv + k] = vp->normal[3 * v + k];
273 }
274 }
275
276 MFTL_TrianglePool *const tp = &obj->tris;
277 for (int t = 0; t < n_t_old; ++t)
278 {
279 const int nt = t_map[t];
280 if (nt == MFTL_ERROR || nt == t) continue;
281
282 tp->rec[nt] = tp->rec[t];
283 for (int k = 0; k < 3; ++k) tp->he[3 * nt + k] = tp->he[3 * t + k];
284
285 tp->area[nt] = tp->area[t];
286 for (int k = 0; k < 3; ++k)
287 {
288 tp->normal[3 * nt + k] = tp->normal[3 * t + k];
289 tp->center[3 * nt + k] = tp->center[3 * t + k];
290 }
291 }
292
293 /* ── 3b. Restore the ownership prefix ─────────────────────────────────────
294 *
295 * The survivors are now [owned][ghost][created]; exchanging the last two
296 * blocks makes the owned entities a prefix again. Every sub-array of a pool is
297 * permuted the same way, which is what one allocation per pool buys -- there is
298 * no list of arrays here that a later addition could be left out of, because
299 * the loop below IS the layout.
300 *
301 * The derived arrays (`verts`, `neighbors`) are deliberately
302 * absent: they are rebuilt at the end of this function, so moving them would be
303 * work thrown away. */
304 if (swap_v || swap_t)
305 {
306 const size_t need_v = swap_v ? (size_t) MFTL_MIN(vpart.n_ghost, vpart.n_created) * 3 * sizeof(double) : 0;
307 const size_t need_t = swap_t ? (size_t) MFTL_MIN(tpart.n_ghost, tpart.n_created) * 3 * sizeof(MFTL_HalfEdge) : 0;
308
309 void *const tmp = MFTLMallocOrDie(MFTL_MAX(MFTL_MAX(need_v, need_t), (size_t) 1), "remesh partition scratch");
310
311 if (swap_v)
312 {
313 const int f = vpart.n_kept_owned, a = vpart.n_ghost, b = vpart.n_created;
314 swap_adjacent_blocks(vp->rec, sizeof(MFTL_Vertex), f, a, b, tmp);
315 swap_adjacent_blocks(vp->pos, 3 * sizeof(double), f, a, b, tmp);
316 swap_adjacent_blocks(vp->normal, 3 * sizeof(double), f, a, b, tmp);
317 }
318
319 if (swap_t)
320 {
321 const int f = tpart.n_kept_owned, a = tpart.n_ghost, b = tpart.n_created;
322 swap_adjacent_blocks(tp->rec, sizeof(MFTL_Triangle), f, a, b, tmp);
323 /* Half-edges move as TRIPLES, which is what preserves both the implicit
324 * indexing invariant and the implicit half-edge global ids. Expressing that
325 * as an element size of three records rather than as three separate swaps
326 * makes it impossible to get the grouping wrong. */
327 swap_adjacent_blocks(tp->he, 3 * sizeof(MFTL_HalfEdge), f, a, b, tmp);
328 swap_adjacent_blocks(tp->area, sizeof(double), f, a, b, tmp);
329 swap_adjacent_blocks(tp->normal, 3 * sizeof(double), f, a, b, tmp);
330 swap_adjacent_blocks(tp->center, 3 * sizeof(double), f, a, b, tmp);
331 }
332
333 free(tmp);
334
335 /* Fold the exchange into the maps, so step 4 remaps straight to the final
336 * index space rather than to the intermediate one. */
337 if (swap_v)
338 {
339 for (int v = 0; v < n_v_old; ++v)
340 {
341 if (v_map[v] != MFTL_ERROR) v_map[v] = partition_remap(&vpart, v_map[v]);
342 }
343 }
344 if (swap_t)
345 {
346 for (int t = 0; t < n_t_old; ++t)
347 {
348 if (t_map[t] != MFTL_ERROR) t_map[t] = partition_remap(&tpart, t_map[t]);
349 }
350 }
351 }
352
353 /* ── 4. Remap the indices the records carry ───────────────────────────────── */
354 for (int h = 0; h < 3 * n_t_new; ++h)
355 {
356 MFTL_HalfEdge *const e = &tp->he[h];
357
358 const int v_new = v_map[e->vertex_idx];
359 MFTL_ASSERT(v_new != MFTL_ERROR && "MFTLRemeshCompactObject: live half-edge targets a removed vertex");
360 e->vertex_idx = v_new;
361
362 if (e->twin_idx == MFTL_ERROR) continue;
363
364 const int t_new = t_map[MFTLHeTriIdx(e->twin_idx)];
365 if (t_new == MFTL_ERROR)
366 {
367 /* A live half-edge twinned to a dead one means an operator removed a
368 * triangle without re-gluing its rim. Turning it into a boundary keeps the
369 * structure traversable, but the surface now has a hole that no flag
370 * records, so this is an assertion first and a fallback second. */
371 MFTL_ASSERT(0 && "MFTLRemeshCompactObject: live half-edge twinned to a removed triangle -- rim was not glued");
372 e->twin_idx = MFTL_ERROR;
373 e->flags |= MFTL_FLAG_BOUNDARY;
374 continue;
375 }
376 e->twin_idx = 3 * t_new + MFTLHeCorner(e->twin_idx);
377 }
378
379 /* ── 5. Clear the transient flags ─────────────────────────────────────────── */
380 for (int v = 0; v < n_v_new; ++v) vp->rec[v].flags &= ~(MFTL_FLAG_DEAD | MFTL_FLAG_VISITED);
381 for (int t = 0; t < n_t_new; ++t) tp->rec[t].flags &= ~(MFTL_FLAG_DEAD | MFTL_FLAG_VISITED);
382 for (int h = 0; h < 3 * n_t_new; ++h) tp->he[h].flags &= ~(MFTL_FLAG_DEAD | MFTL_FLAG_VISITED);
383
384 /* ── 6. Publish the new counts, then re-derive out_he ─────────────────────
385 *
386 * SetCounts before the derivation, so the loop below is bounded by the counts
387 * a later reader will use rather than by the ones that just became wrong.
388 *
389 * The owned count is the two owned runs, whether or not they had to be moved
390 * to become adjacent -- which is also what makes this the point at which the
391 * entities created by the sweep stop being past the prefix and start being
392 * ordinary owned entities. */
393 MFTLObjectSetCounts(obj, n_v_new, vpart.n_kept_owned + vpart.n_created, n_t_new, tpart.n_kept_owned + tpart.n_created);
394
395 for (int h = 0; h < 3 * n_t_new; ++h) vp->rec[MFTLHeOrigin(obj, h)].out_he_idx = h;
396
397 free(scratch);
398
399 if (stats)
400 {
401 stats->n_vertices_removed += n_dead_v;
402 stats->n_orphans_removed += n_orphan_v;
403 }
404
405 MFTLObjectBuildDerived(obj);
406 return removed;
407}

409void MFTLRemeshRenumberGlobalIds(MFTL_Object *obj)
410{
411 MFTL_ASSERT(obj != MFTL_NULL);
412 MFTL_ASSERT(obj->tris.n_local == obj->tris.n && obj->verts.n_local == obj->verts.n &&
413 "MFTLRemeshRenumberGlobalIds: object holds ghosts -- the renumbering is a LOCAL decision about a "
414 "mesh-wide namespace and is only sound when this rank holds all of it");
415
416 const int n_v = obj->verts.n;
417 const int n_t = obj->tris.n;
418
419 for (int v = 0; v < n_v; ++v) obj->verts.rec[v].global_id = v;
420
421 for (int t = 0; t < n_t; ++t)
422 {
423 obj->tris.rec[t].global_id = t;
424 /* Not a free choice: half-edge 3t+k must carry 3 * (triangle gid) + k, which
425 * is what lets redistribution reconstruct half-edge ids instead of sending
426 * them (inv_halfedge_gids_are_implicit). */
427 for (int k = 0; k < 3; ++k) obj->tris.he[3 * t + k].global_id = 3 * t + k;
428 }
429
430 obj->next_vertex_gid = n_v;
431 obj->next_triangle_gid = n_t;
432 obj->next_halfedge_gid = 3 * n_t;
433
434 /* Every id in the body just moved. The assertion above says this rank holds
435 * the whole mesh, so nothing is replicated RIGHT NOW -- but the migration that
436 * puts it back on the grid is the one that has to observe the new numbering,
437 * and it decides per body whether to bother. Tell it. */
438 MFTLObjectMarkPartitionDirty(obj);
439}
