60int MFTLTopologyDetectEvents(MFTL_Ensemble *ens, const MFTL_TopologyOptions *opts, MFTL_TopologyEvent **events, int *n_events)
61{
62 if (ens == MFTL_NULL || opts == MFTL_NULL || events == MFTL_NULL || n_events == MFTL_NULL) return MFTL_ERROR;
63 *events = MFTL_NULL;
64 *n_events = 0;
65 const double dmax = MFTL_MAX(opts->delta_breakup, opts->delta_coalescence);
66 if (!(dmax > 0.0)) return MFTL_SUCCESS; // nothing can be detected, on every rank alike
67
68 const MFTL_Domain *const dom = &ens->domain;
69 for (int o = 0; o < ens->n_local_objects; ++o)
70 if (!MFTLObjectHasTopology(ens->local_objects[o])) MFTLObjectBuildDerived(ens->local_objects[o]);
71
72 MFTL_TopoEventList local = {MFTL_NULL, 0, 0};
73 if (MFTLTopoDetectLocal(ens, opts, &local) != MFTL_SUCCESS)
74 {
75 free(local.e);
76 return MFTL_ERROR;
77 }
78
79 /* A quarter of the bias bump's scale, max(delta, dx): events closer than that
80 * describe one feature and add nothing to the bias but its cost. Keyed on the
81 * smaller active threshold ALONE this shredded one neck into 779 events when
82 * the other threshold was ten times smaller. */
83 double quantum = dmax;
84 if (opts->delta_breakup > 0.0) quantum = fmin(quantum, opts->delta_breakup);
85 if (opts->delta_coalescence > 0.0) quantum = fmin(quantum, opts->delta_coalescence);
86 quantum = 0.25 * fmax(quantum, opts->dx);
87
88 MFTLTopoCanonicaliseEvents(dom, quantum, &local);
89
90 double *const mine = (double *) MFTLMallocOrDie((size_t) MFTL_MAX(local.n, 1) * EVENT_STRIDE * sizeof(double), "topology event records");
91 for (int i = 0; i < local.n; ++i)
92 {
93 double *const r = &mine[EVENT_STRIDE * i];
94 r[0] = local.e[i].body_a;
95 r[1] = local.e[i].body_b;
96 r[2] = local.e[i].kind;
97 r[3] = local.e[i].point[0];
98 r[4] = local.e[i].point[1];
99 r[5] = local.e[i].point[2];
100 r[6] = local.e[i].gap;
101 }
102 free(local.e);
103
104 int n_all = 0;
105 double *const all = MFTLTopoCommAllgather(dom, mine, local.n, EVENT_STRIDE, &n_all);
106 free(mine);
107
108 MFTL_TopoEventList list = {MFTL_NULL, 0, 0};
109 list.cap = MFTL_MAX(n_all, 1);
110 list.e = (MFTL_TopologyEvent *) MFTLMallocOrDie((size_t) list.cap * sizeof(MFTL_TopologyEvent), "topology events");
111 for (int i = 0; i < n_all; ++i)
112 {
113 const double *const r = &all[EVENT_STRIDE * i];
114 MFTL_TopologyEvent *const e = &list.e[list.n++];
115 e->body_a = (int) r[0];
116 e->body_b = (int) r[1];
117 e->kind = (int) r[2];
118 e->point[0] = r[3];
119 e->point[1] = r[4];
120 e->point[2] = r[5];
121 e->gap = r[6];
122 }
123 free(all);
124
125 /* Once more over the whole list: now identical on every rank. */
126 MFTLTopoCanonicaliseEvents(dom, quantum, &list);
127
128 if (list.n == 0)
129 {
130 free(list.e);
131 return MFTL_SUCCESS;
132 }
133 *events = list.e;
134 *n_events = list.n;
135 return MFTL_SUCCESS;
136}

150/**
151 * @brief The size floor again, on what the remesher left: a closed body whose
152 * marching-cubes surface cleared it can still be collapsed below it.
153 *
154 * A satellite a couple of lattice cells across comes out of marching cubes with
155 * a few dozen slivers, and a target edge length comparable to its diameter
156 * collapses that to a tetrahedron -- which the collapse approval then refuses to
157 * dissolve, so it survives as four triangles every one of which faces against
158 * all three of its neighbours. Measured: the pinch-off in the Rayleigh-Taylor
159 * example at np = 4. The count is the owned triangles summed over ranks, one
160 * value per component of THIS reconstruction (the list every rank already
161 * agrees on), so every holder takes the same decision. A deleted body's volume
162 * goes to its cluster's surviving bodies through the correction that follows.
163 */
164static void discard_collapsed(MFTL_Ensemble *ens, MFTL_TopoComponent *comps, int n_comps, int min_tris, int first_new, MFTL_TopologyStats *st)
165{
166 if (min_tris <= 0 || n_comps == 0) return;
167 double *const n = (double *) MFTLCallocOrDie((size_t) n_comps, sizeof(double), "topology remeshed sizes");
168 for (int j = 0; j < n_comps; ++j)
169 {
170 if (comps[j].body < 0 || comps[j].open) continue;
171 const int slot = MFTLEnsembleFindLocalSlot(ens, comps[j].body);
172 if (slot != MFTL_ERROR) n[j] = ens->local_objects[slot]->tris.n_local;
173 }
174 MFTLTopoCommAllreduceDouble(&ens->domain, n, n_comps, 2);
175
176 for (int j = 0; j < n_comps; ++j)
177 {
178 if (comps[j].body < 0 || comps[j].open || n[j] >= (double) min_tris) continue;
179 const int slot = MFTLEnsembleFindLocalSlot(ens, comps[j].body);
180 if (slot != MFTL_ERROR)
181 {
182 MFTL_Object *obj = ens->local_objects[slot];
183 MFTLEnsembleEraseLocalSlot(ens, slot);
184 MFTLObjectDestroy(&obj);
185 }
186 st->n_bodies_discarded++;
187 st->n_bodies_out--;
188 if (comps[j].body >= first_new)
189 st->n_bodies_created--; // its id was handed out and is never used
190 else
191 st->n_bodies_retired++;
192 comps[j].body = -1;
193 }
194 free(n);
195}

197/**
198 * @brief The volume correction's displacement field, left in `verts.normal`,
199 * and the derivative of the enclosed volume along it.
200 *
201 * u is the area-weighted vertex normal; a rim vertex loses the component normal
202 * to the wall it is on, so it slides in the wall and the rim stays there. The
203 * return value is this rank's share of dV/dd = sum_t S_t . (u_0 + u_1 + u_2)/3
204 * (S_t the triangle's area vector), taken over the (triangle, corner) pairs whose
205 * corner this rank OWNS: the halo is vertex-complete, so every such pair is
206 * present exactly once in the job. Corners are gathered seam-consistently, so
207 * it holds on a periodic domain. It is the first variation of the volume for a
208 * closed body and for one closed by the walls alike, because the walls do not
209 * move.
210 */
211static double displacement_field(MFTL_Object *obj)
212{
213 const MFTL_Domain *const dom = obj->domain;
214 MFTLComputeVertexNormals(obj, MFTL_NORMAL_WEIGHT_AREA);
215 for (int v = 0; v < obj->verts.n_local; ++v)
216 {
217 if (!(obj->verts.rec[v].flags & MFTL_FLAG_BOUNDARY)) continue;
218 const double *const p = &obj->verts.pos[3 * v];
219 for (int k = 0; k < 3; ++k)
220 {
221 if (dom->is_periodic[k]) continue;
222 const double tol = 1e-8 * dom->domain_size[k];
223 if (fabs(p[k] - dom->global_min[k]) <= tol || fabs(p[k] - dom->global_max[k]) <= tol) obj->verts.normal[3 * v + k] = 0.0;
224 }
225 }
226
227 double dvdd = 0.0;
228 for (int t = 0; t < obj->tris.n; ++t)
229 {
230 double T[3][3];
231 MFTLGetTriangleCorners(obj, t, T);
232 const double e1[3] = {T[1][0] - T[0][0], T[1][1] - T[0][1], T[1][2] - T[0][2]};
233 const double e2[3] = {T[2][0] - T[0][0], T[2][1] - T[0][1], T[2][2] - T[0][2]};
234 const double S[3] = {0.5 * (e1[1] * e2[2] - e1[2] * e2[1]), 0.5 * (e1[2] * e2[0] - e1[0] * e2[2]), 0.5 * (e1[0] * e2[1] - e1[1] * e2[0])};
235 for (int k = 0; k < 3; ++k)
236 {
237 const int v = obj->tris.verts[3 * t + k];
238 if (v >= obj->verts.n_local) continue;
239 const double *const u = &obj->verts.normal[3 * v];
240 dvdd += (S[0] * u[0] + S[1] * u[1] + S[2] * u[2]) / 3.0;
241 }
242 }
243 return dvdd;
244}

246int MFTLEnsembleReconstructBodies(MFTL_Ensemble *ens, const int *bodies_in, int n_bodies_in, const MFTL_TopologyEvent *events, int n_events,

252
253 if (ens == MFTL_NULL || opts == MFTL_NULL || n_bodies_in < 0 || (n_bodies_in > 0 && bodies_in == MFTL_NULL)) return MFTL_ERROR;
254 if (!(opts->dx > 0.0)) return MFTL_ERROR;

258 const double t0 = MFTLTopoCommTime();

268 /* ── The lattice and the clusters ─────────────────────────────────────── */
269 MFTL_TopoLattice lat;
270 MFTLTopoLatticeInit(&lat, dom, opts->dx);
271 MFTLTopoCommCoords(dom, lat.coords);
272 for (int k = 0; k < 3; ++k) st->lattice_cells[k] = lat.cells[k];
273
274 MFTL_TopoCluster *cl = MFTL_NULL;
275 const int n_cl = MFTLTopoBuildClusters(bodies_in, n_bodies_in, events, n_events, &cl);
276
277 /* The sorted body list and the global size of each body. */
278 int *const bodies = (int *) MFTLMallocOrDie((size_t) n_bodies_in * sizeof(int), "topology body list");
279 memcpy(bodies, bodies_in, (size_t) n_bodies_in * sizeof(int));

283 /* One message for two questions: per body its triangle count (SUM), per

286 for (int i = 0; i < nb; ++i)
287 {

289 if (obj == MFTL_NULL) continue;

293 MFTLTopoCommAllreduceLongLong(dom, sizes, nb + n_cl, 2);
294 for (int c = 0; c < n_cl; ++c)

297 if (cl[c].refused)
298 st->n_clusters_refused++;
299 else
300 st->n_clusters++;
301 }

303 /* Which walls each cluster's rims lie on: six MAX slots per cluster. */
304 long long *const walls = (long long *) MFTLCallocOrDie(6 * (size_t) MFTL_MAX(n_cl, 1), sizeof(long long), "topology rim walls");
305 for (int i = 0; i < nb; ++i)
306 {
307 const MFTL_Object *const obj = MFTLEnsembleFindLocalObject(ens, bodies[i]);
308 if (obj == MFTL_NULL) continue;
309 const int c = MFTLTopoClusterOf(cl, n_cl, bodies[i]);

312 if (m & (1 << b)) walls[6 * c + b] = 1;

315 for (int c = 0; c < n_cl; ++c)
316 {
317 cl[c].rim_walls = 0;
318 for (int b = 0; b < 6; ++b)
319 if (walls[6 * c + b]) cl[c].rim_walls |= 1 << b;
320 }
321 free(walls);

323 /* ── Pseudo-normals: angle-weighted, exact on ghosts too ──────────────────
324 *
325 * Only the cluster bodies are recomputed; the exchange is ensemble-wide, which
326 * for every other body re-copies its owners' current normals onto its ghosts
327 * and changes nothing an owner holds. */
328 for (int o = 0; o < ens->n_local_objects; ++o)
329 {

331 const int c = MFTLTopoClusterOf(cl, n_cl, obj->global_id);
332 if (c < 0 || cl[c].refused) continue;
333 if (!MFTLObjectHasTopology(obj)) MFTLObjectBuildDerived(obj);
334 MFTLComputeVertexNormals(obj, MFTL_NORMAL_WEIGHT_ANGLE);
335 }

339 double *const ext = (double *) MFTLMallocOrDie(12 * (size_t) MFTL_MAX(n_cl, 1) * sizeof(double), "topology extents");
340 MFTLTopoLocalExtents(ens, cl, n_cl, ext);
341 for (int c = 0; c < n_cl; ++c)
342 if (cl[c].refused)
343 for (int i = 0; i < 12; ++i) ext[12 * c + i] = MFTL_DBL_MAX;
344 MFTLTopoCommAllreduceDouble(dom, ext, 12 * n_cl, 0);
345 MFTLTopoResolveExtents(dom, cl, n_cl, ext);
346 free(ext);

352 for (int o = 0; o < ens->n_local_objects; ++o)

530 *
531 * LAST, after the remesh: a freshly extracted surface is all slivers, and
532 * the pass that cleans them up smooths away about half a percent of a jet's
533 * volume on its own -- measured, when the correction ran before it, as a
534 * steady -0.5 per reconstruction the correction was blind to.
535 *
536 * Measured per cluster over the bodies that came out of it, restored by a
537 * uniform offset d along a displacement field u: the area-weighted normals of
538 * the OWNED vertices -- complete fans, the canonical halo guarantees it --
539 * with a rim vertex's wall-normal component removed. The ghost copies are
540 * then refreshed from their owners exactly as a solver's own vertex motion
541 * would be. The step is Newton's: V(d) is a cubic in d whose derivative at 0
542 * is the first variation of the enclosed volume, sum over triangles of
543 * S_t . mean(u) (S_t the area vector), which each rank sums over the corners
544 * it owns. Dividing by the AREA instead -- dV = A d, exact only when u is the
545 * face normal -- converges linearly at about 0.03 per step. Measured on a
546 * satellite remeshed at a target edge length near its size (a 15 % deficit):
547 * 1.3e-4 / 3e-6 / 8e-8 after two / three / four area steps, 2.8e-5 / 1e-9
548 * after two / three Newton steps; the pinched dumbbell reaches 1e-14 after
549 * three. A fixed count, so every rank runs the same collectives: a stopping
550 * test on a reduced double is two ranks disagreeing in the last ulp about
551 * whether to enter another. */

567 }

574 st->volume_after = total;
575 if (iter == n_iter) break;

577 for (int o = 0; o < ens->n_local_objects; ++o)

580 const int c = came_from(comps, n_comps, obj->global_id);

582 const double d = (cl[c].volume_old - va[2 * c]) / va[2 * c + 1];

585 double *const p = &obj->verts.pos[3 * v];
586 for (int k = 0; k < 3; ++k) p[k] += d * obj->verts.normal[3 * v + k];
587 MFTLApplyPeriodicity(dom, p);
588 }
589 }
590 MFTLUpdateGhostPositions(ens);
591 }
592 free(va);
593 MFTLEnsembleComputeTriangleGeometry(ens);
594 }
595 free(comps);
596 MFTLTopoFreeClusters(cl, n_cl);
597 const double t_end = MFTLTopoCommTime();
598
599 double t[5] = {t_rec - t0, t_sep - t_rec, t_red - t_sep, t_end - t_red, t_end - t0};
