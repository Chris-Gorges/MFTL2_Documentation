2 * @file spatial_halo.c
3 * @author Christian Gorges
4 * @date 2026-08-16
5 * @ingroup mftl_communication
6 *
7 * @brief The spatial halo: the parts of body B near this rank's share of body A,
8 * delivered as shadow objects. Design rationale: mftl_communication.h §6.
9 *
10 * Shape of one extension:
11 *
12 * 0. one record per held body (id, extent, radius, envelope)
13 * 1. publish the records to the graph neighbours count allgather + Iallgatherv
14 * 2. per-triangle extents of every sendable body overlapped with 1
15 * 3. match partners (grid or request list), classify and pack triangles
16 * 4. exchange vertices and triangles one MFTLExchangeStart, 2 payloads
17 * 5. rebuild one shadow object per arriving body MFTLBuildTopologyFromIndexList
18 * 6. derive the candidate pairs (near mode); prune the directory
19 *
20 * Work is organised per BODY, not per pair, so a triangle wanted by two
21 * partners on the same destination is packed -- and reported -- once. Two
22 * allocations per call are `[nproc]` send-buffer descriptor arrays (only
23 * neighbour entries are ever filled).
24 */

70/* ═══════════════════════════════════════════════════════════════════════════
71 * EXTENTS ARE CENTRE + HALF-WIDTH, NOT MIN + MAX
72 *
73 * "Near" is the minimum-image distance between centres against the sum of
74 * half-widths. A (lo, hi) box fails both ways on a periodic domain: two shares
75 * touching across the seam are a domain length apart in stored coordinates (a
76 * MISSED contact, decided before the BVH ever sees the triangles), and a share
77 * containing one seam-straddling triangle has a box spanning the whole domain
78 * (every neighbour then ships everything). A share wider than half the domain
79 * degrades to a half-width above L/2, which accepts everything -- extra traffic,
80 * never a miss. On a non-periodic axis the test is the plain interval test.
81 * ═══════════════════════════════════════════════════════════════════════════ */
82

84 * @brief Empty marker for an interval: no owned triangles contributed to it.
85 *
86 * @param h Per-axis half-widths.
87 * @return `true` if the interval carries nothing (a negative half-width).

89static inline bool interval_is_empty(const double *h) { return h[0] < 0.0; }

91/**
92 * @brief Minimum-image proximity of two centre/half-width extents.
93 *
94 * @param domain Supplies the periodicity and the domain sizes.
95 * @param ca Centre of the first extent.
96 * @param ha Half-widths of the first extent.
97 * @param cb Centre of the second extent.
98 * @param hb Half-widths of the second extent.
99 * @return `true` if on every axis the minimum-image distance between the
100 * centres is at most `ha + hb`.
101 */
102static inline bool interval_overlap(const MFTL_Domain *domain, const double *ca, const double *ha, const double *cb, const double *hb)
103{
104 for (int k = 0; k < 3; ++k)
105 {
106 double delta = ca[k] - cb[k];
107 if (domain->is_periodic[k]) delta -= domain->domain_size[k] * nearbyint(delta * domain->inv_domain_size[k]);
108 if (delta < 0.0) delta = -delta;
109 if (delta > ha[k] + hb[k]) return MFTL_FALSE;
110 }
111 return MFTL_TRUE;
112}

114/**
115 * @brief Wraps a scalar back into [global_min, global_max) on a periodic axis.
116 *
117 * @param domain Supplies the bounds and sizes.
118 * @param k Axis, 0..2.
119 * @param v A coordinate.
120 * @return @p v brought into `[global_min[k], global_max[k])`.
121 */
122static inline double wrap_axis(const MFTL_Domain *domain, int k, double v)
123{
124 if (!domain->is_periodic[k] || domain->domain_size[k] <= 0.0) return v;
125 if (v >= domain->global_max[k]) return v - domain->domain_size[k];
126 if (v < domain->global_min[k]) return v + domain->domain_size[k];
127 return v;
128}

130/**
131 * @brief Bounds of the OWNED triangles of @p obj (ghosts would inflate the
132 * extent by a halo layer that their owners publish themselves).
133 *
134 * @param obj The mesh.
135 * @param[out] lo Lower corner over the owned triangles.
136 * @param[out] hi Upper corner.
137 * @param shift Optional per-axis offset applied (with wrap) before
138 * measuring, for the half-domain-shifted frame. May be NULL.
139 */
140static void owned_bounds(const MFTL_Object *obj, double *lo, double *hi, const double *shift)
141{
142 box_clear(lo, hi);
143 if (obj == MFTL_NULL) return;
144
145 const MFTL_Domain *const domain = obj->domain;
146 const int nt = obj->tris.n_local;
147
148 for (int t = 0; t < nt; ++t)
149 {
150 double tlo[3], thi[3];
151 MFTLGetTriangleBounds(obj, t, tlo, thi);
152
153 if (shift != MFTL_NULL && domain != MFTL_NULL)
154 {
155 for (int k = 0; k < 3; ++k)
156 {
157 if (!domain->is_periodic[k] || domain->domain_size[k] <= 0.0) continue;
158
159 double moved = tlo[k] + shift[k];
160 if (moved >= domain->global_max[k])
161 moved -= domain->domain_size[k];
162 else if (moved < domain->global_min[k])
163 moved += domain->domain_size[k];
164
165 const double d = moved - tlo[k]; // what the wrap actually applied
166 tlo[k] = moved;
167 thi[k] += d;
168 }
169 }
170
171 for (int k = 0; k < 3; ++k)
172 {
173 if (tlo[k] < lo[k]) lo[k] = tlo[k];
174 if (thi[k] > hi[k]) hi[k] = thi[k];
175 }
176 }
177}

179/**
180 * @brief The owned extent of @p obj as a centre and half-width per axis.
181 *
182 * Measured in two frames -- raw and displaced by half a domain -- keeping the
183 * tighter per axis: a share straddling the seam is contiguous in the shifted
184 * frame, one straddling the middle in the raw frame, and none can straddle both
185 * while its extent is under L/2 (the minimum-image assumption). The centre of
186 * the shifted frame is mapped back through the shift, which a (lo, hi) pair
187 * could not survive.
188 *
189 * @param obj The mesh.
190 * @param domain Supplies the periodicity and the domain sizes.
191 * @param[out] c Per-axis centre.
192 * @param[out] h Per-axis half-width; negative on every axis when this rank
193 * owns nothing of the body.
194 */
195static void owned_interval(const MFTL_Object *obj, const MFTL_Domain *domain, double *c, double *h)
196{
197 double lo[3], hi[3];
198 owned_bounds(obj, lo, hi, MFTL_NULL);
199
200 if (box_is_empty(lo, hi))
201 {
202 for (int k = 0; k < 3; ++k)
203 {
204 c[k] = 0.0;
205 h[k] = -1.0;
206 }
207 return;
208 }
209
210 double half[3];
211 bool any_periodic = MFTL_FALSE;
212 for (int k = 0; k < 3; ++k)
213 {
214 half[k] = 0.5 * domain->domain_size[k];
215 if (domain->is_periodic[k] && domain->domain_size[k] > 0.0) any_periodic = MFTL_TRUE;
216 }
217
218 double slo[3], shi[3];
219 if (any_periodic) owned_bounds(obj, slo, shi, half);
220
221 for (int k = 0; k < 3; ++k)
222 {
223 double cc = 0.5 * (lo[k] + hi[k]);
224 double hh = 0.5 * (hi[k] - lo[k]);
225
226 if (any_periodic && domain->is_periodic[k] && domain->domain_size[k] > 0.0)
227 {
228 const double sh = 0.5 * (shi[k] - slo[k]);
229 if (sh < hh)
230 {
231 hh = sh;
232 cc = wrap_axis(domain, k, 0.5 * (slo[k] + shi[k]) - half[k]); // back out of the shifted frame
233 }
234 }
235
236 c[k] = wrap_axis(domain, k, cc);
237 h[k] = hh;
238 }
239}

248/**
249 * @brief Grows the three parallel directory arrays to hold @p need entries.
250 * @param h The halo.
251 * @param need Entries required.
252 */
253static void halo_reserve(MFTL_SpatialHalo *h, int need)
254{
255 if (need <= h->capacity) return;
256
257 int cap = (h->capacity > 0) ? h->capacity : 8;
258 while (cap < need) cap *= 2;
259
260 int *const gid = (int *) MFTLMallocOrDie((size_t) cap * sizeof(int), "spatial halo body ids");
261 MFTL_Object **const patch = (MFTL_Object **) MFTLCallocOrDie((size_t) cap, sizeof(MFTL_Object *), "spatial halo patches");
262 double *const rad = (double *) MFTLCallocOrDie((size_t) cap, sizeof(double), "spatial halo radii");
263
264 if (h->num_objects > 0)
265 {
266 memcpy(gid, h->body_gid, (size_t) h->num_objects * sizeof(int));
267 memcpy(patch, h->patch, (size_t) h->num_objects * sizeof(MFTL_Object *));
268 memcpy(rad, h->global_radius, (size_t) h->num_objects * sizeof(double));
269 }
270
271 free(h->body_gid);
272 free(h->patch);
273 free(h->global_radius);
274 h->body_gid = gid;
275 h->patch = patch;
276 h->global_radius = rad;
277 h->capacity = cap;
278}

280/**
281 * @brief Directory slot of @p gid, inserting an empty entry if it is absent.
282 * @param h The halo.
283 * @param gid The body id.
284 * @return The slot. Never #MFTL_ERROR.
285 */
286static int halo_entry(MFTL_SpatialHalo *h, int gid)
287{
288 const int found = MFTLSpatialHaloFindBody(h, gid);
289 if (found != MFTL_ERROR) return found;
290
291 halo_reserve(h, h->num_objects + 1);
292
293 int i = h->num_objects;
294 while (i > 0 && h->body_gid[i - 1] > gid) --i;
295
296 if (i < h->num_objects)
297 {
298 memmove(&h->body_gid[i + 1], &h->body_gid[i], (size_t) (h->num_objects - i) * sizeof(int));
299 memmove(&h->patch[i + 1], &h->patch[i], (size_t) (h->num_objects - i) * sizeof(MFTL_Object *));
300 memmove(&h->global_radius[i + 1], &h->global_radius[i], (size_t) (h->num_objects - i) * sizeof(double));
301 }
302
303 h->body_gid[i] = gid;
304 h->patch[i] = MFTL_NULL;
305 h->global_radius[i] = 0.0;
306 h->num_objects++;
307 return i;
308}

324/**
325 * @brief Appends one candidate pair. Ordering and uniqueness are imposed later.
326 * @param h The halo.
327 * @param a Smaller body id.
328 * @param b Larger body id.

330static void halo_pair_add(MFTL_SpatialHalo *h, int a, int b)
331{
332 if (h->n_pairs == h->cap_pairs)
333 {
334 h->cap_pairs = (h->cap_pairs > 0) ? h->cap_pairs * 2 : 64;
335 MFTL_SpatialHaloPair *const grown =

346/**
347 * @brief Orders pairs by (A, B), so removing duplicates is one linear sweep.
348 * @param x First element.
349 * @param y Second element.
350 * @return Negative, zero or positive in the usual qsort sense.
351 */
352static int pair_cmp(const void *x, const void *y)
353{
354 const MFTL_SpatialHaloPair *const a = (const MFTL_SpatialHaloPair *) x;
355 const MFTL_SpatialHaloPair *const b = (const MFTL_SpatialHaloPair *) y;
356 if (a->body_a != b->body_a) return (a->body_a < b->body_a) ? -1 : 1;
357 if (a->body_b != b->body_b) return (a->body_b < b->body_b) ? -1 : 1;
358 return 0;
359}

361/**
362 * @brief Sorts the pair list and removes duplicates in place. A pair produced
363 * twice (local-local and via a patch) would otherwise report every
364 * contact twice.
365 *
366 * @param h The halo.
367 */
368static void halo_pairs_finish(MFTL_SpatialHalo *h)
369{
370 if (h->n_pairs < 2) return;
371
372 qsort(h->pair, (size_t) h->n_pairs, sizeof(MFTL_SpatialHaloPair), pair_cmp);
373
374 int w = 1;
375 for (int i = 1; i < h->n_pairs; ++i)
376 {
377 if (h->pair[i].body_a == h->pair[w - 1].body_a && h->pair[i].body_b == h->pair[w - 1].body_b) continue;
378 h->pair[w++] = h->pair[i];
379 }
380 h->n_pairs = w;
381}

414int MFTLSpatialHaloResolveRadii(MFTL_Ensemble *ens, MFTL_SpatialHalo *halo)
415{
416 if (ens == MFTL_NULL || halo == MFTL_NULL) return MFTL_ERROR;
417
418 halo_sync_bodies(halo, ens);
419
420 const int b = ens->n_local_objects;
421
422 /* Twelve MIN slots per body: [0..2] lo, [3..5] -hi in the raw frame, [6..8]
423 * lo, [9..11] -hi in the half-domain-shifted frame (upper corners negated so
424 * every slot is a MIN). The smaller extent per axis wins, so a body on the
425 * seam does not measure as domain-wide -- which would inflate every envelope
426 * scaled by its radius and declare contacts between distant surfaces. One
427 * per-body reduction over the holders (MFTLEnsembleReduceBodies()). */
428 enum
429 {
430 STRIDE = 12 /* [0..2] lo, [3..5] -hi, raw | [6..8] lo, [9..11] -hi, shifted */
431 };
432
433 MFTL_BodyReduceOp ops[STRIDE];
434 for (int j = 0; j < STRIDE; ++j) ops[j] = MFTL_BODY_REDUCE_MIN;
435
436 double *const val = (double *) MFTLMallocOrDie((size_t) MFTL_MAX(b * STRIDE, 1) * sizeof(double), "spatial halo extents");
437
438 double half_domain[3];
439 for (int k = 0; k < 3; ++k) half_domain[k] = 0.5 * ens->domain.domain_size[k];
440
441 for (int o = 0; o < b; ++o)
442 {
443 double *const v = &val[(size_t) o * STRIDE];
444 for (int j = 0; j < STRIDE; ++j) v[j] = MFTL_DBL_MAX;
445
446 double lo[3], hi[3];
447
448 owned_bounds(ens->local_objects[o], lo, hi, MFTL_NULL);
449 if (box_is_empty(lo, hi)) continue; /* an identity row: this rank owns nothing of it */
450 for (int k = 0; k < 3; ++k)
451 {
452 v[k] = lo[k];
453 v[3 + k] = -hi[k];
454 }
455
456 owned_bounds(ens->local_objects[o], lo, hi, half_domain);
457 for (int k = 0; k < 3; ++k)
458 {
459 v[6 + k] = lo[k];
460 v[9 + k] = -hi[k];
461 }
462 }
463
464 MFTLEnsembleReduceBodies(ens, val, STRIDE, ops, MFTL_BODY_REDUCE_SCOPE_AUTO);
465
466 for (int o = 0; o < b; ++o)
467 {
468 const double *const v = &val[(size_t) o * STRIDE];
469 const int e = MFTLSpatialHaloFindBody(halo, ens->local_objects[o]->global_id);
470 MFTL_ASSERT(e != MFTL_ERROR && "MFTLSpatialHaloResolveRadii: halo_sync_bodies did not enter a held body");
471
472 if (v[0] == MFTL_DBL_MAX)
473 {
474 halo->global_radius[e] = 0.0; /* every holder that reached the reduction owns nothing of it */
475 continue;
476 }
477
478 /* Half the LONGEST SIDE (not the diagonal): R for a sphere of radius R. */
479 double longest = 0.0;
480 for (int k = 0; k < 3; ++k)
481 {
482 double d = (-v[3 + k]) - v[k];
483
484 if (ens->domain.is_periodic[k] && ens->domain.domain_size[k] > 0.0)
485 {
486 const double d_shifted = (-v[9 + k]) - v[6 + k];
487 if (d_shifted < d) d = d_shifted;
488 }
489
490 if (d > longest) longest = d;
491 }
492 halo->global_radius[e] = 0.5 * longest;
493 }
494
495 free(val);
496
497 /* Publish (body id, radius) one hop along comm_graph: a rank deciding what to
498 * SEND needs the size of the body it sends TOWARDS, which it may hold nothing
499 * of; underestimating it would miss cross-rank contacts. One hop suffices
500 * because the halo only ships along comm_graph. */
501 const MFTL_Domain *const dom = &ens->domain;
502 const int nn = dom->num_neighbors;
503 if (dom->nproc < 2 || nn == 0) return MFTL_SUCCESS;
504
505 double *const send = (double *) MFTLMallocOrDie(2 * (size_t) MFTL_MAX(b, 1) * sizeof(double), "spatial halo radius publish");
506 for (int o = 0; o < b; ++o)
507 {
508 const int g = ens->local_objects[o]->global_id;
509 send[2 * o] = (double) g;
510 send[2 * o + 1] = MFTLSpatialHaloRadius(halo, g);
511 }
512
513 int *const rcount = (int *) MFTLMallocOrDie((size_t) nn * sizeof(int), "spatial halo radius counts");
514 int *const rdispl = (int *) MFTLMallocOrDie((size_t) nn * sizeof(int), "spatial halo radius displacements");
515
516 int n_recv = 0;
517 double *const recv = MFTLNeighborAllgatherDoubles(dom, send, 2 * b, rcount, rdispl, &n_recv);
518 const size_t total = (size_t) n_recv;
519
520 /* MAX: a held body already has the (bit-identical) reduced value. */
521 for (size_t i = 0; i + 1 < total; i += 2)
522 {
523 const int g = (int) recv[i];
524 if (g < 0) continue;
525 const int e = halo_entry(halo, g);
526 if (recv[i + 1] > halo->global_radius[e]) halo->global_radius[e] = recv[i + 1];
527 }
528
529 free(recv);
530 free(rdispl);
531 free(rcount);
532 free(send);
533 return MFTL_SUCCESS;
534}

540/**
541 * @brief Turns the arrivals for one body into a valid, self-contained patch
542 * object (vertices first, then MFTLBuildTopologyFromIndexList()), and only
543 * then marks nothing owned.
544 *
545 * - **Vertex rotation.** The wire carries the TARGET of half-edge 3t+k; the
546 * index list wants the ORIGIN, `origin(3t+k) = target(3t+(k+2)%3)`.
547 * - **Boundary flags are restored, not derived.** The topology build marks every
548 * twin-less half-edge as mesh boundary, but a patch's rim is a cut through a
549 * larger surface, so the transmitted flags overwrite the derived ones.
550 *
551 * @param halo The halo; `halo->patch[entry]` is REPLACED.
552 * @param entry Directory slot of the body, from halo_entry().
553 * @param verts Received vertex records.
554 * @param n_verts Number of received vertices.
555 * @param tris Received triangle records.
556 * @param n_tris Number of received triangles.
557 * @param tbl Epoch-stamped id-to-slot table, reused across bodies.
558 * @param scratch Reusable int buffer for the de-duplication pass.
559 */
560static void rebuild_patch(MFTL_SpatialHalo *halo, int entry, const MFTL_PackedVertex *verts, int n_verts, const MFTL_PackedTriangleComplex *tris, int n_tris,
561 mftl_gid_table *tbl, mftl_int_buffer *scratch)
562{
563 MFTL_Object *patch = halo->patch[entry];
564
565 if (n_tris <= 0 || n_verts < 3)
566 {
567 if (patch != MFTL_NULL) MFTLObjectDestroy(&halo->patch[entry]);
568 return;
569 }
570
571 if (patch == MFTL_NULL)
572 {
573 patch = MFTLObjectCreate(halo->body_gid[entry]);
574 /* Borrowed domain; deliberately NOT in the ensemble's object list. */
575 patch->domain = halo->domain;
576 halo->patch[entry] = patch;
577 }
578
579 /* ── De-duplicate the arriving vertices by global id. A sender sends each
580 * vertex once per destination, but several senders may send the same one;
581 * merging them is what makes the patch connected, so twins link across it. */
582 int max_gid = 0;
583 for (int i = 0; i < n_verts; ++i)
584 {
585 if (verts[i].global_id > max_gid) max_gid = verts[i].global_id;
586 }
587 /* Patch ids are sparse over the body's range: gt_begin() picks sparse mode. */
588 gt_begin(tbl, n_verts, max_gid + 1);
589
590 ib_reset(scratch);
591
592 int nv = 0;
593 for (int i = 0; i < n_verts; ++i)
594 {
595 const int g = verts[i].global_id;
596 if (g < 0 || gt_has(tbl, g)) continue;
597 gt_put(tbl, g, nv++);
598 ib_add(scratch, i); // arrival index of the surviving copy, in slot order
599 }
600
601 if (nv < 3)
602 {
603 MFTLObjectDestroy(&halo->patch[entry]);
604 return;
605 }
606
607 /* ── Vertices ─────────────────────────────────────────────────────────── */
608 MFTLObjectSetCounts(patch, nv, nv, 0, 0);
609
610 int max_v_gid = 0;
611 for (int s = 0; s < nv; ++s)
612 {
613 const MFTL_PackedVertex *const p = &verts[scratch->data[s]];
614 double *const dst = &patch->verts.pos[3 * s];
615 dst[0] = p->pos[0];
616 dst[1] = p->pos[1];
617 dst[2] = p->pos[2];
618
619 MFTL_Vertex *const rec = &patch->verts.rec[s];
620 rec->global_id = p->global_id;
621 rec->owner_proc = p->owner_proc;
622 rec->flags = p->flags;
623 rec->out_he_idx = MFTL_ERROR; // filled in by the topology build
624
625 if (p->global_id > max_v_gid) max_v_gid = p->global_id;
626 }
627
628 /* ── Connectivity ─────────────────────────────────────────────────────── */
629 int *const tri_verts = (int *) MFTLMallocOrDie(3 * (size_t) n_tris * sizeof(int), "spatial halo index list");
630 int n_kept = 0;
631
632 for (int i = 0; i < n_tris; ++i)
633 {
634 const MFTL_PackedTriangleComplex *const p = &tris[i];
635
636 int slot[3];
637 bool ok = MFTL_TRUE;
638 for (int k = 0; k < 3; ++k)
639 {
640 const int g = p->vertex_gid[k];
641 if (g < 0 || !gt_has(tbl, g))
642 {
643 ok = MFTL_FALSE;
644 break;
645 }
646 slot[k] = gt_get(tbl, g);
647 }
648 if (!ok)
649 {
650 MFTL_ASSERT(0 && "spatial halo: a triangle references a vertex that never arrived");
651 continue;
652 }
653
654 /* origin(3t+k) == target(3t+(k+2)%3); the wire carries targets. */
655 for (int k = 0; k < 3; ++k) tri_verts[3 * n_kept + k] = slot[(k + 2) % 3];
656 n_kept++;
657 }
658
659 if (n_kept <= 0)
660 {
661 free(tri_verts);
662 MFTLObjectDestroy(&halo->patch[entry]);
663 return;
664 }
665
666 MFTLBuildTopologyFromIndexList(patch, tri_verts, n_kept);
667 free(tri_verts);
668
669 /* ── Restore transmitted identity over the freshly derived one ─────────── */
670 int max_t_gid = 0;
671 {
672 int w = 0;
673 for (int i = 0; i < n_tris && w < n_kept; ++i)
674 {
675 const MFTL_PackedTriangleComplex *const p = &tris[i];
676
677 bool ok = MFTL_TRUE;
678 for (int k = 0; k < 3; ++k)
679 {
680 const int g = p->vertex_gid[k];
681 if (g < 0 || !gt_has(tbl, g)) ok = MFTL_FALSE;
682 }
683 if (!ok) continue;
684
685 MFTL_Triangle *const rec = &patch->tris.rec[w];
686 rec->global_id = p->global_id;
687 rec->owner_proc = p->owner_proc;
688 rec->flags = p->flags;
689
690 for (int k = 0; k < 3; ++k)
691 {
692 MFTL_HalfEdge *const he = &patch->tris.he[3 * w + k];
693 he->global_id = 3 * p->global_id + k; // the implicit-indexing rule, mesh-wide
694 he->flags = p->he_flags[k];
695 }
696
697 if (p->global_id > max_t_gid) max_t_gid = p->global_id;
698 w++;
699 }
700 }
701
702 /* Id counters bound the ids (consumers size tables from them); not dense. */
703 patch->next_vertex_gid = max_v_gid + 1;
704 patch->next_triangle_gid = max_t_gid + 1;
705 patch->next_halfedge_gid = 3 * (max_t_gid + 1);
706
707 /* ── Nothing here is owned ─────────────────────────────────────────────── */
708 MFTLObjectSetCounts(patch, patch->verts.n, 0, patch->tris.n, 0);
709 MFTLObjectBuildDerived(patch);
710}

712/* ═══════════════════════════════════════════════════════════════════════════
713 * 4. The record exchange, the broad phase and the payload
714 *
715 * Each rank publishes one record per body it HOLDS -- (id, centre[3],
716 * half-width[3], global radius, envelope), nine doubles -- so the payload is
717 * O(local bodies x degree), independent of the ensemble size. Radius and
718 * envelope ride along because deciding whether to send B towards A needs A's
719 * envelope.
720 *
721 * Broad phase: a uniform grid whose cell is at least `2 x widest half-width +
722 * largest envelope` per axis, so overlapping extents are always in adjacent
723 * cells and a query visits 27 cells; O(bodies + records) instead of O(B^2). One
724 * domain-spanning body collapses the grid to one cell (all pairs within the
725 * neighbourhood) -- correct, and the right cost for a few large bodies.
726 * ═══════════════════════════════════════════════════════════════════════════ */
727
728/** @brief Doubles per published body record: gid, centre[3], half[3], radius, tol. */
729#define MFTL_HALO_REC 9
730
731/**
732 * @brief A uniform grid over the domain, one bucket list per cell.
733 *
734 * Cells are at least as wide as the reach on every axis, which makes a 27-cell
735 * query exhaustive. The cell count is capped at `4 x items + 64` so a domain much
736 * larger than the bodies does not allocate proportionally to the domain.
737 */
738typedef struct
739{
740 int n[3]; ///< Cells per axis, at least 1.
741 double inv_w[3]; ///< Reciprocal cell width per axis.
742 double origin[3]; ///< Lower corner of cell (0,0,0).
743 int *head; ///< [n0*n1*n2] first item in the cell, or MFTL_ERROR.
744 int *next; ///< [n_items] next item in the same cell, or MFTL_ERROR.
745 int n_items; ///< Items inserted.
746} body_grid;
747
748/**
749 * @brief The cell holding coordinate @p x along axis @p k, clamped into the grid
750 * in floating point before the cast (an out-of-range or NaN value would
751 * otherwise be undefined behaviour).
752 *
753 * @param g The grid.
754 * @param k Axis.
755 * @param x Coordinate.
756 * @return A cell index in `[0, g->n[k])`.
757 */
758static inline int grid_axis_cell(const body_grid *g, int k, double x)
759{
760 const double f = (x - g->origin[k]) * g->inv_w[k];
761 if (!(f >= 1.0)) return 0;
762 if (f >= (double) g->n[k]) return g->n[k] - 1;
763 return (int) f;
764}
765
766/**
767 * @brief Builds a grid over @p n_items centres, resolving the cell size first.
768 *
769 * @param g Receives the grid. Release with grid_free().
770 * @param dom Supplies the bounds and periodicity.
771 * @param centre First centre; `centre[i*stride + k]` is item i's axis k.
772 * @param stride Doubles between consecutive centres.
773 * @param n_items Items.
774 * @param reach Per-axis width two items' centres can be apart and still be
775 * within reach of each other. The cell is at least this wide.
776 */
777static void grid_build(body_grid *g, const MFTL_Domain *dom, const double *centre, int stride, int n_items, const double reach[3])
778{
779 g->n_items = n_items;
780
781 for (int k = 0; k < 3; ++k)
782 {
783 const double size = dom->domain_size[k];
784 double want = reach[k];
785 if (!(want > 0.0)) want = MFTL_DBL_MAX;
786
787 /* Clamped before the cast: a reach far below the domain size -- a
788 * degenerate body -- would otherwise be an out-of-range int conversion. */
789 int nk = 1;
790 if (size > 0.0 && want < size)
791 {
792 const double q = size / want;
793 nk = (q < (double) INT_MAX) ? (int) q : INT_MAX;
794 }
795 if (nk < 1) nk = 1;
796
797 g->n[k] = nk;
798 g->origin[k] = dom->global_min[k];
799 }
800
801 /* Cap the cell table at O(items): halve the longest axis until it fits
802 * (sizes compared in double, exact below 2^53; the cap stays <= INT_MAX). */
803 const long cap = MFTL_MIN(4L * (long) MFTL_MAX(n_items, 1) + 64L, (long) INT_MAX);
804 while ((double) g->n[0] * (double) g->n[1] * (double) g->n[2] > (double) cap)
805 {
806 int worst = 0;
807 for (int k = 1; k < 3; ++k)
808 {
809 if (g->n[k] > g->n[worst]) worst = k;
810 }
811 if (g->n[worst] <= 1) break;
812 g->n[worst] /= 2;
813 }
814 const long cells = (long) g->n[0] * (long) g->n[1] * (long) g->n[2];
815
816 for (int k = 0; k < 3; ++k)
817 {
818 const double w = (dom->domain_size[k] > 0.0) ? dom->domain_size[k] / (double) g->n[k] : 1.0;
819 g->inv_w[k] = 1.0 / w;
820 }
821
822 g->head = (int *) MFTLMallocOrDie((size_t) cells * sizeof(int), "spatial halo grid heads");
823 for (long i = 0; i < cells; ++i) g->head[i] = MFTL_ERROR;
824 g->next = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(n_items, 1) * sizeof(int), "spatial halo grid links");
825
826 for (int i = 0; i < n_items; ++i)
827 {
828 const double *const c = &centre[(size_t) i * stride];
829 const int idx = grid_axis_cell(g, 0, c[0]) + g->n[0] * (grid_axis_cell(g, 1, c[1]) + g->n[1] * grid_axis_cell(g, 2, c[2]));
830 g->next[i] = g->head[idx];
831 g->head[idx] = i;
832 }
833}

835/** @brief Releases a grid. @param g The grid. */
836static void grid_free(body_grid *g)
837{
838 free(g->head);
839 free(g->next);
840 g->head = MFTL_NULL;
841 g->next = MFTL_NULL;
842}

844/**
845 * @brief The distinct cell indices to visit along one axis around @p ck: three
846 * normally, fewer on an axis of at most three cells, where periodic
847 * wrap-around would visit a cell twice (and a duplicate candidate would
848 * pack a triangle twice).
849 *
850 * @param g The grid.
851 * @param dom Supplies the periodicity.
852 * @param k Axis.
853 * @param ck Cell of the query point on that axis.
854 * @param[out] out Up to 3 distinct cell indices.
855 * @return How many were written.
856 */
857static int grid_axis_cells(const body_grid *g, const MFTL_Domain *dom, int k, int ck, int out[3])
858{
859 const int n = g->n[k];
860
861 if (n <= 3)
862 {
863 if (!dom->is_periodic[k])
864 {
865 int m = 0;
866 for (int i = ck - 1; i <= ck + 1; ++i)
867 {
868 if (i >= 0 && i < n) out[m++] = i;
869 }
870 return m;
871 }
872 for (int i = 0; i < n; ++i) out[i] = i;
873 return n;
874 }
875
876 int m = 0;
877 for (int d = -1; d <= 1; ++d)
878 {
879 int i = ck + d;
880 if (i < 0 || i >= n)
881 {
882 if (!dom->is_periodic[k]) continue;
883 i = (i + n) % n;
884 }
885 out[m++] = i;
886 }
887 return m;
888}

890/**
891 * @brief Appends every item within one cell of @p c to @p out.
892 *
893 * @param g The grid.
894 * @param dom Supplies the periodicity.
895 * @param c Query centre.
896 * @param[out] out Receives item indices, appended.
897 */
898static void grid_collect(const body_grid *g, const MFTL_Domain *dom, const double *c, mftl_int_buffer *out)
899{
900 int base[3];
901 for (int k = 0; k < 3; ++k) base[k] = grid_axis_cell(g, k, c[k]);
902
903 int cx[3], cy[3], cz[3];
904 const int nx = grid_axis_cells(g, dom, 0, base[0], cx);
905 const int ny = grid_axis_cells(g, dom, 1, base[1], cy);
906 const int nz = grid_axis_cells(g, dom, 2, base[2], cz);
907
908 for (int iz = 0; iz < nz; ++iz)
909 {
910 for (int iy = 0; iy < ny; ++iy)
911 {
912 for (int ix = 0; ix < nx; ++ix)
913 {
914 const int idx = cx[ix] + cy[iy] * g->n[0] + cz[iz] * g->n[0] * g->n[1];
915 for (int i = g->head[idx]; i != MFTL_ERROR; i = g->next[i]) ib_add(out, i);
916 }
917 }
918 }
919}

921/**
922 * @brief Inserts @p gid into a sorted, unique int list, growing it on demand.
923 *
924 * @param[in,out] list Sorted list.
925 * @param[in,out] n Entries in use.
926 * @param[in,out] cap Entries allocated.
927 * @param gid The value.
928 * @return The slot @p gid now occupies.
929 */
930static int sorted_insert(int **list, int *n, int *cap, int gid)
931{
932 int lo = 0, hi = *n - 1;
933 while (lo <= hi)
934 {
935 const int mid = lo + ((hi - lo) >> 1);
936 if ((*list)[mid] == gid) return mid;
937 if ((*list)[mid] < gid)
938 lo = mid + 1;
939 else
940 hi = mid - 1;
941 }
942
943 if (*n == *cap)
944 {
945 *cap = (*cap > 0) ? *cap * 2 : 16;
946 int *const grown = (int *) MFTLMallocOrDie((size_t) *cap * sizeof(int), "spatial halo arrival bodies");
947 if (*n > 0) memcpy(grown, *list, (size_t) *n * sizeof(int));
948 free(*list);
949 *list = grown;
950 }
951
952 if (lo < *n) memmove(&(*list)[lo + 1], &(*list)[lo], (size_t) (*n - lo) * sizeof(int));
953 (*list)[lo] = gid;
954 (*n)++;
955 return lo;
956}

958/**
959 * @brief Drops directory entries this rank neither holds nor has a patch for, so
960 * the directory stays bounded by what is near.
961 *
962 * @param h The halo.
963 * @param ens The ensemble, for the "still held here" half of the test.
964 */
965static void halo_prune(MFTL_SpatialHalo *h, const MFTL_Ensemble *ens)
966{
967 int w = 0;
968 for (int e = 0; e < h->num_objects; ++e)
969 {
970 const bool keep = (h->patch[e] != MFTL_NULL) || (MFTLEnsembleFindLocalSlot(ens, h->body_gid[e]) != MFTL_ERROR);
971 if (!keep) continue;
972
973 h->body_gid[w] = h->body_gid[e];
974 h->patch[w] = h->patch[e];
975 h->global_radius[w] = h->global_radius[e];
976 w++;
977 }
978 h->num_objects = w;

981/**
982 * @brief Fills `halo->pair` from this rank's own records and the neighbours'.
983 *
984 * Two sources: LOCAL-LOCAL pairs (two bodies held here whose shares are within
985 * the envelope; no halo involved) and LOCAL-ARRIVED pairs (a neighbour's record
986 * of a LARGER body id within the envelope of a body held here, kept only if
987 * something of it is visible here) -- the mirror of the sender's packing
988 * decision. Plus `(B, B)` for every held body when `self_pairs` is set. The A
989 * side is always held here and has the smaller id.
990 *
991 * @param ens The ensemble.
992 * @param halo Receives the pairs.
993 * @param my_rec `[n_my * MFTL_HALO_REC]` this rank's published records.
994 * @param n_my Records in @p my_rec.
995 * @param nb_rec `[n_nb_rec * MFTL_HALO_REC]` the neighbours' records, or NULL.
996 * @param n_nb_rec Records in @p nb_rec.
997 * @rank_local No MPI.
998 */
999static void build_near_pairs(MFTL_Ensemble *ens, MFTL_SpatialHalo *halo, const double *my_rec, int n_my, const double *nb_rec, int n_nb_rec)
1000{
1001 if (n_my == 0) return;
1002
1003 const MFTL_Domain *const domain = &ens->domain;
1004
1005 double reach[3] = {0.0, 0.0, 0.0};
1006 double max_tol = 0.0;
1007 for (int i = 0; i < n_my; ++i)
1008 {
1009 const double *const r = &my_rec[(size_t) i * MFTL_HALO_REC];
1010 for (int k = 0; k < 3; ++k) reach[k] = MFTL_MAX(reach[k], 2.0 * r[4 + k]);
1011 max_tol = MFTL_MAX(max_tol, r[8]);
1012 }
1013 for (int i = 0; i < n_nb_rec; ++i)
1014 {
1015 const double *const r = &nb_rec[(size_t) i * MFTL_HALO_REC];
1016 for (int k = 0; k < 3; ++k) reach[k] = MFTL_MAX(reach[k], 2.0 * r[4 + k]);
1017 max_tol = MFTL_MAX(max_tol, r[8]);
1018 }
1019 for (int k = 0; k < 3; ++k) reach[k] += max_tol;
1020
1021 body_grid grid;
1022 grid_build(&grid, domain, my_rec + 1, MFTL_HALO_REC, n_my, reach);
1023
1024 mftl_int_buffer cand;
1025 ib_init(&cand, 64);
1026
1027 /* ── Local against local ─────────────────────────────────────────────── */
1028 for (int i = 0; i < n_my; ++i)
1029 {
1030 const double *const ri = &my_rec[(size_t) i * MFTL_HALO_REC];
1031 const int gi = (int) ri[0];
1032
1033 ib_reset(&cand);
1034 grid_collect(&grid, domain, ri + 1, &cand);
1035
1036 for (int ci = 0; ci < cand.size; ++ci)
1037 {
1038 const int j = cand.data[ci];
1039 if (j == i) continue;
1040
1041 const double *const rj = &my_rec[(size_t) j * MFTL_HALO_REC];
1042 const int gj = (int) rj[0];
1043 if (gj <= gi) continue; // each unordered pair once, A = the smaller id
1044
1045 const double m = MFTL_MAX(ri[8], rj[8]);
1046 double wh[3];
1047 for (int k = 0; k < 3; ++k) wh[k] = ri[4 + k] + m;
1048 if (!interval_overlap(domain, ri + 1, wh, rj + 1, rj + 4)) continue;
1049
1050 halo_pair_add(halo, gi, gj);
1051 }
1052 }
1053
1054 /* ── Local against what a neighbour published ────────────────────────── */
1055 for (int r = 0; r < n_nb_rec; ++r)
1056 {
1057 const double *const rr = &nb_rec[(size_t) r * MFTL_HALO_REC];
1058 const int gr = (int) rr[0];
1059
1060 ib_reset(&cand);
1061 grid_collect(&grid, domain, rr + 1, &cand);
1062
1063 for (int ci = 0; ci < cand.size; ++ci)
1064 {
1065 const int i = cand.data[ci];
1066 const double *const ri = &my_rec[(size_t) i * MFTL_HALO_REC];
1067 const int gi = (int) ri[0];
1068 if (gi >= gr) continue; // only the larger id travels, so only it can be the B side
1069
1070 const double m = MFTL_MAX(ri[8], rr[8]);
1071 double wh[3];
1072 for (int k = 0; k < 3; ++k) wh[k] = ri[4 + k] + m;
1073 if (!interval_overlap(domain, ri + 1, wh, rr + 1, rr + 4)) continue;
1074
1075 /* Only if something of B is visible here. */
1076 if (MFTLSpatialHaloPatch(halo, gr) == MFTL_NULL && MFTLEnsembleFindLocalSlot(ens, gr) == MFTL_ERROR) continue;
1077
1078 halo_pair_add(halo, gi, gr);
1079 }
1080 }
1081
1082 /* ── Self pairs, when asked: every body with an owned share here (a neck may
1083 * lie entirely inside one rank). */
1084 if (halo->self_pairs)
1085 {
1086 for (int i = 0; i < n_my; ++i)
1087 {
1088 const int gi = (int) my_rec[(size_t) i * MFTL_HALO_REC];
1089 halo_pair_add(halo, gi, gi);
1090 }
1091 }
1092
1093 ib_free(&cand);
1094 grid_free(&grid);
1095 halo_pairs_finish(halo);
1096}

1098/**
1099 * @brief The implementation behind both public forms of the halo exchange.
1100 *
1101 * @param ens The ensemble.
1102 * @param requests [n_requests] explicit pairs, or #MFTL_NULL in @p near mode.
1103 * @param n_requests Length of @p requests.
1104 * @param tolerance `[n_local_objects]` contact envelope per local slot, or
1105 * #MFTL_NULL for zero. Read only in @p near mode.
1106 * @param halo Filled in.
1107 * @param near #MFTL_TRUE to derive the pairs from the extents, #MFTL_FALSE
1108 * to take them from @p requests.
1109 * @return #MFTL_SUCCESS, or #MFTL_ERROR on a NULL argument.
1110 */
1111static int extend_impl(MFTL_Ensemble *ens, const MFTL_SpatialHaloRequest *requests, int n_requests, const double *tolerance, MFTL_SpatialHalo *halo, bool near)
1112{
1113 if (ens == MFTL_NULL || halo == MFTL_NULL) return MFTL_ERROR;
1114 if (n_requests < 0) return MFTL_ERROR;
1115 if (!near && n_requests > 0 && requests == MFTL_NULL) return MFTL_ERROR;
1116
1117 MFTL_Domain *const domain = &ens->domain;
1118 const int nproc = domain->nproc;
1119 const int nn = domain->num_neighbors;
1120
1121 halo_sync_bodies(halo, ens);
1122 const int nb = ens->n_local_objects;
1123 if (near) halo->n_pairs = 0;
1124
1125 MFTL_ASSERT(MFTL_MPI_PackedVertex != MPI_DATATYPE_NULL && "MFTLExtendSpatialHalo: call MFTLRegisterMpiTypes() first");
1126
1127 /* ── 0. One record per held body with an owned share, ascending by id (so a
1128 * neighbour's block can be binary-searched). */
1129 double *const my_rec = (double *) MFTLMallocOrDie((size_t) MFTL_MAX(nb, 1) * MFTL_HALO_REC * sizeof(double), "spatial halo body records");
1130 int n_my = 0;
1131
1132 for (int o = 0; o < nb; ++o)
1133 {
1134 double c[3], h[3];
1135 owned_interval(ens->local_objects[o], domain, c, h);
1136 if (interval_is_empty(h)) continue; // owns nothing of it: there is nothing to send and nothing to be near
1137
1138 const int e = MFTLSpatialHaloFindBody(halo, ens->local_objects[o]->global_id);
1139 double *const r = &my_rec[(size_t) n_my * MFTL_HALO_REC];
1140 r[0] = (double) ens->local_objects[o]->global_id;
1141 for (int k = 0; k < 3; ++k)
1142 {
1143 r[1 + k] = c[k];
1144 r[4 + k] = h[k];
1145 }
1146 r[7] = (e != MFTL_ERROR) ? halo->global_radius[e] : 0.0;
1147 r[8] = (tolerance != MFTL_NULL) ? tolerance[o] : 0.0;
1148 n_my++;
1149 }
1150
1151 /* ── 1. Discard the patches this call replaces: all of them in near mode

1157 {
1158 if (halo->patch[e] != MFTL_NULL) MFTLObjectDestroy(&halo->patch[e]);
1159 }

1164 {
1165 const int e = MFTLSpatialHaloFindBody(halo, requests[r].body_b);
1166 if (e != MFTL_ERROR && halo->patch[e] != MFTL_NULL) MFTLObjectDestroy(&halo->patch[e]);
1167 }

1170 /* ── 2. Publish the records: count allgather, then a non-blocking allgatherv
1171 * overlapped with step 4 (the only work independent of the answer). */
1172 int *const rcount = (int *) MFTLCallocOrDie((size_t) MFTL_MAX(nn, 1), sizeof(int), "spatial halo record counts");

1177 const int scount = n_my * MFTL_HALO_REC;
1178 const bool exchanging = (nproc > 1 && nn > 0);

1180 if (exchanging)
1181 {

1183

1189 }

1204 if (near)
1205 {

1207 }
1208 else
1209 {
1210 for (int r = 0; r < n_requests; ++r)
1211 {

1219 }
1220 for (int o = 0; o < nb; ++o) req_off[o + 1] += req_off[o];

1224 for (int o = 0; o < nb; ++o) cursor[o] = req_off[o];
1225 for (int r = 0; r < n_requests; ++r)
1226 {
1227 const int ob = MFTLEnsembleFindLocalSlot(ens, requests[r].body_b);
1228 if (ob == MFTL_ERROR) continue;
1229 req_list[cursor[ob]++] = requests[r].body_a;
1230 }
1231 free(cursor);
1232 }

1234 /* ── 4. Per-triangle extents of every sendable body, once, while the records
1235 * fly. The margin is folded into the PARTNER's half-width later, so the inner

1241
1242 if (exchanging)
1243 {

1245 {
1246 if (!is_b[o]) continue;
1247
1248 const MFTL_Object *const obj = ens->local_objects[o];

1267 * needs (erring towards sending more is the safe direction). */

1269 MFTL_SendBuffer *const send_v = (MFTL_SendBuffer *) MFTLCallocOrDie((size_t) nproc, sizeof(MFTL_SendBuffer), "spatial halo vertex buffers");
1270

1272 {
1273 /* Source neighbour of each arriving record. */
1274 int *const rec_src = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(n_nb_rec, 1) * sizeof(int), "spatial halo record sources");
1275 for (int i = 0; i < nn; ++i)
1276 {
1277 const int first = rdispl[i] / MFTL_HALO_REC;
1278 const int last = first + rcount[i] / MFTL_HALO_REC;
1279 for (int r = first; r < last; ++r) rec_src[r] = i;
1280 }

1283 * either side plus the widest envelope. */

1289 {

1291 for (int k = 0; k < 3; ++k) reach[k] = MFTL_MAX(reach[k], 2.0 * rc[4 + k]);
1292 max_tol = MFTL_MAX(max_tol, rc[8]);
1293 }
1294 for (int o = 0; o < n_my; ++o)
1295 {
1296 const double *const rc = &my_rec[(size_t) o * MFTL_HALO_REC];
1297 for (int k = 0; k < 3; ++k) reach[k] = MFTL_MAX(reach[k], 2.0 * rc[4 + k]);
1298 max_tol = MFTL_MAX(max_tol, rc[8]);
1299 }
1300 for (int r = 0; r < n_requests; ++r) max_tol = MFTL_MAX(max_tol, requests[r].margin);
1301 for (int k = 0; k < 3; ++k) reach[k] += max_tol;
1302
1303 grid_build(&grid, domain, (n_nb_rec > 0) ? nb_rec + 1 : MFTL_NULL, MFTL_HALO_REC, n_nb_rec, reach);
1304 }

1307 mftl_int_buffer cand;

1309 int *const mat_cnt = (int *) MFTLCallocOrDie((size_t) MFTL_MAX(nn, 1), sizeof(int), "spatial halo match counts");
1310 int *const mat_off = (int *) MFTLCallocOrDie((size_t) MFTL_MAX(nn, 1) + 1, sizeof(int), "spatial halo match offsets");
1311 mftl_int_buffer matched;
1312 ib_init(&matched, 64);
1313 mftl_int_buffer grouped;
1314 ib_init(&grouped, 64);
1315
1316 int max_nv = 0;
1317 for (int o = 0; o < nb; ++o) max_nv = MFTL_MAX(max_nv, ens->local_objects[o]->verts.n);
1318 unsigned *const vstamp = (unsigned *) MFTLCallocOrDie((size_t) MFTL_MAX(max_nv, 1), sizeof(unsigned), "spatial halo vertex stamps");
1319 unsigned vepoch = 0u;
1320
1321 /* Partner extents of one neighbour (bounded by its record count or by the
1322 * request count, since a caller may name a pair twice). */
1323 double *const ph =
1324 (double *) MFTLMallocOrDie(6 * (size_t) MFTL_MAX(MFTL_MAX(n_nb_rec, n_requests), 1) * sizeof(double), "spatial halo partner extents");
1325
1326 for (int o = 0; o < nb; ++o)
1327 {
1328 if (!is_b[o]) continue;
1329
1330 const MFTL_Object *const obj = ens->local_objects[o];
1331 const int nt = obj->tris.n_local;
1332 if (nt <= 0) continue;
1333
1334 const int gb = obj->global_id;
1335
1336 double my_c[3], my_h[3];

1338 if (interval_is_empty(my_h)) continue;
1339
1340 /* Partners. NEAR: arriving records of SMALLER ids (or the same id with
1341 * self_pairs) within the envelope -- the holder of the smaller id reports,
1342 * so only it needs the copy. REQUEST: the named A bodies, binary-searched
1343 * in each neighbour's sorted block. */
1344 ib_reset(&matched);
1345 for (int i = 0; i < nn; ++i) mat_cnt[i] = 0;
1346
1347 const double my_tol = (tolerance != MFTL_NULL) ? tolerance[o] : 0.0;
1348
1349 if (near)
1350 {
1351 ib_reset(&cand);
1352 grid_collect(&grid, domain, my_c, &cand);

1354 for (int ci = 0; ci < cand.size; ++ci)
1355 {
1356 const int r = cand.data[ci];
1357 const double *const rc = &nb_rec[(size_t) r * MFTL_HALO_REC];
1358 const int ga = (int) rc[0];
1359
1360 if (ga > gb || (ga == gb && !halo->self_pairs)) continue;
1361 if (domain->neighbors[rec_src[r]] == domain->rank) continue; // a thin periodic grid can name this rank
1362
1363 const double m = MFTL_MAX(my_tol, rc[8]);
1364 double wh[3];
1365 for (int k = 0; k < 3; ++k) wh[k] = my_h[k] + m;
1366 if (!interval_overlap(domain, my_c, wh, rc + 1, rc + 4)) continue;
1367
1368 ib_add(&matched, r);
1369 mat_cnt[rec_src[r]]++;
1370 if (m > margin_of[o]) margin_of[o] = m;
1371 }
1372 }
1373 else
1374 {
1375 double wh[3];
1376 for (int k = 0; k < 3; ++k) wh[k] = my_h[k] + margin_of[o];
1377
1378 for (int i = 0; i < nn; ++i)
1379 {
1380 if (domain->neighbors[i] == domain->rank) continue;
1381
1382 const int first = rdispl[i] / MFTL_HALO_REC;
1383 const int nrec = rcount[i] / MFTL_HALO_REC;
1384
1385 for (int pi = req_off[o]; pi < req_off[o + 1]; ++pi)
1386 {
1387 const int ga = req_list[pi];
1388
1389
1390 int lo = 0, hi = nrec - 1, found = MFTL_ERROR;
1391 while (lo <= hi)
1392 {
1393 const int mid = lo + ((hi - lo) >> 1);
1394 const int g = (int) nb_rec[(size_t) (first + mid) * MFTL_HALO_REC];
1395 if (g == ga)
1396 {
1397 found = first + mid;
1398 break;
1399 }
1400 if (g < ga)
1401 lo = mid + 1;
1402 else
1403 hi = mid - 1;
1404 }
1405 if (found == MFTL_ERROR) continue;
1406
1407 const double *const rc = &nb_rec[(size_t) found * MFTL_HALO_REC];
1408 if (!interval_overlap(domain, my_c, wh, rc + 1, rc + 4)) continue;
1409
1410 ib_add(&matched, found);
1411 mat_cnt[i]++;
1412 }
1413 }
1414 }
1415
1416 if (matched.size == 0) continue;
1417
1418 /* Group by neighbour so each destination gets each triangle at most once. */
1419 mat_off[0] = 0;
1420 for (int i = 0; i < nn; ++i) mat_off[i + 1] = mat_off[i] + mat_cnt[i];
1421
1422 ib_reset(&grouped);
1423 for (int i = 0; i < matched.size; ++i) ib_add(&grouped, 0);
1424 {
1425 int *const cur = (int *) MFTLMallocOrDie((size_t) MFTL_MAX(nn, 1) * sizeof(int), "spatial halo group cursors");
1426 for (int i = 0; i < nn; ++i) cur[i] = mat_off[i];
1427 for (int i = 0; i < matched.size; ++i) grouped.data[cur[rec_src[matched.data[i]]]++] = matched.data[i];
1428 free(cur);
1429 }
1430
1431 const double m = margin_of[o];
1432 const double *const bx = &boxes[6 * (size_t) box_off[o]];
1433
1434 for (int i = 0; i < nn; ++i)
1435 {
1436 const int np = mat_off[i + 1] - mat_off[i];
1437 if (np == 0) continue;
1438
1439 const int dest = domain->neighbors[i];
1440
1441 /* Partner extents with the margin folded into the half-width. */
1442 for (int j = 0; j < np; ++j)
1443 {
1444 const double *const rc = &nb_rec[(size_t) grouped.data[mat_off[i] + j] * MFTL_HALO_REC];
1445 for (int k = 0; k < 3; ++k)
1446 {
1447 ph[6 * j + k] = rc[1 + k];
1448 ph[6 * j + 3 + k] = rc[4 + k] + m;
1449 }
1450 }
1451
1452 vepoch++;
1453 if (vepoch == 0u)
1454 {
1455 memset(vstamp, 0, (size_t) MFTL_MAX(max_nv, 1) * sizeof(unsigned));
1456 vepoch = 1u;
1457 }
1458
1459 for (int t = 0; t < nt; ++t)
1460 {
1461 bool wanted = MFTL_FALSE;
1462 for (int j = 0; j < np; ++j)
1463 {
1464 if (interval_overlap(domain, &bx[6 * t], &bx[6 * t + 3], &ph[6 * j], &ph[6 * j + 3]))
1465 {
1466 wanted = MFTL_TRUE; // packed once, however many partners want it
1467 break;
1468 }
1469 }
1470 if (!wanted) continue;
1471
1472 /* --- the triangle --- */
1473 MFTL_SendBuffer *const sbt = &send_t[dest];
1474 if (sbt->count == sbt->capacity)
1475 {
1476 sbt->capacity = (sbt->capacity > 0) ? sbt->capacity * 2 : 256;
1477 MFTL_PackedTriangleComplex *const grown =
1478 (MFTL_PackedTriangleComplex *) MFTLMallocOrDie((size_t) sbt->capacity * sizeof(MFTL_PackedTriangleComplex), "spatial halo triangle send");
1479 if (sbt->count > 0) memcpy(grown, sbt->buffer, (size_t) sbt->count * sizeof(MFTL_PackedTriangleComplex));
1480 free(sbt->buffer);
1481 sbt->buffer = grown;
1482 }
1483
1484 MFTL_PackedTriangleComplex *const pt = &((MFTL_PackedTriangleComplex *) sbt->buffer)[sbt->count++];
1485 pt->global_id = obj->tris.rec[t].global_id;
1486 pt->owner_proc = obj->tris.rec[t].owner_proc;
1487 pt->flags = obj->tris.rec[t].flags;
1488 pt->body_gid = gb;
1489 for (int k = 0; k < 3; ++k)
1490 {
1491 const MFTL_HalfEdge *const he = &obj->tris.he[3 * t + k];
1492 /* The TARGET of half-edge 3t+k (not tris.verts[3t+k], its origin). */
1493 pt->vertex_gid[k] = obj->verts.rec[he->vertex_idx].global_id;
1494 pt->he_flags[k] = he->flags;
1495 }
1496
1497 /* --- its three vertices, each at most once per destination --- */
1498 for (int k = 0; k < 3; ++k)
1499 {
1500 const int v = obj->tris.he[3 * t + k].vertex_idx;
1501 if (vstamp[v] == vepoch) continue;
1502 vstamp[v] = vepoch;
1503
1504 MFTL_SendBuffer *const sbv = &send_v[dest];
1505 if (sbv->count == sbv->capacity)
1506 {
1507 sbv->capacity = (sbv->capacity > 0) ? sbv->capacity * 2 : 512;
1508 MFTL_PackedVertex *const grown =
1509 (MFTL_PackedVertex *) MFTLMallocOrDie((size_t) sbv->capacity * sizeof(MFTL_PackedVertex), "spatial halo vertex send");
1510 if (sbv->count > 0) memcpy(grown, sbv->buffer, (size_t) sbv->count * sizeof(MFTL_PackedVertex));
1511 free(sbv->buffer);
1512 sbv->buffer = grown;
1513 }
1514
1515 MFTL_PackedVertex *const pv = &((MFTL_PackedVertex *) sbv->buffer)[sbv->count++];
1516 const double *const src = &obj->verts.pos[3 * v];
1517 pv->pos[0] = src[0];
1518 pv->pos[1] = src[1];
1519 pv->pos[2] = src[2];
1520 pv->global_id = obj->verts.rec[v].global_id;
1521 pv->owner_proc = obj->verts.rec[v].owner_proc;
1522 pv->flags = obj->verts.rec[v].flags;
1523 pv->body_gid = gb;
1524 }
1525 }
1526 }
1527 }

1530 free(vstamp);

1552 x[0].type = MFTL_MPI_PackedVertex;

1626 for (int i = 0; i < n_blist; ++i)
1627 {
1628 rebuild_patch(halo, entry_of[i], &sv[v_off[i]], v_off[i + 1] - v_off[i], &st[t_off[i]], t_off[i + 1] - t_off[i], &tbl, &scratch);
