21void MFTLRebuildTwins(MFTL_Object *obj)
22{
23 MFTL_ASSERT(obj != MFTL_NULL);
24
25 const int n_he = 3 * obj->tris.n;
26 const int n_vert = obj->verts.n;
27 MFTL_HalfEdge *const he = obj->tris.he;
28
29 if (n_he < 1 || n_vert < 1) return;
30
31 /* Half-edges are bucketed by ORIGIN vertex in an intrusive singly linked list:
32 *
33 * v_heads[u] = first half-edge whose origin is u
34 * he_next[h] = next half-edge sharing h's origin
35 *
36 * The twin of (u -> v) is found by walking the bucket of v for an edge ending
37 * at u; a bucket is as long as the valence. Both arrays share one allocation. */
38 int *scratch = (int *) MFTLMallocOrDie(sizeof(int) * ((size_t) n_vert + (size_t) n_he), "MFTLRebuildTwins");
39
40 int *const v_heads = scratch;
41 int *const he_next = scratch + n_vert;
42
43 for (int v = 0; v < n_vert; ++v) v_heads[v] = MFTL_ERROR;
44
45 /* ── Pass 1: bucket by origin, and reset every twin ────────────────────── */
46 for (int h = 0; h < n_he; ++h)
47 {
48 const int u = he[MFTLHePrevIdx(h)].vertex_idx;
49 MFTL_ASSERT(u >= 0 && u < n_vert && "MFTLRebuildTwins: half-edge origin out of range");
50
51 he_next[h] = v_heads[u];
52 v_heads[u] = h;
53
54 he[h].twin_idx = MFTL_ERROR; // assume boundary until pass 2 finds a partner
55 }
56
57 /* ── Pass 2: pair them off ─────────────────────────────────────────────────
58 *
59 * Each match links both halves, so already-linked half-edges are skipped.
60 * Requiring the candidate to be UNLINKED keeps the relation symmetric on
61 * non-manifold input (three or more half-edges on one undirected edge): the
62 * surplus stays twin-less for MFTLCheckMeshIntegrity() to report. */
63 for (int h = 0; h < n_he; ++h)
64 {
65 if (he[h].twin_idx != MFTL_ERROR) continue; // already linked by its partner
66
67 const int u = he[MFTLHePrevIdx(h)].vertex_idx; // origin
68 const int v = he[h].vertex_idx; // target
69 MFTL_ASSERT(v >= 0 && v < n_vert && "MFTLRebuildTwins: half-edge target out of range");
70
71 for (int cand = v_heads[v]; cand != MFTL_ERROR; cand = he_next[cand])
72 {
73 if (he[cand].vertex_idx != u || he[cand].twin_idx != MFTL_ERROR) continue;
74
75 he[h].twin_idx = cand;
76 he[cand].twin_idx = h;
77 break;
78 }
79 }
80
81 free(scratch);
82
83 /* `tris.neighbors` is a function of the twins, so it is stale by definition. */
84 MFTLObjectInvalidateDerived(obj);
85}

87int MFTLMarkMeshBoundary(MFTL_Object *obj)
88{
89 MFTL_ASSERT(obj != MFTL_NULL);
90 MFTL_ASSERT(obj->tris.n_local == obj->tris.n &&
91 "MFTLMarkMeshBoundary: mesh is partitioned -- the flag can only be derived from a COMPLETE mesh");
92
93 MFTL_Vertex *const vr = obj->verts.rec;
94 MFTL_HalfEdge *const he = obj->tris.he;
95 const int n_verts = obj->verts.n;
96 const int n_he = 3 * obj->tris.n;
97
98 /* Pass 1: clear, so a re-run replaces the previous marking. */
99 for (int i = 0; i < n_verts; ++i) vr[i].flags &= ~MFTL_FLAG_BOUNDARY;
100 for (int i = 0; i < n_he; ++i) he[i].flags &= ~MFTL_FLAG_BOUNDARY;
101
102 /* Pass 2: set on every twin-less half-edge and both its endpoints. */
103 int n_boundary = 0;
104 for (int h = 0; h < n_he; ++h)
105 {
106 if (he[h].twin_idx != MFTL_ERROR) continue;
107
108 he[h].flags |= MFTL_FLAG_BOUNDARY;
109 n_boundary++;
110
111 const int target = he[h].vertex_idx;
112 const int origin = he[MFTLHePrevIdx(h)].vertex_idx;
113
114 MFTL_ASSERT(target >= 0 && target < n_verts && origin >= 0 && origin < n_verts);
115 vr[target].flags |= MFTL_FLAG_BOUNDARY;
116 vr[origin].flags |= MFTL_FLAG_BOUNDARY;
117 }
118
119 return n_boundary;
120}

122int MFTLCheckMeshIntegrity(const MFTL_Object *obj)
123{
124 MFTL_ASSERT(obj != MFTL_NULL);
125
126 int ok = MFTL_SUCCESS;
127 int n_boundary = 0;
128
129 const MFTL_Vertex *const vr = obj->verts.rec;
130 const MFTL_HalfEdge *const he = obj->tris.he;
131
132 const int n_verts = obj->verts.n;
133 const int n_tris = obj->tris.n;
134 const int n_hes = 3 * n_tris;
135
136 /* The next-cycle, the face back-reference and "every triangle has half-edges"
137 * are tautologies under implicit indexing and need no check. */
138
139 /* ── Vertices ─────────────────────────────────────────────────────────── */
140 for (int v = 0; v < n_verts; ++v)
141 {
142 const int oh = vr[v].out_he_idx;
143
144 if (oh == MFTL_ERROR)
145 {
146 fprintf(stderr, "[MeshCheck] Vertex %d has no outgoing half-edge\n", vr[v].global_id);
147 ok = MFTL_ERROR;
148 }
149 else if (oh < 0 || oh >= n_hes)
150 {
151 fprintf(stderr, "[MeshCheck] Vertex %d has out-of-bounds half-edge index %d (max %d)\n", vr[v].global_id, oh, n_hes);
152 ok = MFTL_ERROR;
153 }
154 else if (he[MFTLHePrevIdx(oh)].vertex_idx != v)
155 {
156 /* out_he_idx must be OUTGOING from v. A stale value (a topology edit that
157 * forgot the vertex) makes every one-ring walk from v traverse another fan. */
158 fprintf(stderr, "[MeshCheck] Vertex %d: half-edge %d does not start at it\n", vr[v].global_id, oh);
159 ok = MFTL_ERROR;
160 }
161 }
162
163 /* ── Half-edges ───────────────────────────────────────────────────────── */
164 for (int h = 0; h < n_hes; ++h)
165 {
166 if (he[h].vertex_idx < 0 || he[h].vertex_idx >= n_verts)
167 {
168 fprintf(stderr, "[MeshCheck] Half-edge %d has invalid target vertex index %d\n", he[h].global_id, he[h].vertex_idx);
169 ok = MFTL_ERROR;
170 continue;
171 }
172
173 const int tw = he[h].twin_idx;
174
175 if (tw == MFTL_ERROR)
176 {
177 n_boundary++;
178 continue; // a genuine boundary edge: nothing further to compare
179 }
180
181 if (tw < 0 || tw >= n_hes)
182 {
183 fprintf(stderr, "[MeshCheck] Half-edge %d has out-of-bounds twin index %d\n", he[h].global_id, tw);
184 ok = MFTL_ERROR;
185 continue;
186 }
187
188 if (he[tw].twin_idx != h)
189 {
190 fprintf(stderr, "[MeshCheck] Half-edge %d twin relation is not symmetric\n", he[h].global_id);
191 ok = MFTL_ERROR;
192 }
193
194 /* The origin of h must be the target of its twin, and vice versa. */
195 const int h_origin = he[MFTLHePrevIdx(h)].vertex_idx;
196 const int tw_origin = he[MFTLHePrevIdx(tw)].vertex_idx;
197
198 if (h_origin != he[tw].vertex_idx || he[h].vertex_idx != tw_origin)
199 {
200 fprintf(stderr, "[MeshCheck] Half-edge %d and its twin disagree about their endpoints\n", he[h].global_id);
201 ok = MFTL_ERROR;
202 }
203 }
204
205 if (ok == MFTL_SUCCESS)
206 {
207 /* Twin-less half-edges are legal (open surfaces); report their count. */
208 if (n_boundary == 0)
209 printf("[MeshCheck] Mesh is consistent and closed (%d vertices, %d triangles).\n", n_verts, n_tris);
210 else
211 printf("[MeshCheck] Mesh is consistent and open (%d vertices, %d triangles, %d boundary half-edges).\n", n_verts, n_tris, n_boundary);
212 }
213
214 return ok;
215}

217int MFTLLabelConnectedComponents(const MFTL_Object *obj, int *labels)
218{
219 MFTL_ASSERT(obj != MFTL_NULL);
220
221 const int nt = obj->tris.n;
222 if (nt <= 0) return 0;
223
224 /* One allocation: the labels when the caller wants none, then the queue. */
225 int *const scratch = (int *) MFTLMallocOrDie((size_t) (labels ? nt : 2 * nt) * sizeof(int), "MFTLLabelConnectedComponents");
226 int *const lab = labels ? labels : scratch + nt;
227 int *const queue = scratch;
228
229 for (int t = 0; t < nt; ++t) lab[t] = MFTL_ERROR;
230
231 const MFTL_HalfEdge *const he = obj->tris.he;
232 int n_comp = 0;
233
234 /* Seeds in ascending index: a component's label is the rank of its smallest
235 * triangle, a function of the pool alone. */
236 for (int seed = 0; seed < nt; ++seed)
237 {
238 if (lab[seed] != MFTL_ERROR) continue;
239
240 int head = 0, tail = 0;
241 queue[tail++] = seed;
242 lab[seed] = n_comp;
243
244 while (head < tail)
245 {
246 const int t = queue[head++];
247 for (int k = 0; k < 3; ++k)
248 {
249 const int tw = he[3 * t + k].twin_idx;
250 if (tw == MFTL_ERROR) continue;
251 const int u = MFTLHeTriIdx(tw);
252 MFTL_ASSERT(u >= 0 && u < nt && "MFTLLabelConnectedComponents: twin outside the pool");
253 if (lab[u] != MFTL_ERROR) continue;
254 lab[u] = n_comp;
255 queue[tail++] = u;
256 }
257 }
258 n_comp++;
259 }
260
261 free(scratch);
262 return n_comp;
263}
