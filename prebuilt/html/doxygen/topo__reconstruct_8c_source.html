1/**
2 * @file topo_reconstruct.c
3 * @author Christian Gorges
4 * @date 2026-09-25
5 * @ingroup mftl_topology_change
6 *
7 * @brief The field and the surface: rasterisation, the overlap exchange, the
8 * bias, and marching cubes on this rank's cells.
9 *
10 * ## WHY EACH RANK RASTERISES ONLY WHAT IT OWNS
11 *
12 * The value at a lattice node is the minimum over EVERY triangle within the
13 * cutoff, and those triangles may be owned by any of up to eight ranks around
14 * the node. Two ways to get it:
15 *
16 * PULL the triangles: grow a spatial halo of every neighbour's triangles near
17 * this box and rasterise everything locally. Triangles near a corner are
18 * shipped to seven ranks and rasterised by eight.
19 *
20 * PUSH the values: each rank rasterises the triangles it OWNS -- each exactly
21 * once in the job -- onto a padded window, and sends every node another rank
22 * needs. MFTLSdfMerge() is commutative and associative, so the merged value is
23 * the same whatever order the contributions arrive in, and bit-identical to a
24 * serial rasterisation of the whole mesh.
25 *
26 * PUSH is what a CFD code does with a ghost-cell halo, and it moves one double
27 * per boundary node instead of a triangle per nearby triangle.
28 *
29 * ## WHICH NODES A RANK NEEDS
30 *
31 * Its own box of nodes and one more layer on every side: marching cubes runs on
32 * its own cells AND one layer of the neighbours' cells, so that the triangles
33 * around every vertex it owns are present -- the vertex-complete halo, built by
34 * the extraction rather than by a later exchange.
35 */

52static inline int wrap_index(const MFTL_TopoLattice *lat, int k, int g)
53{
54 if (!lat->periodic[k]) return g;
55 const int n = lat->n_global[k];
56 int w = g % n;
57 if (w < 0) w += n;
58 return w;
59}

61int MFTLTopoCellOwner(const MFTL_Domain *dom, const MFTL_TopoLattice *lat, long long cell)
62{
63 const long long nx = lat->n_global[0], ny = lat->n_global[1];
64 const int ci = (int) (cell % nx), cj = (int) ((cell / nx) % ny), ck = (int) (cell / (nx * ny));
65 const int c[3] = {ci / lat->per_rank[0], cj / lat->per_rank[1], ck / lat->per_rank[2]};
66 return dom->rank_map[c[0] + c[1] * dom->cuts[0] + c[2] * dom->cuts[0] * dom->cuts[1]];
67}

75void MFTLTopoRasterise(const MFTL_Ensemble *ens, MFTL_TopoCluster *cl, int n_cl, const MFTL_TopoLattice *lat)
76{
77 for (int o = 0; o < ens->n_local_objects; ++o)
78 {
79 const MFTL_Object *const obj = ens->local_objects[o];
80 const int c = MFTLTopoClusterOf(cl, n_cl, obj->global_id);
81 if (c < 0 || !cl[c].has_window) continue;
82 /* OWNED triangles only: each triangle is rasterised by exactly one rank in
83 * the job, so the merge sees every contribution exactly once. */
84 MFTLRasterizeSignedDistance(&cl[c].grid, obj, obj->tris.n_local, lat->r_cut);
85 }

88/**
89 * @brief The rank-cells along one axis whose NEED range -- box nodes plus one
90 * layer -- contains wrapped node @p g.
91 * @return How many were written to @p out (at most 4).
92 */
93static int needing_cells(const MFTL_Domain *dom, const MFTL_TopoLattice *lat, int k, int g, int out[4])
94{
95 const int n = lat->per_rank[k], cuts = dom->cuts[k], N = lat->cells[k];
96 int m = 0;
97 for (int c = g / n - 2; c <= g / n + 2 && m < 4; ++c)
98 {
99 int cw = c;
100 if (lat->periodic[k])
101 {
102 cw %= cuts;
103 if (cw < 0) cw += cuts;
104 }
105 else if (c < 0 || c >= cuts)
106 continue;
107
108 const int lo = cw * n - 1, hi = (cw + 1) * n + 1; // inclusive need range, unwrapped
109 bool in;
110 if (lat->periodic[k])
111 {
112 int d = (g - lo) % N;
113 if (d < 0) d += N;
114 in = d <= hi - lo || (hi - lo + 1 >= N);
115 }
116 else
117 in = g >= lo && g <= hi;
118 if (!in) continue;
119
120 bool dup = MFTL_FALSE;
121 for (int i = 0; i < m; ++i)
122 if (out[i] == cw) dup = MFTL_TRUE;
123 if (!dup) out[m++] = cw;
124 }
125 return m;
126}

128bool MFTLTopoPackNodeExchange(const MFTL_Domain *dom, const MFTL_TopoLattice *lat, const MFTL_TopoCluster *cl, int n_cl, int *dest_counts, double **records)
129{
130 const int np = dom->nproc;
131 for (int r = 0; r < np; ++r) dest_counts[r] = 0;
132 bool far = MFTL_FALSE;
133
134 /* Two passes over the known nodes: count per destination, then fill. The
135 * node lists are short -- a band of r_cut around the surface -- so walking
136 * twice beats a growable buffer per destination. */
137 int *cursor = MFTL_NULL;
138 double *buf = MFTL_NULL;
139
140 for (int pass = 0; pass < 2; ++pass)
141 {
142 for (int c = 0; c < n_cl; ++c)
143 {
144 if (!cl[c].has_window) continue;
145 const MFTL_SdfGrid *const g = &cl[c].grid;
146 const int nx = g->dims[0], ny = g->dims[1], nz = g->dims[2];
147
148 for (int lz = 0; lz < nz; ++lz)
149 for (int ly = 0; ly < ny; ++ly)
150 for (int lx = 0; lx < nx; ++lx)
151 {
152 const double v = g->phi[((size_t) lz * ny + ly) * nx + lx];
153 if (v == MFTL_SDF_UNKNOWN) continue;
154
155 const int gw[3] = {wrap_index(lat, 0, g->lo[0] + lx), wrap_index(lat, 1, g->lo[1] + ly), wrap_index(lat, 2, g->lo[2] + lz)};
156 int cx[4], cy[4], cz[4];
157 const int mx = needing_cells(dom, lat, 0, gw[0], cx);
158 const int my = needing_cells(dom, lat, 1, gw[1], cy);
159 const int mz = needing_cells(dom, lat, 2, gw[2], cz);
160
161 const double key = (double) (((long long) gw[2] * lat->n_global[1] + gw[1]) * lat->n_global[0] + gw[0]);
162
163 /* At most 64 (cell, cell, cell) combinations, few of them distinct
164 * ranks; de-duplicate so a rank is sent a node once. */
165 int sent_to[64];
166 int n_sent = 0;
167 for (int a = 0; a < mx; ++a)
168 for (int b = 0; b < my; ++b)
169 for (int d = 0; d < mz; ++d)
170 {
171 const int r = dom->rank_map[cx[a] + cy[b] * dom->cuts[0] + cz[d] * dom->cuts[0] * dom->cuts[1]];
172 if (r == dom->rank) continue;
173 bool dup = MFTL_FALSE;
174 for (int i = 0; i < n_sent; ++i)
175 if (sent_to[i] == r) dup = MFTL_TRUE;
176 if (dup) continue;
177 sent_to[n_sent++] = r;
178
179 if (pass == 0)
180 {
181 dest_counts[r]++;
182 if (!MFTLTopoIsNeighbour(dom, r)) far = MFTL_TRUE;
183 }
184 else
185 {
186 double *const rec = &buf[3 * (size_t) cursor[r]++];
187 rec[0] = (double) c;
188 rec[1] = key;
189 rec[2] = v;
190 }
191 }
192 }
193 }
194
195 if (pass == 0)
196 {
197 int total = 0;
198 cursor = (int *) MFTLMallocOrDie((size_t) np * sizeof(int), "topology node cursors");
199 for (int r = 0; r < np; ++r)
200 {
201 cursor[r] = total;
202 total += dest_counts[r];
203 }
204 buf = (double *) MFTLMallocOrDie(3 * (size_t) MFTL_MAX(total, 1) * sizeof(double), "topology node records");
205 }
206 }
207
208 free(cursor);
209 *records = buf;
210 return far;
211}

213void MFTLTopoMergeNodes(const MFTL_TopoLattice *lat, MFTL_TopoCluster *cl, int n_cl, const double *records, int n_records)
214{
215 const long long nx = lat->n_global[0], ny = lat->n_global[1];
216 for (int i = 0; i < n_records; ++i)
217 {
218 const int c = (int) records[3 * i];
219 MFTL_ASSERT(c >= 0 && c < n_cl);
220 if (c < 0 || c >= n_cl || !cl[c].has_window) continue;
221
222 const long long key = (long long) records[3 * i + 1];
223 const int g[3] = {(int) (key % nx), (int) ((key / nx) % ny), (int) (key / (nx * ny))};
224 MFTL_SdfGrid *const grid = &cl[c].grid;
225 const int l0 = MFTLSdfGridLocal(grid, 0, g[0]), l1 = MFTLSdfGridLocal(grid, 1, g[1]), l2 = MFTLSdfGridLocal(grid, 2, g[2]);
226 if (l0 < 0 || l1 < 0 || l2 < 0) continue; // outside this rank's window: no cell of its needs it
227
228 double *const node = &grid->phi[((size_t) l2 * grid->dims[1] + l1) * grid->dims[0] + l0];
229 *node = MFTLSdfMerge(*node, records[3 * i + 2]);
230 }
231}

233/* ═══════════════════════════════════════════════════════════════════════════
234 * 3. The bias
235 *
236 * A breakup event opens the thread it sits in: +eps_b on the field, which moves
237 * the zero level set INTO the liquid by eps_b and cuts any thread thinner than
238 * 2 * eps_b. A coalescence event closes the film: -eps_c, which fills any gap
239 * thinner than 2 * eps_c. The bump is 1 within half its radius of the event
240 * point and falls smoothly to 0 at the radius, so the reconstructed surface
241 * rejoins the unbiased one without a step.
242 *
243 * Overlapping bumps of one sign are combined by MAX rather than summed: two
244 * events a lattice cell apart describe the same feature and must not open it
245 * twice as wide.
246 * ═══════════════════════════════════════════════════════════════════════════ */
247
248static inline double bump(double s)
249{
250 if (s <= 0.5) return 1.0;
251 if (s >= 1.0) return 0.0;
252 const double t = (s - 0.5) / 0.5;
253 return 1.0 - t * t * (3.0 - 2.0 * t);
254}

256void MFTLTopoApplyBias(const MFTL_Domain *dom, const MFTL_TopoLattice *lat, MFTL_TopoCluster *cl, int n_cl, const MFTL_TopologyEvent *events,
257 const MFTL_TopologyOptions *opts)
258{
259 const double dx_max = fmax(lat->spacing[0], fmax(lat->spacing[1], lat->spacing[2]));
260
261 for (int c = 0; c < n_cl; ++c)
262 {
263 if (!cl[c].has_window || cl[c].n_events == 0) continue;
264 MFTL_SdfGrid *const g = &cl[c].grid;
265 const size_t n = (size_t) g->dims[0] * g->dims[1] * g->dims[2];
266
267 double *const open = (double *) MFTLCallocOrDie(n, sizeof(double), "topology bias open");
268 double *const close = (double *) MFTLCallocOrDie(n, sizeof(double), "topology bias close");
269
270 for (int i = 0; i < cl[c].n_events; ++i)
271 {
272 const MFTL_TopologyEvent *const e = &events[cl[c].events[i]];
273 const bool breakup = (e->kind == MFTL_TOPOLOGY_BREAKUP);
274 const double delta = breakup ? opts->delta_breakup : opts->delta_coalescence;
275 const double eps = opts->bias_factor * 0.5 * delta;
276 const double rho = opts->bias_radius_factor * fmax(delta, dx_max);
277 double *const acc = breakup ? open : close;
278
279 /* Every image of the event point that reaches the window. */
280 int lo[3], hi[3];
281 for (int k = 0; k < 3; ++k)
282 {
283 lo[k] = (int) ceil((e->point[k] - rho - lat->origin[k]) / lat->spacing[k]);
284 hi[k] = (int) floor((e->point[k] + rho - lat->origin[k]) / lat->spacing[k]);
285 }
286 for (int gz = lo[2]; gz <= hi[2]; ++gz)
287 {
288 const int l2 = MFTLSdfGridLocal(g, 2, gz);
289 if (l2 < 0) continue;
290 const double z = MFTLSdfGridCoord(g, 2, gz) - e->point[2];
291 for (int gy = lo[1]; gy <= hi[1]; ++gy)
292 {
293 const int l1 = MFTLSdfGridLocal(g, 1, gy);
294 if (l1 < 0) continue;
295 const double y = MFTLSdfGridCoord(g, 1, gy) - e->point[1];
296 for (int gx = lo[0]; gx <= hi[0]; ++gx)
297 {
298 const int l0 = MFTLSdfGridLocal(g, 0, gx);
299 if (l0 < 0) continue;
300 const double x = MFTLSdfGridCoord(g, 0, gx) - e->point[0];
301 const double w = bump(sqrt(x * x + y * y + z * z) / rho);
302 const size_t idx = ((size_t) l2 * g->dims[1] + l1) * g->dims[0] + l0;
303 if (eps * w > acc[idx]) acc[idx] = eps * w;
304 }
305 }
306 }
307 }
308
309 for (size_t i = 0; i < n; ++i)
310 if (g->phi[i] != MFTL_SDF_UNKNOWN) g->phi[i] += open[i] - close[i];
311
312 free(open);
313 free(close);
314 }
315 (void) dom;
316}

322void MFTLTopoExtract(const MFTL_Domain *dom, const MFTL_TopoLattice *lat, MFTL_TopoCluster *cl, int n_cl, double snap)
323{
324 (void) dom;
325 int lo[3], hi[3];
326 for (int k = 0; k < 3; ++k)
327 {
328 const int n = lat->per_rank[k], N = lat->cells[k];
329 lo[k] = lat->coords[k] * n - 1;
330 hi[k] = (lat->coords[k] + 1) * n; // inclusive: own cells plus one on each side
331 if (lat->periodic[k])
332 {
333 if (hi[k] - lo[k] + 1 > N)
334 {
335 /* One rank across the whole periodic axis: every cell is its own, and
336 * the layer would wrap onto them. */
337 lo[k] = lat->coords[k] * n;
338 hi[k] = lo[k] + N - 1;
339 }
340 }
341 else
342 {
343 lo[k] = MFTL_MAX(lo[k], 0);
344 hi[k] = MFTL_MIN(hi[k], N - 1);
345 }
346 }
347
348 for (int c = 0; c < n_cl; ++c)
349 {
350 if (!cl[c].has_window) continue;
351 MFTLExtractLevelSet(&cl[c].grid, lo, hi, snap, &cl[c].surf);
352 }
