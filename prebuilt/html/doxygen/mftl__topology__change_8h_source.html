1/**
2 * @file mftl_topology_change.h
3 * @author Christian Gorges
4 * @date 2026-09-25
5 * @ingroup mftl_topology_change
6 *
7 * @brief Topology changes -- breakup and coalescence -- by distributed
8 * signed-distance rasterisation and marching cubes, on the partition as
9 * it stands.
10 *
11 * ## WHAT HAPPENS
12 *
13 * 1. DETECT. Pairs of triangles with opposing normals closer than a threshold,
14 * within one body and between two (MFTLEnsembleIntersectObjects with
15 * self-proximity). A pair whose gap lies INSIDE the liquid is a thread or
16 * sheet about to BREAK; one whose gap lies outside is a film about to
17 * RUPTURE and let two surfaces COALESCE.
18 * 2. AGREE. Every rank learns every event, and the bodies they touch are
19 * grouped into CLUSTERS by union-find: bodies joined by a coalescence event
20 * are reconstructed together.
21 * 3. RECONSTRUCT each cluster's bodies WHOLE from their signed distance field
22 * on a global, wall-aligned lattice, with a smooth local bias at every
23 * event that opens the thread or closes the film -- so the outcome is
24 * decided by the event, not by whether a lattice node happens to fall in a
25 * gap. Marching cubes then triangulates the zero level set.
26 * 4. SEPARATE the result into connected components, one body each.
27 * 5. REDISTRIBUTE to the canonical partition, and optionally remesh.
28 *
29 * ## NOTHING MIGRATES
30 *
31 * Every step runs on the partition the caller has. Each rank rasterises its
32 * OWNED triangles onto a window around its own box and the neighbours merge the
33 * overlap by a commutative rule, so every rank ends up with the exact field on
34 * its box. Each rank then runs marching cubes on its own cells plus one layer of
35 * its neighbours', which builds the ghost halo directly. A surface vertex is
36 * named by the lattice EDGE it lies on and produced from bit-identical values on
37 * every rank that produces it, so the ranks' pieces fit together with no
38 * stitching and no negotiation. Components are labelled by propagating the
39 * smallest triangle key across ghost triangles until nothing changes.
40 *
41 * The reconstructed surface is therefore independent of the process count up to
42 * the last bit of the pseudo-normal sums (see test_topology_change.c).
43 *
44 * ## WHAT IT COSTS, AND WHAT IT CHANGES
45 *
46 * A reconstructed body is re-triangulated at the lattice spacing EVERYWHERE, not
47 * only at the event: vertices are new, ids are new, and its surface moves by
48 * the marching cubes error, O(dx^2) times the curvature, with a local volume
49 * change of order bias x area at each event. Bodies not in a cluster are not
50 * touched at all. A rim on a domain wall is reproduced ON the wall, as the
51 * surface's intersection with the wall plane at lattice resolution; a rim
52 * anywhere else cannot be represented by a level set, so a cluster containing
53 * one is refused rather than corrupted.
54 *
55 * Collective throughout: every rank calls with the same options.
56 */

89/**
90 * @struct MFTL_TopologyOptions
91 * @brief Everything a topology change needs besides the ensemble.
92 *
93 * Start from MFTLTopologyDefaultOptions() and set the three lengths: a
94 * zero-initialised struct has zero thresholds and detects nothing.
95 */
96typedef struct MFTL_TopologyOptions
97{
98 double delta_breakup; ///< A liquid thread or sheet thinner than this breaks. Absolute length.
99 double delta_coalescence; ///< A film between surfaces thinner than this ruptures. Absolute length.
100 double normal_tol_deg; ///< Two triangles oppose each other when n_A . n_B <= -cos(this). Default 60.
101
102 /**
103 * @brief Target lattice spacing for the reconstruction. Absolute length.
104 *
105 * The spacing actually used divides every axis into a whole number of cells
106 * per rank -- see MFTLTopologyLattice() -- so it may differ slightly per axis.
107 * Around the target edge length of the remesher that follows is right: finer
108 * wastes time on a surface the remesher coarsens anyway, coarser loses
109 * features the detection did not flag.
110 */
111 double dx;
112
113 /**
114 * @brief Bias magnitude, as a multiple of half the threshold of the event.
115 *
116 * A breakup event adds `+bias_factor * delta_breakup / 2` to the field around
117 * its gap, which opens any thread of radius below that; a coalescence event
118 * subtracts `bias_factor * delta_coalescence / 2`, which closes any film
119 * thinner than twice that. Above 1 so that a feature just at the threshold is
120 * decided with margin rather than by rounding. Default 1.5.
121 */
122 double bias_factor;
123
124 /**
125 * @brief Radius of the bias bump around each event point, in units of
126 * max(delta, dx). Default 1.5.
127 *
128 * The bump has full weight out to half its radius, i.e. 0.75 delta by
129 * default -- beyond the surface of any thread thinner than delta, whose event
130 * points lie on its axis -- and falls smoothly to zero at the radius. Larger
131 * spends the bias on the bodies either side of the feature. Measured on the
132 * pinched dumbbell of test_topology_change.c (bulbs of radius 0.45, lattice
133 * 0.067), volume lost per bulb: 2.9 % at 1.5 against 4.0 % at 2 with
134 * delta_breakup = 0.12, and 6.7 % against 11 % with delta_breakup = 0.2 --
135 * which is also why delta should be set from the feature size the flow can
136 * resolve, not generously.
137 */
138 double bias_radius_factor;
139
140 /** @brief Marching cubes snap distance, relative to dx. See MFTLExtractLevelSet(). Default 0.02. */
141 double mc_snap;
142
143 /**
144 * @brief Closed components with fewer triangles than this are deleted.
145 *
146 * A satellite smaller than a few lattice cells cannot be represented and
147 * would be collapsed into a degenerate shape by the remesher. The count is
148 * taken TWICE: on the marching-cubes surface, and again after the `remesh`
149 * pass when one runs -- a satellite that clears the floor as a few dozen MC
150 * slivers can come out of a remesh at a comparable target edge length as a
151 * tetrahedron, which the collapse approval then refuses to dissolve. A body
152 * deleted at either point gives its volume to its cluster's surviving bodies
153 * through `conserve_volume`. 0 keeps everything. An OPEN component is never
154 * deleted. Default 16.
155 */
156 int min_body_triangles;
157
158 /**
159 * @brief Restore each cluster's enclosed volume after the reconstruction.
160 *
161 * Default ON. Marching cubes cuts chords across a curved surface, so every
162 * reconstruction loses O(dx^2 * curvature) of volume per unit area -- on the
163 * liquid jet about one percent of the jet PER reconstruction, measured as 60
164 * of a final 81 volume units over a run of 45 reconstructions without it. The
165 * liquid a cluster encloses is therefore measured before and after, and the
166 * difference is put back by a uniform offset along the area-weighted vertex
167 * normals (a rim vertex on a wall moves only within the wall), in three
168 * Newton steps on the exact first variation of the volume -- 1e-9 relative
169 * from a 15 % deficit, measured. It runs AFTER the optional `remesh`, which
170 * moves volume too. It also puts back what the bias moved, which is what a solver
171 * wants: a thread that pinches off does not destroy liquid, and a film that
172 * ruptures does not create any.
173 *
174 * The volume of an open surface is measured together with the walls its rims
175 * lie on, as the flux of (x_k - x_k,min) e_k along an axis whose far wall
176 * carries no rim. A cluster for which no such axis exists -- rims on both
177 * walls of every non-periodic axis -- is left uncorrected.
178 */
179 bool conserve_volume;
180
181 /**
182 * @brief Remesh the ensemble after a reconstruction, with these parameters.
183 *
184 * #MFTL_NULL for none. Marching cubes output has slivers and edges from a few
185 * percent of dx up to sqrt(2) dx, and a solver almost always wants it
186 * remeshed before the next step; the call is skipped when nothing changed.
187 * Only MFTLRemeshEnsemble() is run -- all bodies take part, as they would in
188 * the caller's own remeshing pass.
189 */
190 const struct MFTL_RemeshParameters *remesh;
191} MFTL_TopologyOptions;

230/**
231 * @brief The reconstruction lattice for a domain and a target spacing.
232 *
233 * Axis k has `cells[k] = cuts[k] * max(1, round(L_k / (cuts[k] * dx)))` cells
234 * of `spacing[k] = L_k / cells[k]`, anchored at `global_min[k]`. Two properties
235 * follow and the whole reconstruction rests on them: a node plane lies on every
236 * domain wall, so a surface meeting a wall is cut exactly there, and a node
237 * plane lies on every rank's cut plane, so every lattice cell belongs to

239 *
240 * @param domain The domain.
241 * @param dx Target spacing, > 0.
242 * @param[out] spacing Actual spacing per axis.
243 * @param[out] cells Cells per axis.
244 * @rank_local Pure function of the domain.

267/**
268 * @brief Reconstructs the listed bodies from their signed distance field,
269 * biased at the listed events, and splits the result into bodies.
270 *
271 * The core of MFTLEnsembleApplyTopologyChanges(), exposed because it is useful
272 * on its own -- re-gridding a body, or reconstructing one with no events at all
273 * to test the pipeline -- and because it is where every distributed property of
274 * the module lives.
275 *
276 * CLUSTERS are the connected components of the listed bodies under the events
277 * between two of them -- of either kind: a coalescence joins two bodies, and a
278 * liquid film between a drop and a bubble inside it is a breakup event between
279 * two bodies that bound one liquid from opposite sides. Each cluster's bodies
280 * are rasterised together, so the field is the signed distance to the UNION of
281 * their surfaces, signed by the nearest one -- which is correct for nested
282 * bodies as well, where a min over per-body fields would not be.
283 * Ids: in each cluster the largest resulting component keeps the id of the
284 * largest body that went in, the second the second, and so on; components
285 * beyond that get new ids, and bodies beyond that are retired. So the jet a
286 * drop pinches off stays the jet, and the pool a drop falls into stays the pool.
287 *
288 * Closes with MFTLRedistribute() to the canonical partition, current triangle
289 * geometry and global counts -- and then, when `opts->remesh` is set, one
290 * MFTLRemeshEnsemble() over the whole ensemble.
291 *
292 * @param ens The ensemble. After the call every #MFTL_Object pointer of a
293 * listed body is invalid -- use MFTLEnsembleFindLocalObject() --
294 * and so are all local indices, as after any redistribution.
295 * @param bodies `[n_bodies]` body ids to reconstruct, identical on every rank.
296 * @param n_bodies Their number.
297 * @param events `[n_events]` events, identical on every rank; each biases the
298 * field of the cluster its bodies are in. May be #MFTL_NULL.
299 * @param n_events Their number.
300 * @param opts The options.
301 * @param[out] stats Optional.
302 * @return #MFTL_SUCCESS, or #MFTL_ERROR (on every rank).
303 * @collective
304 */
305int MFTLEnsembleReconstructBodies(MFTL_Ensemble *ens, const int *bodies, int n_bodies, const MFTL_TopologyEvent *events, int n_events,
306 const MFTL_TopologyOptions *opts, MFTL_TopologyStats *stats);
