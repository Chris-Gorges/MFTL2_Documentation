29/**
30 * @brief One level of 1-to-4 subdivision, in place.
31 *
32 * v2
33 * / \
34 * m2 -- m1
35 * / \ / \
36 * v0 -- m0 -- v1
37 *
38 * Each edge's midpoint (the plain average, NOT projected: projection happens
39 * once, after the last level) is created once: `midpoint_map[h]` is written for
40 * a half-edge and its twin together, so the second triangle finds it by a
41 * direct array read -- "the other side of this edge" is already an index, no
42 * hash map needed. The sub-triangles keep their parent's rotational order, which
43 * propagates the outward winding. Requires a closed mesh.
44 *
45 * @param object The mesh; every triangle becomes four.
46 * @return #MFTL_SUCCESS, or #MFTL_ERROR on an open mesh or a topology failure.
47 */
48static int tessellate_sphere(MFTL_Object *object)
49{
50 const int old_nv = object->verts.n;
51 const int old_nt = object->tris.n;
52 const int old_nhe = 3 * old_nt;
53 const int old_nedges = old_nhe / 2; // closed mesh: every edge has exactly two half-edges
54
55 const int new_nv = old_nv + old_nedges;
56 const int new_nt = 4 * old_nt;
57
58 const int rank = object->domain->rank;
59
60 /* ── 1. Midpoints, appended after the old vertices (growth preserves them). */
61 MFTLObjectSetCounts(object, new_nv, new_nv, old_nt, old_nt);
62
63 int *midpoint_map = (int *) MFTLMallocOrDie((size_t) old_nhe * sizeof(int), "tessellate_sphere midpoint map");
64 for (int h = 0; h < old_nhe; ++h) midpoint_map[h] = MFTL_ERROR;
65
66 double *const pos = object->verts.pos;
67 MFTL_Vertex *const vr = object->verts.rec;
68 const MFTL_HalfEdge *const he = object->tris.he;
69
70 int next_mid = old_nv;
71
72 for (int h = 0; h < old_nhe; ++h)
73 {
74 if (midpoint_map[h] != MFTL_ERROR) continue; // already created via the twin
75
76 const int twin = he[h].twin_idx;
77 MFTL_ASSERT(twin != MFTL_ERROR && "tessellate_sphere: open mesh (boundary half-edge) is not supported");
78 if (twin == MFTL_ERROR)
79 {
80 free(midpoint_map);
81 return MFTL_ERROR;
82 }
83
84 const int v_end = he[h].vertex_idx;
85 const int v_start = he[twin].vertex_idx; // target of the twin == origin of h
86
87 /* No minimum image needed: the mesh is built about the origin and only
88 * translated (and wrapped) at the end. */
89 pos[3 * next_mid + 0] = 0.5 * (pos[3 * v_start + 0] + pos[3 * v_end + 0]);
90 pos[3 * next_mid + 1] = 0.5 * (pos[3 * v_start + 1] + pos[3 * v_end + 1]);
91 pos[3 * next_mid + 2] = 0.5 * (pos[3 * v_start + 2] + pos[3 * v_end + 2]);
92
93 vr[next_mid].global_id = object->next_vertex_gid++;
94 vr[next_mid].owner_proc = rank;
95 vr[next_mid].out_he_idx = MFTL_ERROR; // assigned when the topology is rebuilt
96 vr[next_mid].flags = 0;
97
98 midpoint_map[h] = next_mid;
99 midpoint_map[twin] = next_mid;
100 next_mid++;
101 }
102
103 MFTL_ASSERT(next_mid == new_nv && "tessellate_sphere: midpoint count does not match the Euler formula");
104
105 /* ── 2. Four sub-triangles per triangle, in the parent's rotational order. */
106 int *tri_verts = (int *) MFTLMallocOrDie(3 * (size_t) new_nt * sizeof(int), "tessellate_sphere index list");
107
108 int out = 0;
109 for (int t = 0; t < old_nt; ++t)
110 {
111 const int h0 = 3 * t, h1 = 3 * t + 1, h2 = 3 * t + 2;
112
113 const int v0 = he[h2].vertex_idx; // origin of h0
114 const int v1 = he[h0].vertex_idx;
115 const int v2 = he[h1].vertex_idx;
116
117 const int m0 = midpoint_map[h0]; // on edge v0-v1
118 const int m1 = midpoint_map[h1]; // on edge v1-v2
119 const int m2 = midpoint_map[h2]; // on edge v2-v0
120
121 const int sub[4][3] = {
122 {v0, m0, m2}, // corner 0
123 {v1, m1, m0}, // corner 1
124 {v2, m2, m1}, // corner 2
125 {m0, m1, m2} // centre
126 };
127
128 for (int j = 0; j < 4; ++j)
129 {
130 tri_verts[3 * out + 0] = sub[j][0];
131 tri_verts[3 * out + 1] = sub[j][1];
132 tri_verts[3 * out + 2] = sub[j][2];
133 ++out;
134 }
135 }
136
137 free(midpoint_map);
138
139 /* ── 3. Rebuild the topology from the index list. */
140 const int rc = MFTLBuildTopologyFromIndexList(object, tri_verts, new_nt);
141 free(tri_verts);
142
143 return rc;
144}

146/**
147 * @brief Builds the regular icosahedron inscribed in the unit sphere: the 12
148 * vertices \f$(0, \pm1, \pm\varphi)\f$ and cyclic permutations, scaled by
149 * \f$1/\sqrt{1+\varphi^2}\f$ (so X = 0.5257, Z = 0.8507), and 20 faces
150 * wound counter-clockwise as seen from OUTSIDE.
151 *
152 * @param object The (empty) mesh to fill.
153 * @return #MFTL_SUCCESS, or #MFTL_ERROR.
154 */
155static int generate_icosahedron(MFTL_Object *object)
156{
157 const double X = 0.525731112119133606;
158 const double Z = 0.850650808352039932;
159
160 const double positions[12 * 3] = {-X, 0.0, Z, X, 0.0, Z, -X, 0.0, -Z, X, 0.0, -Z, 0.0, Z, X, 0.0, Z, -X,
161 0.0, -Z, X, 0.0, -Z, -X, Z, X, 0.0, -Z, X, 0.0, Z, -X, 0.0, -Z, -X, 0.0};
162
163 const int tri_verts[20 * 3] = {0, 1, 4, 0, 4, 9, 9, 4, 5, 4, 8, 5, 4, 1, 8, 8, 1, 10, 8, 10, 3, 5, 8, 3, 5, 3, 2, 2, 3, 7,
164 7, 3, 10, 7, 10, 6, 7, 6, 11, 11, 6, 0, 0, 6, 1, 6, 10, 1, 9, 11, 0, 9, 2, 11, 9, 5, 2, 7, 11, 2};
165
166 return MFTLBuildIndexedMesh(object, positions, 12, tri_verts, 20);
167}

169int MFTLGenerateSphereLocal(MFTL_Object *object, double center[3], double radius, int refinements)
170{
171 MFTL_ASSERT(object != MFTL_NULL);
172 MFTL_ASSERT(object->domain != MFTL_NULL && "MFTLGenerateSphereLocal: object has no domain -- add it to an ensemble first");
173 MFTL_ASSERT(radius >= 0.0 && "MFTLGenerateSphereLocal: radius must not be negative");
174 MFTL_ASSERT(refinements >= 0 && "MFTLGenerateSphereLocal: refinements must not be negative");
175
176 if (object == MFTL_NULL || object->domain == MFTL_NULL || center == MFTL_NULL) return MFTL_ERROR;
177 if (radius < 0.0 || refinements < 0) return MFTL_ERROR;
178
179 MFTL_Domain *domain = object->domain;
180 int rc = MFTL_SUCCESS;
181
182 /* Rank 0 builds; the others stay empty until the first redistribution. */
183 if (domain->rank == 0)
184 {
185 rc = generate_icosahedron(object);
186
187 for (int i = 0; rc == MFTL_SUCCESS && i < refinements; ++i) rc = tessellate_sphere(object);
188
189 if (rc == MFTL_SUCCESS)
190 {
191 /* Project onto the sphere, scale, translate, wrap. */
192 double *const pos = object->verts.pos;
193 const int nv = object->verts.n;
194
195 for (int v = 0; v < nv; ++v)
196 {
197 double *p = &pos[3 * v];
198 MFTLNormalizeVector(p);
199 p[0] = center[0] + radius * p[0];
200 p[1] = center[1] + radius * p[1];
201 p[2] = center[2] + radius * p[2];
202 MFTLApplyPeriodicity(domain, p);
203 }
204
205 /* Closed, so this marks nothing; asserted. */
206 const int n_boundary = MFTLMarkMeshBoundary(object);
207 MFTL_ASSERT(n_boundary == 0 && "MFTLGenerateSphereLocal: the generated sphere is not closed");
208 (void) n_boundary;
209
210 /* After the projection, which moved every vertex: the first
211 * MFTLRedistribute() routes by these centres. */
212 MFTLComputeTriangleGeometry(object, object->tris.n);
213 }
214 }
215 else
216 {
217 /* Empty: mark the (vacuous) derived arrays valid. */
218 MFTLObjectBuildDerived(object);
219 }
220
221 return rc;
222}

224/* ═══════════════════════════════════════════════════════════════════════════
225 * Publication: the collective half, separate so that N bodies cost two
226 * collectives instead of 2N in strict sequence on rank 0's critical path (at
227 * 5 us per collective, 10 s before the first step at a million bodies; not
228 * measurable on a single node). The counters are rank 0's and are BROADCAST;
229 * the counts are everybody's and are REDUCED -- one fused reduction would be
230 * correct only while every non-root object is empty.
231 * ════════════════════════════════════════════════════════════════════════════ */
232
233/**
234 * @brief Settles the ID counters and global counts of @p n bodies in two collectives.
235 *
236 * @param domain The MPI world. All of @p objects must share it.
237 * @param objects [n] bodies to publish, in the same order on every rank.
238 * @param n Bodies. Zero is a no-op, but every rank must still call.
239 * @return #MFTL_SUCCESS, or #MFTL_ERROR on an allocation-free invalid argument.
240 */
241static int publish_bodies(MFTL_Domain *domain, MFTL_Object *const *objects, int n)
242{
243 if (domain == MFTL_NULL || (n > 0 && objects == MFTL_NULL)) return MFTL_ERROR;
244 if (n <= 0) return MFTL_SUCCESS;
245
246 /* One allocation, two slices: [3n counters | 2n local counts | 2n global]. */
247 int *const buf = (int *) MFTLMallocOrDie((size_t) (7 * n) * sizeof(int), "generated body publication");
248 int *const gids = buf;
249 int *const local_counts = buf + 3 * n;
250 int *const global_counts = buf + 5 * n;
251
252 for (int o = 0; o < n; ++o)
253 {
254 const MFTL_Object *const obj = objects[o];
255 MFTL_ASSERT(obj != MFTL_NULL);
256 gids[3 * o + 0] = obj->next_vertex_gid;
257 gids[3 * o + 1] = obj->next_triangle_gid;
258 gids[3 * o + 2] = obj->next_halfedge_gid;
259
260 /* Ownership is a prefix: n_local is the owned count. */
261 local_counts[2 * o + 0] = obj->tris.n_local;
262 local_counts[2 * o + 1] = obj->verts.n_local;
263 }
264
265 MPI_Bcast(gids, 3 * n, MPI_INT, 0, domain->comm_world);
266 MPI_Allreduce(local_counts, global_counts, 2 * n, MPI_INT, MPI_SUM, domain->comm_world);
267
268 for (int o = 0; o < n; ++o)
269 {
270 MFTL_Object *const obj = objects[o];
271 obj->next_vertex_gid = gids[3 * o + 0];
272 obj->next_triangle_gid = gids[3 * o + 1];
273 obj->next_halfedge_gid = gids[3 * o + 2];
274
275 obj->global_n_triangles = global_counts[2 * o + 0];
276 obj->global_n_vertices = global_counts[2 * o + 1];
277 }
278
279 free(buf);
280 return MFTL_SUCCESS;
281}

283int MFTLPublishGeneratedBodies(MFTL_Ensemble *ens)
284{
285 MFTL_ASSERT(ens != MFTL_NULL);
286 if (ens == MFTL_NULL) return MFTL_ERROR;
287

291int MFTLGenerateSphere(MFTL_Object *object, double center[3], double radius, int refinements)
292{
293 const int rc = MFTLGenerateSphereLocal(object, center, radius, refinements);
294 if (rc == MFTL_ERROR && (object == MFTL_NULL || object->domain == MFTL_NULL)) return rc;
295
296 /* One body: one broadcast, one reduction. */
297 MFTL_Object *one = object;
298 (void) publish_bodies(object->domain, &one, 1);
299
300 return rc;
301}
