1/**
2 * @file mftl_memory.h
3 * @author Christian Gorges
4 * @date 2026-08-14
5 * @ingroup mftl_memory
6 *
7 * @brief Allocation policy, cache-line-aligned pool blocks, and array growth.
8 *
9 * Layer 0. Two kinds of storage, handled differently:
10 *
11 * - **Pool blocks** (the vertex and triangle pools of an #MFTL_Object). The AoS
12 * records and every SoA array of a pool describe the same entities, so they
13 * must always have the same capacity. A pool is therefore ONE allocation, cut
14 * into cache-line-aligned sub-arrays by MFTLLayoutAdd(): one pointer to free,
15 * one capacity, and no way to grow half of it. Only `src/object/object.c`
16 * calls MFTLLayoutAdd().
17 * - **Independent arrays** (object lists, per-neighbour buffers, algorithm
18 * scratch) use MFTLMallocOrDie() / MFTLCallocOrDie() / MFTLReallocOrDie() and
19 * the growth macros #MFTL_ENSURE_CAPACITY and #MFTL_RESERVE_ADDITIONAL.
20 *
21 * Allocation failure is fatal everywhere: in an MPI job an out-of-memory rank
22 * cannot recover, and a NULL buffer carried on corrupts data far from the cause.
23 */
24

37/**
38 * @brief `malloc` that aborts instead of returning NULL.
39 *
40 * @param bytes Number of bytes to allocate.
41 * @param what Human-readable name of the caller, printed in the fatal message.
42 * @return A pointer to @p bytes uninitialised bytes. Never NULL -- the process
43 * exits instead.
44 */
45static inline void *MFTLMallocOrDie(size_t bytes, const char *what)
46{
47 void *p = malloc(bytes);
48 if (!p)
49 {
50 fprintf(stderr, "FATAL ERROR: %s allocation failed for %lu bytes.\n", what, (unsigned long) bytes);
51 exit(EXIT_FAILURE);
52 }
53 return p;
54}

75/**
76 * @brief `realloc` that aborts instead of returning NULL.
77 *
78 * For growable scratch arrays whose sizes move together, where the growth
79 * macros would need one capacity counter per array.
80 *
81 * @param ptr Existing block, or NULL.

88 void *p = realloc(ptr, bytes);

92 exit(EXIT_FAILURE);

101/**
102 * @brief Rounds @p x up to the next multiple of @p a.
103 *
104 * @param x Value to round.
105 * @param a Alignment. **Must be a power of two** -- the implementation is a mask,
106 * not a modulo, so a non-power-of-two silently returns nonsense.
107 * @return The smallest multiple of @p a that is not less than @p x.
108 */
109static inline size_t MFTLAlignUp(size_t x, size_t a) { return (x + a - 1u) & ~(a - 1u); }
110
111/**
112 * @brief Allocates @p bytes, 64-byte aligned, aborting on failure.
113 *
114 * C11 `aligned_alloc` requires the size to be a multiple of the alignment, so the
115 * request is rounded up here.
116 *
117 * @param bytes Requested size. Rounded up internally; 0 is treated as 1.
118 * @param what Human-readable name of the caller, printed in the fatal message.
119 * @return A pointer aligned to #MFTL_CACHE_LINE bytes. Never NULL. Release it
120 * with plain `free()`.
121 */
122static inline void *MFTLAlignedAllocOrDie(size_t bytes, const char *what)
123{
124 const size_t padded = MFTLAlignUp(bytes ? bytes : 1u, MFTL_CACHE_LINE);
125 void *p = aligned_alloc(MFTL_CACHE_LINE, padded);
126 if (!p)
127 {
128 fprintf(stderr, "FATAL ERROR: %s aligned allocation failed for %lu bytes.\n", what, (unsigned long) padded);
129 exit(EXIT_FAILURE);
130 }
131 return p;
132}

134/**
135 * @brief Reserves one sub-array inside a pool block and returns its byte offset.
136 *
137 * Call it once per field, in order, starting from a zeroed @p cursor; afterwards
138 * @p cursor holds the total block size:
139 * \f$o_0 = 0,\; o_{i+1} = L\lceil (o_i + n_i s_i)/L \rceil\f$ with
140 * \f$L = \f$ #MFTL_CACHE_LINE.
141 *
142 * size_t cursor = 0;
143 * const size_t o_rec = MFTLLayoutAdd(&cursor, cap, sizeof(MFTL_Vertex));
144 * const size_t o_pos = MFTLLayoutAdd(&cursor, 3 * cap, sizeof(double));
145 * ...
146 * unsigned char *block = MFTLAlignedAllocOrDie(cursor, "vertex pool");
147 *
148 * Because the capacity is a property of the block, "the AoS grew but an SoA
149 * array did not" cannot be expressed.
150 *
151 * @param cursor Running block size in bytes, updated in place. Start at 0.
152 * @param count Number of elements in this sub-array.
153 * @param elem_size Size of one element in bytes.
154 * @return Byte offset of the sub-array from the start of the block. Always a
155 * multiple of #MFTL_CACHE_LINE.
156 */
157static inline size_t MFTLLayoutAdd(size_t *cursor, size_t count, size_t elem_size)
158{
159 const size_t offset = *cursor;
160 *cursor = MFTLAlignUp(offset + count * elem_size, MFTL_CACHE_LINE);
161 return offset;
162}

164/**
165 * @brief Geometric growth policy: at least @p needed, at least MFTL_INIT_CAPACITY,
166 * otherwise double until it fits.
167 *
168 * A mesh that grows entity by entity therefore performs \f$O(\log n)\f$
169 * reallocations.
170 *
171 * @param current Present capacity. May be 0.
172 * @param needed Capacity that must fit afterwards.
173 * @return @p current if it already suffices, otherwise the smallest
174 * \f$c_0 2^k \ge\f$ @p needed, with \f$c_0\f$ = @p current, or
175 * #MFTL_INIT_CAPACITY when @p current is 0. Saturates to exactly
176 * @p needed near `INT_MAX` instead of overflowing to a negative size.
177 */
178static inline int MFTLGrowCapacity(int current, int needed)
179{
180 if (needed <= current) return current;
181 int cap = (current > 0) ? current : MFTL_INIT_CAPACITY;
182 while (cap < needed)
183 {
184 /* int overflow would turn a huge request into a negative capacity, i.e. a
185 * silently undersized allocation. Saturate instead and let the allocator
186 * report the real problem. */
187 if (cap > (2147483647 / 2)) return needed;
188 cap *= 2;
189 }
190 return cap;
191}

194 * Growth macros for independent arrays. Never for pool storage: a pool's

198/**
199 * @brief Implementation shared by #MFTL_ENSURE_CAPACITY and #MFTL_RESERVE_ADDITIONAL.
200 *
201 * Reallocates only when @p target_cap exceeds the present capacity, aborts on
202 * failure, and writes the new capacity back through @p capacity.
203 *
204 * @param ptr Lvalue holding the array pointer. Updated in place.
205 * @param capacity Pointer to the capacity counter. Updated in place.
206 * @param target_cap Capacity wanted, in elements.
207 * @param type Element type, used for `sizeof` and for the cast.
208 *
209 * @internal Not part of the public API; use one of the two wrappers.
210 */
211#define _MFTL_REALLOC_CORE(ptr, capacity, target_cap, type) \
212 do \
213 { \
214 int _req_cap = (target_cap); \
215 if (_req_cap > (*(capacity))) \
216 { \
217 void *_new_ptr = realloc((ptr), (size_t) _req_cap * sizeof(type)); \
218 if (!_new_ptr && _req_cap > 0) \
219 { \
220 fprintf(stderr, "FATAL ERROR: MFTL Alloc failed for %lu bytes.\n", (unsigned long) ((size_t) _req_cap * sizeof(type))); \
221 exit(EXIT_FAILURE); \
222 } \
223 (ptr) = (type *) _new_ptr; \
224 (*(capacity)) = _req_cap; \
225 } \
226 } while (0)

228/**
229 * @brief Geometric growth (&times;2). For incremental appends inside a loop.
230 *
231 * Grows only when `*count + needed` exceeds `*capacity`, following
232 * MFTLGrowCapacity()'s doubling schedule, so \f$n\f$ appends cost
233 * \f$O(\log n)\f$ reallocations.
234 *
235 * @param ptr Lvalue holding the array pointer. Updated in place.
236 * @param count Pointer to the element counter. Read, never written.
237 * @param capacity Pointer to the capacity counter. Updated in place.
238 * @param needed Additional elements that must fit.
239 * @param type Element type.
240 *
241 * @warning Independent arrays only; never pool storage (see mftl_object.h).
242 */
243#define MFTL_ENSURE_CAPACITY(ptr, count, capacity, needed, type) \
244 do \
245 { \
246 if ((*(count)) + (needed) > (*(capacity))) \
247 { \
248 int _new_cap = MFTLGrowCapacity(*(capacity), (*(count)) + (needed)); \
249 _MFTL_REALLOC_CORE(ptr, capacity, _new_cap, type); \
250 } \
251 } while (0)
252
253/**
254 * @brief Exact growth. For bulk loads where the final count is known up front.
255 *
256 * Allocates precisely `*count + needed` elements: a bulk load knows its size, and
257 * geometric slack would never be reclaimed.
258 *
259 * @param ptr Lvalue holding the array pointer. Updated in place.
260 * @param count Pointer to the element counter. Read, never written.
261 * @param capacity Pointer to the capacity counter. Updated in place.
262 * @param needed Additional elements that must fit.
263 * @param type Element type.
264 *
265 * @warning Independent arrays only. See #MFTL_ENSURE_CAPACITY.
266 */
267#define MFTL_RESERVE_ADDITIONAL(ptr, count, capacity, needed, type) \
268 do \
269 { \
270 if ((*(count)) + (needed) > (*(capacity))) \
271 { \
272 int _new_cap = (*(count)) + (needed); \
273 _MFTL_REALLOC_CORE(ptr, capacity, _new_cap, type); \
274 } \
275 } while (0)
