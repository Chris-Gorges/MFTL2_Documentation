1/**
2 * @file topo_detect.c
3 * @author Christian Gorges
4 * @date 2026-09-25
5 * @ingroup mftl_topology_change
6 *
7 * @brief Detection: which pairs of triangles are a thin feature, and of which kind.
8 *
9 * ## THE PREDICATE
10 *
11 * Two triangles A and B -- of one body or of two -- are an EVENT when
12 *
13 * 1. they share no vertex (MFTLEnsembleIntersectObjects' self filter);
14 * 2. their unit normals OPPOSE: n_A . n_B <= -cos(theta);
15 * 3. each lies on the same side of the other, as seen along its normal:
16 * s_A = n_A . (c_B - c_A) and s_B = n_B . (c_A - c_B)
17 * both negative means each is BEHIND the other -- the gap between them is
18 * inside the body, a thread or sheet: BREAKUP. Both positive means each is
19 * in FRONT of the other -- the gap is outside, a film: COALESCENCE. Mixed
20 * signs are a fold seen edge-on, not a thin feature, and are not events;
21 * 4. their distance is within the threshold of THAT kind.
22 *
23 * Condition 3 is what the original design's detection criteria left implicit, and it is
24 * the one that makes the same machinery serve both directions of a topology
25 * change: the RT cap rolling onto its own stem and a jet's neck thinning are
26 * both "two opposing triangles within delta", and only the side the gap is on
27 * tells them apart.
28 *
29 * ## THE CANONICAL LIST
30 *
31 * Detection is local and many pairs describe one feature -- a neck is found
32 * once per triangle pair around it, and a pair across a rank boundary by both
33 * ranks' owners in general. Every event point is therefore snapped to a cell of
34 * a quarter of max(smaller active threshold, dx) -- the lattice spacing too,
35 * because a bias bump is at least 1.5 dx wide and events closer than that are
36 * one feature (a quantum on the threshold alone shredded one jet neck into 779
37 * events) -- and events are merged on (bodies, kind, cell),
38 * keeping the smallest gap. After the gather, the same canonicalisation over the
39 * whole list makes it a pure function of the geometry: identical on every rank
40 * and at every process count.
41 */

64static void push_event(MFTL_TopoEventList *l, const MFTL_TopologyEvent *e)
65{

74/** @brief Unit normal and centroid of corners @p T; returns false for a degenerate triangle. */
75static bool normal_and_centre(const double T[3][3], double n[3], double c[3])
76{
77 const double e1[3] = {T[1][0] - T[0][0], T[1][1] - T[0][1], T[1][2] - T[0][2]};
78 const double e2[3] = {T[2][0] - T[0][0], T[2][1] - T[0][1], T[2][2] - T[0][2]};
79 n[0] = e1[1] * e2[2] - e1[2] * e2[1];
80 n[1] = e1[2] * e2[0] - e1[0] * e2[2];
81 n[2] = e1[0] * e2[1] - e1[1] * e2[0];
82 const double mag = sqrt(n[0] * n[0] + n[1] * n[1] + n[2] * n[2]);
83 if (!(mag > 0.0)) return MFTL_FALSE;
84 for (int k = 0; k < 3; ++k)
85 {
86 n[k] /= mag;
87 c[k] = (T[0][k] + T[1][k] + T[2][k]) / 3.0;
88 }
89 return MFTL_TRUE;
90}

92static void detect_pair(const MFTL_BVH *bvhA, int pa, const MFTL_BVH *bvhB, int pb, void *ud)
93{
94 detect_ctx *const ctx = (detect_ctx *) ud;
95
96 double TA[3][3], TB[3][3], nA[3], nB[3], cA[3], cB[3];
97 MFTL_BVHPrimVerts(bvhA, pa, TA);
98 MFTL_BVHPrimVerts(bvhB, pb, TB);
99 if (!normal_and_centre(TA, nA, cA) || !normal_and_centre(TB, nB, cB)) return;
100
101 /* B in the periodic image nearest A: the traversal tried every image, but the
102 * callback is handed the stored corners. */
103 double d[3];
104 MFTLGetMinimumImageVector(ctx->dom, d, cA, cB);
105 const double shift[3] = {cA[0] + d[0] - cB[0], cA[1] + d[1] - cB[1], cA[2] + d[2] - cB[2]};
106 for (int j = 0; j < 3; ++j)
107 for (int k = 0; k < 3; ++k) TB[j][k] += shift[k];
108
109 if (nA[0] * nB[0] + nA[1] * nB[1] + nA[2] * nB[2] > -ctx->cos_tol) return;
110
111 const double sA = nA[0] * d[0] + nA[1] * d[1] + nA[2] * d[2];
112 const double sB = -(nB[0] * d[0] + nB[1] * d[1] + nB[2] * d[2]);
113
114 int kind;
115 double delta;
116 if (sA < 0.0 && sB < 0.0)
117 {
118 kind = MFTL_TOPOLOGY_BREAKUP;
119 delta = ctx->delta_b;
120 }
121 else if (sA > 0.0 && sB > 0.0)
122 {
123 kind = MFTL_TOPOLOGY_COALESCENCE;
124 delta = ctx->delta_c;
125 }
126 else
127 return;
128 if (!(delta > 0.0)) return;
129
130 const double gap = sqrt(MFTL_TriTriDistanceSq(TA, TB));
131 if (gap > delta) return;
132
133 int ta, tb;
134 const MFTL_Object *const oa = MFTL_BVHPrimTriangle(bvhA, pa, &ta);
135 const MFTL_Object *const ob = MFTL_BVHPrimTriangle(bvhB, pb, &tb);
136
137 MFTL_TopologyEvent ev;
138 ev.body_a = MFTL_MIN(oa->global_id, ob->global_id);
139 ev.body_b = MFTL_MAX(oa->global_id, ob->global_id);
140 ev.kind = kind;
141 ev.gap = gap;
142 for (int k = 0; k < 3; ++k) ev.point[k] = cA[k] + 0.5 * d[k];
143 MFTLApplyPeriodicity(ctx->dom, ev.point);
144 push_event(ctx->out, &ev);
145}

147/**
148 * @brief The cheap half of the predicate, run before the exact narrow phase:
149 * only OPPOSING normals can be an event.
150 *
151 * On a smooth surface nearly every triangle within the envelope of another has
152 * a nearly parallel normal, and the exact proximity test costs two orders of
153 * magnitude more than this dot product. See MFTL_BVHBuildParams::pair_filter.
154 */
155static bool opposing_normals(const MFTL_BVH *bvhA, int pa, const MFTL_BVH *bvhB, int pb, void *ud)
156{
157 const detect_ctx *const ctx = (const detect_ctx *) ud;
158 double TA[3][3], TB[3][3], nA[3], nB[3], cA[3], cB[3];
159 MFTL_BVHPrimVerts(bvhA, pa, TA);
160 MFTL_BVHPrimVerts(bvhB, pb, TB);
161 if (!normal_and_centre(TA, nA, cA) || !normal_and_centre(TB, nB, cB)) return MFTL_FALSE;
162 return nA[0] * nB[0] + nA[1] * nB[1] + nA[2] * nB[2] <= -ctx->cos_tol;
163}

165int MFTLTopoDetectLocal(MFTL_Ensemble *ens, const MFTL_TopologyOptions *opts, MFTL_TopoEventList *out)
166{
167 detect_ctx ctx;
168 ctx.dom = &ens->domain;
169 ctx.cos_tol = cos(opts->normal_tol_deg * MFTL_PI / 180.0);
170 ctx.delta_b = opts->delta_breakup;
171 ctx.delta_c = opts->delta_coalescence;
172 ctx.out = out;
173
174 /* The envelope is ABSOLUTE and the same for every body: a thread breaks at
175 * delta whatever the size of the drop it belongs to. */
176 MFTL_BVHBuildParams p = MFTL_BVHDefaultParams();
177 p.proximity_factor = 0.0;
178 p.proximity_floor = MFTL_MAX(opts->delta_breakup, opts->delta_coalescence);
179 p.self_pairs = MFTL_TRUE;
180 p.pair_filter = opposing_normals;
181
182 return MFTLEnsembleIntersectObjects(ens, &p, detect_pair, &ctx);
183}

189/**
190 * @brief Canonical order: bodies, kind, snapped point, then gap.
191 *
192 * After snapping, two events describing one cell carry BIT-identical points
193 * (the same arithmetic on the same integers), so the point itself is the merge
194 * key and the comparator needs no state.
195 */
196static int cmp_event(const void *x, const void *y)
197{
198 const MFTL_TopologyEvent *a = (const MFTL_TopologyEvent *) x, *b = (const MFTL_TopologyEvent *) y;
199 if (a->body_a != b->body_a) return (a->body_a < b->body_a) ? -1 : 1;
200 if (a->body_b != b->body_b) return (a->body_b < b->body_b) ? -1 : 1;
201 if (a->kind != b->kind) return (a->kind < b->kind) ? -1 : 1;
202 for (int k = 0; k < 3; ++k)
203 if (a->point[k] != b->point[k]) return (a->point[k] < b->point[k]) ? -1 : 1;
204 if (a->gap != b->gap) return (a->gap < b->gap) ? -1 : 1;
205 return 0;
206}

208void MFTLTopoCanonicaliseEvents(const MFTL_Domain *dom, double quantum, MFTL_TopoEventList *list)
209{
210 if (list->n == 0) return;
211
212 /* Snap each point to the centre of its cell: the point then no longer depends
213 * on which of the pairs describing one feature was found first. */
214 for (int i = 0; i < list->n; ++i)
215 {
216 for (int k = 0; k < 3; ++k)
217 {
218 const double c = floor((list->e[i].point[k] - dom->global_min[k]) / quantum);
219 list->e[i].point[k] = dom->global_min[k] + (c + 0.5) * quantum;
220 }
221 MFTLApplyPeriodicity(dom, list->e[i].point);
222 }
223
224 qsort(list->e, (size_t) list->n, sizeof(MFTL_TopologyEvent), cmp_event);
225
226 /* Same bodies, kind and cell: keep the first, which has the smallest gap. */
227 int w = 0;
228 for (int i = 0; i < list->n; ++i)
229 {
230 if (w > 0)
231 {
232 const MFTL_TopologyEvent *p = &list->e[w - 1], *q = &list->e[i];
233 if (p->body_a == q->body_a && p->body_b == q->body_b && p->kind == q->kind && p->point[0] == q->point[0] && p->point[1] == q->point[1] &&
234 p->point[2] == q->point[2])
235 continue;
236 }
237 list->e[w++] = list->e[i];
238 }
239 list->n = w;
240}
