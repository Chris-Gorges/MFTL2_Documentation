125/**
126 * @brief Appends `printf`-formatted text, growing the buffer as needed.
127 * @param b The buffer.
128 * @param fmt `printf` format string; the arguments follow.
129 */
130static void sb_addf(sbuf *b, const char *fmt, ...)
131{
132 va_list ap, ap2;
133 va_start(ap, fmt);
134 va_copy(ap2, ap);
135
136 const int need = vsnprintf(MFTL_NULL, 0, fmt, ap);
137 va_end(ap);
138
139 if (need > 0)
140 {
141 if (b->len + (size_t) need + 1u > b->cap)
142 {
143 while (b->cap < b->len + (size_t) need + 1u) b->cap *= 2u;
144 char *grown = (char *) realloc(b->s, b->cap);
145 if (!grown)
146 {
147 fprintf(stderr, "FATAL ERROR: MFTL vtu header allocation failed.\n");
148 exit(EXIT_FAILURE);
149 }
150 b->s = grown;
151 }
152 vsnprintf(b->s + b->len, b->cap - b->len, fmt, ap2);
153 b->len += (size_t) need;
154 }
155 va_end(ap2);
156}

158/* ═══════════════════════════════════════════════════════════════════════════
159 * 2. ANALYSE: which cells, which points, which of them straddle a seam
160 *
161 * Cells: owned triangles (`[0, tris.n_local)`), or all with `include_ghosts`.
162 * Points: every vertex a written triangle references (possibly a ghost), found
163 * by one marking pass and a prefix scan; unreferenced vertices are dropped. A
164 * seam vertex is thus written by every rank with a triangle on it -- cracked in
165 * the point topology, seamless in the picture.
166 *
167 * Periodic unwrapping: vertices are stored wrapped, so a triangle straddling a
168 * periodic face would be drawn across the whole domain. It gets three PRIVATE
169 * points at its unwrapped corners p0, p0 + e1, p0 + e2 (minimum-image edges),
170 * which may lie outside the box. A triangle wraps iff some minimum-image edge
171 * differs from its raw difference, i.e. iff a periodic correction was applied:
172 * `d - L*nearbyint(d/L)` returns `d` bit-exactly when the rounding is 0, and
173 * differs by about L otherwise. The test is exact, with no tolerance.
174 * ═══════════════════════════════════════════════════════════════════════════ */
175

205/**
206 * @brief Appends one triangle's three unwrapped corners (9 doubles).
207 * @param b The buffer.
208 * @param p0 Corner 0.
209 * @param e1 Edge vector to corner 1, seam-consistent.
210 * @param e2 Edge vector to corner 2, seam-consistent.
211 */
212static void db_add_corners(dbuf *b, const double p0[3], const double e1[3], const double e2[3])
213{
214 MFTL_ENSURE_CAPACITY(b->data, &b->n, &b->cap, 9, double);
215 double *o = b->data + b->n;
216 for (int k = 0; k < 3; ++k)
217 {
218 o[0 + k] = p0[k];
219 o[3 + k] = p0[k] + e1[k];
220 o[6 + k] = p0[k] + e2[k];
221 }
222 b->n += 9;
223}

237/**
238 * @brief Fills @p p for one object and appends its unwrapped corners to @p wpos.
239 *
240 * Two loop shapes, because periodicity is loop-invariant: (a) non-periodic --
241 * mark connectivity only; (b) periodic -- also compute each triangle's
242 * minimum-image edges once and apply the exact wrap test.
243 *
244 * @param obj The mesh.
245 * @param include_ghosts Count ghost triangles too.
246 * @param[out] p Piece description: which vertices become shared
247 * points, which triangles wrap, and the two totals.
248 * @param[out] wpos Receives the unwrapped corners of the wrapping
249 * triangles, nine doubles each.
250 */
251static void analyze_piece(const MFTL_Object *obj, bool include_ghosts, piece *p, dbuf *wpos)
252{
253 const int nv = obj->verts.n;
254 const int nt = include_ghosts ? obj->tris.n : obj->tris.n_local;
255 const MFTL_Domain *const dom = obj->domain;
256 const bool periodic = (dom != MFTL_NULL) && (dom->is_periodic[0] || dom->is_periodic[1] || dom->is_periodic[2]);
257
258 p->n_cells = nt;
259 p->n_wrapped = 0;
260 p->wpos_off = wpos->n;
261
262 for (int v = 0; v < nv; ++v) p->remap[v] = MFTL_ERROR;
263
264 const bool compact = MFTLObjectHasTopology(obj);
265
266 if (!periodic)
267 {
268 /* (a) Marking only. With the derived array this is a flat pass over 12 bytes
269 * per triangle; without it, three loads with a 16-byte stride. */
270 if (compact)
271 {
272 const int *restrict tv = obj->tris.verts;
273 for (int t = 0; t < nt; ++t)
274 {
275 p->remap[tv[3 * t + 0]] = 0;
276 p->remap[tv[3 * t + 1]] = 0;
277 p->remap[tv[3 * t + 2]] = 0;
278 }
279 }
280 else
281 {
282 for (int t = 0; t < nt; ++t)
283 {
284 int v[3];
285 MFTLGetTriangleVertices(obj, t, v);
286 p->remap[v[0]] = 0;
287 p->remap[v[1]] = 0;
288 p->remap[v[2]] = 0;
289 }
290 }
291 if (nt > 0) memset(p->wflag, 0, (size_t) nt);
292 }
293 else
294 {
295 /* (b) Compute each triangle's seam-consistent edges once during analysis. */
296 const double *const pos = obj->verts.pos;
297 const int *const tv = compact ? obj->tris.verts : MFTL_NULL;
298 for (int t = 0; t < nt; ++t)
299 {
300 int v[3];
301 if (compact)
302 {
303 v[0] = tv[3 * t + 0];
304 v[1] = tv[3 * t + 1];
305 v[2] = tv[3 * t + 2];
306 }
307 else
308 {
309 MFTLGetTriangleVertices(obj, t, v);
310 }
311 p->remap[v[0]] = 0;
312 p->remap[v[1]] = 0;
313 p->remap[v[2]] = 0;
314
315 const double *const p0 = &pos[3 * v[0]];
316 const double *const p1 = &pos[3 * v[1]];
317 const double *const p2 = &pos[3 * v[2]];
318
319 double e1[3], e2[3];
320 MFTLGetMinimumImageVector(dom, e1, p0, p1);
321 MFTLGetMinimumImageVector(dom, e2, p0, p2);
322
323 int wrapped = 0;
324 for (int k = 0; k < 3; ++k)
325 {
326 if (e1[k] != p1[k] - p0[k] || e2[k] != p2[k] - p0[k]) wrapped = 1;
327 }
328
329 p->wflag[t] = (unsigned char) wrapped;
330 if (wrapped)
331 {
332 db_add_corners(wpos, p0, e1, e2);
333 p->n_wrapped++;
334 }
335 }
336 }
337
338 /* Prefix scan: turn the marks into compact indices. */
339 int next = 0;
340 for (int v = 0; v < nv; ++v)
341 {
342 if (p->remap[v] >= 0) p->remap[v] = next++;
343 }
344 p->n_marked = next;
345 p->n_points = next + 3 * p->n_wrapped;
346}

405/**
406 * @brief Points of one object: the referenced vertices, then the private
407 * unwrapped corners (emit_conn() addresses them as `n_marked + 3*w`, so
408 * the order is load-bearing).
409 *
410 * @param obj The mesh.
411 * @param p Piece description from analyze_piece().
412 * @param wpos Unwrapped corners collected by analyze_piece().
413 * @param[out] out Destination coordinate array, `3 * p->n_points` doubles.
414 */
415static void emit_points(const MFTL_Object *obj, const piece *p, const dbuf *wpos, double *out)
416{
417 const int nv = obj->verts.n;
418 const double *const pos = obj->verts.pos;
419
420 if (p->n_marked == nv)
421 {
422 /* Every vertex referenced (the common case): the compaction is the identity. */
423 memcpy(out, pos, (size_t) (3 * nv) * sizeof(double));
424 out += 3 * nv;
425 }
426 else
427 {
428 for (int v = 0; v < nv; ++v)
429 {
430 if (p->remap[v] < 0) continue;
431 *out++ = pos[3 * v + 0];
432 *out++ = pos[3 * v + 1];
433 *out++ = pos[3 * v + 2];
434 }
435 }
436
437 if (p->n_wrapped > 0) memcpy(out, wpos->data + p->wpos_off, (size_t) (9 * p->n_wrapped) * sizeof(double));
438}

440/**
441 * @brief Connectivity of one object, shifted into the file's point numbering.
442 *
443 * @param obj The mesh.
444 * @param p Piece description from analyze_piece().
445 * @param point_base Point index this object's block starts at.
446 * @param[out] out Destination connectivity array, `3 * p->n_cells` ints.
447 */
448static void emit_conn(const MFTL_Object *obj, const piece *p, int point_base, int *out)
449{
450 const int nt = p->n_cells;
451 const int priv0 = point_base + p->n_marked;
452 int w = 0;
453
454 if (MFTLObjectHasTopology(obj))
455 {
456 const int *restrict tv = obj->tris.verts;
457 for (int t = 0; t < nt; ++t)
458 {
459 if (p->wflag[t])
460 {
461 out[3 * t + 0] = priv0 + 3 * w + 0;
462 out[3 * t + 1] = priv0 + 3 * w + 1;
463 out[3 * t + 2] = priv0 + 3 * w + 2;
464 ++w;
465 }
466 else
467 {
468 out[3 * t + 0] = point_base + p->remap[tv[3 * t + 0]];
469 out[3 * t + 1] = point_base + p->remap[tv[3 * t + 1]];
470 out[3 * t + 2] = point_base + p->remap[tv[3 * t + 2]];
471 }
472 }
473 }
474 else
475 {
476 for (int t = 0; t < nt; ++t)
477 {
478 if (p->wflag[t])
479 {
480 out[3 * t + 0] = priv0 + 3 * w + 0;
481 out[3 * t + 1] = priv0 + 3 * w + 1;
482 out[3 * t + 2] = priv0 + 3 * w + 2;
483 ++w;
484 }
485 else
486 {
487 int v[3];
488 MFTLGetTriangleVertices(obj, t, v);
489 out[3 * t + 0] = point_base + p->remap[v[0]];
490 out[3 * t + 1] = point_base + p->remap[v[1]];
491 out[3 * t + 2] = point_base + p->remap[v[2]];
492 }
493 }
494 }

497/**
498 * @brief A per-point Int32 array: owner rank or global id. Private corners
499 * inherit the values of the vertices they stand in for.
500 *
501 * Reads the AoS records with a 16-byte stride; an SoA owner array would cost
502 * 4 B per vertex permanently for an output path dominated by the disk write.
503 *
504 * @param obj The mesh.
505 * @param p Piece description from analyze_piece().
506 * @param want_gid `true` for the global id, `false` for the owning rank.
507 * @param[out] out Destination int array, `p->n_points` entries.

510{

512 const MFTL_Vertex *const rec = obj->verts.rec;
513

515 {
516 if (p->remap[v] < 0) continue;
517 *out++ = want_gid ? rec[v].global_id : rec[v].owner_proc;
518 }
519
520 /* The private corners inherit the values of the vertices they stand in for. */
521 const int nt = p->n_cells;

573/**
574 * @brief Writes everything up to and including the `_` of the appended section
575 * (binary), or everything up to the first DataArray (ascii, where the
576 * caller interleaves data and closing tags itself).
577 *
578 * @param[out] h Receives the complete XML header text.
579 * @param blocks The block table.
580 * @param rel_offsets Byte offset of each block inside the appended section.
581 * @param n_points Points in this piece.
582 * @param n_cells Cells in this piece.
583 * @param binary `true` for raw appended data, `false` for inline ASCII.
584 */
585static void build_header(sbuf *h, const io_block *blocks, const unsigned long long *rel_offsets, long long n_points, long long n_cells, bool binary)
586{
587 static const int point_arrays[] = {BLK_VOWNER, BLK_VGID};
588 static const int cell_arrays[] = {BLK_TOWNER, BLK_BODYID, BLK_TGID};
589
590 sb_addf(h, "<?xml version=\"1.0\"?>\n");
591 sb_addf(h, "<VTKFile type=\"UnstructuredGrid\" version=\"1.0\" byte_order=\"%s\" header_type=\"UInt64\">\n", host_byte_order());
592 sb_addf(h, " <UnstructuredGrid>\n");
593 sb_addf(h, " <Piece NumberOfPoints=\"%lld\" NumberOfCells=\"%lld\">\n", n_points, n_cells);
594
595 /* In ascii mode the data lives inside the tags, so this function only emits
596 * the binary skeleton; the ascii backend calls the tag helper itself. */
597 if (!binary) return;
598
599 sb_addf(h, " <Points>\n");
600 emit_dataarray_tag(h, &blocks[BLK_POINTS], true, rel_offsets[BLK_POINTS]);
601 sb_addf(h, " </Points>\n");
602
603 sb_addf(h, " <Cells>\n");
604 emit_dataarray_tag(h, &blocks[BLK_CONN], true, rel_offsets[BLK_CONN]);
605 emit_dataarray_tag(h, &blocks[BLK_OFFSETS], true, rel_offsets[BLK_OFFSETS]);
606 emit_dataarray_tag(h, &blocks[BLK_TYPES], true, rel_offsets[BLK_TYPES]);
607 sb_addf(h, " </Cells>\n");
608
609 const char *ps = first_enabled(blocks, point_arrays, 2);
610 sb_addf(h, " <PointData%s%s%s>\n", ps ? " Scalars=\"" : "", ps ? ps : "", ps ? "\"" : "");
611 for (int i = 0; i < 2; ++i)
612 {
613 if (blocks[point_arrays[i]].enabled) emit_dataarray_tag(h, &blocks[point_arrays[i]], true, rel_offsets[point_arrays[i]]);
614 }
615 sb_addf(h, " </PointData>\n");
616
617 const char *cs = first_enabled(blocks, cell_arrays, 3);
618 sb_addf(h, " <CellData%s%s%s>\n", cs ? " Scalars=\"" : "", cs ? cs : "", cs ? "\"" : "");
619 for (int i = 0; i < 3; ++i)
620 {
621 if (blocks[cell_arrays[i]].enabled) emit_dataarray_tag(h, &blocks[cell_arrays[i]], true, rel_offsets[cell_arrays[i]]);
622 }
623 sb_addf(h, " </CellData>\n");
624
625 sb_addf(h, " </Piece>\n");
626 sb_addf(h, " </UnstructuredGrid>\n");
627 sb_addf(h, " <AppendedData encoding=\"raw\">\n _");
628}

637/**
638 * @brief Formats one block as decimal text, one tuple per line, in chunks.
639 * @param fp Destination stream.
640 * @param b The block.
641 * @return #MFTL_SUCCESS, or #MFTL_ERROR on a write failure.
642 */
643static int ascii_emit(FILE *fp, const io_block *b)
644{
645 const size_t n = b->bytes / scalar_size(b->kind);
646 char buf[8192];
647 size_t len = 0;
648
649 for (size_t i = 0; i < n; ++i)
650 {
651 if (len + 64u > sizeof buf)
652 {
653 if (fwrite(buf, 1, len, fp) != len) return MFTL_ERROR;
654 len = 0;
655 }
656 int w = 0;
657 switch (b->kind)
658 {
659 case SC_F64: w = snprintf(buf + len, sizeof buf - len, "%.10e", ((const double *) b->data)[i]); break;
660 case SC_I32: w = snprintf(buf + len, sizeof buf - len, "%d", ((const int *) b->data)[i]); break;
661 case SC_U8: w = snprintf(buf + len, sizeof buf - len, "%u", (unsigned) ((const unsigned char *) b->data)[i]); break;
662 }
663 if (w < 0) return MFTL_ERROR;
664 len += (size_t) w;
665 buf[len++] = (((i + 1u) % (size_t) b->ncomp) == 0u) ? '\n' : ' ';
666 }
667
668 if (len > 0 && fwrite(buf, 1, len, fp) != len) return MFTL_ERROR;
669 return MFTL_SUCCESS;
670}

672/**
673 * @brief Writes one block's raw bytes, preceded by its UInt64 length.
674 *
675 * @param fp Destination stream.
676 * @param b The block.
677 * @return #MFTL_SUCCESS, or #MFTL_ERROR on a write failure.
678 */
679static int binary_emit(FILE *fp, const io_block *b)
680{
681 const unsigned long long n = (unsigned long long) b->bytes;
682 if (fwrite(&n, sizeof n, 1, fp) != 1) return MFTL_ERROR;
683 if (b->bytes > 0 && fwrite(b->data, 1, b->bytes, fp) != b->bytes) return MFTL_ERROR;
684 return MFTL_SUCCESS;
685}

687/**
688 * @brief Backend A: this rank writes its own `.vtu`, with no collective at all.
689 * @param prefix File name prefix; the rank number and extension are appended.
690 * @param rank This rank, used in the file name.
691 * @param blocks The block table, fully populated.
692 * @param n_points Points in this piece.
693 * @param n_cells Cells in this piece.
694 * @param binary `true` for raw appended data, `false` for inline ASCII.
695 * @return #MFTL_SUCCESS, or #MFTL_ERROR on any I/O failure.
696 */
697static int write_piece_file(const char *prefix, int rank, io_block *blocks, long long n_points, long long n_cells, bool binary)
698{
699 char path[1024];
700 const int np = snprintf(path, sizeof path, "%s_%05d.vtu", prefix, rank);
701 if (np < 0 || (size_t) np >= sizeof path) return MFTL_ERROR;
702
703 FILE *fp = fopen(path, binary ? "wb" : "w");
704 if (!fp)
705 {
706 fprintf(stderr, "[Rank %d] MFTLWriteVtu: could not open %s.\n", rank, path);
707 return MFTL_ERROR;
708 }
709
710 char *const iobuf = (char *) malloc(1u << 20);
711 if (iobuf) setvbuf(fp, iobuf, _IOFBF, 1u << 20);
712
713 int status = MFTL_SUCCESS;
714
715 unsigned long long rel[BLK_COUNT];
716 unsigned long long acc = 0;
717 for (int i = 0; i < BLK_COUNT; ++i)
718 {
719 rel[i] = acc;
720 if (blocks[i].enabled) acc += 8u + (unsigned long long) blocks[i].bytes;
721 }
722
723 sbuf h;
724 sb_init(&h);
725 build_header(&h, blocks, rel, n_points, n_cells, binary);
726
727 if (fwrite(h.s, 1, h.len, fp) != h.len) status = MFTL_ERROR;
728
729 if (binary)
730 {
731 for (int i = 0; i < BLK_COUNT && status == MFTL_SUCCESS; ++i)
732 {
733 if (blocks[i].enabled) status = binary_emit(fp, &blocks[i]);
734 }
735 if (status == MFTL_SUCCESS && fwrite(k_footer, 1, strlen(k_footer), fp) != strlen(k_footer)) status = MFTL_ERROR;
736 }
737 else
738 {
739 /* ASCII: the numbers live inside the tags, so the structure is interleaved.
740 * Same block descriptors, same order, different punctuation. */
741 static const int point_arrays[] = {BLK_VOWNER, BLK_VGID};
742 static const int cell_arrays[] = {BLK_TOWNER, BLK_BODYID, BLK_TGID};
743 sbuf t;
744 sb_init(&t);
745
746/* Emits one enabled block as an ASCII <DataArray> with its numbers inline.
747 * Documented with @def near the top of this file, because a doc comment inside a
748 * function body is attributed to the FUNCTION rather than to the macro. */
749#define ASCII_ARRAY(idx) \
750 do \
751 { \
752 t.len = 0; \
753 t.s[0] = '\0'; \
754 emit_dataarray_tag(&t, &blocks[idx], false, 0); \
755 if (fwrite(t.s, 1, t.len, fp) != t.len) status = MFTL_ERROR; \
756 if (status == MFTL_SUCCESS) status = ascii_emit(fp, &blocks[idx]); \
757 if (status == MFTL_SUCCESS && fprintf(fp, " </DataArray>\n") < 0) status = MFTL_ERROR; \
758 } while (0)
759
760 fprintf(fp, " <Points>\n");
761 ASCII_ARRAY(BLK_POINTS);
762 fprintf(fp, " </Points>\n <Cells>\n");
763 ASCII_ARRAY(BLK_CONN);
764 ASCII_ARRAY(BLK_OFFSETS);
765 ASCII_ARRAY(BLK_TYPES);
766 fprintf(fp, " </Cells>\n");
767
768 const char *ps = first_enabled(blocks, point_arrays, 2);
769 fprintf(fp, " <PointData%s%s%s>\n", ps ? " Scalars=\"" : "", ps ? ps : "", ps ? "\"" : "");
770 for (int i = 0; i < 2; ++i)
771 {
772 if (blocks[point_arrays[i]].enabled) ASCII_ARRAY(point_arrays[i]);
773 }
774 fprintf(fp, " </PointData>\n");
775
776 const char *cs = first_enabled(blocks, cell_arrays, 3);
777 fprintf(fp, " <CellData%s%s%s>\n", cs ? " Scalars=\"" : "", cs ? cs : "", cs ? "\"" : "");
778 for (int i = 0; i < 3; ++i)
779 {
780 if (blocks[cell_arrays[i]].enabled) ASCII_ARRAY(cell_arrays[i]);
781 }
782 fprintf(fp, " </CellData>\n");
783 fprintf(fp, " </Piece>\n </UnstructuredGrid>\n</VTKFile>\n");
784#undef ASCII_ARRAY
785
786 sb_free(&t);
787 }
788
789 sb_free(&h);
790
791 if (ferror(fp)) status = MFTL_ERROR;
792 if (fclose(fp) != 0) status = MFTL_ERROR;
793 free(iobuf);
794 return status;
795}

855/* ═══════════════════════════════════════════════════════════════════════════
856 * 6. Backend B: one shared file (MPI-IO)
857 *
858 * [0, H) XML header (rank 0)
859 * H + rel[b] UInt64 length of block b (rank 0)
860 * H + rel[b] + 8 + my_offset[b] this rank's slice of block b (every rank)
861 * H + total footer (rank 0)
862 *
863 * Every rank knows H (identical header text) and its offsets (one exclusive
864 * scan), so no offsets are distributed. The slices are written with
865 * MPI_File_write_at_all, once per enabled block on every rank (count zero if
866 * nothing to contribute), so the MPI library can aggregate the many small
867 * non-contiguous slices into a few large stripe-aligned writes (two-phase
868 * collective buffering) -- the independent MPI_File_write_at would not.
869 * ═══════════════════════════════════════════════════════════════════════════ */
870
871/**
872 * @brief The one round of communication of the shared layout: an exclusive scan
873 * of {points, cells} (where this rank's slices go) and an all-reduce (the
874 * totals the header quotes), posted together as non-blocking collectives.
875 *
876 * @param comm The communicator.
877 * @param my_points This rank's point count.
878 * @param my_cells This rank's cell count.
879 * @param[out] base Exclusive-scan result: {points, cells} before this rank.
880 * @param[out] total Totals over all ranks.
881 */
882static void scan_placement(MPI_Comm comm, long long my_points, long long my_cells, long long base[2], long long total[2])
883{
884 long long mine[2] = {my_points, my_cells};
885 total[0] = total[1] = 0;
886
887 MPI_Request req[2];
888 MPI_Iexscan(mine, base, 2, MPI_LONG_LONG, MPI_SUM, comm, &req[0]);
889 MPI_Iallreduce(mine, total, 2, MPI_LONG_LONG, MPI_SUM, comm, &req[1]);
890 MPI_Waitall(2, req, MPI_STATUSES_IGNORE);
891
892 /* MPI leaves rank 0's exscan result UNDEFINED; its prefix is zero. */
893 int rank = 0;
894 MPI_Comm_rank(comm, &rank);
895 if (rank == 0) base[0] = base[1] = 0;
896}
897
898/**
899 * @brief Backend B: every rank writes into ONE `.vtu` through MPI-IO.
900 * @param comm The communicator.
901 * @param rank This rank.
902 * @param prefix File name prefix.
903 * @param blocks The block table, fully populated.
904 * @param base Exclusive-scan offsets from scan_placement().
905 * @param total Global totals from scan_placement().
906 * @return #MFTL_SUCCESS, or #MFTL_ERROR on any I/O failure.
907 */
908static int write_shared_file(MPI_Comm comm, int rank, const char *prefix, io_block *blocks, const long long base[2], const long long total[2])
909{
910 /* Int32 limits on the GLOBAL totals (identical everywhere, so returning is a
911 * collective decision): point ids, and offsets = 3 x cumulative cells. */
912 if (total[0] > 2147483647LL || 3 * total[1] > 2147483647LL)
913 {
914 if (rank == 0) fprintf(stderr, "MFTLWriteVtu: %lld points / %lld cells exceed the Int32 connectivity and offset limits.\n", total[0], total[1]);
915 return MFTL_ERROR;
916 }
917
918 /* Place every block. `unit` is bytes per point or per cell, so this is the
919 * only arithmetic the scan result feeds. */
920 unsigned long long rel[BLK_COUNT];
921 unsigned long long acc = 0;
922 int local_bad = 0;
923 for (int i = 0; i < BLK_COUNT; ++i)
924 {
925 rel[i] = acc;
926 if (!blocks[i].enabled) continue;
927 const long long n_glob = blocks[i].per_cell ? total[1] : total[0];
928 const long long n_base = blocks[i].per_cell ? base[1] : base[0];
929 blocks[i].global_bytes = blocks[i].unit * (size_t) n_glob;
930 blocks[i].my_offset = blocks[i].unit * (size_t) n_base;
931 acc += 8u + (unsigned long long) blocks[i].global_bytes;
932
933 if (blocks[i].bytes > 2147483647u)
934 {
935 fprintf(stderr, "[Rank %d] MFTLWriteVtu: block '%s' is %llu bytes, above the MPI count limit.\n", rank, blocks[i].name, (unsigned long long) blocks[i].bytes);
936 local_bad = 1;
937 }
938 }
939
940 char path[1024];
941 const int np = snprintf(path, sizeof path, "%s.vtu", prefix);
942 if (np < 0 || (size_t) np >= sizeof path) local_bad = 1;
943
944 /* Local checks, agreed before the collective MPI_File_open. */
945 int any_bad = 0;
946 MPI_Allreduce(&local_bad, &any_bad, 1, MPI_INT, MPI_MAX, comm);
947 if (any_bad) return MFTL_ERROR;
948
949 sbuf h;
950 sb_init(&h);
951 build_header(&h, blocks, rel, total[0], total[1], true);
952 const MPI_Offset header_len = (MPI_Offset) h.len;
953 const MPI_Offset footer_at = header_len + (MPI_Offset) acc;
954
955 MPI_File fh;
956 /* File error handlers default to MPI_ERRORS_RETURN, so the return codes are
957 * meaningful without installing one. */
958 if (MPI_File_open(comm, path, MPI_MODE_CREATE | MPI_MODE_WRONLY, MPI_INFO_NULL, &fh) != MPI_SUCCESS)
959 {
960 if (rank == 0) fprintf(stderr, "MFTLWriteVtu: MPI_File_open failed for %s.\n", path);
961 sb_free(&h);
962 return MFTL_ERROR;
963 }
964
965 int status = MFTL_SUCCESS;
966
967 /* Truncate: overwriting a longer file would leave trailing bytes. */
968 if (MPI_File_set_size(fh, footer_at + (MPI_Offset) strlen(k_footer)) != MPI_SUCCESS) status = MFTL_ERROR;
969
970 if (rank == 0 && status == MFTL_SUCCESS)
971 {
972 if (MPI_File_write_at(fh, 0, h.s, (int) h.len, MPI_BYTE, MPI_STATUS_IGNORE) != MPI_SUCCESS) status = MFTL_ERROR;
973
974 /* The per-block length prefixes: eight bytes each, at most nine of them. */
975 for (int i = 0; i < BLK_COUNT && status == MFTL_SUCCESS; ++i)
976 {
977 if (!blocks[i].enabled) continue;
978 const unsigned long long n = (unsigned long long) blocks[i].global_bytes;
979 if (MPI_File_write_at(fh, header_len + (MPI_Offset) rel[i], &n, 8, MPI_BYTE, MPI_STATUS_IGNORE) != MPI_SUCCESS) status = MFTL_ERROR;
980 }
981
982 if (status == MFTL_SUCCESS && MPI_File_write_at(fh, footer_at, (void *) k_footer, (int) strlen(k_footer), MPI_BYTE, MPI_STATUS_IGNORE) != MPI_SUCCESS)
983 status = MFTL_ERROR;
984 }
985
986 /* Collective, once per enabled block, on every rank (zero counts included). */
987 for (int i = 0; i < BLK_COUNT; ++i)
988 {
989 if (!blocks[i].enabled) continue;
990 const MPI_Offset at = header_len + (MPI_Offset) rel[i] + 8 + (MPI_Offset) blocks[i].my_offset;
991 if (MPI_File_write_at_all(fh, at, blocks[i].data, (int) blocks[i].bytes, MPI_BYTE, MPI_STATUS_IGNORE) != MPI_SUCCESS) status = MFTL_ERROR;
992 }
993
994 if (MPI_File_close(&fh) != MPI_SUCCESS) status = MFTL_ERROR;
995 sb_free(&h);
996 return status;
997}

999/* ═══════════════════════════════════════════════════════════════════════════
1000 * 7. MFTLWriteVtu
1001 * ═══════════════════════════════════════════════════════════════════════════ */
1002
1003int MFTLWriteVtu(MFTL_Ensemble *ensemble, const char *filename_prefix, const MFTL_VtkOptions *options)
1004{
1005 if (!ensemble || !filename_prefix) return MFTL_ERROR;
1006
1007 const MFTL_VtkOptions opt = options ? *options : MFTLVtkDefaults();
1008 const MFTL_Domain *const dom = &ensemble->domain;
1009 const int rank = dom->rank;
1010 const int nproc = dom->nproc;
1011 const int n_obj = ensemble->n_local_objects;
1012 const bool binary = (opt.encoding == MFTL_VTK_BINARY);
1013 const bool shared = (opt.layout == MFTL_IO_SINGLE_FILE);
1014
1015 if (shared && !binary)
1016 {
1017 if (rank == 0) fprintf(stderr, "MFTLWriteVtu: MFTL_IO_SINGLE_FILE requires MFTL_VTK_BINARY.\n");
1018 return MFTL_ERROR;
1019 }
1020
1021 /* ── ANALYSE ──────────────────────────────────────────────────────────── */
1022
1023 piece *pieces = (piece *) MFTLCallocOrDie((size_t) (n_obj > 0 ? n_obj : 1), sizeof(piece), "MFTL vtu pieces");
1024 dbuf wpos;
1025 db_init(&wpos);
1026
1027 long long my_points = 0, my_cells = 0;
1028 for (int o = 0; o < n_obj; ++o)
1029 {
1030 const MFTL_Object *obj = ensemble->local_objects[o];
1031 const int nv = obj->verts.n;
1032 const int nt = opt.include_ghosts ? obj->tris.n : obj->tris.n_local;
1033
1034 pieces[o].remap = (int *) MFTLMallocOrDie((size_t) (nv > 0 ? nv : 1) * sizeof(int), "MFTL vtu point remap");
1035 pieces[o].wflag = (unsigned char *) MFTLMallocOrDie((size_t) (nt > 0 ? nt : 1), "MFTL vtu wrap flags");
1036
1037 analyze_piece(obj, opt.include_ghosts, &pieces[o], &wpos);
1038
1039 my_points += pieces[o].n_points;
1040 my_cells += pieces[o].n_cells;
1041 }
1042
1043 /* ── ASSEMBLE ─────────────────────────────────────────────────────────── */
1044
1045 io_block blocks[BLK_COUNT];
1046 memset(blocks, 0, sizeof blocks);
1047
1048/* Fills in one row of the block table. Documented with @def near the top of this
1049 * file; see ASCII_ARRAY for why. */
1050#define DECL(idx, nm, ty, kd, nc, pc, en, un) \
1051 blocks[idx].name = (nm); \
1052 blocks[idx].vtk_type = (ty); \
1053 blocks[idx].kind = (kd); \
1054 blocks[idx].ncomp = (nc); \
1055 blocks[idx].per_cell = (pc); \
1056 blocks[idx].enabled = (en); \
1057 blocks[idx].unit = (un)
1058
1059 DECL(BLK_POINTS, "Points", "Float64", SC_F64, 3, false, true, 3 * sizeof(double));
1060 DECL(BLK_CONN, "connectivity", "Int32", SC_I32, 1, true, true, 3 * sizeof(int));
1061 DECL(BLK_OFFSETS, "offsets", "Int32", SC_I32, 1, true, true, sizeof(int));
1062 DECL(BLK_TYPES, "types", "UInt8", SC_U8, 1, true, true, 1);
1063 DECL(BLK_VOWNER, "VertexOwner", "Int32", SC_I32, 1, false, opt.write_owner, sizeof(int));
1064 DECL(BLK_VGID, "VertexGID", "Int32", SC_I32, 1, false, opt.write_global_id, sizeof(int));
1065 DECL(BLK_TOWNER, "TriangleOwner", "Int32", SC_I32, 1, true, opt.write_owner, sizeof(int));
1066 DECL(BLK_BODYID, "BodyID", "Int32", SC_I32, 1, true, opt.write_body_id, sizeof(int));
1067 DECL(BLK_TGID, "TriangleGID", "Int32", SC_I32, 1, true, opt.write_global_id, sizeof(int));
1068#undef DECL
1069
1070 for (int i = 0; i < BLK_COUNT; ++i)
1071 {
1072 if (!blocks[i].enabled) continue;
1073 blocks[i].bytes = blocks[i].unit * (size_t) (blocks[i].per_cell ? my_cells : my_points);
1074 blocks[i].data = MFTLMallocOrDie(blocks[i].bytes > 0 ? blocks[i].bytes : 1, "MFTL vtu data block");
1075 }
1076
1077 /* Shared layout: one Piece, so numbering needs this rank's base indices.
1078 * Per-rank layout: bases are zero. */
1079 long long base[2] = {0, 0}, total[2] = {0, 0};
1080 if (shared) scan_placement(dom->comm_world, my_points, my_cells, base, total);
1081
1082 {
1083 long long p_off = base[0]; /* running point base of the current object */
1084 long long c_off = base[1];
1085
1086 for (int o = 0; o < n_obj; ++o)
1087 {
1088 const MFTL_Object *obj = ensemble->local_objects[o];
1089 const piece *p = &pieces[o];
1090 const long long lp = p_off - base[0]; /* this object's slice within my blocks */
1091 const long long lc = c_off - base[1];
1092
1093 emit_points(obj, p, &wpos, (double *) blocks[BLK_POINTS].data + 3 * lp);
1094 emit_conn(obj, p, (int) p_off, (int *) blocks[BLK_CONN].data + 3 * lc);
1095
1096 int *const off = (int *) blocks[BLK_OFFSETS].data + lc;
1097 for (int t = 0; t < p->n_cells; ++t) off[t] = (int) (3 * (c_off + t + 1));
1098
1099 memset((unsigned char *) blocks[BLK_TYPES].data + lc, MFTL_VTK_TRIANGLE, (size_t) p->n_cells);
1100
1101 if (blocks[BLK_VOWNER].enabled) emit_point_ints(obj, p, false, (int *) blocks[BLK_VOWNER].data + lp);
1102 if (blocks[BLK_VGID].enabled) emit_point_ints(obj, p, true, (int *) blocks[BLK_VGID].data + lp);
1103
1104 if (blocks[BLK_TOWNER].enabled)
1105 {
1106 int *const d = (int *) blocks[BLK_TOWNER].data + lc;
1107 for (int t = 0; t < p->n_cells; ++t) d[t] = obj->tris.rec[t].owner_proc;
1108 }
1109 if (blocks[BLK_BODYID].enabled)
1110 {
1111 int *const d = (int *) blocks[BLK_BODYID].data + lc;
1112 for (int t = 0; t < p->n_cells; ++t) d[t] = obj->global_id;
1113 }
1114 if (blocks[BLK_TGID].enabled)
1115 {
