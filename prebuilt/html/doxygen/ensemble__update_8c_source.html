64/**
65 * @brief The hot loop: geometry into the triangle pool, statistics into @p acc.
66 *
67 * Min/max use `fmin`/`fmax`: with an if-form clang rejects the loop as a
68 * reduction before anything else ("value that could not be identified as
69 * reduction"). They lower to single instructions on AArch64 but need NaN fix-up
70 * on x86-64, so the measured 4-5 % is platform dependent.
71 *
72 * @param obj The mesh. `tris.area`, `tris.normal` and `tris.center` are
73 * written for the first @p n_tris triangles.
74 * @param n_tris Triangles to process, from index 0.
75 * @param[out] acc The nine running accumulators, written back once at the end.
76 */
77static void update_kernel(MFTL_Object *obj, int n_tris, update_accumulators *acc)
78{
79 MFTL_ASSERT(MFTLObjectHasTopology(obj) && "MFTLEnsembleUpdate: stale tris.verts -- call MFTLObjectBuildDerived first");
80 MFTL_ASSERT(n_tris <= obj->tris.n && "MFTLEnsembleUpdate: count exceeds the triangle pool");
81
82 /* Three contiguous output streams (SoA). */
83 double *const restrict out_area = obj->tris.area;
84 double *const restrict out_normal = obj->tris.normal;
85 double *const restrict out_center = obj->tris.center;
86
87 /* The centroid wrap stays the branchy MFTLApplyPeriodicity(): a branchless
88 * form measured 1.34-1.37x slower in this (non-vectorised) loop. */
89 const MFTL_Domain *const dom = obj->domain;
90
91 double min_area = acc->min_area, max_area = acc->max_area, sum_area = 0.0;
92 double min_q = acc->min_q, max_q = acc->max_q, sum_q = 0.0;
93 double min_esq = acc->min_esq, max_esq = acc->max_esq, sum_esq = 0.0;
94
95 MFTL_FOREACH_TRIANGLE_GEOMETRY(
96 obj, n_tris,
97
98 /* Unnormalised normal; its magnitude is twice the area. */
99 const double nx = e1y * e2z - e1z * e2y;
100 const double ny = e1z * e2x - e1x * e2z;
101 const double nz = e1x * e2y - e1y * e2x;
102
103 /* Unclamped, as in MFTLGetTriangleProperties(): a clamp would floor the
104 * area and let q exceed 1. */
105 const double mag = MFTL_SQRT(nx * nx + ny * ny + nz * nz);
106 const double area = 0.5 * mag;
107
108 /* The three edges are e1, e2 and (e2 - e1). */
109 const double e3x = e2x - e1x;
110 const double e3y = e2y - e1y;
111 const double e3z = e2z - e1z;
112
113 const double a_sq = e1x * e1x + e1y * e1y + e1z * e1z;
114 const double b_sq = e2x * e2x + e2y * e2y + e2z * e2z;
115 const double c_sq = e3x * e3x + e3y * e3y + e3z * e3z;
116
117 /* Mean-ratio quality, dimensionless: the only scale-free degeneracy test
118 * is against zero. */
119 const double denom = a_sq + b_sq + c_sq;
120 const double q = (denom > 0.0) ? MFTL_FOUR_SQRT3 * area / denom : 0.0;
121
122 min_area = fmin(min_area, area); max_area = fmax(max_area, area); sum_area += area;
123 min_q = fmin(min_q, q); max_q = fmax(max_q, q); sum_q += q;
124 min_esq = fmin(min_esq, fmin(a_sq, fmin(b_sq, c_sq))); max_esq = fmax(max_esq, fmax(a_sq, fmax(b_sq, c_sq))); sum_esq += denom;
125
126 const double inv_mag = (mag > 0.0) ? 1.0 / mag : 0.0; /* exact at every scale; see MFTLGetTriangleProperties */
127 out_area[t] = area;
128 out_normal[3 * t + 0] = nx * inv_mag;
129 out_normal[3 * t + 1] = ny * inv_mag;
130 out_normal[3 * t + 2] = nz * inv_mag;
131
132 /* Centroid from seam-consistent offsets, then wrapped (MFTLGetOwnerProc()). */
133 double c[3] = {p0x + (e1x + e2x) * MFTL_ONE_THIRD, p0y + (e1y + e2y) * MFTL_ONE_THIRD, p0z + (e1z + e2z) * MFTL_ONE_THIRD};
134 MFTLApplyPeriodicity(dom, c);
135 out_center[3 * t + 0] = c[0];
136 out_center[3 * t + 1] = c[1];
137 out_center[3 * t + 2] = c[2];);
138
139 acc->min_area = min_area;
140 acc->max_area = max_area;
141 acc->sum_area = sum_area;
142 acc->min_q = min_q;
143 acc->max_q = max_q;
144 acc->sum_q = sum_q;
145 acc->min_esq = min_esq;
146 acc->max_esq = max_esq;
147 acc->sum_esq = sum_esq;
148}

150/**
151 * @brief Turns one reduced record into the nine public statistics.
152 *
153 * Averages divide by GLOBAL counts (never an average of per-rank averages). A
154 * body with no triangles anywhere reports zeros instead of the identities.
155 *
156 * @param[out] qs Destination.
157 * @param mn MFTL_UPD_N_MIN reduced minima for this body.
158 * @param mx MFTL_UPD_N_MAX reduced maxima for this body.
159 * @param sm MFTL_UPD_N_SUM reduced sums for this body.
160 */
161static void unpack_slot(MFTL_ObjectQualityStatistics *qs, const double *mn, const double *mx, const double *sm)
162{
163 const double n_tris_g = sm[3];
164 const double n_edges_g = sm[4];
165
166 /* Averages against the GLOBAL counts. */
167 qs->avg_triangle_area = (n_tris_g > 0.0) ? sm[0] / n_tris_g : 0.0;
168 qs->avg_aspect_ratio = (n_tris_g > 0.0) ? sm[1] / n_tris_g : 0.0;
169 qs->avg_edge_length_sq = (n_edges_g > 0.0) ? sm[2] / n_edges_g : 0.0;
170
171 /* No triangles anywhere: report zeros, not the DBL_MAX identities. */
172 const bool empty = (n_tris_g <= 0.0);
173 qs->min_triangle_area = empty ? 0.0 : mn[0];
174 qs->max_triangle_area = empty ? 0.0 : mx[0];
175 qs->min_aspect_ratio = empty ? 0.0 : mn[1];
176 qs->max_aspect_ratio = empty ? 0.0 : mx[1];
177 qs->min_edge_length_sq = empty ? 0.0 : mn[2];
178 qs->max_edge_length_sq = empty ? 0.0 : mx[2];
179}

181void MFTLEnsembleUpdate(MFTL_Ensemble *ensemble)
182{
183 MFTL_ASSERT(ensemble != MFTL_NULL);
184 if (ensemble == MFTL_NULL) return;
185
186 /* Guard on the GLOBAL count: a collective follows. */
187 if (ensemble->n_bodies_global == 0) return;
188
189 const int n_local = ensemble->n_local_objects;
190
191 /* One record per LOCAL body, indexed by slot:
192 *
193 * [0..2] MIN area, quality, squared edge length
194 * [3..5] MAX the same three
195 * [6..10] SUM area, quality, squared edge length, triangle count, half-edge count */
196 static const MFTL_BodyReduceOp ops[MFTL_UPD_N_SLOTS] = {
197 MFTL_BODY_REDUCE_MIN, MFTL_BODY_REDUCE_MIN, MFTL_BODY_REDUCE_MIN, MFTL_BODY_REDUCE_MAX, MFTL_BODY_REDUCE_MAX, MFTL_BODY_REDUCE_MAX,
198 MFTL_BODY_REDUCE_SUM, MFTL_BODY_REDUCE_SUM, MFTL_BODY_REDUCE_SUM, MFTL_BODY_REDUCE_SUM, MFTL_BODY_REDUCE_SUM};
199
200 double *const rows =
201 (double *) MFTLMallocOrDie((size_t) MFTL_MAX(n_local * MFTL_UPD_N_SLOTS, 1) * sizeof(double), "MFTLEnsembleUpdate accumulators");
202
203 /* ── The hot loop, once per LOCAL object ───────────────────────────────── */
204 for (int o = 0; o < n_local; ++o)
205 {
206 MFTL_Object *obj = ensemble->local_objects[o];
207 MFTL_ASSERT(obj != MFTL_NULL);
208
209 const int n_tris = obj->tris.n_local;
210
211 update_accumulators acc = {MFTL_DBL_MAX, -MFTL_DBL_MAX, 0.0, MFTL_DBL_MAX, -MFTL_DBL_MAX, 0.0, MFTL_DBL_MAX, -MFTL_DBL_MAX, 0.0};
212
213 /* No owned triangles here: contribute the identities, not zeros. */
214 if (n_tris > 0) update_kernel(obj, n_tris, &acc);
215
216 double *const r = &rows[(size_t) o * MFTL_UPD_N_SLOTS];
217 r[0] = acc.min_area;
218 r[1] = acc.min_q;
219 r[2] = acc.min_esq;
220 r[3] = acc.max_area;
221 r[4] = acc.max_q;
222 r[5] = acc.max_esq;
223 r[6] = acc.sum_area;
224 r[7] = acc.sum_q;
225 r[8] = acc.sum_esq;
226 r[9] = (double) n_tris;
227 r[10] = (double) (3 * n_tris); /* edges visited */
228 }
229
230 /* One blocking reduction over each body's holders for the whole ensemble
231 * (per-body overlap would multiply the message count). */
232 MFTLEnsembleReduceBodies(ensemble, rows, MFTL_UPD_N_SLOTS, ops, MFTL_BODY_REDUCE_SCOPE_AUTO);
233
234 for (int o = 0; o < n_local; ++o)
235 {
236 const double *const r = &rows[(size_t) o * MFTL_UPD_N_SLOTS];
237 unpack_slot(&ensemble->local_objects[o]->quality, &r[0], &r[3], &r[6]);
238 }
239
240 free(rows);
241}

244{

255/* The per-vertex counterpart needs an EXCHANGE, not a reduction: owned normals
256 * are exact locally (vertex-complete halo); only the ghost half is missing. */
257
258int MFTLEnsembleUpdateVertexNormals(MFTL_Ensemble *ensemble, MFTL_NormalWeighting weighting)
259{
260 MFTL_ASSERT(ensemble != MFTL_NULL);
261 if (ensemble == MFTL_NULL) return MFTL_ERROR;
262 /* Guard on the GLOBAL count: the exchange below is collective. */
263 if (ensemble->n_bodies_global == 0) return MFTL_SUCCESS;
264
265 /* Exact on owned vertices; partial sums on ghosts until the exchange. */
266 for (int o = 0; o < ensemble->n_local_objects; ++o) MFTLComputeVertexNormals(ensemble->local_objects[o], weighting);
267
268 /* One request/reply exchange for the whole ensemble (per object would
269 * multiply the message count for the same bytes). */
270 return MFTLUpdateGhostVertexNormals(ensemble);
271}

273/* ═══════════════════════════════════════════════════════════════════════════
274 * Why this loop does not vectorise (docs/analysis.md §1.6)
275 *
276 * clang's cost model rejects it (-Rpass-missed=loop-vectorize), and forcing it
277 * with `#pragma clang loop vectorize(enable)` fails outright. The nine-way
278 * reduction is NOT the cause: splitting it into independent accumulator sets
279 * changed nothing. One sqrt and one reciprocal per triangle are about a quarter
280 * of the pass and neither is removable (the kernel needs both |n| and n/|n|).
281 * Measured traps: splitting the statistics into a second pass is 25 % slower;
282 * a per-component output layout gains 1.3 %, not worth breaking `tris.normal`.
283 * ═══════════════════════════════════════════════════════════════════════════ */
