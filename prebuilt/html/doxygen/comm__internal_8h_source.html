57 *

213/* ═══════════════════════════════════════════════════════════════════════════
214 * 3. Global id -> local slot table
215 *
216 * "Do I already hold global id g, and in which slot?" is asked once per arriving
217 * entity. Ids are dense per BODY, but a rank holds a scattered subset of a
218 * body's range, so a direct-address array is sized by the global count while
219 * holding the local one (measured: 15-62 MB per rank per step at np = 2-8 for
220 * one 327 680-triangle body, 97 % empty at np = 8). The table therefore has two
221 * modes chosen per use from the ratio of id range to entries: DIRECT
222 * (`tag[gid]`) when the ids are dense enough, SPARSE (open addressing, linear
223 * probing, load factor <= 1/2) otherwise. This is the library's one hash table.
224 *
225 * - EPOCH stamps instead of clearing: a slot is live only if its tag carries the
226 * current epoch, so retiring every entry is `epoch++`.
227 * - No deletion ever, so a probe stops at the first stale slot.
228 * - Capacity is a power of two in both modes, so a mode change never
229 * reallocates; growth discards instead of rehashing (every entry is retired
230 * anyway).
231 * ═══════════════════════════════════════════════════════════════════════════ */
232
233/**
234 * @brief Global-id to local-slot map with a direct and a sparse (hashed) mode.
235 */
236typedef struct
237{
238 /**
239 * `[capacity]` epoch in the high 32 bits, global id in the low 32: one load
240 * and one compare decide "live, and mine".
241 */
242 uint64_t *tag;
243 int *value; ///< [capacity] the local slot. Touched only on a hit.
244 int capacity; ///< Slots allocated. In sparse mode a power of two, so `capacity - 1` is the probe mask.
245 unsigned epoch; ///< Bumped by gt_begin(). Unsigned: wraparound is defined and handled.
246 bool direct; ///< Indexing mode: `tag[gid]` if set, hashed with linear probing if not.
247} mftl_gid_table;
248
249/** @brief The tag a live entry for @p gid carries in epoch @p epoch. */
250static inline uint64_t gt_tag(unsigned epoch, int gid) { return ((uint64_t) epoch << 32) | (uint32_t) gid; }
251
252/** @brief Zeroes a table without allocating. */
253static inline void gt_init(mftl_gid_table *t)
254{
255 t->tag = MFTL_NULL;
256 t->value = MFTL_NULL;
257 t->capacity = 0;
258 t->epoch = 0u;
259 t->direct = MFTL_FALSE;
260}
261
262/** @brief Releases a table. */
263static inline void gt_free(mftl_gid_table *t)
264{
265 free(t->tag);
266 free(t->value);
267 gt_init(t);
268}

270/**
271 * @brief Scatters a global id over a sparse table's index space (multiplicative
272 * avalanche hash).
273 *
274 * The identity index clusters badly: a partition's ids are many dense runs that
275 * alias under the mask, and linear probing then degrades to quadratic insertion
276 * (measured 127 ms against 0.76 ms on a 327 680-triangle body).
277 *
278 * @param gid The global id.
279 * @return A well-mixed 32-bit value.
280 */
281static inline unsigned gt_hash(int gid)
282{
283 unsigned x = (unsigned) gid;
284 x *= 0x9e3779b1u;
285 return x ^ (x >> 16);
286}

288/**
289 * @brief Direct addressing is used when `range <= 4 * n_entries + 64`: then it
290 * costs at most about twice the memory of the sparse table for a
291 * one-load lookup.
292 */
293#define MFTL_GT_DIRECT_RATIO 4
294
295/**
296 * @brief Sizes the table for @p n_entries insertions drawn from `[0, range)`,
297 * picks an indexing mode, and retires everything in it.
298 *
299 * @param t The table.
300 * @param n_entries How many gt_put() calls will follow.
301 * @param range One past the largest global id that will be presented. Only
302 * used to decide the mode and, in direct mode, the size.
303 */
304static inline void gt_begin(mftl_gid_table *t, int n_entries, int range)
305{
306 const bool direct = (range <= MFTL_GT_DIRECT_RATIO * n_entries + 64);
307
308 /* A power of two in both modes: one table serves every body of a call, and a
309 * size rule shared by both modes lets it survive mode changes without
310 * reallocating (sizing direct mode exactly measured 19x the table bytes on a
311 * thousand-droplet ensemble). The epoch bump retires entries of either mode. */
312 const int need = direct ? range : 2 * n_entries + 2;
313 int want = 16;
314 while (want < need) want <<= 1;
315
316 if (want > t->capacity)
317 {
318 /* Discarded, not rehashed: every entry is retired anyway. */
319 free(t->tag);
320 free(t->value);
321 t->tag = (uint64_t *) MFTLCallocOrDie((size_t) want, sizeof(uint64_t), "MFTL gid table tags");
322 t->value = (int *) MFTLMallocOrDie((size_t) want * sizeof(int), "MFTL gid table values");
323 t->capacity = want;
324 t->epoch = 0u;
325 cs_alloc((size_t) want * (sizeof(uint64_t) + sizeof(int)));
326 MFTLCommStats.n_gid_table_bytes += (long) want * (long) (sizeof(uint64_t) + sizeof(int));
327 if (direct) MFTLCommStats.n_gid_table_direct++;
328 else MFTLCommStats.n_gid_table_sparse++;
329 }
330 else if (direct)
331 {
332 MFTLCommStats.n_gid_table_direct++;
333 }
334 else
335 {
336 MFTLCommStats.n_gid_table_sparse++;
337 }
338
339 t->direct = direct;
340
341 /* Epochs start at 1 (a zero tag is always absent). On unsigned wrap-around,
342 * after 2^32 uses, the table is wiped once. */
343 t->epoch++;
344 if (t->epoch == 0u)
345 {
346 memset(t->tag, 0, (size_t) t->capacity * sizeof(uint64_t));
347 t->epoch = 1u;
348 }
349}

351/**
352 * @brief Finds the slot @p gid occupies, or the free slot it would occupy.
353 *
354 * Sparse mode stops at the first slot not of the current epoch (sound because
355 * nothing is ever deleted). The mode branch is constant per use and predicts
356 * perfectly.
357 *
358 * @param t The table.
359 * @param gid The global id.
360 * @param[out] found Set to `true` if the returned slot holds @p gid.
361 * @return The slot index.
362 */
363static inline int gt_probe(const mftl_gid_table *t, int gid, bool *found)
364{
365 const uint64_t want = gt_tag(t->epoch, gid);
366
367 if (t->direct)
368 {
369 /* An id beyond the table is ABSENT, as in sparse mode (the spatial halo
370 * legitimately probes ids of vertices that were not sent). */
371 if (gid >= t->capacity)
372 {
373 *found = false;
374 return 0;
375 }
376 *found = (t->tag[gid] == want);
377 return gid;
378 }
379
380 const int mask = t->capacity - 1;
381 const uint64_t live = (uint64_t) t->epoch << 32;
382 int i = (int) (gt_hash(gid) & (unsigned) mask);
383
384 for (;;)
385 {
386 const uint64_t tag = t->tag[i];
387 if (tag == want)
388 {
389 *found = true;
390 return i;
391 }
392 if ((tag & 0xffffffff00000000ULL) != live)
393 {
394 *found = false;
395 return i; // stale or never written: this is where the id would go
396 }
397 i = (i + 1) & mask;
398 }
399}

401/**
402 * @brief Has @p gid been stored since the last gt_begin()?
403 * @param t The table.
404 * @param gid The global id. Must be non-negative; an id above the declared range
405 * is simply absent.
406 * @return `true` if gt_put() stored a value for @p gid in this epoch.

408static inline bool gt_has(const mftl_gid_table *t, int gid)
409{
410 MFTL_ASSERT(gid >= 0 && "gt_has: negative global id");
411 bool found = false;
412 (void) gt_probe(t, gid, &found);
413 return found;
414}

416/**
417 * @brief Reads the value stored for @p gid.
418 * @param t The table.
419 * @param gid The global id.
420 * @return The stored value. **Undefined unless gt_has() is true** -- asserted in
421 * Debug, unchecked in Release, because every caller has just tested it.
422 */
423static inline int gt_get(const mftl_gid_table *t, int gid)
424{
425 bool found = false;
426 const int i = gt_probe(t, gid, &found);
427 MFTL_ASSERT(found && "gt_get: global id not present in this epoch");
428 (void) found;
429 return t->value[i];
430}

432/**
433 * @brief Stores @p value under @p gid for the current epoch.
434 * @param t The table.
435 * @param gid The global id. Must be non-negative and below the declared range.
436 * @param value The local slot.

438static inline void gt_put(mftl_gid_table *t, int gid, int value)
