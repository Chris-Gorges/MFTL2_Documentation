1/**
2 * @file vertex_geometry.c
3 * @author Christian Gorges
4 * @date 2026-08-14
5 * @ingroup mftl_geometry
6 *
7 * @brief Area- and angle-weighted vertex normals.
8 *
9 * A SCATTER over triangles: each triangle gathers its three current positions
10 * through `tris.verts` and adds its weighted face normal into its three
11 * vertices; one streaming pass then normalises. The alternative GATHER (walk
12 * each vertex's fan) serialises dependent loads (he -> twin -> next) and
13 * measured 3.4x slower; a vertex-to-triangle CSR gather measured a dead heat
14 * with the scatter while costing 16 B/triangle and a rebuild per topology
15 * change (docs/analysis.md §1.5). `verts.normal` stays xyz-interleaved: the
16 * read-modify-write of three contiguous doubles is four memory operations per
17 * vertex against six for split arrays (1.20-1.25x in examples/layout_benchmark.c).
18 */

29/**
30 * @brief Scatters every face's unnormalised normal \f$\vect{e}_1\times\vect{e}_2
31 * = 2A\hat{\vect{n}}\f$ into its three vertices (area weighting).
32 *
33 * @param obj The mesh. `verts.normal` is accumulated into; the caller zeroes it.
34 */
35static void scatter_face_normals(MFTL_Object *obj)
36{
37 const int n_tris = obj->tris.n;
38 double *restrict vn = obj->verts.normal;
39
40 MFTL_FOREACH_TRIANGLE_GEOMETRY(
41 obj, n_tris,
42 /* e1 x e2 = 2 A n_hat: accumulating it raw IS area weighting. */
43 const double nx = e1y * e2z - e1z * e2y;
44 const double ny = e1z * e2x - e1x * e2z;
45 const double nz = e1x * e2y - e1y * e2x;
46
47 vn[3 * v0 + 0] += nx;
48 vn[3 * v0 + 1] += ny;
49 vn[3 * v0 + 2] += nz;
50
51 vn[3 * v1 + 0] += nx;
52 vn[3 * v1 + 1] += ny;
53 vn[3 * v1 + 2] += nz;
54
55 vn[3 * v2 + 0] += nx;
56 vn[3 * v2 + 1] += ny;
57 vn[3 * v2 + 2] += nz;);
58}

60/**
61 * @brief Scatters each face's UNIT normal times its interior angle at each corner.
62 *
63 * All three corners share \f$s = \norm{\vect{e}_1\times\vect{e}_2}\f$, so with
64 * \f$d_{ij} = \vect{e}_i\cdot\vect{e}_j\f$:
65 *
66 * alpha_0 = atan2(s, d12), alpha_1 = atan2(s, d11 - d12), alpha_2 = atan2(s, d22 - d12)
67 *
68 * (\f$\vect{e}_1\cdot(\vect{e}_1-\vect{e}_2)\f$ is the dot product of the two
69 * edges at corner 1, and similarly at corner 2). `atan2` is well conditioned
70 * for every angle, unlike `acos` near 0 and \f$\pi\f$. Zero-area faces have no
71 * unit normal and are skipped.
72 *
73 * @param obj The mesh. `verts.normal` is accumulated into; the caller zeroes it.
74 */
75static void scatter_angle_weighted_normals(MFTL_Object *obj)
76{
77 const int n_tris = obj->tris.n;
78 double *restrict vn = obj->verts.normal;
79
80 // clang-format off
81 MFTL_FOREACH_TRIANGLE_GEOMETRY(
82 obj, n_tris,
83 const double nx = e1y * e2z - e1z * e2y;
84 const double ny = e1z * e2x - e1x * e2z;
85 const double nz = e1x * e2y - e1y * e2x;
86 const double mag = MFTL_SQRT(nx * nx + ny * ny + nz * nz);
87 if (!(mag > 0.0)) continue;
88
89 const double inv = 1.0 / mag;
90 const double ux = nx * inv, uy = ny * inv, uz = nz * inv;
91
92 const double d12 = e1x * e2x + e1y * e2y + e1z * e2z;
93 const double d11 = e1x * e1x + e1y * e1y + e1z * e1z;
94 const double d22 = e2x * e2x + e2y * e2y + e2z * e2z;
95 const double a0 = atan2(mag, d12);
96 const double a1 = atan2(mag, d11 - d12);
97 const double a2 = atan2(mag, d22 - d12);
98
99 vn[3 * v0 + 0] += a0 * ux;
100 vn[3 * v0 + 1] += a0 * uy;
101 vn[3 * v0 + 2] += a0 * uz;
102
103 vn[3 * v1 + 0] += a1 * ux;
104 vn[3 * v1 + 1] += a1 * uy;
105 vn[3 * v1 + 2] += a1 * uz;
106
107 vn[3 * v2 + 0] += a2 * ux;
108 vn[3 * v2 + 1] += a2 * uy;
109 vn[3 * v2 + 2] += a2 * uz;);
110 // clang-format on
111}

113/**
114 * @brief Scales every accumulated vector to unit length, in place.
115 *
116 * A vertex with no incident triangle stays exactly zero -- a detectable "not
117 * computed" sentinel. The guard is a test against ZERO, not a small constant:
118 * area-weighted sums scale as a length squared, so any absolute threshold
119 * would zero the normals of a valid mesh with small enough triangles.
120 *
121 * @param obj The mesh. Each `verts.normal` entry is normalised in place.
122 */
123static void normalise_vertex_normals(MFTL_Object *obj)
124{
125 const int n_verts = obj->verts.n;
126 double *restrict vn = obj->verts.normal;
127
128 for (int v = 0; v < n_verts; ++v)
129 {
130 const double x = vn[3 * v + 0];
131 const double y = vn[3 * v + 1];
132 const double z = vn[3 * v + 2];
133
134 const double len_sq = x * x + y * y + z * z;
135 const double inv = (len_sq > 0.0) ? 1.0 / MFTL_SQRT(len_sq) : 0.0;
136
137 vn[3 * v + 0] = x * inv;
138 vn[3 * v + 1] = y * inv;
139 vn[3 * v + 2] = z * inv;
140 }
141}

143void MFTLComputeVertexNormals(MFTL_Object *obj, MFTL_NormalWeighting weighting)
144{
145 MFTL_ASSERT(obj != MFTL_NULL);
146
147 MFTL_ASSERT((obj->derived & MFTL_DERIVED_TOPOLOGY) != 0u && "MFTLComputeVertexNormals: tris.verts is stale -- MFTLObjectBuildDerived first");
148
149 if (obj->verts.n == 0) return;
150
151 /* Clear every vertex, ghosts included: a ghost keeping last step's value
152 * would look fresh. */
153 memset(obj->verts.normal, 0, 3 * (size_t) obj->verts.n * sizeof(double));
154
155 if (weighting == MFTL_NORMAL_WEIGHT_ANGLE)
156 scatter_angle_weighted_normals(obj);
157 else
158 scatter_face_normals(obj);
159 normalise_vertex_normals(obj);
160}
