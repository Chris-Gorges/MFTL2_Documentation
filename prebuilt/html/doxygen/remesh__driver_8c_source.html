1/**
2 * @file remesh_driver.c
3 * @author Christian Gorges
4 * @date 2026-08-15
5 * @ingroup mftl_remeshing
6 *
7 * @brief The remeshing pass: sweep order, convergence, and what is refreshed
8 * when.
9 *
10 * ## The order of the sweeps
11 *
12 * DETECT -> SPLIT -> COLLAPSE -> COMPACT -> FLIP -> DETECT -> SMOOTH
13 *
14 * - SPLIT before COLLAPSE: splitting cannot fail (no link condition, no fold
15 * test), so the collapse sweep then sees edges that are genuinely short rather
16 * than short relative to over-long neighbours.
17 * - COMPACT right after COLLAPSE: flip and smoothing never see a dead entity, so
18 * the smoother can use MFTLComputeVertexNormals() over the whole pool without
19 * accumulating dead triangles into live normals.
20 * - FLIP before SMOOTH: the valence criterion reads the connectivity the
21 * previous sweeps produced; smoothing does not change it.
22 * - DETECT twice per iteration: splits and collapses change which edges exist,
23 * and a vertex the smoother wrongly believes free (e.g. the midpoint of a
24 * split rim edge) slides off its curve.
25 *
26 * Each sweep bounds its loop by the half-edge count taken BEFORE it: the halves
27 * of a split edge are not re-examined in the same sweep (a long edge would
28 * otherwise burst into many tiny ones), but in the next iteration, after
29 * smoothing.
30 */

51/**
52 * @brief Splits every approved edge, once per undirected edge.
53 *
54 * The parallel freeze gate is tested first (two flag tests, against an
55 * approval's fan walk). Refusals are NOT counted as blocked work: a rank sees
56 * ghosts with incomplete one-rings, approves a collapse the owner would reject,
57 * and would report phantom blocked operations (3 on a converged np = 4 sphere).
58 * Convergence is instead certified by `count_hidden_edges` in remesh_parallel.c,
59 * which needs no fan walk.
60 *
61 * @param obj The mesh.
62 * @param params Supplies `max_edge_length_sq`.
63 * @return Splits performed.
64 */
65
66static int sweep_split(MFTL_Object *obj, const MFTL_RemeshParameters *params)
67{
68 const int n_he = 3 * obj->tris.n; // snapshot: see the file header
69 int count = 0;
70
71 for (int h = 0; h < n_he; ++h)
72 {
73 if (freeze_blocks_pair(obj, h)) continue;
74 if (!MFTLSplitApproved(obj, h, params)) continue;
75 if (MFTLSplitEdge(obj, h) != MFTL_ERROR) count++;
76 }
77 return count;
78}

80/**
81 * @brief Collapses every approved edge, once per undirected edge.
82 * @param obj The mesh.
83 * @param params Supplies `min_edge_length_sq` and the safety thresholds.
84 * @param[out] stats May be #MFTL_NULL. The rejection counters are accumulated here.
85 * @return Collapses performed.
86 */
87static int sweep_collapse(MFTL_Object *obj, const MFTL_RemeshParameters *params, MFTL_RemeshStats *stats)
88{
89 const int n_he = 3 * obj->tris.n;
90 int count = 0;
91
92 for (int h = 0; h < n_he; ++h)
93 {
94 if (freeze_blocks_collapse(obj, h)) continue;
95
96 int reason = MFTL_REMESH_REJECT_NONE;
97
98 if (MFTLCollapseApproved(obj, h, params, &reason))
99 {
100 if (MFTLCollapseEdge(obj, h) == MFTL_SUCCESS) count++;
101 continue;
102 }
103
104 if (!stats) continue;
105
106 switch (reason)
107 {
108 case MFTL_REMESH_REJECT_LINK: stats->n_collapse_rejected_link++; break;
109 case MFTL_REMESH_REJECT_FOLD: stats->n_collapse_rejected_fold++; break;
110 case MFTL_REMESH_REJECT_FEATURE: stats->n_collapse_rejected_feature++; break;
111 default: break;
112 }
113 }
114 return count;
115}

117/**
118 * @brief Flips every edge whose flip strictly lowers the valence deviation.
119 * @param obj The mesh.
120 * @param params Supplies `max_normal_deviation_cos`.
121 * @param[out] stats May be #MFTL_NULL.
122 * @return Flips performed.
123 */
124static int sweep_flip(MFTL_Object *obj, const MFTL_RemeshParameters *params, MFTL_RemeshStats *stats)
125{
126 const int n_he = 3 * obj->tris.n;
127 int count = 0;
128
129 for (int h = 0; h < n_he; ++h)
130 {
131 if (freeze_blocks_pair(obj, h)) continue;
132
133 int reason = MFTL_REMESH_REJECT_NONE;
134
135 if (MFTLFlipApproved(obj, h, params, &reason))
136 {
137 if (MFTLFlipEdge(obj, h) == MFTL_SUCCESS) count++;
138 }
139 else if (stats && reason == MFTL_REMESH_REJECT_FOLD)
140 {
141 stats->n_flip_rejected_fold++;
142 }
143 }
144 return count;
145}

147/**
148 * @brief Clears the per-sweep flags so the next iteration starts from scratch.
149 *
150 * @param obj The mesh. #MFTL_FLAG_VISITED is cleared on every entity, so the
151 * next sweep starts from a clean slate.
152 */
153static void clear_visited(MFTL_Object *obj)
154{
155 const int n_he = 3 * obj->tris.n;
156 for (int h = 0; h < n_he; ++h) obj->tris.he[h].flags &= ~MFTL_FLAG_VISITED;
157 for (int v = 0; v < obj->verts.n; ++v) obj->verts.rec[v].flags &= ~MFTL_FLAG_VISITED;
158}

164double MFTLMeanEdgeLength(const MFTL_Object *obj)
165{
166 MFTL_ASSERT(obj != MFTL_NULL);
167
168 const int n_he = 3 * obj->tris.n;
169 if (n_he == 0) return 0.0;
170
171 double sum = 0.0;
172 int count = 0;
173
174 /* Over UNDIRECTED edges. Averaging half-edges gives the same answer on a closed
175 * mesh and a wrong one on an open surface, where a boundary edge has one half
176 * and an interior edge has two -- so the boundary would be weighted half. */
177 for (int h = 0; h < n_he; ++h)
178 {
179 const int tw = obj->tris.he[h].twin_idx;
180 if (tw != MFTL_ERROR && h > tw) continue;
181
182 sum += MFTLGetEdgeLength(obj, h);
183 count++;
184 }
185
186 return (count > 0) ? sum / (double) count : 0.0;
187}

189double MFTLEnsembleMeanEdgeLength(MFTL_Ensemble *ens)
190{
191 MFTL_ASSERT(ens != MFTL_NULL);
192
193 double acc[2] = {0.0, 0.0};
194
195 for (int o = 0; o < ens->n_local_objects; ++o)
196 {
197 const MFTL_Object *const obj = ens->local_objects[o];
198 const int n_he = 3 * obj->tris.n;
199
200 for (int h = 0; h < n_he; ++h)
201 {
202 const int t = MFTLHeTriIdx(h);
203 if (t >= obj->tris.n_local) continue; // a ghost's owner counts it
204
205 const int tw = obj->tris.he[h].twin_idx;
206 if (tw != MFTL_ERROR)
207 {
208 const int to = MFTLHeTriIdx(tw);
209
210 /* Both mine: the smaller half-edge counts, as in the serial form. Split
211 * across ranks: the smaller triangle global id counts, which is a rule
212 * both sides evaluate to the same answer without asking each other. */
213 if (to < obj->tris.n_local)
214 {
215 if (h > tw) continue;
216 }
217 else if (obj->tris.rec[t].global_id > obj->tris.rec[to].global_id)
218 {
219 continue;
220 }
221 }
222
223 acc[0] += MFTLGetEdgeLength(obj, h);
224 acc[1] += 1.0;
225 }
226 }
227
228 MFTLRemeshAgreeDouble(&ens->domain, acc, 2);
229 return (acc[1] > 0.0) ? acc[0] / acc[1] : 0.0;
230}

232int MFTLRemeshIterate(MFTL_Object *obj, const MFTL_RemeshParameters *params, MFTL_RemeshStats *st, int iterations)
233{
234 int total_ops = 0;
235
236 if (obj->tris.n == 0) return 0;
237
238 /* Every operator reads and writes the half-edge records; the derived arrays are
239 * rebuilt by the compaction that follows the collapse sweep. Building them here
240 * makes the object valid for the very first MFTLDetectFeatures. */
241 if (!MFTLObjectHasTopology(obj)) MFTLObjectBuildDerived(obj);
242
243 for (int iter = 0; iter < iterations; ++iter)
244 {
245 st->iterations = iter + 1;
246
247 MFTLDetectFeatures(obj, params->feature_angle_threshold, params->feature_corner_angle, st);
248
249 const int n_split = sweep_split(obj, params);
250 const int n_collapse = sweep_collapse(obj, params, st);
251
252 /* Unconditional, not "if anything died": the split sweep left the derived
253 * arrays invalid too, and MFTLRemeshCompactObject rebuilds them on the way
254 * out whether or not it moved anything. */
255 MFTLRemeshCompactObject(obj, st);
256
257 const int n_flip = sweep_flip(obj, params, st);
258 if (n_flip > 0) MFTLObjectBuildDerived(obj);
259
260 clear_visited(obj);
261
262 /* Detected AGAIN, against the mesh that is about to be smoothed rather than
263 * the one the sweeps started from. The three sweeps in between create
264 * vertices that the first classification never saw: a split of a rim edge
265 * produces a midpoint that IS on the rim, and a vertex the smoother believes
266 * to be free gets the 2D tangent-plane treatment and slides straight off it.
267 * The surface then grows a serrated notch, which is invisible to every
268 * structural check -- the result is a perfectly valid manifold with a
269 * perfectly valid boundary, just not the boundary it started with.
270 *
271 * The cost is one geometry pass and two linear passes per iteration. Handing
272 * the classification forward from the first call instead would mean every
273 * operator maintaining it incrementally, which is the arrangement that leaves
274 * a smoother three sweeps out of date. */
275 MFTLDetectFeatures(obj, params->feature_angle_threshold, params->feature_corner_angle, st);
276
277 MFTLSmoothTangential(obj, params, params->smooth_iterations);
278
279 st->n_split += n_split;
280 st->n_collapse += n_collapse;
281 st->n_flip += n_flip;
282 total_ops += n_split + n_collapse + n_flip;
283
284 if (n_split + n_collapse + n_flip <= params->min_convergence_ops) break;
285 }
286
287 return total_ops;
288}

290int MFTLRemeshObject(MFTL_Object *obj, const MFTL_RemeshParameters *params, MFTL_RemeshStats *stats)
291{
292 MFTL_ASSERT(obj != MFTL_NULL && params != MFTL_NULL);
293 MFTL_ASSERT(obj->tris.n_local == obj->tris.n && obj->verts.n_local == obj->verts.n &&
294 "MFTLRemeshObject: object holds ghosts. Collapsing an edge whose far endpoint is owned elsewhere "
295 "needs agreement, so a 'serial' remesh of a partitioned object silently diverges between ranks. "
296 "The distributed entry point is MFTLRemeshEnsemble.");
297
298 /* The stats describe THIS call, so they are zeroed here rather than left to the
299 * caller. Accumulating into whatever the caller's struct happened to hold would
300 * make a second pass over the same mesh report the sum of both. */
301 MFTL_RemeshStats local;
302 MFTL_RemeshStats *const st = stats ? stats : &local;
303 memset(st, 0, sizeof(*st));
304
305 if (obj->tris.n == 0) return MFTL_SUCCESS;
306
307 MFTLRemeshIterate(obj, params, st, params->iterations);
308
309 if (params->renumber_global_ids) MFTLRemeshRenumberGlobalIds(obj);
310
311 /* Postcondition: the caller gets an object it can stream immediately. The
312 * derived topology is already current -- compaction and the flip sweep rebuild
313 * it, and neither smoothing nor the id renumbering changes connectivity -- so
314 * only `tris.area/normal/center` are left to fill; the operators leave them
315 * stale by design because none of them read them. */
316 if (!MFTLObjectHasTopology(obj)) MFTLObjectBuildDerived(obj);
317 MFTLComputeTriangleGeometry(obj, obj->tris.n);
318
319 return MFTL_SUCCESS;
320}

322int MFTLRemeshEnsemble(MFTL_Ensemble *ens, const MFTL_RemeshParameters *params, MFTL_RemeshStats *stats)
323{
324 MFTL_ASSERT(ens != MFTL_NULL && params != MFTL_NULL);
325
326 /* One rank stays on the path below, and not merely as an optimisation: the
327 * phase schedule migrates, renumbers ids collectively and rebuilds the halo,
328 * all of which are identities at nproc == 1 but none of which are bit-level
329 * identities -- a redistribution permutes the pools. Keeping the serial answer
330 * reproducible is worth the branch. */
331 if (ens->domain.nproc > 1)
332 {
333 /* The parameters are agreed ONCE per call, at the first collective of
334 * whichever path runs (MFTLRemeshAgreeParams()). Every path's first
335 * collective is that check, so two ranks that read different strategies
336 * still meet in it and both refuse. */
337 /* AUTO resolves COLLECTIVELY -- MFTLRemeshChooseStrategy measures the bodies
338 * with two Allreduces and every rank therefore reads the same ratio. That is
339 * the load-bearing property of this branch, not the choice itself: two ranks
340 * entering different drivers is an immediate deadlock, and nothing further
341 * down would report it. */
342 MFTL_RemeshParallelStrategy strategy = params->parallel_strategy;
343
344 if (strategy != MFTL_REMESH_PARALLEL_AUTO)
345 {
346 if (strategy == MFTL_REMESH_PARALLEL_DISPERSED) return MFTLRemeshDispersedPass(ens, params, stats, "MFTLRemeshEnsemble");
347 return MFTLRemeshSpanningPass(ens, params, stats, "MFTLRemeshEnsemble");
348 }
349
350 /* AUTO measures the bodies to decide, and the dispersed driver needs exactly
351 * the same measurement to route them -- so it is taken ONCE here and handed
352 * on. Doing it the obvious way (choose, then call the public driver) makes
353 * the pass pay four per-body reductions where two do, and the first caller
354 * then discards the centroids the second one recomputes.
355 *
356 * GLOBAL body count for the guard: everything below is collective. The
357 * measurement itself is indexed by LOCAL SLOT -- see MFTL_RemeshMeasurement. */
358 if (ens->n_bodies_global < 1) return MFTLRemeshSpanningPass(ens, params, stats, MFTL_NULL);
359
360 MFTL_RemeshMeasurement measure;
361 MFTLRemeshMeasurementInit(&measure, ens);
362
363 MFTLEnsembleComputeTriangleGeometry(ens);
364 if (MFTLRemeshAgreeParams(&ens->domain, params, "MFTLRemeshEnsemble") != MFTL_SUCCESS)
365 {
366 MFTLRemeshMeasurementFree(&measure);
367 if (stats) memset(stats, 0, sizeof(*stats));
368 return MFTL_ERROR;
369 }
370 strategy = MFTLRemeshChooseStrategyMeasured(ens, params, &measure);
371
372 const int rc = (strategy == MFTL_REMESH_PARALLEL_DISPERSED) ? MFTLRemeshEnsembleDispersedMeasured(ens, params, stats, &measure)
373 : MFTLRemeshSpanningPass(ens, params, stats, MFTL_NULL);
374 MFTLRemeshMeasurementFree(&measure);
375 return rc;
376 }
377
378 /* One rank: the band test alone, no MPI. */
379 if (ens->n_bodies_global > 0 && MFTLRemeshAgreeParams(&ens->domain, params, "MFTLRemeshEnsemble") != MFTL_SUCCESS)
380 {
381 if (stats) memset(stats, 0, sizeof(*stats));
382 return MFTL_ERROR;
383 }
384
385 if (stats)
386 {
387 memset(stats, 0, sizeof(*stats));
388
389 /* A serial pass IS converged, by the only definition the field has: the mesh
390 * is as converged as the serial remesher would have made it, because it is
391 * what the serial remesher made. Leaving the flag false because "no phase
392 * driver ran" would make every nproc == 1 run report a convergence failure,
393 * which is what a solver logs and acts on. */
394 stats->converged = MFTL_TRUE;
395 stats->n_phases = 1;
396 stats->strategy = MFTL_REMESH_PARALLEL_SPANNING;
397 stats->n_coarse_groups = 1;
398 stats->n_distinct_placements = 1; // one rank, one placement, and it is the canonical one
399 }
400
401 int status = MFTL_SUCCESS;
402
403 for (int i = 0; i < ens->n_local_objects; ++i)
404 {
405 MFTL_RemeshStats one;
406 memset(&one, 0, sizeof(one));
407
408 if (MFTLRemeshObject(ens->local_objects[i], params, &one) != MFTL_SUCCESS) status = MFTL_ERROR;
409
410 if (!stats) continue;
411
412 remesh_stats_accumulate(stats, &one);
413 }
414
415 MFTLEnsembleUpdateGlobalCounts(ens);
416 return status;
417}
