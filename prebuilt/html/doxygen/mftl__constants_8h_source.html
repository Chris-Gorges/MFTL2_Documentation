105/**
106 * @name Tolerances
107 *
108 * The library itself uses only #MFTL_VERYVERYSMALL, and only as a
109 * **dimensionless** threshold -- always multiplied by the product of the
110 * quantities it compares, e.g. \f$|\vect{a}\times\vect{b}|^2 >
111 * \varepsilon\,|\vect{a}|^2|\vect{b}|^2\f$, a test on \f$\sin^2\theta\f$
112 * alone. No library decision compares a dimensional quantity against an absolute
113 * constant, so results are invariant under a change of length unit
114 * (`test_remeshing_is_independent_of_the_length_unit` checks this bit for bit).
115 * #MFTL_EPSILON and #MFTL_VERYSMALL are absolute and are provided for
116 * application code and tests only.
117 *
118 * @warning None of them may be used to clamp a vector magnitude before
119 * normalisation: that floors the magnitude and returns vectors that are
120 * shorter than unit. See MFTLNormalizeVector().
121 * @{
122 */
123/** @brief Absolute comparison tolerance, \f$10^{-9}\f$. Tests and applications only. */
124#define MFTL_EPSILON 1e-9
125/** @brief Absolute small-quantity tolerance, \f$10^{-12}\f$. Tests and applications only. */
126#define MFTL_VERYSMALL 1e-12
127/** @brief Dimensionless degeneracy threshold, \f$10^{-15}\f$ (about \f$5\,\varepsilon_\mathrm{mach}\f$), applied to squared sines. */
128#define MFTL_VERYVERYSMALL 1e-15

154/**
155 * @brief Upper bound on the valence handled by the one-ring walkers.
156 *
157 * Fan walks write into caller-supplied stack buffers of this size and the
158 * rotation loops are bounded by it, so a corrupt (non-manifold) fan terminates.
159 * Overflow asserts in Debug and truncates in Release; the remesher's collapse and
160 * flip approvals refuse any operation that would exceed it (the regular valence
161 * of a triangulated surface is 6).
162 */
163#define MFTL_MAX_VALENCE 32
164
165/**
166 * @brief Alignment, in bytes, of every sub-array of a pool block.
167 *
168 * MFTLLayoutAdd() rounds each sub-array start up to this boundary, so two
169 * arrays that different kernels write (e.g. `tris.area` and `tris.normal`) never
170 * share a cache line, and every sub-array starts on a vector-aligned address.
171 * 64 B is the line size of current x86-64 and AArch64 cores; a 128-byte-line
172 * target would only weaken the separation, not break anything.
173 */
174#define MFTL_CACHE_LINE 64u
175
176/**
177 * @brief Smallest capacity a geometric growth step produces (MFTLGrowCapacity()).
178 *
179 * A pool grown geometrically from empty follows the ladder \f$128 \cdot 2^k\f$,
180 * so incremental growth costs \f$O(\log n)\f$ reallocations.
181 */
182#define MFTL_INIT_CAPACITY 128
183
184/**
185 * @brief Default waste factor above which MFTLObjectShrinkToFit() reallocates.
186 *
187 * A pool is shrunk when `capacity > slack * n`, to the smallest ladder capacity
188 * \f$128\cdot2^k \ge n\f$. Both growth and shrink leave \f$C \in [n, 2n)\f$
189 * (for \f$n \ge 128\f$), so a gate of 2 would fire on ANY decrease and a count
190 * dithering around a power of two would reallocate every step. A gate of 4 is the
191 * product of the two factor-2 headrooms: after a resize at \f$n_0\f$,
192 * \f$C < 2n_0\f$, so \f$C > 4n_1\f$ requires \f$n_1 < n_0/2\f$, and the number
193 * of reallocations over a run is \f$O(\log(n_{max}/n_{min}))\f$.
194 *
195 * Pinned by `test_shrink_to_fit_does_not_thrash_against_the_grow_ladder`.
196 */
197#define MFTL_SHRINK_SLACK_DEFAULT 4.0
