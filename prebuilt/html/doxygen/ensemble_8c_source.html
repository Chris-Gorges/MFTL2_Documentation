22/**
23 * @brief Builds the MPI topology of a domain, once, so no hot path calls an MPI
24 * topology function again.
25 *
26 * - `comm_cart`: the Cartesian communicator (no reordering), which defines the
27 * grid-coordinate-to-rank mapping;
28 * - `rank_map`: that mapping as a flat array (owner lookup is per entity);
29 * - `comm_graph`: a symmetric distributed graph over the distinct ranks of the
30 * 26 surrounding cells, for neighbourhood collectives whose message count is
31 * bounded by the degree rather than the process count. On thin periodic grids
32 * several offsets alias one rank (listed once, with its first offset in
33 * `neighbor_offset`), and the graph may contain this rank itself.
34 *
35 * On failure the communicators created so far are reset to `MPI_COMM_NULL` so
36 * MFTLEnsembleDestroy() can clean up.
37 *
38 * @param domain The domain; `comm_world` must be set.
39 * @return #MFTL_SUCCESS, or #MFTL_ERROR if a communicator could not be created.
40 */
41static int domain_init_comm(MFTL_Domain *domain)
42{
43 if (!domain) return MFTL_ERROR;
44
45 // --- 1. Cartesian communicator ---
46 int periods[3] = {(int) domain->is_periodic[0], (int) domain->is_periodic[1], (int) domain->is_periodic[2]};
47 // Reordering is disabled so ranks keep the same order as the input communicator.
48 const int reorder = 0;
49
50 if (MPI_Cart_create(domain->comm_world, 3, domain->cuts, periods, reorder, &domain->comm_cart) != MPI_SUCCESS)
51 {
52 fprintf(stderr, "MFTL Error (Rank %d): Failed to create Cartesian communicator.\n", domain->rank);
53 domain->comm_cart = MPI_COMM_NULL;
54 return MFTL_ERROR;
55 }
56
57 // --- 2. Rank map and grid factors ---
58 // grid_factors turns the per-axis division in MFTLGetOwnerProc into a multiply.
59 for (int d = 0; d < 3; ++d)
60 {
61 domain->grid_factors[d] = (domain->domain_size[d] > 0.0) ? (double) domain->cuts[d] / domain->domain_size[d] : 0.0;
62 }
63
64 domain->rank_map = (int *) MFTLMallocOrDie((size_t) domain->nproc * sizeof(int), "MFTL domain rank_map");
65
66 for (int k = 0; k < domain->cuts[2]; ++k)
67 {
68 for (int j = 0; j < domain->cuts[1]; ++j)
69 {
70 for (int i = 0; i < domain->cuts[0]; ++i)
71 {
72 int coords[3] = {i, j, k};
73 int rank_out;
74 MPI_Cart_rank(domain->comm_cart, coords, &rank_out);
75 domain->rank_map[i + j * domain->cuts[0] + k * domain->cuts[0] * domain->cuts[1]] = rank_out;
76 }
77 }
78 }
79
80 // --- 3. Collect the (up to) 26 surrounding neighbours ---
81 int my_coords[3];
82 MPI_Cart_coords(domain->comm_cart, domain->rank, 3, my_coords);
83
84 int unique_neighbors[26];
85 signed char unique_offsets[26][3];
86 int num_unique = 0;
87
88 for (int dz = -1; dz <= 1; ++dz)
89 {
90 for (int dy = -1; dy <= 1; ++dy)
91 {
92 for (int dx = -1; dx <= 1; ++dx)
93 {
94 if (dx == 0 && dy == 0 && dz == 0) continue; // skip myself
95
96 int nc[3] = {my_coords[0] + dx, my_coords[1] + dy, my_coords[2] + dz};
97 bool valid = MFTL_TRUE;
98
99 for (int i = 0; i < 3; ++i)
100 {
101 if (periods[i])
102 {
103 // Periodic wrap-around; the + cuts handles negative results correctly.
104 nc[i] = (nc[i] + domain->cuts[i]) % domain->cuts[i];
105 }
106 else if (nc[i] < 0 || nc[i] >= domain->cuts[i])
107 {
108 valid = MFTL_FALSE;
109 break;
110 }
111 }
112
113 if (!valid) continue;
114
115 int neighbor_rank;
116 MPI_Cart_rank(domain->comm_cart, nc, &neighbor_rank);
117
118 /* De-duplicate: on thin periodic grids (e.g. cuts = {2,1,1}) several
119 * offsets map to one rank, which the graph must list once. */
120 bool present = MFTL_FALSE;
121 for (int n = 0; n < num_unique; ++n)
122 {
123 if (unique_neighbors[n] == neighbor_rank)
124 {
125 present = MFTL_TRUE;
126 break;
127 }
128 }
129 if (!present)
130 {
131 /* Keep the (first) OFFSET with the rank: the per-body reduction tests
132 * holder sets against it, and it needs no seam correction. */
133 unique_offsets[num_unique][0] = (signed char) dx;
134 unique_offsets[num_unique][1] = (signed char) dy;
135 unique_offsets[num_unique][2] = (signed char) dz;
136 unique_neighbors[num_unique++] = neighbor_rank;
137 }
138 }
139 }
140 }
141
142 // --- 4. Distributed graph communicator ---
143 // For an adjacent graph the sources of incoming messages are the same as the
144 // destinations of outgoing messages.
145 if (MPI_Dist_graph_create_adjacent(domain->comm_world, num_unique, unique_neighbors, MPI_UNWEIGHTED, num_unique, unique_neighbors, MPI_UNWEIGHTED,
146 MPI_INFO_NULL, reorder, &domain->comm_graph) != MPI_SUCCESS)
147 {
148 fprintf(stderr, "MFTL Error (Rank %d): Failed to create distributed graph communicator.\n", domain->rank);
149 domain->comm_graph = MPI_COMM_NULL;
150 return MFTL_ERROR;
151 }
152
153 // --- 5. Read the neighbour list back from MPI ---
154 int indegree = 0, outdegree = 0, weighted = 0;
155 MPI_Dist_graph_neighbors_count(domain->comm_graph, &indegree, &outdegree, &weighted);
156 MFTL_ASSERT(indegree == outdegree && "MFTL assumes a symmetric neighbour graph");
157
158 domain->num_neighbors = indegree;
159 domain->neighbors = (int *) MFTLMallocOrDie((size_t) (indegree > 0 ? indegree : 1) * sizeof(int), "MFTL domain neighbours");
160 int *dest = (int *) MFTLMallocOrDie((size_t) (outdegree > 0 ? outdegree : 1) * sizeof(int), "MFTL domain neighbours (dest)");
161
162 MPI_Dist_graph_neighbors(domain->comm_graph, indegree, domain->neighbors, MPI_UNWEIGHTED, outdegree, dest, MPI_UNWEIGHTED);
163 free(dest);
164
165 /* MPI may report the neighbours in its own order: match offsets by RANK. */
166 for (int n = 0; n < indegree; ++n)
167 {
168 domain->neighbor_offset[n][0] = 0;
169 domain->neighbor_offset[n][1] = 0;
170 domain->neighbor_offset[n][2] = 0;
171 for (int u = 0; u < num_unique; ++u)
172 {
173 if (unique_neighbors[u] != domain->neighbors[n]) continue;
174 domain->neighbor_offset[n][0] = unique_offsets[u][0];
175 domain->neighbor_offset[n][1] = unique_offsets[u][1];
176 domain->neighbor_offset[n][2] = unique_offsets[u][2];
177 break;
178 }
179 }
180
181 return MFTL_SUCCESS;
182}

184MFTL_Ensemble *MFTLEnsembleCreate(MPI_Comm comm, int cuts[3], double global_min[3], double global_max[3], bool is_periodic[3], int initial_object_capacity)
185{
186 if (!cuts || !global_min || !global_max) return MFTL_NULL;
187
188 MFTL_Ensemble *ens = (MFTL_Ensemble *) calloc(1, sizeof(MFTL_Ensemble));
189 if (!ens) return MFTL_NULL;
190
191 MFTL_Domain *dom = &ens->domain;
192
193 /* MPI_COMM_NULL is not necessarily 0 (Open MPI: a pointer to a static object),
194 * so the handles are set explicitly for the cleanup path. */
195 dom->comm_world = MPI_COMM_NULL;
196 dom->comm_cart = MPI_COMM_NULL;
197 dom->comm_graph = MPI_COMM_NULL;
198
199 // --- 1. Duplicate the world communicator so the library owns its own ---
200 if (MPI_Comm_dup(comm, &dom->comm_world) != MPI_SUCCESS)
201 {
202 fprintf(stderr, "MFTL Error: Failed to duplicate the MPI communicator.\n");
203 dom->comm_world = MPI_COMM_NULL;
204 MFTLEnsembleDestroy(&ens);
205 return MFTL_NULL;
206 }
207 MPI_Comm_size(dom->comm_world, &dom->nproc);
208 MPI_Comm_rank(dom->comm_world, &dom->rank);
209
210 // --- 2. Validate the process grid and the bounds ---
211 /* The product in 64 bits, so an int overflow cannot fake a match. */
212 if (cuts[0] < 1 || cuts[1] < 1 || cuts[2] < 1 || (long long) cuts[0] * cuts[1] * cuts[2] != (long long) dom->nproc)
213 {
214 if (dom->rank == 0)
215 {
216 fprintf(stderr, "MFTL Error: Cuts {%d, %d, %d} do not match MPI size %d.\n", cuts[0], cuts[1], cuts[2], dom->nproc);
217 }
218 MFTLEnsembleDestroy(&ens);
219 return MFTL_NULL;
220 }
221
222 /* A zero-length axis is legal; an inverted or non-finite one is not.
223 * `!(a >= b)` also rejects NaN. */
224 for (int i = 0; i < 3; ++i)
225 {
226 if (!(global_max[i] >= global_min[i]) || !isfinite(global_min[i]) || !isfinite(global_max[i]))
227 {
228 if (dom->rank == 0) fprintf(stderr, "MFTL Error: invalid domain bounds on axis %d: [%g, %g].\n", i, global_min[i], global_max[i]);
229 MFTLEnsembleDestroy(&ens);
230 return MFTL_NULL;
231 }
232 }
233
234 // --- 3. Copy bounds and flags ---
235 for (int i = 0; i < 3; ++i)
236 {
237 dom->cuts[i] = cuts[i];
238 dom->global_min[i] = global_min[i];
239 dom->global_max[i] = global_max[i];
240 dom->domain_size[i] = global_max[i] - global_min[i];
241 dom->inv_domain_size[i] = (dom->domain_size[i] > 0.0) ? 1.0 / dom->domain_size[i] : 0.0;
242 dom->is_periodic[i] = is_periodic ? is_periodic[i] : MFTL_FALSE;
243 }
244
245 // --- 4. Build the MPI topology ---
246 if (domain_init_comm(dom) != MFTL_SUCCESS)
247 {
248 MFTLEnsembleDestroy(&ens);
249 return MFTL_NULL;
250 }
251
252 // --- 5. Set up the object list ---
253 ens->capacity_objects = (initial_object_capacity > 0) ? initial_object_capacity : 1;
254 ens->local_objects = (MFTL_Object **) MFTLCallocOrDie((size_t) ens->capacity_objects, sizeof(MFTL_Object *), "MFTL ensemble object list");
255 ens->n_local_objects = 0;
256
257 return ens;
258}

260void MFTLEnsembleDestroy(MFTL_Ensemble **p_ens)
261{
262 if (!p_ens || !(*p_ens)) return;
263 MFTL_Ensemble *ens = *p_ens;
264
265 // 1. Destroy every object the ensemble owns
266 for (int i = 0; i < ens->n_local_objects; ++i) MFTLObjectDestroy(&ens->local_objects[i]);
267 free(ens->local_objects);
268
269 // 1a. And the communication module's scratch, if a redistribution built one
270 MFTLCommWorkspaceDestroy(&ens->comm_workspace);
271
272 // 2. Release the domain resources
273 MFTL_Domain *dom = &ens->domain;
274 if (dom->comm_graph != MPI_COMM_NULL) MPI_Comm_free(&dom->comm_graph);
275 if (dom->comm_cart != MPI_COMM_NULL) MPI_Comm_free(&dom->comm_cart);
276 if (dom->comm_world != MPI_COMM_NULL) MPI_Comm_free(&dom->comm_world);
277 free(dom->neighbors);
278 free(dom->rank_map);
279
280 // 3. Free the ensemble itself
281 free(ens);
282 *p_ens = MFTL_NULL;
283}

285int MFTLEnsembleAddObject(MFTL_Ensemble *ensemble, MFTL_Object *object)
286{
287 if (!ensemble || !object) return MFTL_ERROR;
288
289 object->domain = &ensemble->domain;
290
291 /* The ensemble hands out the body id, so ids are DENSE in [0, n_bodies_global)
292 * and identical on every rank as long as every rank adds the same bodies in
293 * the same order (checked by the redistribution's Debug checksum). A dense id
294 * is an array index, which is why no per-body lookup needs a hash. */
295 object->global_id = ensemble->n_bodies_global++;
296
297 /* Ids ascend in add order, so appending keeps the directory sorted. */
298 return MFTLEnsembleInsertLocalObject(ensemble, object);
299}

307int MFTLEnsembleFindLocalSlot(const MFTL_Ensemble *ens, int body_gid)
308{
309 MFTL_ASSERT(ens != MFTL_NULL);
310 if (ens == MFTL_NULL) return MFTL_ERROR;
311
312 int lo = 0, hi = ens->n_local_objects - 1;
313 while (lo <= hi)
314 {
315 const int mid = lo + ((hi - lo) >> 1);
316 const int g = ens->local_objects[mid]->global_id;
317 if (g == body_gid) return mid;
318 if (g < body_gid)
319 lo = mid + 1;
320 else
321 hi = mid - 1;
322 }
323 return MFTL_ERROR;
324}

332int MFTLEnsembleInsertLocalObject(MFTL_Ensemble *ens, MFTL_Object *object)
333{
334 if (ens == MFTL_NULL || object == MFTL_NULL) return MFTL_ERROR;
335
336 /* Geometric growth: bodies are added one at a time. */
337 MFTL_ENSURE_CAPACITY(ens->local_objects, &ens->n_local_objects, &ens->capacity_objects, 1, MFTL_Object *);
338
339 /* Scan from the END: both callers insert in ascending id, so the common case
340 * moves nothing. */
341 int i = ens->n_local_objects;
342 while (i > 0 && ens->local_objects[i - 1]->global_id > object->global_id) --i;
343
344 MFTL_ASSERT((i == 0 || ens->local_objects[i - 1]->global_id != object->global_id) && "MFTLEnsembleInsertLocalObject: that body is already here");
345
346 if (i < ens->n_local_objects) memmove(&ens->local_objects[i + 1], &ens->local_objects[i], (size_t) (ens->n_local_objects - i) * sizeof(MFTL_Object *));
347
348 ens->local_objects[i] = object;
349 ens->n_local_objects++;
350 return MFTL_SUCCESS;
351}

353void MFTLEnsembleEraseLocalSlot(MFTL_Ensemble *ens, int slot)
354{
355 MFTL_ASSERT(ens != MFTL_NULL);
356 if (ens == MFTL_NULL || slot < 0 || slot >= ens->n_local_objects) return;
357
358 if (slot + 1 < ens->n_local_objects)
359 {
360 memmove(&ens->local_objects[slot], &ens->local_objects[slot + 1], (size_t) (ens->n_local_objects - slot - 1) * sizeof(MFTL_Object *));
361 }
362 ens->n_local_objects--;
363}

365void MFTLEnsembleUpdateGlobalCounts(MFTL_Ensemble *ensemble)
366{
367 if (!ensemble) return;
368
369 /* Guard on the GLOBAL count: collectives follow, and an early return on the
370 * local count would leave the other ranks waiting. */
371 if (ensemble->n_bodies_global == 0)
372 {
373 ensemble->total_global_triangles = 0;
374 return;
375 }
376
377 const int n_local = ensemble->n_local_objects;
378
379 /* Two sums per body over its HOLDERS, indexed by local slot. Doubles hold
380 * integer counts exactly up to 2^53. */
381 static const MFTL_BodyReduceOp ops[2] = {MFTL_BODY_REDUCE_SUM, MFTL_BODY_REDUCE_SUM};
382
383 double *const counts = (double *) MFTLMallocOrDie((size_t) MFTL_MAX(2 * n_local, 1) * sizeof(double), "MFTLEnsembleUpdateGlobalCounts");
384
385 /* The local half is `n_local`, thanks to the ownership prefix. */
386 long local_tris = 0;
387 for (int o = 0; o < n_local; ++o)
388 {
389 MFTL_Object *const obj = ensemble->local_objects[o];
390 MFTL_ASSERT(obj != MFTL_NULL);
391 counts[2 * o + 0] = (double) obj->tris.n_local;
392 counts[2 * o + 1] = (double) obj->verts.n_local;
393 local_tris += obj->tris.n_local;
394 }
395
396 MFTLEnsembleReduceBodies(ensemble, counts, 2, ops, MFTL_BODY_REDUCE_SCOPE_AUTO);
397
398 for (int o = 0; o < n_local; ++o)
399 {
400 MFTL_Object *const obj = ensemble->local_objects[o];
401 obj->global_n_triangles = (int) counts[2 * o + 0];
402 obj->global_n_vertices = (int) counts[2 * o + 1];
403 }
404
405 free(counts);
406
407 /* The ensemble total is one scalar reduction. */
408 long total = 0;
409 MPI_Allreduce(&local_tris, &total, 1, MPI_LONG, MPI_SUM, ensemble->domain.comm_world);
410 ensemble->total_global_triangles = (int) total;
411}

417double MFTLDomainGetHMin(const MFTL_Domain *domain)
418{
419 MFTL_ASSERT(domain != MFTL_NULL);
420
421 double h_min = MFTL_DBL_MAX;
422 for (int k = 0; k < 3; ++k)
423 {
424 /* An uncut or zero-extent axis has no interior plane and constrains nothing. */
425 if (domain->cuts[k] < 2 || domain->domain_size[k] <= 0.0) continue;
426 h_min = MFTL_MIN(h_min, domain->domain_size[k] / (double) domain->cuts[k]);
427 }
428
429 return (h_min == MFTL_DBL_MAX) ? 0.0 : h_min;
430}
