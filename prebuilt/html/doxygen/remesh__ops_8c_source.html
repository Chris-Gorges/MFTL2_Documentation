1/**
2 * @file remesh_ops.c
3 * @author Christian Gorges
4 * @date 2026-08-15
5 * @ingroup mftl_remeshing
6 *
7 * @brief The three topological operators and their approval predicates.
8 *
9 * ## THE SHAPE EVERY OPERATOR HAS, AND WHY
10 *
11 * 1. READ the patch into locals: vertex indices, rim twin indices, rim flags.
12 * 2. ALLOCATE whatever new entities are needed.
13 * 3. WRITE the triangles, place the flags, re-link the rim.
14 * 4. REPAIR the affected vertices' out_he_idx.
15 *
16 * Step 1 must complete before step 2 because allocation can move both pool
17 * blocks, and step 3 must not begin before step 2 for the same reason. That is
18 * the one rule the AoS-in-a-pool layout imposes on a remesher, and it is cheap:
19 * a patch is a dozen integers.
20 *
21 * Step 3 is where implicit indexing shows up. When triangle t is rewritten, the
22 * logical edge that occupied slot 1 may end up in slot 2, so the rim FLAGS have
23 * to be carried explicitly -- a half-edge record does not move with its edge,
24 * only its contents do. MFTL_FLAG_FEATURE and MFTL_FLAG_BOUNDARY are both
25 * properties of the EDGE; losing one during a split is invisible until a ridge
26 * develops a gap several timesteps later.
27 *
28 * ## WHY THE PREDICATES ARE THIS LONG
29 *
30 * Every clause in MFTLCollapseApproved corresponds to a mesh that is still a
31 * valid data structure and no longer a valid surface. None of them announce
32 * themselves: a violated link condition produces a non-manifold vertex whose
33 * one-ring walk silently visits half the fan, and a fold-over produces an
34 * inverted triangle that the feature detector then reads as a sharp edge and the
35 * smoother locks in place forever. The predicates are the whole correctness
36 * argument of the module; the execution functions assume it has been made.
37 */

53/**
54 * @brief Is @p h the half of its undirected edge the operators are offered?
55 *
56 * Operators act on EDGES, so each must be offered once per sweep, or an edge is
57 * split twice or a flip undone by its own twin. The smaller index wins; a
58 * boundary half-edge has no twin and is always canonical.
59 *
60 * @param obj The mesh.
61 * @param h Half-edge index.
62 * @return `true` for the canonical half-edge.
63 */
64static inline bool he_is_canonical(const MFTL_Object *obj, int h)
65{
66 const int tw = obj->tris.he[h].twin_idx;
67 return (tw == MFTL_ERROR) || (h < tw);
68}

70/**
71 * @brief Common preconditions: the edge and both endpoints are live and untouched.
72 *
73 * @param obj The mesh.
74 * @param h Half-edge index.
75 * @return `true` if @p h is alive, untouched this sweep, and its triangle is too.
76 */
77static bool edge_is_operable(const MFTL_Object *obj, int h)
78{
79 if (h < 0 || h >= 3 * obj->tris.n) return MFTL_FALSE;
80 if (tri_is_dead(obj, MFTLHeTriIdx(h))) return MFTL_FALSE;
81 if (obj->tris.he[h].flags & (MFTL_FLAG_DEAD | MFTL_FLAG_VISITED)) return MFTL_FALSE;
82
83 const int a = MFTLHeOrigin(obj, h);
84 const int b = MFTLHeTarget(obj, h);
85 if (obj->verts.rec[a].flags & (MFTL_FLAG_DEAD | MFTL_FLAG_VISITED)) return MFTL_FALSE;
86 if (obj->verts.rec[b].flags & (MFTL_FLAG_DEAD | MFTL_FLAG_VISITED)) return MFTL_FALSE;
87
88 return MFTL_TRUE;
89}

121/**
122 * @brief Number of entries of @p a that also appear in @p b.
123 *
124 * @param a First sorted-or-not index list.
125 * @param na Length of @p a.
126 * @param b Second index list.
127 * @param nb Length of @p b.
128 * @return Number of indices appearing in both. This is the link condition's
129 * counting step: two endpoints must share exactly two neighbours (one at
130 * a mesh boundary edge).
131 */
132static int count_common(const int *a, int na, const int *b, int nb)
133{
134 int common = 0;
135 for (int i = 0; i < na; ++i)
136 {
137 for (int j = 0; j < nb; ++j)
138 {
139 if (a[i] == b[j])
140 {
141 common++;
142 break;
143 }
144 }
145 }
146 return common;
147}

166bool MFTLSplitApproved(const MFTL_Object *obj, int h, const MFTL_RemeshParameters *params)
167{
168 MFTL_ASSERT(obj != MFTL_NULL && params != MFTL_NULL);
169
170 if (!edge_is_operable(obj, h)) return MFTL_FALSE;
171 if (!he_is_canonical(obj, h)) return MFTL_FALSE;
172
173 return MFTLGetEdgeLengthSquared(obj, h) > params->max_edge_length_sq;
174}

176int MFTLSplitEdge(MFTL_Object *obj, int h)
177{
178 MFTL_ASSERT(obj != MFTL_NULL);
179 if (h < 0 || h >= 3 * obj->tris.n) return MFTL_ERROR;
180 if (tri_is_dead(obj, MFTLHeTriIdx(h))) return MFTL_ERROR;
181 obj->derived &= ~MFTL_DERIVED_GEOMETRY;
182
183 /* ── 1. Read the patch ─────────────────────────────────────────────────────
184 *
185 * Upper triangle T1 = (A, B, C) with h = A->B; lower T2 = (B, A, D) across the
186 * twin, if there is one. Everything captured here is a value: after step 2 the
187 * pool blocks may live somewhere else entirely. */
188 const MFTL_HalfEdge *he = obj->tris.he;
189
190 const int t1 = MFTLHeTriIdx(h);
191 const int hn = MFTLHeNextIdx(h); // B->C
192 const int hp = MFTLHePrevIdx(h); // C->A
193
194 const int va = he[hp].vertex_idx;
195 const int vb = he[h].vertex_idx;
196 const int vc = he[hn].vertex_idx;
197
198 const int w_bc = he[hn].twin_idx;
199 const int w_ca = he[hp].twin_idx;
200
201 const int f_ab = he[h].flags;
202 const int f_bc = he[hn].flags;
203 const int f_ca = he[hp].flags;
204
205 const int ht = he[h].twin_idx;
206
207 int t2 = MFTL_ERROR;
208 int vd = MFTL_ERROR;
209 int w_ad = MFTL_ERROR;
210 int w_db = MFTL_ERROR;
211 int f_ba = 0;
212 int f_ad = 0;
213 int f_db = 0;
214
215 if (ht != MFTL_ERROR)
216 {
217 const int hnt = MFTLHeNextIdx(ht); // A->D
218 const int hpt = MFTLHePrevIdx(ht); // D->B
219
220 t2 = MFTLHeTriIdx(ht);
221 vd = he[hnt].vertex_idx;
222 w_ad = he[hnt].twin_idx;
223 w_db = he[hpt].twin_idx;
224 f_ba = he[ht].flags;
225 f_ad = he[hnt].flags;
226 f_db = he[hpt].flags;
227 }
228
229 double pm[3];
230 midpoint_periodic(obj->domain, MFTLVertexPos(obj, va), MFTLVertexPos(obj, vb), pm);
231
232 /* ── 2. Allocate. `he` is dangling from here on. ──────────────────────────── */
233 const int vm = remesh_new_vertex(obj, pm);
234 const int t3 = remesh_new_triangle(obj);
235 const int t4 = (ht != MFTL_ERROR) ? remesh_new_triangle(obj) : MFTL_ERROR;
236
237 MFTL_HalfEdge *const hw = obj->tris.he;
238
239 /* ── 3a. Upper half: T1 becomes (A, M, C), T3 is (M, B, C) ─────────────────
240 *
241 * The sub-edges A-M and M-B are both halves of the edge that was split, so
242 * they inherit its flags -- including MFTL_FLAG_BOUNDARY, which is what keeps
243 * an open surface's rim recognisable, and MFTL_FLAG_FEATURE, which is what
244 * keeps a ridge continuous across the new vertex. */
245 tri_write(obj, t1, va, vm, vc);
246 hw[3 * t1 + 0].flags = f_ab; // A->M
247 hw[3 * t1 + 1].flags = 0; // M->C, brand new and interior by construction
248 hw[3 * t1 + 2].flags = f_ca; // C->A
249
250 tri_write(obj, t3, vm, vb, vc);
251 hw[3 * t3 + 0].flags = f_ab; // M->B
252 hw[3 * t3 + 1].flags = f_bc; // B->C
253 hw[3 * t3 + 2].flags = 0; // C->M
254
255 he_pair(obj, 3 * t1 + 1, 3 * t3 + 2); // M->C <-> C->M
256 he_pair_outer(obj, 3 * t1 + 2, w_ca);
257 he_pair_outer(obj, 3 * t3 + 1, w_bc);
258
259 /* ── 3b. Lower half: T2 becomes (B, M, D), T4 is (M, A, D) ────────────────── */
260 if (ht != MFTL_ERROR)
261 {
262 tri_write(obj, t2, vb, vm, vd);
263 hw[3 * t2 + 0].flags = f_ba; // B->M
264 hw[3 * t2 + 1].flags = 0; // M->D
265 hw[3 * t2 + 2].flags = f_db; // D->B
266
267 tri_write(obj, t4, vm, va, vd);
268 hw[3 * t4 + 0].flags = f_ba; // M->A
269 hw[3 * t4 + 1].flags = f_ad; // A->D
270 hw[3 * t4 + 2].flags = 0; // D->M
271
272 he_pair(obj, 3 * t2 + 1, 3 * t4 + 2); // M->D <-> D->M
273 he_pair_outer(obj, 3 * t2 + 2, w_db);
274 he_pair_outer(obj, 3 * t4 + 1, w_ad);
275
276 /* The split edge itself, now two edges each with two sides. */
277 he_pair(obj, 3 * t1 + 0, 3 * t4 + 0); // A->M <-> M->A
278 he_pair(obj, 3 * t3 + 0, 3 * t2 + 0); // M->B <-> B->M
279 }
280 else
281 {
282 /* A mesh boundary edge splits into two mesh boundary edges. The flags said so
283 * already (f_ab carries MFTL_FLAG_BOUNDARY); only the twins need stating. */
284 hw[3 * t1 + 0].twin_idx = MFTL_ERROR;
285 hw[3 * t3 + 0].twin_idx = MFTL_ERROR;
286 obj->verts.rec[vm].flags |= MFTL_FLAG_BOUNDARY;
287 }
288
289 /* ── 4. Outgoing half-edges ───────────────────────────────────────────────── */
290 obj->verts.rec[va].out_he_idx = 3 * t1 + 0; // A->M
291 obj->verts.rec[vm].out_he_idx = 3 * t1 + 1; // M->C
292 obj->verts.rec[vc].out_he_idx = 3 * t1 + 2; // C->A
293 obj->verts.rec[vb].out_he_idx = 3 * t3 + 1; // B->C (slot 0 is M->B, which starts at M)
294 if (ht != MFTL_ERROR) obj->verts.rec[vd].out_he_idx = 3 * t2 + 2; // D->B
295
296 /* The two halves are exempt from further operations this sweep: without this a
297 * long edge is split, and one of its halves -- still longer than the target on
298 * a badly graded mesh -- is split again by the same sweep, which is how a
299 * remesher produces a local explosion of tiny triangles in one pass. Marking
300 * the VERTEX prevents the collapse sweep from removing what this one just made,
301 * which is the classic split/collapse ping-pong. */
302 hw[3 * t1 + 0].flags |= MFTL_FLAG_VISITED;
303 hw[3 * t3 + 0].flags |= MFTL_FLAG_VISITED;
304 if (ht != MFTL_ERROR)
305 {
306 hw[3 * t4 + 0].flags |= MFTL_FLAG_VISITED;
307 hw[3 * t2 + 0].flags |= MFTL_FLAG_VISITED;
308 }
309 obj->verts.rec[vm].flags |= MFTL_FLAG_VISITED;
310
311 /* M inherits the split edge's classification, so the operator's output is
312 * self-consistent without waiting for the next MFTLDetectFeatures: splitting a
313 * feature edge puts M on that curve with exactly the two sub-edges as its
314 * feature edges, which is the definition of a ridge point. Leaving it clear
315 * would mark M free, and a free vertex on a feature curve is one the smoother
316 * moves off it. */
317 if (f_ab & MFTL_FLAG_FEATURE) obj->verts.rec[vm].flags |= MFTL_FLAG_FEATURE;
318
319 MFTLObjectInvalidateDerived(obj);
320 return vm;
321}

327/**
328 * @brief Where the surviving vertex of a collapse ends up.
329 *
330 * Its own position if it is a corner, or if it alone lies on a feature curve or
331 * boundary: collapsing a face vertex INTO a ridge vertex is legal, but the
332 * midpoint would drag the ridge into the face and erode it one collapse at a
333 * time while every approval clause passes. Otherwise the minimum-image
334 * midpoint -- exact on a straight feature and \f$O(h^2)\f$ off a curved one,
335 * acceptable because the edge is short by construction.
336 *
337 * @param obj The mesh.
338 * @param a The vertex that would be retired.
339 * @param b The vertex that would survive.
340 * @param[out] out The survivor's new position.
341 */
342static void collapse_target_position(const MFTL_Object *obj, int a, int b, double out[3])
343{
344 const bool a_is_pinned = vertex_on_feature(obj, a) || MFTLVertexIsMeshBoundary(obj, a);
345 const bool b_is_pinned = vertex_on_feature(obj, b) || MFTLVertexIsMeshBoundary(obj, b);
346
347 /* A corner outranks a ridge point. Collapsing along a feature curve INTO its
348 * corner is legal and useful, but the corner is the one thing on that curve
349 * that has a definite position -- averaging it with a neighbour rounds it off. */
350 if ((obj->verts.rec[b].flags & MFTL_FLAG_CORNER) || (b_is_pinned && !a_is_pinned))
351 {
352 const double *pb = MFTLVertexPos(obj, b);
353 out[0] = pb[0];
354 out[1] = pb[1];
355 out[2] = pb[2];
356 return;
357 }
358
359 midpoint_periodic(obj->domain, MFTLVertexPos(obj, a), MFTLVertexPos(obj, b), out);
360}

362bool MFTLCollapseApproved(const MFTL_Object *obj, int h, const MFTL_RemeshParameters *params, int *reject_reason)
363{
364 MFTL_ASSERT(obj != MFTL_NULL && params != MFTL_NULL);
365
366 int reason = MFTL_REMESH_REJECT_NONE;
367 bool ok = MFTL_FALSE;
368
369 if (!edge_is_operable(obj, h)) goto done;
370
371 const int a = MFTLHeOrigin(obj, h); // retired
372 const int b = MFTLHeTarget(obj, h); // survives
373
374 /* ── Wanted? ─────────────────────────────────────────────────────────────── */
375 if (MFTLGetEdgeLengthSquared(obj, h) >= params->min_edge_length_sq) goto done;
376
377 /* ── Features and boundaries ──────────────────────────────────────────────
378 *
379 * A corner is defined by being pinned, so it cannot be the vertex that goes.
380 * A vertex on a feature curve may only collapse into another one, or the curve
381 * is dragged off itself -- and the same statement for a mesh boundary, where
382 * the consequence is that the surface's rim moves inwards. */
383 reason = MFTL_REMESH_REJECT_FEATURE;
384 if (obj->verts.rec[a].flags & MFTL_FLAG_CORNER) goto done;
385 if (vertex_on_feature(obj, a) && !vertex_on_feature(obj, b)) goto done;
386 if (MFTLVertexIsMeshBoundary(obj, a) && !MFTLVertexIsMeshBoundary(obj, b)) goto done;
387
388 /* ── The link condition FOR CURVES, which is a separate statement ─────────
389 *
390 * Both endpoints on a feature curve, but the edge between them NOT on it: the
391 * edge is a chord across the surface joining two points of the curve, and
392 * collapsing it merges them into one. On an open patch that means two rim
393 * vertices joined by an interior edge, and the collapse pinches the rim inwards
394 * -- the surface grows a serrated notch that nothing reports, because the
395 * result is still a perfectly valid manifold with a perfectly valid boundary.
396 * On a cube it means two points on DIFFERENT sharp edges, merged into the
397 * middle of the face between them.
398 *
399 * The vertex link condition above cannot see this. It counts shared
400 * NEIGHBOURS, and these two have exactly the two they are supposed to have; the
401 * degeneracy is in the curve, not in the surface. This is the same clause the
402 * classical link condition states by giving a boundary vertex a virtual
403 * neighbour outside the mesh. */
404 if (!(obj->tris.he[h].flags & MFTL_FLAG_FEATURE) && vertex_on_feature(obj, a) && vertex_on_feature(obj, b)) goto done;
405 if (obj->tris.he[h].twin_idx != MFTL_ERROR && MFTLVertexIsMeshBoundary(obj, a) && MFTLVertexIsMeshBoundary(obj, b)) goto done;
406
407 /* ── Link condition ───────────────────────────────────────────────────────
408 *
409 * |N(a) INTERSECT N(b)| must be exactly 2 (the two wing vertices), or exactly 1
410 * at a mesh boundary edge, which has one wing. More than that means a and b are
411 * joined by a path OTHER than through their shared triangles, and collapsing
412 * welds two sheets of the surface into one non-manifold vertex. The data
413 * structure survives it; every subsequent one-ring walk from that vertex
414 * silently visits only part of the fan. */
415 reason = MFTL_REMESH_REJECT_LINK;
416
417 int ring_a[MFTL_REMESH_RING_MAX];
418 int ring_b[MFTL_REMESH_RING_MAX];
419 const int na = ring_vertices(obj, a, ring_a);
420 const int nb = ring_vertices(obj, b, ring_b);
421 if (na == MFTL_ERROR || nb == MFTL_ERROR)
422 {
423 reason = MFTL_REMESH_REJECT_OTHER;
424 goto done;
425 }
426
427 const bool boundary_edge = (obj->tris.he[h].twin_idx == MFTL_ERROR);
428 const int common = count_common(ring_a, na, ring_b, nb);
429 if (common != (boundary_edge ? 1 : 2)) goto done;
430
431 /* ── Valence ──────────────────────────────────────────────────────────────
432 *
433 * The merged vertex inherits the union of both neighbourhoods minus a and b
434 * themselves. Capping it at MFTL_MAX_VALENCE is not a quality preference: the
435 * library's fan walkers stop there, so a vertex above it becomes untraversable
436 * and every later query about it is quietly truncated. */
437 const int merged_valence = na + nb - common - 2;
438 if (merged_valence < 3 || merged_valence > MFTL_MAX_VALENCE) goto done;
439
440 /* The wing vertices each lose one incident triangle. Below valence 4 that
441 * leaves a vertex with two triangles sharing two edges -- a pocket, not a
442 * surface. */
443 for (int i = 0; i < na; ++i)
444 {
445 for (int j = 0; j < nb; ++j)
446 {
447 if (ring_a[i] != ring_b[j]) continue;
448
449 int ring_w[MFTL_REMESH_RING_MAX];
450 const int nw = ring_vertices(obj, ring_a[i], ring_w);
451 if (nw == MFTL_ERROR || nw < 4) goto done;
452 break;
453 }
454 }
455
456 /* ── Stretch ──────────────────────────────────────────────────────────────
457 *
458 * Every edge of the merged vertex must stay inside the band, or the next split
459 * sweep undoes this collapse and the two sweeps oscillate for as long as the
460 * caller is willing to iterate. */
461 reason = MFTL_REMESH_REJECT_OTHER;
462
463 double pm[3];
464 collapse_target_position(obj, a, b, pm);
465
466 for (int pass = 0; pass < 2; ++pass)
467 {
468 const int *ring = pass ? ring_b : ring_a;
469 const int n = pass ? nb : na;
470
471 for (int i = 0; i < n; ++i)
472 {
473 if (ring[i] == a || ring[i] == b) continue;
474
475 double d[3];
476 MFTLGetMinimumImageVector(obj->domain, d, pm, MFTLVertexPos(obj, ring[i]));
477 if (d[0] * d[0] + d[1] * d[1] + d[2] * d[2] > params->max_edge_length_sq) goto done;
478 }
479 }
480
481 /* ── Fold-over ────────────────────────────────────────────────────────────
482 *
483 * Every triangle that SURVIVES the collapse is re-evaluated with both endpoints
484 * moved to the merge point, and its normal must not turn by more than the
485 * threshold. The two triangles being removed are skipped -- they degenerate to
486 * a line by construction, which is the point of removing them.
487 *
488 * Omitting this clause is not a missing nicety, it is the head of a feedback
489 * loop: the inverted
490 * triangle a collapse produces presents a ~180 degree dihedral angle, the
491 * feature detector reads that as a sharp edge, and the smoother then LOCKS
492 * every vertex of it. The inversion becomes permanent, and it seeds more of
493 * them on the next pass. */
494 reason = MFTL_REMESH_REJECT_FOLD;
495
496 const int t_keep_1 = MFTLHeTriIdx(h);
497 const int t_keep_2 = boundary_edge ? MFTL_ERROR : MFTLHeTriIdx(obj->tris.he[h].twin_idx);
498
499 for (int pass = 0; pass < 2; ++pass)
500 {
501 int fan[MFTL_REMESH_RING_MAX];
502 const int nf = ring_triangles(obj, pass ? b : a, fan);
503 if (nf == MFTL_ERROR)
504 {
505 reason = MFTL_REMESH_REJECT_OTHER;
506 goto done;
507 }
508
509 for (int i = 0; i < nf; ++i)
510 {
511 const int t = fan[i];
512 if (t == t_keep_1 || t == t_keep_2) continue;
513
514 double n_old[3];
515 double n_new[3];
516 tri_area_vector_substituted(obj, t, MFTL_ERROR, MFTL_ERROR, MFTL_NULL, n_old);
517 tri_area_vector_substituted(obj, t, a, b, pm, n_new);
518
519 if (area_vector_cos(n_old, n_new) < params->max_normal_deviation_cos) goto done;
520 }
521 }
522
523 reason = MFTL_REMESH_REJECT_NONE;
524 ok = MFTL_TRUE;
525
526done:
527 if (reject_reason) *reject_reason = ok ? MFTL_REMESH_REJECT_NONE : reason;
528 return ok;
529}

531/**
532 * @brief Re-links the two rim half-edges of a removed triangle to each other.
533 *
534 * Removing a triangle merges two of its edges into one, so the neighbours across
535 * those edges become directly adjacent. Either side may be absent, in which case
536 * the survivor becomes a mesh boundary half-edge and is flagged as one -- the
537 * flag is what tells a later pass that this is the edge of the SURFACE rather
538 * than of the partition, and it cannot be re-derived after the fact.
539 *
540 * The feature bit is OR-ed rather than picked, so a ridge that ran through one of
541 * the two edges still runs through the merged one.
542 *
543 * @param obj The mesh.
544 * @param x Rim half-edge just written into a new slot.
545 * @param y Its partner outside the patch, or #MFTL_ERROR.
546 */
547static void glue_rim(MFTL_Object *obj, int x, int y)
548{
549 MFTL_HalfEdge *const he = obj->tris.he;
550
551 const int fx = (x != MFTL_ERROR) ? he[x].flags : 0;
552 const int fy = (y != MFTL_ERROR) ? he[y].flags : 0;
553 const int feature = (fx | fy) & MFTL_FLAG_FEATURE;
554
555 if (x != MFTL_ERROR)
556 {
557 he[x].twin_idx = y;
558 he[x].flags |= feature;
559 if (y == MFTL_ERROR) he[x].flags |= MFTL_FLAG_BOUNDARY;
560 }
561 if (y != MFTL_ERROR)
562 {
563 he[y].twin_idx = x;
564 he[y].flags |= feature;
565 if (x == MFTL_ERROR) he[y].flags |= MFTL_FLAG_BOUNDARY;
566 }
567}

569/**
570 * @brief Flags triangle @p t and its three half-edges dead.
571 *
572 * @param obj The mesh.
573 * @param t Triangle index. Flagged #MFTL_FLAG_DEAD along with its three
574 * half-edges; the compaction removes it later.
575 */
576static void kill_triangle(MFTL_Object *obj, int t)
577{
578 obj->tris.rec[t].flags |= MFTL_FLAG_DEAD;
579 for (int k = 0; k < 3; ++k) obj->tris.he[3 * t + k].flags |= MFTL_FLAG_DEAD;
580}
581
582int MFTLCollapseEdge(MFTL_Object *obj, int h)
583{
584 MFTL_ASSERT(obj != MFTL_NULL);
585 if (h < 0 || h >= 3 * obj->tris.n) return MFTL_ERROR;
586 if (tri_is_dead(obj, MFTLHeTriIdx(h))) return MFTL_ERROR;
587 obj->derived &= ~MFTL_DERIVED_GEOMETRY;
588
589 /* A collapse allocates nothing, so unlike the split this pointer stays valid
590 * for the whole function. */
591 MFTL_HalfEdge *const he = obj->tris.he;
592
593 const int t1 = MFTLHeTriIdx(h);
594 const int hn = MFTLHeNextIdx(h); // B->C
595 const int hp = MFTLHePrevIdx(h); // C->A
596
597 const int a = he[hp].vertex_idx; // retired
598 const int b = he[h].vertex_idx; // survives
599
600 /* Both fans, captured while the topology still describes them. The union of
601 * their triangles is exactly the patch this operation can disturb, which is
602 * what makes the out_he repair at the end bounded and complete. */
603 int fan_a[MFTL_REMESH_RING_MAX];
604 int fan_b[MFTL_REMESH_RING_MAX];
605 const int n_fan_a = MFTLGetVertexOneRingHalfedges(obj, a, fan_a, MFTL_REMESH_RING_MAX, MFTL_FALSE);
606 const int n_fan_b = MFTLGetVertexOneRingHalfedges(obj, b, fan_b, MFTL_REMESH_RING_MAX, MFTL_FALSE);
607 if (n_fan_a <= 0 || n_fan_a >= MFTL_REMESH_RING_MAX) return MFTL_ERROR;
608 if (n_fan_b <= 0 || n_fan_b >= MFTL_REMESH_RING_MAX) return MFTL_ERROR;
609
610 double pm[3];
611 collapse_target_position(obj, a, b, pm);
612
613 /* ── 1. Every half-edge that ENDS at A now ends at B ──────────────────────
614 *
615 * For an outgoing half-edge o of A, prev(o) is the incoming one, and the set
616 * {prev(o)} over the fan is every half-edge targeting A -- including the
617 * closing edge of an open fan, which exists only in the incoming direction and
618 * which a walk over outgoing edges alone would miss. */
619 for (int i = 0; i < n_fan_a; ++i) he[MFTLHePrevIdx(fan_a[i])].vertex_idx = b;
620
621 /* ── 2. Remove the incident triangles ─────────────────────────────────────── */
622 const int ht = he[h].twin_idx;
623
624 glue_rim(obj, he[hn].twin_idx, he[hp].twin_idx);
625 kill_triangle(obj, t1);
626
627 if (ht != MFTL_ERROR)
628 {
629 const int hnt = MFTLHeNextIdx(ht); // A->D
630 const int hpt = MFTLHePrevIdx(ht); // D->B
631
632 glue_rim(obj, he[hnt].twin_idx, he[hpt].twin_idx);
633 kill_triangle(obj, MFTLHeTriIdx(ht));
634 }
635
636 /* ── 3. Retire A, move B ──────────────────────────────────────────────────── */
637 obj->verts.rec[a].flags |= MFTL_FLAG_DEAD;
638 obj->verts.rec[b].flags |= (obj->verts.rec[a].flags & MFTL_FLAG_BOUNDARY);
639 obj->verts.rec[b].flags |= MFTL_FLAG_VISITED;
640
641 obj->verts.pos[3 * b + 0] = pm[0];
642 obj->verts.pos[3 * b + 1] = pm[1];
643 obj->verts.pos[3 * b + 2] = pm[2];
644
645 /* ── 4. Repair out_he over the patch ──────────────────────────────────────
646 *
647 * Applying the global rule (`out_he[origin(x)] = x` for every live half-edge x)
648 * to the patch alone. Deriving it rather than repairing each affected vertex
649 * individually is one correctness argument instead of one per case -- and the
650 * cases are exactly where a hand repair goes wrong: B's replacement must come
651 * from a triangle that survived, C's old outgoing edge died with T1, and on an
652 * open fan neither has an obvious substitute. */
653 for (int pass = 0; pass < 2; ++pass)
654 {
655 const int *fan = pass ? fan_b : fan_a;
656 const int n = pass ? n_fan_b : n_fan_a;
657
658 for (int i = 0; i < n; ++i)
659 {
660 const int t = MFTLHeTriIdx(fan[i]);
661 if (tri_is_dead(obj, t)) continue;
662
663 for (int k = 0; k < 3; ++k) obj->verts.rec[MFTLHeOrigin(obj, 3 * t + k)].out_he_idx = 3 * t + k;
664 }
665 }
666
667 MFTLObjectInvalidateDerived(obj);
668 return MFTL_SUCCESS;
669}

672 * FLIP
673 * ═══════════════════════════════════════════════════════════════════════════ */

675/**
676 * @brief The valence a vertex would ideally have: 6 inside, 4 on a mesh boundary.
677 *
678 * @param obj The mesh.
679 * @param v Vertex index.
680 * @return 6 for an interior vertex, 4 on a mesh boundary -- the valence of a
681 * regular triangulation of the plane, and of its half.
682 */
683static inline int optimal_valence(const MFTL_Object *obj, int v) { return MFTLVertexIsMeshBoundary(obj, v) ? 4 : 6; }
684
685/**
686 * @brief Squared deviation of one vertex's valence from its optimum.
687 * @param val Actual valence.
688 * @param opt Optimal valence from optimal_valence().
689 * @return \\f$(\\mathrm{val} - \\mathrm{opt})^2\\f$.
690 */
691static inline int valence_defect(int val, int opt) { return (val - opt) * (val - opt); }
692
693bool MFTLFlipApproved(const MFTL_Object *obj, int h, const MFTL_RemeshParameters *params, int *reject_reason)
694{
695 MFTL_ASSERT(obj != MFTL_NULL && params != MFTL_NULL);
696
697 if (reject_reason) *reject_reason = MFTL_REMESH_REJECT_NONE;
698
699 if (!edge_is_operable(obj, h)) return MFTL_FALSE;
700 if (!he_is_canonical(obj, h)) return MFTL_FALSE;
701
702 const int ht = obj->tris.he[h].twin_idx;
703 if (ht == MFTL_ERROR) return MFTL_FALSE; // a boundary edge has no quad to re-cut
704 if (tri_is_dead(obj, MFTLHeTriIdx(ht))) return MFTL_FALSE;
705
706 /* A feature edge is a feature because of where it runs; flipping it moves it. */
707 if (obj->tris.he[h].flags & (MFTL_FLAG_FEATURE | MFTL_FLAG_BOUNDARY)) return MFTL_FALSE;
708
709 const int va = MFTLHeOrigin(obj, h);
710 const int vb = MFTLHeTarget(obj, h);
711 const int vc = MFTLHeTarget(obj, MFTLHeNextIdx(h));
712 const int vd = MFTLHeTarget(obj, MFTLHeNextIdx(ht));
713
714 int ring[MFTL_REMESH_RING_MAX];
715
716 const int val_a = ring_vertices(obj, va, ring);
717 if (val_a == MFTL_ERROR) return MFTL_FALSE;
718 const int val_b = ring_vertices(obj, vb, ring);
719 if (val_b == MFTL_ERROR) return MFTL_FALSE;
720
721 /* A and B each lose a neighbour. Below 4 they would drop under the minimum a
722 * manifold interior vertex can have. */
723 if (val_a <= 3 || val_b <= 3) return MFTL_FALSE;
724
725 const int val_c = ring_vertices(obj, vc, ring);
726 if (val_c == MFTL_ERROR) return MFTL_FALSE;
727
728 /* If C and D are already joined, the flip creates a SECOND edge between them:
729 * two distinct undirected edges with the same endpoints, which no half-edge
730 * traversal can tell apart. Cheap to test here, unrecoverable afterwards. */
731 for (int i = 0; i < val_c; ++i)
732 {
733 if (ring[i] == vd) return MFTL_FALSE;
734 }
735
736 const int val_d = ring_vertices(obj, vd, ring);
737 if (val_d == MFTL_ERROR) return MFTL_FALSE;
738
739 if (val_c + 1 > MFTL_MAX_VALENCE || val_d + 1 > MFTL_MAX_VALENCE) return MFTL_FALSE;
740
741 /* ── The valence criterion ────────────────────────────────────────────────
742 *
743 * Evaluated before the geometry, and not only because comparing four integers
744 * is cheaper than two cross products: it is what makes "wanted but refused"
745 * a distinguishable outcome. A flip the valence criterion declines is the
746 * criterion working; a flip it wants and the geometry vetoes is a fold, and
747 * only the second is worth a counter.
748 *
749 * STRICTLY smaller, never equal: two flips that leave the defect unchanged
750 * would alternate for as many iterations as the caller allows, and each one
751 * reports itself as work done, so the convergence test never fires either. */
752 const int opt_a = optimal_valence(obj, va);
753 const int opt_b = optimal_valence(obj, vb);
754 const int opt_c = optimal_valence(obj, vc);
755 const int opt_d = optimal_valence(obj, vd);
756
757 const int before = valence_defect(val_a, opt_a) + valence_defect(val_b, opt_b) + valence_defect(val_c, opt_c) + valence_defect(val_d, opt_d);
758 const int after = valence_defect(val_a - 1, opt_a) + valence_defect(val_b - 1, opt_b) + valence_defect(val_c + 1, opt_c) + valence_defect(val_d + 1, opt_d);
759
760 if (after >= before) return MFTL_FALSE;
761
762 /* ── Geometry: would the new diagonal fold the quad? ──────────────────────
763 *
764 * The valence criterion is purely combinatorial and will happily re-cut a
765 * strongly non-convex quad, which produces a sliver overlapping its neighbour.
766 * The test is that both new faces still face the way the pair used to. */
767 const int t1 = MFTLHeTriIdx(h);
768 const int t2 = MFTLHeTriIdx(ht);
769
770 double n1[3];
771 double n2[3];
772 tri_area_vector_substituted(obj, t1, MFTL_ERROR, MFTL_ERROR, MFTL_NULL, n1);
773 tri_area_vector_substituted(obj, t2, MFTL_ERROR, MFTL_ERROR, MFTL_NULL, n2);
774
775 /* The pair's reference normal. It cancels only on a quad folded back onto
776 * itself, and then its direction is rounding noise -- zero it, relative to the
777 * two it is built from, so the fold test refuses rather than guesses. */
778 double n_ref[3] = {n1[0] + n2[0], n1[1] + n2[1], n1[2] + n2[2]};
779 {
780 const double ref_sq = n_ref[0] * n_ref[0] + n_ref[1] * n_ref[1] + n_ref[2] * n_ref[2];
781 const double parts_sq = (n1[0] * n1[0] + n1[1] * n1[1] + n1[2] * n1[2]) + (n2[0] * n2[0] + n2[1] * n2[1] + n2[2] * n2[2]);
782 if (!(ref_sq > MFTL_REMESH_DEGENERATE_SIN2 * parts_sq)) n_ref[0] = n_ref[1] = n_ref[2] = 0.0;
783 }
784
785 const double *pa = MFTLVertexPos(obj, va);
786 const double *pb = MFTLVertexPos(obj, vb);
787 const double *pc = MFTLVertexPos(obj, vc);
788 const double *pd = MFTLVertexPos(obj, vd);
789
790 /* (A,D,C) and (D,B,C): the same quad cut along the other diagonal. */
791 double ad[3];
792 double ac[3];
793 double db[3];
794 double dc[3];
795 MFTLGetMinimumImageVector(obj->domain, ad, pa, pd);
796 MFTLGetMinimumImageVector(obj->domain, ac, pa, pc);
797 MFTLGetMinimumImageVector(obj->domain, db, pd, pb);
798 MFTLGetMinimumImageVector(obj->domain, dc, pd, pc);
799
800 double m1[3];
801 double m2[3];
802 fold_area_vector(ad, ac, m1);
803 fold_area_vector(db, dc, m2);
804
805 if (area_vector_cos(n_ref, m1) < params->max_normal_deviation_cos || area_vector_cos(n_ref, m2) < params->max_normal_deviation_cos)
806 {
807 if (reject_reason) *reject_reason = MFTL_REMESH_REJECT_FOLD;
808 return MFTL_FALSE;
809 }
810
811 return MFTL_TRUE;
812}

814int MFTLFlipEdge(MFTL_Object *obj, int h)
815{
816 MFTL_ASSERT(obj != MFTL_NULL);
817 if (h < 0 || h >= 3 * obj->tris.n) return MFTL_ERROR;
818
819 MFTL_HalfEdge *const he = obj->tris.he;
820
821 const int ht = he[h].twin_idx;
822 if (ht == MFTL_ERROR) return MFTL_ERROR;
823 obj->derived &= ~MFTL_DERIVED_GEOMETRY;
824
825 const int t1 = MFTLHeTriIdx(h);
826 const int t2 = MFTLHeTriIdx(ht);
827
828 const int hn = MFTLHeNextIdx(h); // B->C
829 const int hp = MFTLHePrevIdx(h); // C->A
830 const int hnt = MFTLHeNextIdx(ht); // A->D
831 const int hpt = MFTLHePrevIdx(ht); // D->B
832
833 const int va = he[hp].vertex_idx;
834 const int vb = he[h].vertex_idx;
835 const int vc = he[hn].vertex_idx;
836 const int vd = he[hnt].vertex_idx;
837
838 /* Rim twins and flags, captured before the rewrite moves every logical edge to
839 * a different slot. */
840 const int w_bc = he[hn].twin_idx;
841 const int w_ca = he[hp].twin_idx;
842 const int w_ad = he[hnt].twin_idx;
843 const int w_db = he[hpt].twin_idx;
844
845 const int f_bc = he[hn].flags;
846 const int f_ca = he[hp].flags;
847 const int f_ad = he[hnt].flags;
848 const int f_db = he[hpt].flags;
849
850 /* The quad is A -> D -> B -> C. Cutting it along the other diagonal gives
851 * (A, D, C) and (D, B, C), both still counter-clockwise. */
852 tri_write(obj, t1, va, vd, vc);
853 he[3 * t1 + 0].flags = f_ad; // A->D
854 he[3 * t1 + 1].flags = 0; // D->C, the new diagonal
855 he[3 * t1 + 2].flags = f_ca; // C->A
856
857 tri_write(obj, t2, vd, vb, vc);
858 he[3 * t2 + 0].flags = f_db; // D->B
859 he[3 * t2 + 1].flags = f_bc; // B->C
860 he[3 * t2 + 2].flags = 0; // C->D
861
862 he_pair(obj, 3 * t1 + 1, 3 * t2 + 2); // D->C <-> C->D
863 he_pair_outer(obj, 3 * t1 + 0, w_ad);
864 he_pair_outer(obj, 3 * t1 + 2, w_ca);
865 he_pair_outer(obj, 3 * t2 + 0, w_db);
866 he_pair_outer(obj, 3 * t2 + 1, w_bc);
867
868 /* A and B lost the edge they were joined by, so whichever of their outgoing
869 * half-edges pointed along it no longer exists. Setting all four is cheaper
870 * than working out which two needed it. */
871 obj->verts.rec[va].out_he_idx = 3 * t1 + 0; // A->D
872 obj->verts.rec[vd].out_he_idx = 3 * t1 + 1; // D->C
873 obj->verts.rec[vc].out_he_idx = 3 * t1 + 2; // C->A
874 obj->verts.rec[vb].out_he_idx = 3 * t2 + 1; // B->C
875
876 /* The new diagonal is off limits for the rest of the sweep. The valence
877 * criterion's strict inequality already makes an immediate flip-back
878 * impossible, but only while the four valences are unchanged -- a later flip in
879 * the same quad's neighbourhood can restore the condition. */
880 he[3 * t1 + 1].flags |= MFTL_FLAG_VISITED;
881 he[3 * t2 + 2].flags |= MFTL_FLAG_VISITED;
882
883 MFTLObjectInvalidateDerived(obj);
884 return MFTL_SUCCESS;
885}
