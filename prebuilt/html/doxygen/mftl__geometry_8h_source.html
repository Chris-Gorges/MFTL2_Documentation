60/**
61 * @brief The shortest vector from @p p1 to @p p2 under the minimum-image convention.
62 *
63 * On each periodic axis
64 *
65 * \f[
66 * \Delta_k = (p_2 - p_1)_k - L_k\,\mathrm{nearbyint}\!\left(\frac{(p_2-p_1)_k}{L_k}\right),
67 * \f]
68 *
69 * so \f$|\Delta_k| \le L_k/2\f$ for ANY displacement (the comparison form
70 * `if (d > L/2) d -= L` is only correct for \f$|d| < 3L/2\f$). Ties at
71 * \f$\pm L/2\f$ round to even. One multiply, one round and one fused
72 * multiply-subtract per axis.
73 *
74 * @param domain Supplies the per-axis periodicity, sizes and reciprocals.
75 * @param[out] vec_out The shortest displacement from @p p1 to @p p2.
76 * @param p1 Start position.
77 * @param p2 End position.
78 */
79static inline void MFTLGetMinimumImageVector(const MFTL_Domain *domain, double vec_out[3], const double p1[3], const double p2[3])
80{
81 vec_out[0] = p2[0] - p1[0];
82 vec_out[1] = p2[1] - p1[1];
83 vec_out[2] = p2[2] - p1[2];
84
85 for (int i = 0; i < 3; ++i)
86 {
87 if (domain->is_periodic[i]) vec_out[i] -= domain->domain_size[i] * nearbyint(vec_out[i] * domain->inv_domain_size[i]);
88 }
89}
90
91/**
92 * @brief Wraps a position into the periodic domain box by at most one period.
93 *
94 * Correct when the position is less than one period outside the box, which a
95 * position wrapped at every step always is. The compare-and-shift form beats a
96 * branchless one (measured 1.34-1.37x) because the branches are perfectly
97 * predicted and non-periodic axes do no work.
98 *
99 * @param domain Supplies the per-axis periodicity and bounds.
100 * @param pos Position, wrapped in place into \f$[\min, \max)\f$ on periodic
101 * axes. Non-periodic axes are untouched.
102 */
103static inline void MFTLApplyPeriodicity(const MFTL_Domain *domain, double pos[3])
104{
105 for (int i = 0; i < 3; ++i)
106 {
107 if (domain->is_periodic[i])
108 {
109 if (pos[i] >= domain->global_max[i])
110 pos[i] -= domain->domain_size[i];
111 else if (pos[i] < domain->global_min[i])
112 pos[i] += domain->domain_size[i];
113 }
114 }
115}
116
117/**
118 * @brief The rank whose Cartesian cell contains @p pos.
119 *
120 * Cell \f$i_k = \lfloor (x_k - \min_k)\,\mathrm{cuts}_k / L_k \rfloor\f$, clamped to
121 * \f$[0, \mathrm{cuts}_k)\f$, then a `rank_map` lookup. @p pos must be WRAPPED:
122 * coordinates outside the box are clamped to the boundary cell, a plausible
123 * wrong answer on a periodic axis.
124 *
125 * This is the inner statement of redistribution's owner classification (once
126 * per vertex and per triangle per redistributed step), so it carries nothing
127 * conditional. The remesher's shifted cut planes deliberately do NOT live here --
128 * adding the shift and its periodicity branch measured 1.70x slower (1.79 ->
129 * 3.05 ns/entity); they reach redistribution through #MFTL_OwnerMap instead.
130 *
131 * @param domain The MPI/physical world.
132 * @param pos A position, already wrapped into the domain box.
133 * @return The rank owning the cell containing @p pos.
134 */
135static inline int MFTLGetOwnerProc(const MFTL_Domain *domain, const double pos[3])
136{
137 int c[3];
138 for (int k = 0; k < 3; ++k)
139 {
140 int i = (int) ((pos[k] - domain->global_min[k]) * domain->grid_factors[k]);
141 if (i < 0) i = 0;
142 if (i >= domain->cuts[k]) i = domain->cuts[k] - 1;
143 c[k] = i;
144 }
145 return domain->rank_map[c[0] + c[1] * domain->cuts[0] + c[2] * domain->cuts[0] * domain->cuts[1]];
146}

165 * @brief Squared length of half-edge @p h. Prefer this wherever the sqrt can be avoided.
166 * @param obj The mesh.

190/**
191 * @brief Area, unit normal and centroid of triangle @p t.
192 *
193 * With minimum-image edges \f$\vect{e}_1 = \vect{p}_1 - \vect{p}_0\f$,
194 * \f$\vect{e}_2 = \vect{p}_2 - \vect{p}_0\f$ and
195 * \f$\vect{n} = \vect{e}_1 \times \vect{e}_2\f$:
196 * \f$A = \tfrac12\norm{\vect{n}}\f$, \f$\hat{\vect{n}} = \vect{n}/\norm{\vect{n}}\f$
197 * (zero for a degenerate triangle), \f$\vect{c} = \vect{p}_0 +
198 * \tfrac13(\vect{e}_1+\vect{e}_2)\f$ wrapped into the box. Reads `tris.verts`
199 * when current, the half-edge records otherwise. Loops over many triangles
200 * should use #MFTL_FOREACH_TRIANGLE_GEOMETRY or MFTLComputeTriangleGeometry().
201 *
202 * @param obj The mesh.
203 * @param t Triangle index.
204 * @param[out] area_out Area, or #MFTL_NULL.
205 * @param[out] normal_out Unit normal (zero if degenerate), or #MFTL_NULL.
206 * @param[out] center_out Centroid, WRAPPED into the domain; the true centroid
207 * even for a triangle straddling a periodic face. Or
208 * #MFTL_NULL.
209 */
210static inline void MFTLGetTriangleProperties(const MFTL_Object *obj, int t, double *area_out, double normal_out[3], double center_out[3])
211{
212 int v[3];
213 if ((obj->derived & MFTL_DERIVED_TOPOLOGY) != 0u)
214 {
215 v[0] = obj->tris.verts[3 * t + 0];
216 v[1] = obj->tris.verts[3 * t + 1];
217 v[2] = obj->tris.verts[3 * t + 2];
218 }
219 else
220 {
221 MFTLGetTriangleVertices(obj, t, v);
222 }
223
224 const double *p0 = MFTLVertexPos(obj, v[0]);
225
226 double e1[3];
227 double e2[3];
228 MFTLGetMinimumImageVector(obj->domain, e1, p0, MFTLVertexPos(obj, v[1]));
229 MFTLGetMinimumImageVector(obj->domain, e2, p0, MFTLVertexPos(obj, v[2]));
230
231 /* |e1 x e2| is twice the area. The magnitude is UNCLAMPED and the reciprocal
232 * guarded by a test against zero, so every triangle with a representable
233 * non-zero cross product gets an exactly unit normal and the true area (a
234 * clamp would floor the area and shorten small normals; see
235 * MFTLNormalizeVector()). The expression must be identical in every geometry
236 * kernel: test_triangle_geometry_paths_are_bit_identical compares them. */
237 double cr[3];
238 cr[0] = e1[1] * e2[2] - e1[2] * e2[1];
239 cr[1] = e1[2] * e2[0] - e1[0] * e2[2];
240 cr[2] = e1[0] * e2[1] - e1[1] * e2[0];
241
242 const double mag = MFTL_SQRT(cr[0] * cr[0] + cr[1] * cr[1] + cr[2] * cr[2]);
243
244 if (area_out) *area_out = 0.5 * mag;
245
246 if (normal_out)
247 {
248 const double inv = (mag > 0.0) ? 1.0 / mag : 0.0;
249 normal_out[0] = cr[0] * inv;
250 normal_out[1] = cr[1] * inv;
251 normal_out[2] = cr[2] * inv;
252 }
253
254 if (center_out)
255 {
256 center_out[0] = p0[0] + (e1[0] + e2[0]) * MFTL_ONE_THIRD;
257 center_out[1] = p0[1] + (e1[1] + e2[1]) * MFTL_ONE_THIRD;
258 center_out[2] = p0[2] + (e1[2] + e2[2]) * MFTL_ONE_THIRD;
259 MFTLApplyPeriodicity(obj->domain, center_out);
260 }
261}

263/**
264 * @brief The three corners of triangle @p t, mutually UNWRAPPED.
265 *
266 * Corner 0 is the stored (wrapped) position; corners 1 and 2 are corner 0 plus
267 * the minimum-image edges, so they may lie just outside the domain box. A
268 * triangle at \f$x = L-\epsilon\f$ and \f$x = +\epsilon\f$ thus comes out small
269 * rather than domain-spanning. The input of the BVH, the spatial halo and the
270 * signed-distance rasteriser.
271 *
272 * @param obj The mesh.
273 * @param t Triangle index. Bounds-asserted in Debug.
274 * @param[out] out Row k is corner k.
275 */
276static inline void MFTLGetTriangleCorners(const MFTL_Object *obj, int t, double out[3][3])
277{
278 MFTL_ASSERT(t >= 0 && t < obj->tris.n && "MFTLGetTriangleCorners: triangle index out of range");
279
280 int v[3];
281 if ((obj->derived & MFTL_DERIVED_TOPOLOGY) != 0u)
282 {
283 v[0] = obj->tris.verts[3 * t + 0];
284 v[1] = obj->tris.verts[3 * t + 1];
285 v[2] = obj->tris.verts[3 * t + 2];
286 }
287 else
288 {
289 MFTLGetTriangleVertices(obj, t, v);
290 }
291
292 const double *const p0 = MFTLVertexPos(obj, v[0]);
293 double e1[3];
294 double e2[3];
295 MFTLGetMinimumImageVector(obj->domain, e1, p0, MFTLVertexPos(obj, v[1]));
296 MFTLGetMinimumImageVector(obj->domain, e2, p0, MFTLVertexPos(obj, v[2]));
297
298 for (int k = 0; k < 3; ++k)
299 {
300 out[0][k] = p0[k];
301 out[1][k] = p0[k] + e1[k];
302 out[2][k] = p0[k] + e2[k];
303 }
304}

306/**
307 * @brief Tight axis-aligned bounds of triangle @p t, from its unwrapped corners.
308 *
309 * No tolerance is added; callers inflate by their own. The box of a
310 * seam-straddling triangle correctly extends outside the domain.
311 *
312 * @param obj The mesh.
313 * @param t Triangle index.
314 * @param[out] lo Lower corner of the box.
315 * @param[out] hi Upper corner of the box.
316 */
317static inline void MFTLGetTriangleBounds(const MFTL_Object *obj, int t, double lo[3], double hi[3])
318{
319 double c[3][3];
320 MFTLGetTriangleCorners(obj, t, c);
321
322 for (int k = 0; k < 3; ++k)
323 {
324 lo[k] = c[0][k];
325 hi[k] = c[0][k];
326 }
327 for (int j = 1; j < 3; ++j)
328 {
329 for (int k = 0; k < 3; ++k)
330 {
331 if (c[j][k] < lo[k]) lo[k] = c[j][k];
332 if (c[j][k] > hi[k]) hi[k] = c[j][k];
333 }
334 }
335}

368/**
369 * @brief Whether triangle @p t's cached `tris.center` still matches its vertices.
370 *
371 * Stale means off by more than \f$10^{-12}(|c_k| + L_k)\f$ on some axis k,
372 * compared through the seam, with \f$c\f$ the recomputed centre and \f$L_k\f$
373 * the domain length. The tolerance is relative, never absolute: rounding
374 * differences between two evaluations are ulps of \f$|c_k|\f$, and a centre
375 * error below \f$10^{-12}L_k\f$ can change the owner only of a triangle that
376 * close to a cut plane. A NaN or garbage centre is stale.
377 *
378 * @param obj The mesh; needs a domain.
379 * @param t Triangle index.
380 * @return `true` if the cached centre is current.
381 */
382static inline bool MFTLTriangleCentreIsCurrent(const MFTL_Object *obj, int t)
383{
384 double fresh[3];
385 MFTLGetTriangleCenter(obj, t, fresh);
386
387 /* Both are wrapped positions, so compare THROUGH the seam, not across it. */
388 double d[3];
389 MFTLGetMinimumImageVector(obj->domain, d, &obj->tris.center[3 * t], fresh);
390 for (int k = 0; k < 3; ++k)
391 {
392 if (!(fabs(d[k]) <= 1e-12 * (fabs(fresh[k]) + obj->domain->domain_size[k]))) return MFTL_FALSE;
393 }
394 return MFTL_TRUE;
395}
396
397/**
398 * @brief Whether every triangle's cached `tris.center`, ghosts included, still
399 * matches its vertices (MFTLTriangleCentreIsCurrent()).
400 *
401 * The precondition of every migration that routes by centre (MFTLRedistribute(),
402 * the remesher's shifted placements): a rank routes its ghosts too, and a stale
403 * ghost centre makes two ranks disagree about a destination. A full geometry
404 * pass -- use it inside `MFTL_ASSERT` only.
405 *
406 * @param obj The mesh.
407 * @return `true` if every centre is current.
408 */
409static inline bool MFTLTriangleCentresAreCurrent(const MFTL_Object *obj)
410{
411 for (int t = 0; t < obj->tris.n; ++t)
412 {
413 if (!MFTLTriangleCentreIsCurrent(obj, t)) return MFTL_FALSE;
414 }
415 return MFTL_TRUE;
416}

422/**
423 * @brief Which feature of a triangle a closest point lies on (its Voronoi region).
424 *
425 * Corner k is `tri[k]`; edge k runs from corner k to corner (k+1) mod 3, i.e. it
426 * is half-edge 3t+k of an MFTL triangle. A sign decided from a closest point on
427 * an edge or a vertex needs the pseudo-normal of that feature (Bærentzen &
428 * Aanæs 2005), not the face normal.
429 */
430typedef enum MFTL_TriangleFeature
431{
432 MFTL_TRI_FEATURE_FACE = 0, ///< Strictly inside the triangle.
433 MFTL_TRI_FEATURE_VERTEX0 = 1, ///< Corner 0. Corner k is `MFTL_TRI_FEATURE_VERTEX0 + k`.
434 MFTL_TRI_FEATURE_VERTEX1 = 2, ///< Corner 1.
435 MFTL_TRI_FEATURE_VERTEX2 = 3, ///< Corner 2.
436 MFTL_TRI_FEATURE_EDGE0 = 4, ///< Interior of edge 0 (corner 0 -> 1). Edge k is `MFTL_TRI_FEATURE_EDGE0 + k`.
437 MFTL_TRI_FEATURE_EDGE1 = 5, ///< Interior of edge 1 (corner 1 -> 2).
438 MFTL_TRI_FEATURE_EDGE2 = 6 ///< Interior of edge 2 (corner 2 -> 0).
439} MFTL_TriangleFeature;
440
441/**
442 * @brief Closest point of a (closed, solid) triangle to a point, and its feature.
443 *
444 * Ericson's region test (*Real-Time Collision Detection*, §5.1.5): six dot
445 * products select one of the seven Voronoi regions and only that region's
446 * formula is evaluated; corners are returned bit-exactly. A triangle with
447 * \f$|\vect{ab}\times\vect{ac}|^2 \le 10^{-15}|\vect{ab}|^2|\vect{ac}|^2\f$
448 * (a scale-free \f$\sin^2\f$ test) has no face region and is answered as the
449 * nearest of its three edges.
450 *
451 * @param p Query point.
452 * @param tri The three corners, in one periodic image (the caller
453 * applies any minimum-image shift).
454 * @param[out] closest The closest point, or #MFTL_NULL.
455 * @param[out] feature Its feature, or #MFTL_NULL.
456 * @return The SQUARED distance from @p p to the triangle.
457 *
458 * @rank_local Pure function.
459 */
460double MFTLClosestPointOnTriangle(const double p[3], const double tri[3][3], double closest[3], MFTL_TriangleFeature *feature);

466/**
467 * @brief Writes `tris.area`, `tris.normal` and `tris.center` for the first
468 * @p count triangles.
469 *
470 * With current `tris.verts` it streams compact indices with periodicity
471 * unswitched outside the loop; before derived connectivity exists it walks the
472 * half-edge records. The two paths are bit-identical
473 * (`test_triangle_geometry_paths_are_bit_identical`).
474 *
475 * @param obj The mesh.
476 * @param count `obj->tris.n_local` for owned triangles, `obj->tris.n` to include
477 * ghosts; only the latter sets #MFTL_DERIVED_GEOMETRY.
478 *
479 * @warning Before a redistribution the count must be `obj->tris.n`: ghosts are
480 * routed by their cached centre, and a stale ghost centre makes two
481 * ranks disagree about a destination (an owned triangle then silently
482 * loses an edge neighbour). MFTLRedistribute() recomputes a body whose
483 * bit is clear rather than trust it.
484 *
485 * @rank_local MFTLEnsembleUpdate() is the collective counterpart that also
486 * reduces quality statistics.
487 */
488void MFTLComputeTriangleGeometry(MFTL_Object *obj, int count);

490/**
491 * @brief How MFTLComputeVertexNormals() weights the faces around a vertex.
492 */
493typedef enum MFTL_NormalWeighting
494{
495 /**
496 * Weight each face by its AREA: \f$\vect{N}_v \propto \sum_{f \ni v}
497 * (\vect{e}_1\times\vect{e}_2)_f = 2\sum_f A_f\hat{\vect{n}}_f\f$, free given
498 * the cross product. Parallel to the gradient of the enclosed volume with
499 * respect to the vertex position, \f$\partial V/\partial\vect{x}_v =
500 * \tfrac13\sum_f A_f\hat{\vect{n}}_f\f$ -- the normal a volume-conserving
501 * front-tracking scheme wants. The default for moving vertices.
502 */
503 MFTL_NORMAL_WEIGHT_AREA = 0,
504
505 /**
506 * Weight each face by its INTERIOR ANGLE at the vertex (Thürmer & Wüthrich
507 * 1998): \f$\vect{N}_v \propto \sum_f \alpha_f(v)\,\hat{\vect{n}}_f\f$.
508 * Invariant under re-triangulation of the fan, and the pseudo-normal for which
509 * \f$\sign((\vect{x}-\vect{v})\cdot\vect{N}_v)\f$ is provably the correct
510 * inside/outside test when \f$\vect{v}\f$ is the closest surface point to
511 * \f$\vect{x}\f$ (Bærentzen & Aanæs 2005). Used for signed distances. One
512 * `atan2` per corner; degenerate faces contribute nothing.
513 */
514 MFTL_NORMAL_WEIGHT_ANGLE = 1
515} MFTL_NormalWeighting;
516
517/**
518 * @brief Unit vertex normals for every vertex of the pool, into `verts.normal`.
519 *
520 * One flat pass over ALL triangles (ghosts included) scatters each weighted face
521 * normal into its three corners, then one pass normalises; a vertex without
522 * incident triangles gets the zero vector. Requires current `tris.verts`.
523 *
524 * On a distributed mesh an OWNED vertex is exact without communication,
525 * because the redistribution halo is vertex-complete (every triangle incident
526 * to an owned vertex is present locally). A GHOST vertex misses its far-side
527 * faces and holds a normalised PARTIAL sum -- plausible-looking and wrong.
528 * MFTLEnsembleUpdateVertexNormals() adds the exchange that makes ghosts exact.
529 *
530 * @param obj The mesh; `verts.normal` is overwritten for every vertex.
531 * @param weighting #MFTL_NORMAL_WEIGHT_AREA or #MFTL_NORMAL_WEIGHT_ANGLE.
532 *
533 * @rank_local
534 */
535void MFTLComputeVertexNormals(MFTL_Object *obj, MFTL_NormalWeighting weighting);

541/**
542 * @brief Value of a lattice node no triangle came within the cutoff of: sign and

552/**
553 * @struct MFTL_SdfGrid
554 * @brief A dense WINDOW onto a (possibly periodic) Cartesian lattice of signed
555 * distances.
556 *
557 * The LATTICE is global: node g on axis k sits at `origin[k] + g*spacing[k]`,
558 * and on a periodic axis g and g + `n_global[k]` are the same node. Integer node
559 * coordinates are therefore an identity every rank computes identically, which
560 * lets a distributed marching cubes name a surface vertex by its lattice EDGE
561 * with no negotiation.
562 *
563 * The WINDOW is what one caller stores: `dims[k]` consecutive nodes from global
564 * index `lo[k]` (which may be negative or past the end on a periodic axis;
565 * indices are taken modulo `n_global`, and `dims <= n_global`).
566 * MFTLSdfGridInit() makes the window the whole lattice, non-periodic.
567 *
568 * `phi` is negative INSIDE (opposite the outward normals), positive outside,
569 * #MFTL_SDF_UNKNOWN where nothing was rasterised; x varies fastest.
570 */
571typedef struct MFTL_SdfGrid
572{
573 double origin[3]; ///< Coordinate of GLOBAL lattice node 0 on each axis.
574 double spacing[3]; ///< Node spacing per axis. May differ between axes.
575 int n_global[3]; ///< Nodes in the whole lattice per axis; the wrap period on a periodic axis.
576 bool periodic[3]; ///< Whether global indices wrap modulo `n_global` on this axis.
577 int lo[3]; ///< Global (unwrapped) index of the window's node 0.
578 int dims[3]; ///< Nodes the window stores per axis.
579 double *phi; ///< [dims[0]*dims[1]*dims[2]] values, x fastest.
580} MFTL_SdfGrid;

615/**
616 * @brief Window index of global node @p g on axis @p k, or -1 if the window
617 * does not hold it.
618 * @param grid The grid.
619 * @param k Axis.
620 * @param g Global node index, unwrapped: any integer on a periodic axis.
621 * @return The local index in `[0, dims[k])`, or -1.
622 */
623static inline int MFTLSdfGridLocal(const MFTL_SdfGrid *grid, int k, int g)
624{
625 int l = g - grid->lo[k];
626 if (grid->periodic[k])
627 {
628 const int n = grid->n_global[k];
629 l %= n;
630 if (l < 0) l += n;
631 }
632 return (l >= 0 && l < grid->dims[k]) ? l : -1;
633}

635/**
636 * @brief Coordinate of global node @p g on axis @p k, in the frame @p g lives in.
637 *
638 * Unwrapped on purpose: node `g` and node `g + n_global` have coordinates one
639 * period apart, which is what lets a triangle near the seam measure its distance
640 * to a node in its OWN periodic image.
641 *
642 * @param grid The grid.
643 * @param k Axis.
644 * @param g Global node index, unwrapped.
645 * @return `origin[k] + g * spacing[k]`.
646 */
647static inline double MFTLSdfGridCoord(const MFTL_SdfGrid *grid, int k, int g) { return grid->origin[k] + (double) g * grid->spacing[k]; }

649/**
650 * @brief Combines two contributions to one node: the smaller magnitude wins, a
651 * tie goes to the NEGATIVE value.
652 *
653 * A total order, hence commutative and associative: the merged field does not
654 * depend on the order in which triangles -- or ranks -- contribute, which makes
655 * a distributed rasterisation bit-identical to a serial one. #MFTL_SDF_UNKNOWN
656 * is the identity.
657 *
658 * @param current The node's value so far.
659 * @param candidate A new contribution.
660 * @return The merged value.
661 */
662static inline double MFTLSdfMerge(double current, double candidate)
663{
664 const double ac = fabs(current), an = fabs(candidate);
665 if (an < ac) return candidate;
666 if (an == ac && candidate < current) return candidate;
667 return current;
668}

670/**
671 * @brief Rasterises the signed distance to triangles `[0, n_tris)` onto every
672 * window node within @p cutoff of one of them.
673 *
674 * For each triangle T and each node \f$\vect{x}\f$ in T's box dilated by the
675 * cutoff, the closest point \f$\vect{q}\f$ and its feature are found with
676 * MFTLClosestPointOnTriangle(); the value is
677 * \f$\pm\norm{\vect{x}-\vect{q}}\f$ with the sign of
678 * \f$(\vect{x}-\vect{q})\cdot\vect{n}\f$, \f$\vect{n}\f$ the pseudo-normal of
679 * the feature: the face normal; for an edge the sum of the two incident unit
680 * face normals (T's alone at a mesh rim); for a vertex `verts.normal`, which
681 * must hold ANGLE-weighted normals. Values are merged with MFTLSdfMerge(). For a
682 * closed, consistently oriented, non-self-intersecting surface this yields the
683 * exact signed distance (Bærentzen & Aanæs 2005); area-weighted vertex normals
684 * can give the wrong sign near thin, sharply folded sheets.
685 *
686 * **Cutoff.** Nodes farther than @p cutoff from every triangle stay
687 * #MFTL_SDF_UNKNOWN. Every corner of a lattice cell crossed by the surface lies
688 * within one cell diagonal of it, so a cutoff above the diagonal
689 * (\f$\sqrt3\,\Delta x\f$ on a cubic lattice) provides every value marching
690 * cubes needs, with no flood fill and no global pass.
691 *
692 * **Periodicity.** Triangles are unwrapped around corner 0 and reach nodes by
693 * UNWRAPPED global index, so each distance is measured in the triangle's own
694 * image and the window's modular indexing stores it where it belongs.
695 *
696 * @param grid The grid; values are MERGED, so several objects or ranks may
697 * rasterise into one.
698 * @param obj The mesh: current `tris.verts`, and angle-weighted
699 * `verts.normal` for every corner of the triangles used, ghosts
700 * included (MFTLEnsembleUpdateVertexNormals() with
701 * #MFTL_NORMAL_WEIGHT_ANGLE on a distributed mesh).
702 * @param n_tris Rasterise `[0, n_tris)`; `tris.n_local` in a distributed
703 * rasterisation, so no triangle is counted by two ranks.
704 * @param cutoff Distance within which nodes are written; > 0.
705 * @return The number of node updates that changed a value.
706 * @rank_local No MPI.
707 */
708long MFTLRasterizeSignedDistance(MFTL_SdfGrid *grid, const MFTL_Object *obj, int n_tris, double cutoff);
