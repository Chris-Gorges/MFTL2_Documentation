22/**
23 * @brief Is @p tri_verts a usable index list: every index in range and no
24 * triangle with a repeated corner (a self-edge that could twin with
25 * itself)? Checked in Release, before anything is written.
26 *
27 * @param tri_verts [3 * n_triangles] vertex indices.
28 * @param n_triangles Number of triangles.
29 * @param n_vertices Size of the vertex pool the indices refer to.
30 * @return `true` if every triangle has three distinct, in-range corners.
31 */
32static bool index_list_is_valid(const int *tri_verts, int n_triangles, int n_vertices)
33{
34 for (int t = 0; t < n_triangles; ++t)
35 {
36 const int a = tri_verts[3 * t + 0], b = tri_verts[3 * t + 1], c = tri_verts[3 * t + 2];
37 if (a < 0 || a >= n_vertices || b < 0 || b >= n_vertices || c < 0 || c >= n_vertices) return MFTL_FALSE;
38 if (a == b || b == c || c == a) return MFTL_FALSE;
39 }
40 return MFTL_TRUE;
41}

43/**
44 * @brief Builds triangles, half-edges, twins, the boundary flag and the derived
45 * arrays from a VALIDATED index list.
46 *
47 * @param object Object with a filled vertex pool.
48 * @param tri_verts [3 * n_triangles] vertex indices, validated by the caller.
49 * @param n_triangles Number of triangles, >= 1.
50 */
51static void build_topology(MFTL_Object *object, const int *tri_verts, int n_triangles)
52{
53 const int n_vertices = object->verts.n;
54 const int rank = (object->domain != MFTL_NULL) ? object->domain->rank : 0;
55
56 /* Grows the triangle pool (half-edges and derived arrays included). */
57 MFTLObjectSetCounts(object, n_vertices, object->verts.n_local, n_triangles, n_triangles);
58
59 /* Dense ids: triangles and half-edges restart at zero (the previous
60 * generation is retired); vertex ids belong to the caller. */
61 object->next_triangle_gid = n_triangles;
62 object->next_halfedge_gid = 3 * n_triangles;
63
64 MFTL_Vertex *const vr = object->verts.rec;
65 MFTL_Triangle *const tr = object->tris.rec;
66 MFTL_HalfEdge *const he = object->tris.he;
67
68 /* Re-derived below; a stale value would index a retired half-edge. */
69 for (int v = 0; v < n_vertices; ++v) vr[v].out_he_idx = MFTL_ERROR;
70
71 for (int t = 0; t < n_triangles; ++t)
72 {
73 tr[t].global_id = t;
74 tr[t].owner_proc = rank;
75 tr[t].flags = 0;
76
77 for (int k = 0; k < 3; ++k)
78 {
79 const int origin = tri_verts[3 * t + k];
80 const int target = tri_verts[3 * t + (k + 1) % 3];
81
82 const int h = 3 * t + k; // implicit indexing: no he_idx, next_idx or triangle_idx to write
83 he[h].global_id = h;
84 he[h].vertex_idx = target; // half-edges store their TARGET
85 he[h].twin_idx = MFTL_ERROR; // MFTLRebuildTwins links them below
86 he[h].flags = 0;
87
88 /* Any outgoing half-edge is valid (fan walks rewind to the start). */
89 vr[origin].out_he_idx = h;
90 }
91 }
92
93 MFTLRebuildTwins(object);
94
95 /* The mesh is complete here, so "no twin" means "mesh boundary". */
96 MFTLMarkMeshBoundary(object);
97
98 MFTLObjectBuildDerived(object);
99}

101int MFTLBuildTopologyFromIndexList(MFTL_Object *object, const int *tri_verts, int n_triangles)
102{
103 MFTL_ASSERT(object != MFTL_NULL);
104 MFTL_ASSERT(tri_verts != MFTL_NULL);
105 MFTL_ASSERT(object->verts.n >= 3 && "MFTLBuildTopologyFromIndexList: the vertex pool must be filled first");
106
107 if (object == MFTL_NULL || tri_verts == MFTL_NULL) return MFTL_ERROR;
108 if (n_triangles < 1 || object->verts.n < 3) return MFTL_ERROR;
109 if (!index_list_is_valid(tri_verts, n_triangles, object->verts.n)) return MFTL_ERROR;
110
111 build_topology(object, tri_verts, n_triangles);
112 return MFTL_SUCCESS;
113}

115int MFTLBuildIndexedMesh(MFTL_Object *object, const double *positions, int n_vertices, const int *tri_verts, int n_triangles)
116{
117 MFTL_ASSERT(object != MFTL_NULL);
118 MFTL_ASSERT(object->tris.n == 0 && object->verts.n == 0 && "MFTLBuildIndexedMesh: object is not empty");
119
120 if (object == MFTL_NULL || positions == MFTL_NULL || tri_verts == MFTL_NULL) return MFTL_ERROR;
121 if (n_vertices < 3 || n_triangles < 1) return MFTL_ERROR;
122 if (!index_list_is_valid(tri_verts, n_triangles, n_vertices)) return MFTL_ERROR;
123
124 const int rank = (object->domain != MFTL_NULL) ? object->domain->rank : 0;
125
126 MFTLObjectSetCounts(object, n_vertices, n_vertices, 0, 0);
127
128 /* The caller's layout is the pool's layout (xyz interleaved). */
129 memcpy(object->verts.pos, positions, 3 * (size_t) n_vertices * sizeof(double));
130
131 MFTL_Vertex *const vr = object->verts.rec;
132 for (int v = 0; v < n_vertices; ++v)
133 {
134 vr[v].global_id = v;
135 vr[v].owner_proc = rank;
136 vr[v].out_he_idx = MFTL_ERROR; // filled in by the topology pass
137 vr[v].flags = 0;
138 }
139 object->next_vertex_gid = n_vertices;
140
141 build_topology(object, tri_verts, n_triangles);
142
143 /* A new mesh leaves with current geometry, so the first MFTLRedistribute()
144 * routes by real centres. Without a domain nothing can route it yet. */
145 if (object->domain != MFTL_NULL) MFTLComputeTriangleGeometry(object, object->tris.n);
146 return MFTL_SUCCESS;
147}
